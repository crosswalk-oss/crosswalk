// Messaging helpers: channels (project team channels and direct conversations), unread tracking and deep links.
// Deep links use plain hashes the router passes through as the view name:
//   #messages            → the inbox of conversations
//   #messages-1042       → PRJ-1042's team channel (created on first open)
//   #messages-dm-u-pc1   → direct conversation with that person (created on first message)
import type { AppState, Channel, ID, Message } from '@/core/types'
import { nowISO, uid } from '@/core/util'

/** Link to a project's team channel. Accepts 'prj-1042' or 'PRJ-1042'. */
export const messagesHref = (projectIdOrCode?: string) => (projectIdOrCode ? `#messages-${projectIdOrCode.replace(/\D/g, '')}` : '#messages')
/** Link to a direct conversation with a person. */
export const dmHref = (personId: ID) => `#messages-dm-${personId}`

export type MessageTarget = { kind: 'project'; projectId: ID } | { kind: 'direct'; personId: ID } | { kind: 'none' }

/** Parse the router's view name ('messages', 'messages-1042', 'messages-dm-u-pc1'). */
export function parseMessagesView(view: string): MessageTarget {
  const dm = view.match(/^messages-dm-(.+)$/)
  if (dm) return { kind: 'direct', personId: dm[1] }
  const pj = view.match(/^messages-(\d+)$/)
  if (pj) return { kind: 'project', projectId: `prj-${pj[1]}` }
  return { kind: 'none' }
}
export const isMessagesView = (view: string) => view === 'messages' || view.startsWith('messages-')

export const isMember = (c: Channel, me: ID) => c.memberIds.includes(me)
export const channelsFor = (s: Pick<AppState, 'channels'>, me: ID) => s.channels.filter((c) => isMember(c, me))
export const messagesIn = (s: Pick<AppState, 'messages'>, channelId: ID) => s.messages.filter((m) => m.channelId === channelId)
export const isUnread = (m: Message, me: ID) => m.fromId !== me && !m.readBy.includes(me)

export function unreadIn(s: Pick<AppState, 'messages'>, channelId: ID, me: ID): number {
  let n = 0
  for (const m of s.messages) if (m.channelId === channelId && isUnread(m, me)) n++
  return n
}

/** Unread messages across every channel the person belongs to (nav badge). */
export function unreadCount(s: Pick<AppState, 'channels' | 'messages'>, me: ID): number {
  const mine = new Set(channelsFor(s, me).map((c) => c.id))
  let n = 0
  for (const m of s.messages) if (mine.has(m.channelId) && isUnread(m, me)) n++
  return n
}

export const otherMember = (c: Channel, me: ID) => c.memberIds.find((x) => x !== me) ?? me

export function channelTitle(s: Pick<AppState, 'people' | 'projects'>, c: Channel, me: ID): string {
  if (c.kind === 'direct') return s.people.find((p) => p.id === otherMember(c, me))?.name ?? 'Direct message'
  const p = s.projects.find((x) => x.id === c.projectId)
  return p ? `${p.code} · ${p.name}` : c.name
}

export const findProjectChannel = (s: Pick<AppState, 'channels'>, projectId: ID) => s.channels.find((c) => c.kind === 'project' && c.projectId === projectId)
export const findDirect = (s: Pick<AppState, 'channels'>, a: ID, b: ID) =>
  s.channels.find((c) => c.kind === 'direct' && c.memberIds.length === 2 && c.memberIds.includes(a) && c.memberIds.includes(b))

/** Project team channel; created with the project team when missing, and `me` joins it on open. */
export function ensureProjectChannel(d: AppState, projectId: ID, me: ID): Channel | undefined {
  const p = d.projects.find((x) => x.id === projectId)
  if (!p) return
  let c = findProjectChannel(d, projectId)
  if (!c) {
    const members = Array.from(new Set([...(Object.values(p.team).filter(Boolean) as string[]), me]))
    c = { id: uid('ch'), kind: 'project', projectId, memberIds: members, name: `${p.code} team` }
    d.channels.push(c)
  } else if (!c.memberIds.includes(me)) c.memberIds.push(me)
  return c
}

export function ensureDirect(d: AppState, me: ID, other: ID): Channel {
  let c = findDirect(d, me, other)
  if (!c) {
    c = { id: uid('ch'), kind: 'direct', memberIds: [me, other], name: '' }
    d.channels.push(c)
  }
  return c
}

/** Post a message. Direct messages also raise a notification for the recipient. */
export function sendMessage(d: AppState, channelId: ID, fromId: ID, text: string): Message | undefined {
  const c = d.channels.find((x) => x.id === channelId)
  const body = text.trim()
  if (!c || !body) return
  if (!c.memberIds.includes(fromId)) c.memberIds.push(fromId)
  const m: Message = { id: uid('msg'), channelId, fromId, at: nowISO(), text: body, readBy: [fromId] }
  d.messages.push(m)
  const from = d.people.find((p) => p.id === fromId)?.name ?? 'Someone'
  if (c.kind === 'direct') {
    for (const to of c.memberIds.filter((x) => x !== fromId)) d.notifications.push({ id: uid('ntf'), at: m.at, toId: to, text: `${from}: ${body.length > 80 ? body.slice(0, 77) + '…' : body}`, link: dmHref(fromId), read: false, kind: 'message' })
  }
  return m
}

/** Mark every message in a channel read for `me`. Returns how many changed. */
export function markRead(d: AppState, channelId: ID, me: ID): number {
  let n = 0
  for (const m of d.messages) if (m.channelId === channelId && isUnread(m, me)) { m.readBy.push(me); n++ }
  // message notifications for that conversation are read too
  const c = d.channels.find((x) => x.id === channelId)
  if (c?.kind === 'direct') for (const nt of d.notifications) if (nt.toId === me && nt.kind === 'message' && nt.link === dmHref(otherMember(c, me))) nt.read = true
  return n
}

export const lastMessage = (s: Pick<AppState, 'messages'>, channelId: ID): Message | undefined => {
  let last: Message | undefined
  for (const m of s.messages) if (m.channelId === channelId && (!last || m.at > last.at)) last = m
  return last
}

// Messages (#messages): project team channels and direct conversations, kept inside Crosswalk next to the records
// they are about. Unread tracking per person (readBy). Deep links: #messages-1042 (project team), #messages-dm-<person>.
import { useEffect, useMemo, useRef, useState } from 'react'
import type { AppState, Channel, Message, Person } from '@/core/types'
import { mutate, personName, useMe, useStore } from '@/core/store'
import { go, projectHref } from '@/core/router'
import { cx, fmtDate } from '@/core/util'
import { Avatar, Badge, Button, Empty, Field, Input, Modal, PageHeader, SegButtons, Select, TextArea } from '@/ui/kit'
import { toast } from '@/ui/toast'
import { Icon } from '@/ui/icons'
import { ProgramTag } from '@/views/shared/pickers'
import { channelTitle, channelsFor, dmHref, ensureDirect, ensureProjectChannel, findDirect, findProjectChannel, lastMessage, markRead, messagesHref, messagesIn, otherMember, parseMessagesView, sendMessage, unreadCount, unreadIn } from './lib'

const timeOf = (iso: string) => new Date(iso).toLocaleTimeString('en-US', { hour: 'numeric', minute: '2-digit' })
const dayOf = (iso: string) => iso.slice(0, 10)

function shortWhen(iso: string, asOf: string) {
  const d = dayOf(iso)
  return d === asOf ? timeOf(iso) : fmtDate(d).replace(/, \d{4}$/, '')
}

const hrefFor = (c: Channel, me: string) => (c.kind === 'project' ? messagesHref(c.projectId) : dmHref(otherMember(c, me)))

export default function Messages({ view = 'messages' }: { view?: string }) {
  const s = useStore()
  const me = useMe()
  const target = parseMessagesView(view)
  const [q, setQ] = useState('')
  const [newOpen, setNewOpen] = useState(false)
  const mine = useMemo(() => channelsFor(s, me.id).map((c) => ({ c, last: lastMessage(s, c.id), unread: unreadIn(s, c.id, me.id) }))
    .sort((a, b) => (b.last?.at ?? '').localeCompare(a.last?.at ?? '')), [s.channels, s.messages, me.id]) // eslint-disable-line react-hooks/exhaustive-deps

  // Resolve the open conversation from the route.
  const projectChannel = target.kind === 'project' ? findProjectChannel(s, target.projectId) : undefined
  const directChannel = target.kind === 'direct' ? findDirect(s, me.id, target.personId) : undefined
  const selected: Channel | undefined = target.kind === 'project' ? projectChannel : target.kind === 'direct' ? directChannel : mine[0]?.c
  const dmPerson = target.kind === 'direct' ? s.people.find((p) => p.id === target.personId) : undefined
  const targetProject = target.kind === 'project' ? s.projects.find((p) => p.id === target.projectId) : undefined

  // Opening a project's team channel creates it (with the project team) or adds you to it.
  useEffect(() => {
    if (target.kind !== 'project' || !targetProject) return
    const c = findProjectChannel(s, target.projectId)
    if (c && c.memberIds.includes(me.id)) return
    mutate((d) => { ensureProjectChannel(d, target.projectId, me.id) }, c ? undefined : { entity: 'channel', entityId: target.projectId, projectId: target.projectId, action: 'create', summary: `Opened the ${targetProject.code} team channel` })
  }, [target.kind === 'project' ? target.projectId : '', me.id, !!targetProject]) // eslint-disable-line react-hooks/exhaustive-deps

  // Reading a conversation marks it read for you.
  const unreadHere = selected ? unreadIn(s, selected.id, me.id) : 0
  useEffect(() => { if (selected && unreadHere > 0 && selected.memberIds.includes(me.id)) mutate((d) => { markRead(d, selected.id, me.id) }) }, [selected?.id, unreadHere, me.id]) // eslint-disable-line react-hooks/exhaustive-deps

  const list = mine.filter(({ c }) => !q || channelTitle(s, c, me.id).toLowerCase().includes(q.toLowerCase()))
  const projects = list.filter((x) => x.c.kind === 'project')
  const directs = list.filter((x) => x.c.kind === 'direct')
  const total = unreadCount(s, me.id)
  const showThread = target.kind !== 'none'
  const threadOpen = !!(selected || dmPerson)

  const row = ({ c, last, unread }: (typeof mine)[number]) => {
    const active = selected?.id === c.id
    const p = c.kind === 'project' ? s.projects.find((x) => x.id === c.projectId) : undefined
    const other = c.kind === 'direct' ? s.people.find((x) => x.id === otherMember(c, me.id)) : undefined
    return (
      <li key={c.id}>
        <a href={hrefFor(c, me.id)} className={cx('flex items-start gap-2.5 border-l-2 px-3 py-2', active ? 'border-marker bg-marker/10' : 'border-transparent hover:bg-sunken/60')}>
          {other ? <Avatar name={other.name} size={28} /> : <span className="mt-0.5 flex h-7 w-7 shrink-0 items-center justify-center rounded-sm bg-sunken font-mono text-[10px] font-bold text-muted">#</span>}
          <span className="min-w-0 flex-1">
            <span className="flex items-center gap-1.5">
              {p && <ProgramTag programId={p.programId} />}
              <span className={cx('min-w-0 truncate text-[13px]', unread ? 'font-bold' : 'font-semibold')}>{p ? p.code : other?.name ?? c.name}</span>
              {last && <span className="ml-auto shrink-0 text-[11px] text-faint">{shortWhen(last.at, s.settings.asOf)}</span>}
            </span>
            <span className="flex items-center gap-1.5">
              <span className={cx('min-w-0 flex-1 truncate text-[12px]', unread ? 'text-ink' : 'text-muted')}>{last ? `${last.fromId === me.id ? 'You' : personName(s, last.fromId).split(' ')[0]}: ${last.text}` : p ? p.name : 'No messages yet'}</span>
              {unread > 0 && <span className="shrink-0 rounded-full bg-marker px-1.5 text-[10.5px] font-bold tabular-nums text-markerink">{unread}</span>}
            </span>
          </span>
        </a>
      </li>
    )
  }

  return (
    <div className="flex flex-col gap-4">
      <PageHeader eyebrow="Work" title="Messages"
        subtitle={<>Project team channels and direct conversations, kept with the project record. {total > 0 ? <b className="text-ink">{total} unread.</b> : 'All caught up.'}</>}
        actions={<Button variant="primary" icon={<Icon name="plus" className="h-4 w-4" />} onClick={() => setNewOpen(true)}>New message</Button>} />

      <div className="grid min-h-[560px] grid-cols-1 overflow-hidden rounded-md border border-line bg-surface lg:grid-cols-[300px_minmax(0,1fr)]">
        <aside className={cx('min-w-0 flex-col border-line lg:flex lg:border-r', showThread ? 'hidden' : 'flex')}>
          <div className="border-b border-line p-2.5"><Input aria-label="Search conversations" placeholder="Search conversations" value={q} onChange={(e) => setQ(e.target.value)} className="h-8" /></div>
          <div className="flex-1 overflow-y-auto lg:max-h-[640px]">
            <div className="eyebrow px-3 pb-1 pt-3">Project channels <span className="font-normal normal-case tracking-normal text-faint">{projects.length}</span></div>
            {projects.length ? <ul>{projects.map(row)}</ul> : <p className="px-3 pb-2 text-[12px] text-muted">Open a project and choose “Message the team”.</p>}
            <div className="eyebrow flex items-center px-3 pb-1 pt-3">Direct messages <span className="ml-1 font-normal normal-case tracking-normal text-faint">{directs.length}</span><button className="ml-auto text-[11.5px] font-semibold normal-case tracking-normal text-link" onClick={() => setNewOpen(true)}>+ New</button></div>
            {directs.length ? <ul>{directs.map(row)}</ul> : <p className="px-3 pb-2 text-[12px] text-muted">No direct conversations yet.</p>}
          </div>
          <p className="border-t border-line px-3 py-2 text-[11px] leading-snug text-faint">Messages are delivered in Crosswalk. Email copies to Outlook need the backend mail integration (Configuration → Workflow notifications).</p>
        </aside>

        <section className={cx('min-w-0 flex-col', showThread ? 'flex' : 'hidden lg:flex')}>
          {threadOpen ? <Thread key={selected?.id ?? dmPerson?.id} s={s} me={me} channel={selected} dmPerson={dmPerson} />
            : target.kind === 'project' && !targetProject ? <div className="p-6"><Empty title="Project not found">That project number does not exist.</Empty></div>
              : <div className="flex flex-1 items-center justify-center p-6"><Empty title="No conversation selected" action={<Button onClick={() => setNewOpen(true)}>Start a conversation</Button>}>Pick a channel on the left, or message anyone in the organization.</Empty></div>}
        </section>
      </div>

      <NewMessage open={newOpen} onClose={() => setNewOpen(false)} s={s} me={me} />
    </div>
  )
}

function Thread({ s, me, channel, dmPerson }: { s: AppState; me: Person; channel?: Channel; dmPerson?: Person }) {
  const [text, setText] = useState('')
  const [showMembers, setShowMembers] = useState(false)
  const [addId, setAddId] = useState('')
  const bottom = useRef<HTMLDivElement>(null)
  const msgs = channel ? messagesIn(s, channel.id).slice().sort((a, b) => a.at.localeCompare(b.at)) : []
  const project = channel?.kind === 'project' ? s.projects.find((p) => p.id === channel.projectId) : undefined
  const other = channel?.kind === 'direct' ? s.people.find((p) => p.id === otherMember(channel, me.id)) : dmPerson
  const members = channel ? channel.memberIds.map((id) => s.people.find((p) => p.id === id)).filter((p): p is Person => !!p) : other ? [me, other] : [me]
  useEffect(() => { bottom.current?.scrollIntoView?.({ block: 'end' }) }, [msgs.length, channel?.id])

  const send = () => {
    const body = text.trim()
    if (!body) return
    mutate((d) => {
      const c = channel ? d.channels.find((x) => x.id === channel.id)! : ensureDirect(d, me.id, other!.id)
      sendMessage(d, c.id, me.id, body)
    }, { entity: 'message', entityId: channel?.id ?? `dm-${other?.id}`, projectId: project?.id, action: 'send', summary: `Message to ${project ? `${project.code} team` : other?.name}: ${body.length > 60 ? body.slice(0, 57) + '…' : body}` })
    setText('')
  }
  const addMember = () => {
    if (!channel || !addId) return
    mutate((d) => { const c = d.channels.find((x) => x.id === channel.id); if (c && !c.memberIds.includes(addId)) c.memberIds.push(addId) },
      { entity: 'channel', entityId: channel.id, projectId: project?.id, action: 'add-member', summary: `Added ${personName(s, addId)} to ${channel.name || 'conversation'}` })
    toast(`${personName(s, addId)} added to the channel`)
    setAddId('')
  }

  // group consecutive messages from the same person within 10 minutes; date dividers between days
  const blocks: { day?: string; head: boolean; m: Message }[] = msgs.map((m, i) => {
    const prev = msgs[i - 1]
    const newDay = !prev || dayOf(prev.at) !== dayOf(m.at)
    const head = newDay || prev.fromId !== m.fromId || Date.parse(m.at) - Date.parse(prev.at) > 10 * 60e3
    return { day: newDay ? dayOf(m.at) : undefined, head, m }
  })

  return (
    <div className="flex min-h-[560px] flex-1 flex-col">
      <header className="flex flex-wrap items-center gap-2 border-b border-line px-4 py-2.5">
        <a href="#messages" className="rounded p-1 text-muted hover:bg-sunken lg:hidden" aria-label="Back to conversations"><Icon name="arrow" className="h-4 w-4 rotate-180" /></a>
        {other && !project && <Avatar name={other.name} size={28} />}
        <div className="min-w-0 flex-1">
          <h2 className="truncate font-display text-[15px] font-semibold">{project ? <span className="flex items-center gap-2"><ProgramTag programId={project.programId} /><span className="font-mono">{project.code}</span><span className="truncate">{project.name}</span></span> : other?.name}</h2>
          <div className="truncate text-[11.5px] text-muted">{project ? `Project team · ${members.length} members` : other ? `${other.title}${other.company ? ` · ${other.company}` : ''} · ${other.email}` : ''}</div>
        </div>
        {project && <a href={projectHref(project.id)} className="text-[12px] font-semibold text-link">Open project →</a>}
        {channel && <Button size="sm" variant="ghost" onClick={() => setShowMembers(!showMembers)}>{showMembers ? 'Hide members' : `Members (${members.length})`}</Button>}
      </header>
      {showMembers && channel && (
        <div className="flex flex-col gap-2 border-b border-line bg-sunken/40 px-4 py-2.5">
          <div className="flex flex-wrap gap-1.5">{members.map((p) => <span key={p.id} className="flex items-center gap-1.5 rounded border border-line bg-surface px-1.5 py-0.5 text-[12px]"><Avatar name={p.name} size={18} />{p.name}<span className="text-faint">{p.title}</span></span>)}</div>
          {channel.kind === 'project' && (
            <div className="flex flex-wrap items-center gap-2">
              <select aria-label="Add a member" className="inp h-8 w-auto max-w-full" value={addId} onChange={(e) => setAddId(e.target.value)}>
                <option value="">Add someone to this channel…</option>
                {s.people.filter((p) => !channel.memberIds.includes(p.id)).map((p) => <option key={p.id} value={p.id}>{p.name} · {p.title}{p.company ? ` (${p.company})` : ''}</option>)}
              </select>
              <Button size="sm" disabled={!addId} onClick={addMember}>Add</Button>
            </div>
          )}
        </div>
      )}
      <div className="flex-1 overflow-y-auto px-4 py-3 lg:max-h-[540px]">
        {msgs.length === 0 && <div className="py-10"><Empty title={project ? `Start the ${project.code} team conversation` : `Message ${other?.name ?? ''}`}>{project ? 'Everyone on the project team sees what you post here.' : 'Only the two of you see this conversation.'}</Empty></div>}
        {blocks.map(({ day, head, m }) => (
          <div key={m.id}>
            {day && <div className="my-3 flex items-center gap-3 text-[11px] font-semibold uppercase tracking-wide text-faint"><span className="h-px flex-1 bg-line" />{day === s.settings.asOf ? 'Today' : fmtDate(day)}<span className="h-px flex-1 bg-line" /></div>}
            <div className={cx('group flex gap-2.5 rounded px-1 hover:bg-sunken/40', head ? 'mt-2 pt-1' : 'mt-0')}>
              <div className="w-8 shrink-0">{head && <Avatar name={personName(s, m.fromId)} size={30} />}</div>
              <div className="min-w-0 flex-1 pb-0.5">
                {head && <div className="flex flex-wrap items-baseline gap-x-2"><b className="text-[13px]">{m.fromId === me.id ? 'You' : personName(s, m.fromId)}</b><span className="text-[11px] text-faint">{s.people.find((p) => p.id === m.fromId)?.title}</span><span className="text-[11px] text-faint">{timeOf(m.at)}</span></div>}
                <p className="whitespace-pre-wrap break-words text-[13px] leading-relaxed">{m.text}</p>
              </div>
            </div>
          </div>
        ))}
        {msgs.length > 0 && (() => {
          const last = msgs[msgs.length - 1]
          const seen = last.readBy.filter((x) => x !== last.fromId).map((x) => (x === me.id ? 'you' : personName(s, x).split(' ')[0]))
          return last.fromId === me.id ? <div className="mt-1 pl-11 text-[11px] text-faint">{seen.length ? `Seen by ${seen.join(', ')}` : 'Delivered'}</div> : null
        })()}
        <div ref={bottom} />
      </div>
      <form className="flex items-end gap-2 border-t border-line p-3" onSubmit={(e) => { e.preventDefault(); send() }}>
        <TextArea id="message-text" aria-label="Message" rows={2} className="min-h-[44px] flex-1 resize-none" placeholder={project ? `Message the ${project.code} team` : `Message ${other?.name ?? ''}`} value={text}
          onChange={(e) => setText(e.target.value)} onKeyDown={(e) => { if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); send() } }} />
        <Button type="submit" variant="primary" disabled={!text.trim()}>Send</Button>
      </form>
    </div>
  )
}

function NewMessage({ open, onClose, s, me }: { open: boolean; onClose: () => void; s: AppState; me: Person }) {
  const [mode, setMode] = useState<'direct' | 'project'>('direct')
  const [q, setQ] = useState('')
  const [projectId, setProjectId] = useState('')
  useEffect(() => { if (open) { setQ(''); setProjectId('') } }, [open])
  const people = s.people.filter((p) => p.id !== me.id && (!q || `${p.name} ${p.title} ${p.company ?? ''}`.toLowerCase().includes(q.toLowerCase())))
  const pick = (href: string) => { onClose(); go(href) }
  return (
    <Modal open={open} onClose={onClose} title="New message"
      footer={mode === 'project' ? <><Button variant="ghost" onClick={onClose}>Cancel</Button><Button variant="primary" disabled={!projectId} onClick={() => pick(messagesHref(projectId))}>Open team channel</Button></> : <Button variant="ghost" onClick={onClose}>Cancel</Button>}>
      <div className="flex flex-col gap-3">
        <SegButtons<'direct' | 'project'> value={mode} onChange={setMode} options={[{ id: 'direct', label: 'Direct message' }, { id: 'project', label: 'Project team' }]} />
        {mode === 'direct' ? (
          <>
            <Input autoFocus aria-label="Find a person" placeholder="Find a person by name, title or company" value={q} onChange={(e) => setQ(e.target.value)} />
            <ul className="max-h-[320px] overflow-y-auto rounded border border-line">
              {people.map((p) => {
                const existing = findDirect(s, me.id, p.id)
                return (
                  <li key={p.id} className="border-b border-line/60 last:border-0">
                    <button className="flex w-full items-center gap-2.5 px-3 py-2 text-left hover:bg-sunken/60" onClick={() => pick(dmHref(p.id))}>

// Portfolio planning model (V2): benefit-cost ranking, fit-to-funding scenarios and capacity.
import type { AppState, Driver, ID, Project, Risk } from '@/core/types'
import { activeBudget, position } from '@/core/engine/position'
import { estimateTotals, spreadByYear } from '@/core/engine/estimates'
import { expectedCost, isActive, riskScore } from '@/core/engine/risk'
import { templateFor } from '@/core/engine/gates'
import { sum } from '@/core/util'

export interface Weights { driver: Record<Driver, number>; hca: number; dac: number }
export const DEFAULT_WEIGHTS: Weights = {
  driver: { 'Safety / compliance': 1.5, Reliability: 1.2, Environmental: 1.1, Capacity: 1.0, 'Customer / new business': 0.9, Relocation: 0.8 },
  hca: 1.25,
  dac: 1.1,
}

export interface ScenarioItem { include: boolean; defer: number; mandatory?: boolean }
export interface Scenario {
  id: string
  name: string
  createdAt: string
  basis: 'budget' | 'eac'
  weights: Weights
  items: Record<ID, ScenarioItem>
  capacityLimit: number
}

export interface Candidate {
  project: Project
  stageName: string
  /** Committed = control budget in force; scope is authorized and spend is under contract. */
  committed: boolean
  proposed: boolean // no budget yet (estimate only)
  mandatoryDefault: boolean
  costBasis: 'control budget' | 'planning budget' | 'estimate'
  byYear: Record<string, number> // planned spend by year on the chosen basis
  remaining: number // forecast spend not yet incurred
  riskReduced: number // inherent − residual expected cost of active threats
  inherent: number
  residual: number
  constructionWindow?: { start: string; end: string }
}

/** Residual expected cost: scale the inherent expected cost by residual / inherent score. */
export function residualExpected(r: Risk): number {
  const inh = riskScore(r)
  const res = r.residual.likelihood * r.residual.consequence
  return expectedCost(r) * (inh ? res / inh : 1)
}

export function riskReducedFor(s: Pick<AppState, 'risks'>, pid: string): { inherent: number; residual: number; reduced: number } {
  const rs = s.risks.filter((r) => r.projectId === pid && isActive(r) && r.kind === 'threat')
  const inherent = sum(rs.map(expectedCost))
  const residual = sum(rs.map(residualExpected))
  return { inherent: Math.round(inherent), residual: Math.round(residual), reduced: Math.round(inherent - residual) }
}

const ms = (p: Project, ...names: string[]) => { for (const n of names) { const m = p.milestones.find((x) => x.name === n); if (m) return m.actual ?? m.forecast } return undefined }

export function candidates(s: AppState, basis: 'budget' | 'eac' = 'budget'): Candidate[] {
  const asOfYear = Number(s.settings.asOf.slice(0, 4))
  const out: Candidate[] = []
  for (const p of s.projects) {
    if (p.status === 'closed' || p.status === 'cancelled') continue
    const tpl = templateFor(s, p)
    const b = activeBudget(s, p.id)
    const pos = position(s, p.id)
    let byYear: Record<string, number> = {}
    let costBasis: Candidate['costBasis'] = 'estimate'
    if (b) {
      byYear = { ...b.byYear }
      costBasis = b.kind === 'control' ? 'control budget' : 'planning budget'
      if (basis === 'eac' && pos.eac && b.total) {
        // spread the forecast overrun/underrun across the remaining years
        const delta = pos.eac - pos.currentBudget
        const fut = Object.keys(byYear).filter((y) => Number(y) >= asOfYear)
        const futTotal = sum(fut.map((y) => byYear[y]))
        if (fut.length && futTotal > 0) for (const y of fut) byYear[y] = Math.round(byYear[y] + (delta * byYear[y]) / futTotal)
      }
    } else {
      const est = s.estimates.filter((e) => e.projectId === p.id && e.status !== 'rejected' && e.status !== 'superseded').sort((a, c) => a.version - c.version).pop()
      const total = est ? estimateTotals(est).total : 0
      const start = ms(p, 'Business case approved', 'Plan approved') ?? s.settings.asOf
      const end = ms(p, 'In service', 'In service & closed') ?? start
      byYear = total ? spreadByYear(total, start, end) : {}
    }
    const r = riskReducedFor(s, p.id)
    const cs = ms(p, 'Construction start', 'Construction authorized')
    const ce = ms(p, 'In service', 'In service & closed')
    const spentYears = Object.keys(byYear).filter((y) => Number(y) < asOfYear)
    const remaining = Math.max(0, sum(Object.values(byYear)) - Math.max(pos.actualToDate, sum(spentYears.map((y) => byYear[y]))))
    out.push({
      project: p,
      stageName: tpl.stages[p.stageIndex]?.name ?? '—',
      committed: b?.kind === 'control',
      proposed: !b,
      mandatoryDefault: p.driver === 'Safety / compliance',
      costBasis,
      byYear,
      remaining,
      riskReduced: r.reduced,
      inherent: r.inherent,
      residual: r.residual,
      constructionWindow: cs && ce ? { start: cs, end: ce } : undefined,
    })
  }
  return out
}

export interface Ranked extends Candidate { perMillion: number; weight: number; score: number; mandatory: boolean; rank: number }

/** Benefit-cost: risk $ reduced per $M of remaining spend, times driver and consequence weights. Normalized so the best = 100. */
export function rank(cs: Candidate[], sc: Pick<Scenario, 'weights' | 'items'>): Ranked[] {
  const raw = cs.map((c) => {
    const denom = Math.max(c.remaining, 250_000) / 1e6
    const perMillion = c.riskReduced / denom
    const w = (sc.weights.driver[c.project.driver] ?? 1) * (c.project.location.hca ? sc.weights.hca : 1) * (c.project.location.dac ? sc.weights.dac : 1)
    const mandatory = sc.items[c.project.id]?.mandatory ?? c.mandatoryDefault
    return { ...c, perMillion, weight: w, score: perMillion * w, mandatory, rank: 0 }
  })
  const max = Math.max(...raw.map((r) => r.score), 1)
  raw.forEach((r) => { r.score = Math.round((r.score / max) * 1000) / 10 })
  raw.sort((a, b) => Number(b.mandatory) - Number(a.mandatory) || b.score - a.score)
  raw.forEach((r, i) => { r.rank = i + 1 })
  return raw
}

export const itemOf = (sc: Scenario, pid: string): ScenarioItem => sc.items[pid] ?? { include: true, defer: 0 }

/** Shift future-year spend (years after the as-of year) by `defer` years; current and past years are sunk. */
export function shifted(byYear: Record<string, number>, defer: number, asOfYear: number): Record<string, number> {
  if (!defer) return byYear
  const out: Record<string, number> = {}
  for (const [y, v] of Object.entries(byYear)) {
    const n = Number(y)
    const k = String(n > asOfYear ? n + defer : n)
    out[k] = (out[k] || 0) + v
  }
  return out
}

export interface Fit {
  years: number[]
  byProgram: Record<ID, Record<number, { authorized: number; planned: number }>>
  total: Record<number, { authorized: number; planned: number }>
  overYears: number // program-years over authorization
  overAmount: number
  riskFunded: number
  deferred: number
  excluded: number
}

export function fit(s: AppState, cs: Candidate[], sc: Scenario, years: number[]): Fit {
  const asOfYear = Number(s.settings.asOf.slice(0, 4))
  const byProgram: Fit['byProgram'] = {}
  const total: Fit['total'] = {}
  for (const y of years) total[y] = { authorized: 0, planned: 0 }
  for (const pr of s.programs) {
    byProgram[pr.id] = {}
    for (const y of years) {
      const a = s.funding.find((f) => f.programId === pr.id && f.year === y)?.authorized ?? 0
      byProgram[pr.id][y] = { authorized: a, planned: 0 }
      total[y].authorized += a
    }
  }
  let riskFunded = 0, deferred = 0, excluded = 0
  for (const c of cs) {
    const it = itemOf(sc, c.project.id)
    if (!it.include) { excluded++; continue }
    if (it.defer) deferred++
    riskFunded += c.riskReduced
    const sh = shifted(c.byYear, it.defer, asOfYear)
    for (const [yk, v] of Object.entries(sh)) {
      const y = Number(yk)
      if (!byProgram[c.project.programId]?.[y]) continue
      byProgram[c.project.programId][y].planned += v
      total[y].planned += v
    }
  }
  let overYears = 0, overAmount = 0
  for (const pid of Object.keys(byProgram)) for (const y of years) {
    const c = byProgram[pid][y]
    if (y >= asOfYear && c.planned > c.authorized) { overYears++; overAmount += c.planned - c.authorized }
  }
  return { years, byProgram, total, overYears, overAmount: Math.round(overAmount), riskFunded, deferred, excluded }
}

/**
 * Suggest a fit: in each program-year over authorization (from next year on), defer the lowest-ranked
 * flexible project (not committed, not mandatory) by one more year, up to `maxDefer`. Deterministic and bounded.
 */
export function suggestFit(s: AppState, cs: Candidate[], sc: Scenario, years: number[], maxDefer = 3): { scenario: Scenario; moves: string[] } {
  const asOfYear = Number(s.settings.asOf.slice(0, 4))
  const next: Scenario = JSON.parse(JSON.stringify(sc))
  const moves: string[] = []
  const ranked = rank(cs, next)
  const stuck = new Set<string>()
  const rejected = new Set<string>()
  for (let guard = 0; guard < 300; guard++) {
    const f = fit(s, cs, next, years)
    let target: { pid: string; y: number } | null = null
    outer: for (const y of years.filter((x) => x > asOfYear)) for (const pr of s.programs) {
      const c = f.byProgram[pr.id][y]
      if (c.planned > c.authorized * 1.0001 && !stuck.has(pr.id + '|' + y)) { target = { pid: pr.id, y }; break outer }
    }
    if (!target) break
    const tg = target
    const pick = [...ranked].reverse().find((r) => {
      const it = itemOf(next, r.project.id)
      if (rejected.has(r.project.id + '|' + tg.y) || r.project.programId !== tg.pid || r.committed || r.mandatory || !it.include || it.defer >= maxDefer) return false
      return (shifted(r.byYear, it.defer, asOfYear)[String(tg.y)] ?? 0) > 0
    })
    if (!pick) { stuck.add(tg.pid + '|' + tg.y); continue }
    const it = itemOf(next, pick.project.id)
    next.items[pick.project.id] = { ...it, defer: it.defer + 1 }
    // keep a deferral only if it lowers total over-authorization (otherwise it just moves the problem)
    if (fit(s, cs, next, years).overAmount >= f.overAmount) { next.items[pick.project.id] = it; rejected.add(pick.project.id + '|' + tg.y); continue }
    moves.push(`${pick.project.code} deferred +${it.defer + 1} yr (score ${pick.score}) to relieve ${s.programs.find((p) => p.id === tg.pid)?.code} ${tg.y}`)
  }
  return { scenario: next, moves }
}

/** Projects in construction per district per year under the scenario (deferral shifts the window). */
export function capacity(cs: Candidate[], sc: Scenario, years: number[], asOfYear: number): { districts: string[]; grid: Record<string, Record<number, string[]>> } {
  const grid: Record<string, Record<number, string[]>> = {}
  for (const c of cs) {
    const it = itemOf(sc, c.project.id)
    if (!it.include || !c.constructionWindow) continue
    const d = c.project.location.district
    grid[d] ||= Object.fromEntries(years.map((y) => [y, []]))
    let y0 = Number(c.constructionWindow.start.slice(0, 4)), y1 = Number(c.constructionWindow.end.slice(0, 4))
    if (it.defer && y0 > asOfYear) { y0 += it.defer; y1 += it.defer } else if (it.defer && y1 > asOfYear) y1 += it.defer
    for (let y = y0; y <= y1; y++) if (grid[d][y]) grid[d][y].push(c.project.code)
  }
  return { districts: Object.keys(grid).sort(), grid }
}

export function newScenario(name: string, at: string, from?: Scenario): Scenario {
  return { id: 'scn_' + Math.random().toString(36).slice(2, 9), name, createdAt: at, basis: from?.basis ?? 'budget', weights: JSON.parse(JSON.stringify(from?.weights ?? DEFAULT_WEIGHTS)), items: JSON.parse(JSON.stringify(from?.items ?? {})), capacityLimit: from?.capacityLimit ?? 3 }
}

export function phaseOf(s: AppState, p: Project): number {

/** Overall health roll-up: red if cost is red or two dimensions are red; amber if any red or two ambers. */
export function overallHealth(h: Project['health']): RAG {
  const v = Object.values(h)
  const reds = v.filter((x) => x === 'red').length
  const ambers = v.filter((x) => x === 'amber').length
  if (h.cost === 'red' || reds >= 2) return 'red'
  if (reds >= 1 || ambers >= 2) return 'amber'
  return 'green'
}

export interface ProjectRow {
  p: Project
  programCode: string
  stages: string[]
  stageName: string
  phase: number
  pos: Position
  budget: number
  budgetBasis: 'control' | 'planning' | 'estimate' | 'none'
  eac: number
  variance: number // eac - budget (positive = overrun)
  variancePct: number
  next?: Milestone
  nextSlip: number
  readiness: number
  blockers: string[]
  pmName: string
  overall: RAG
  nextReview?: GateReview
}

export function projectRow(s: AppState, p: Project): ProjectRow {
  const t = templateFor(s, p)
  const pos = position(s, p.id)
  let budget = pos.currentBudget
  let eac = pos.eac
  let basis: ProjectRow['budgetBasis'] = pos.budgetKind ?? 'none'
  if (!budget) {
    const e = latestEstimate(s, p.id)
    if (e) { budget = estimateTotals(e).total; eac = budget; basis = 'estimate' }
  }
  const next = nextMilestone(p)
  const r = p.status === 'active' ? readiness(s, p) : { pct: 100, blockers: [] as string[] }
  const variance = basis === 'estimate' || basis === 'none' ? 0 : eac - budget
  return {
    p,
    programCode: s.programs.find((x) => x.id === p.programId)?.code ?? '—',
    stages: t.stages.map((x) => x.name),
    stageName: t.stages[p.stageIndex]?.name ?? '—',
    phase: phaseOf(s, p),
    pos,
    budget,
    budgetBasis: basis,
    eac,
    variance,
    variancePct: budget ? variance / budget : 0,
    next,
    nextSlip: next ? daysBetween(next.baseline, next.forecast) : 0,
    readiness: r.pct,
    blockers: r.blockers,
    pmName: personName(s, p.team.pm),
    overall: overallHealth(p.health),
    nextReview: s.gateReviews.filter((g) => g.projectId === p.id && g.status === 'scheduled').sort((a, b) => a.scheduled.localeCompare(b.scheduled))[0],
  }
}

// ───────────────────────────── Funding position (authorized vs budget vs EAC vs actual)
export interface FundingFigures { authorized: number; budget: number; eac: number; actual: number }

/**
 * Year figures: authorized = funding envelopes for the year; budget = active budget's spread for the year;
 * forecast = actuals to date in the year + remaining year budget scaled by the project's EAC/budget ratio;
 * actual = ERP cost periods in the year.
 */
export function yearFigures(s: AppState, projects: Project[], programIds: string[], year: number): FundingFigures {
  const y = String(year)
  const authorized = sum(s.funding.filter((f) => f.year === year && programIds.includes(f.programId)).map((f) => f.authorized))
  let budget = 0, eac = 0, actual = 0
  for (const p of projects) {
    const b = activeBudget(s, p.id)
    const pos = position(s, p.id)
    const by = b?.byYear[y] ?? 0
    const act = sum(s.costPeriods.filter((c) => c.projectId === p.id && c.period.startsWith(y)).map((c) => c.actual))
    const ratio = pos.currentBudget ? pos.eac / pos.currentBudget : 1
    budget += by
    actual += act
    eac += p.status === 'closed' ? act : act + Math.max(0, by - act) * ratio
  }
  return { authorized, budget: Math.round(budget), eac: Math.round(eac), actual: Math.round(actual) }
}

export function allInFigures(s: AppState, projects: Project[], programIds: string[]): FundingFigures {
  const pos = projects.map((p) => position(s, p.id))
  return {
    authorized: sum(s.funding.filter((f) => programIds.includes(f.programId)).map((f) => f.authorized)),
    budget: sum(pos.map((x) => x.currentBudget)),
    eac: sum(pos.map((x) => x.eac)),
    actual: sum(pos.map((x) => x.actualToDate)),
  }
}

export interface AttentionItem { id: string; projectId: string; code: string; kind: string; label: string; detail: string; severity: 'bad' | 'warn'; href: string; date?: string }

export function needsAttention(s: AppState, projects: Project[]): AttentionItem[] {
  const out: AttentionItem[] = []
  const asOf = s.settings.asOf
  const ids = new Set(projects.map((p) => p.id))
  const code = (pid: string) => s.projects.find((p) => p.id === pid)?.code ?? pid
  const num = (pid: string) => pid.replace(/\D/g, '')
  for (const p of projects) {
    if (p.status !== 'active') continue
    const reds = (Object.keys(p.health) as (keyof Project['health'])[]).filter((k) => p.health[k] === 'red')
    if (reds.length) out.push({ id: 'h-' + p.id, projectId: p.id, code: p.code, kind: 'Red health', label: p.name, detail: `Red on ${reds.join(', ')}`, severity: 'bad', href: `#p-${num(p.id)}` })
    const late = s.deliverables.filter((d) => d.projectId === p.id && d.stageIndex === p.stageIndex && d.due && d.due < asOf && !['accepted', 'waived', 'submitted'].includes(d.status))
    if (late.length) out.push({ id: 'dl-' + p.id, projectId: p.id, code: p.code, kind: 'Overdue deliverables', label: p.name, detail: `${late.length} gate deliverable${late.length > 1 ? 's' : ''} past due (oldest ${daysBetween(late.map((d) => d.due!).sort()[0], asOf)} days)`, severity: 'warn', href: `#p-${num(p.id)}-lifecycle`, date: late.map((d) => d.due!).sort()[0] })
  }
  for (const w of s.workflows) {
    if (w.status !== 'running' || !w.projectId || !ids.has(w.projectId)) continue
    for (const a of w.active) {
      if (!a.due) continue
      const days = daysBetween(asOf, a.due)
      if (days > 3) continue
      const def = s.workflowDefs.find((d) => d.id === w.defId)
      const step = def?.nodes.find((n) => n.id === a.nodeId)?.label ?? 'Step'
      out.push({ id: 'wf-' + w.id + a.nodeId, projectId: w.projectId, code: code(w.projectId), kind: days < 0 ? 'Overdue workflow step' : 'Workflow step due', label: w.title, detail: `${step} · ${a.assigneeIds.map((x) => personName(s, x)).join(', ')} · ${days < 0 ? `${-days} days overdue` : days === 0 ? 'due today' : `due in ${days} day${days > 1 ? 's' : ''}`}`, severity: days < 0 ? 'bad' : 'warn', href: '#inbox', date: a.due })
    }
  }
  for (const pm of s.permits) {
    if (!ids.has(pm.projectId) || pm.status !== 'issued' || !pm.expires) continue
    const days = daysBetween(asOf, pm.expires)
    if (days > 30) continue
    out.push({ id: 'pm-' + pm.id, projectId: pm.projectId, code: code(pm.projectId), kind: days < 0 ? 'Permit expired' : 'Permit expiring', label: `${pm.type} ${pm.number}`, detail: `${pm.agency} · ${days < 0 ? `expired ${-days} days ago` : `expires in ${days} days`}`, severity: days < 7 ? 'bad' : 'warn', href: `#p-${num(pm.projectId)}-permits`, date: pm.expires })
  }
  for (const sp of s.safetyPlans) {
    if (!ids.has(sp.projectId) || sp.status === 'superseded') continue
    const proj = s.projects.find((p) => p.id === sp.projectId)
    if (!proj || proj.status !== 'active') continue
    for (const t of sp.tickets) {
      const days = daysBetween(asOf, t.expires)
      if (days > 7) continue
      out.push({ id: 'tk-' + t.id, projectId: sp.projectId, code: code(sp.projectId), kind: t.type === '811 locate' ? (days < 0 ? '811 ticket expired' : '811 ticket expiring') : days < 0 ? 'Ticket expired' : 'Ticket expiring', label: `${t.type} ${t.number}`, detail: `${sp.workPackage} · ${days < 0 ? `expired ${-days} day${-days > 1 ? 's' : ''} ago — stop excavation until renewed` : days === 0 ? 'expires today' : `expires in ${days} day${days > 1 ? 's' : ''}`}`, severity: days <= 2 ? 'bad' : 'warn', href: `#p-${num(sp.projectId)}-safety`, date: t.expires })
    }
  }
  for (const a of s.actions) {
    if (!ids.has(a.projectId) || a.status !== 'open' || a.due >= asOf) continue
    out.push({ id: 'act-' + a.id, projectId: a.projectId, code: code(a.projectId), kind: 'Overdue gate condition', label: a.title, detail: `${personName(s, a.ownerId)} · due ${a.due}`, severity: 'warn', href: `#p-${num(a.projectId)}-lifecycle`, date: a.due })
  }
  for (const v of s.variances) {
    if (!ids.has(v.projectId) || v.status !== 'required') continue
    out.push({ id: 'var-' + v.id, projectId: v.projectId, code: code(v.projectId), kind: 'Variance explanation due', label: v.description, detail: 'Threshold breached; narrative and root cause required before period close', severity: 'warn', href: `#p-${num(v.projectId)}-baselines` })
  }
  const rank = (x: AttentionItem) => (x.severity === 'bad' ? 0 : 1)
  return out.sort((a, b) => rank(a) - rank(b) || (a.date ?? '9').localeCompare(b.date ?? '9'))
}

// ───────────────────────────── Safety leading indicators
export function safetyIndicators(s: AppState, projects: Project[]) {
  const ids = new Set(projects.map((p) => p.id))
  const asOf = s.settings.asOf
  const from7 = addDays(asOf, -7), from30 = addDays(asOf, -30)
  const tb7 = s.tailboards.filter((t) => ids.has(t.projectId) && t.date > from7 && t.date <= asOf)
  const tbPrior = s.tailboards.filter((t) => ids.has(t.projectId) && t.date > addDays(asOf, -14) && t.date <= from7)
  const sifOpen = s.safetyEvents.filter((e) => ids.has(e.projectId) && e.sifPotential && e.status === 'open')
  const nearMiss30 = s.safetyEvents.filter((e) => ids.has(e.projectId) && e.type === 'near-miss' && e.date > from30)
  const obs30 = s.safetyEvents.filter((e) => ids.has(e.projectId) && e.type === 'observation' && e.date > from30)
  const incidents30 = s.safetyEvents.filter((e) => ids.has(e.projectId) && e.type === 'incident' && e.date > from30)
  const plansInReview = s.safetyPlans.filter((x) => ids.has(x.projectId) && x.status === 'in-review')
  const byProject = projects
    .map((p) => ({ p, tb: tb7.filter((t) => t.projectId === p.id).length, attendees: sum(tb7.filter((t) => t.projectId === p.id).map((t) => t.attendees.length)), sif: sifOpen.filter((e) => e.projectId === p.id).length }))
    .filter((x) => x.tb || x.sif)
  return { tb7, tbPrior, sifOpen, nearMiss30, obs30, incidents30, plansInReview, byProject }
}

/** Days from one go decision to the next, per program and stage (SG-12). */
export function stageDurations(s: AppState) {
  const out: { programId: string; stageIndex: number; days: number; projectId: string }[] = []
  for (const p of s.projects) {
    const ds = s.gateDecisions.filter((g) => g.projectId === p.id && (g.outcome === 'go' || g.outcome === 'go-with-conditions')).sort((a, b) => a.date.localeCompare(b.date))
    let prev = p.dates.initiated
    for (const g of ds) {
      out.push({ programId: p.programId, stageIndex: g.stageIndex, days: daysBetween(prev, g.date), projectId: p.id })
      prev = g.date
    }
  }
  return out
}

function StageFunnel({ s, rows, programIds }: { s: AppState; rows: ProjectRow[]; programIds: string[] }) {

  const max = Math.max(...totals, 1)
  const phaseTotals = PHASES.map((_, i) => rows.filter((r) => r.phase === i))
  return (
    <div className="flex flex-col gap-3">
      <div className="flex flex-wrap items-center justify-between gap-2">

          <span className="inline-flex items-center gap-1.5"><span className="inline-block h-2.5 w-2.5 rounded-sm border border-dashed border-faint" />Closed</span>

function ContingencyByProgram({ s, rows, programIds }: { s: AppState; rows: ProjectRow[]; programIds: string[] }) {

    const orig = sum(rs.map((r) => r.pos.contingencyOriginal))
    const used = sum(rs.map((r) => r.pos.contingencyUsed))
    const pending = sum(s.contingencyDraws.filter((d) => d.status === 'in-review' && rs.some((r) => r.p.id === d.projectId)).map((d) => d.amount))
    const exp = sum(rs.map((r) => r.pos.riskExposure))
    return { pr, orig, used, rem: orig - used, pending, exp, rs }
  })
  return (
    <div className="flex flex-col gap-3">
      <div className="scroll-x">
        <table className="tbl">
          <thead><tr><th>Program</th><th className="num">Original</th><th className="num">Used</th><th className="num">Pending</th><th className="num">Remaining</th><th style={{ minWidth: 140 }}>Remaining vs exposure</th></tr></thead>
          <tbody>
            {data.map((d) => {
              const remPct = d.orig ? d.rem / d.orig : 0
              const covered = d.exp <= 0 ? 1 : d.rem / d.exp
              return (

const esc = (v: unknown) => String(v ?? '').replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')

/** Data the status report and its AI prompt share. */
export function statusFacts(s: AppState, p: Project) {
  const pos = position(s, p.id)
  const t = templateFor(s, p)
  const month = s.settings.asOf.slice(0, 7)
  const periods = s.costPeriods.filter((c) => c.projectId === p.id).sort((a, b) => a.period.localeCompare(b.period))
  const last = periods[periods.length - 1]
  const cumPV = sum(periods.map((c) => c.planned)), cumEV = sum(periods.map((c) => c.earned)), cumAC = sum(periods.map((c) => c.actual))
  const from = addDays(s.settings.asOf, -31)
  const changes = s.changes.filter((c) => c.projectId === p.id && (c.requestedAt >= from || (c.decidedAt ?? '') >= from))
  const risks = s.risks.filter((r) => r.projectId === p.id && isActive(r)).sort((a, b) => riskScore(b) - riskScore(a)).slice(0, 5)
  const r = readiness(s, p)
  const events = s.safetyEvents.filter((e) => e.projectId === p.id && e.date >= from)
  const tbs = s.tailboards.filter((x) => x.projectId === p.id && x.date >= from)
  const variances = s.variances.filter((v) => v.projectId === p.id && v.status !== 'accepted')
  const review = s.gateReviews.filter((g) => g.projectId === p.id && g.status === 'scheduled').sort((a, b) => a.scheduled.localeCompare(b.scheduled))[0]
  return { pos, t, month, last, cpi: cumAC ? cumEV / cumAC : undefined, spi: cumPV ? cumEV / cumPV : undefined, changes, risks, r, events, tbs, variances, review, next: nextMilestone(p) }
}

  const [bId, setB] = useState<string | undefined>(defB?.id)
  const [slot, setSlot] = useState<'A' | 'B'>('B')
  const [showClose, setShowClose] = useState(false)
  const [open, setOpen] = useState<string | null>(null)
  const [take, setTake] = useState(false)
  const [rebase, setRebase] = useState(false)
  useEffect(() => { const f = takeFocusSnapshot(); if (f && snaps.some((x) => x.id === f)) setOpen(f) }, []) // eslint-disable-line react-hooks/exhaustive-deps
  useEffect(() => { if (!snaps.some((x) => x.id === aId)) setA(defA?.id); if (!snaps.some((x) => x.id === bId)) setB(defB?.id) }, [snaps.length]) // eslint-disable-line react-hooks/exhaustive-deps

  const A = snaps.find((x) => x.id === aId)
  const B = snaps.find((x) => x.id === bId)
  const approved = s.changes.filter((c) => c.projectId === projectId && c.status === 'approved')
  const pendingRebase = s.changes.find((c) => c.projectId === projectId && c.type === 'scope' && c.title.startsWith('Re-baseline') && (c.status === 'in-review' || c.status === 'draft'))

  if (!snaps.length) {
    return (
      <div className="flex flex-col gap-4">
        <Empty title="No baselines yet" action={<Button onClick={() => setTake(true)} icon={<Icon name="plus" className="h-4 w-4" />}>Take snapshot</Button>}>
          {p.code} is in {stageLabel(s, p, p.stageIndex)}. The first snapshot is frozen automatically at Gate 1; the Original baseline is frozen when the control budget is authorized.
        </Empty>
        <TakeSnapshotModal open={take} onClose={() => setTake(false)} projectId={projectId} />
      </div>
    )
  }

  const pick = (id: string) => { if (slot === 'A') { setA(id); setSlot('B') } else setB(id) }
  return (
    <div className="flex flex-col gap-4">
      <div className="grid grid-cols-2 gap-3 lg:grid-cols-5">
        <Stat label="Original baseline" value={money(original?.budget.total)} sub={original ? fmtDate(original.date) : 'Not yet authorized'} />
        <Stat label="Current baseline" value={money(current?.budget.total)} sub={current ? `${fmtDate(current.date)} · ${moneyDelta((current?.budget.total ?? 0) - (original?.budget.total ?? 0))} vs Original` : '—'} />
        <Stat label="Forecast (EAC)" value={money(pos.eac)} tone={current && pos.eac > current.budget.total ? 'bad' : undefined} sub={current ? `${moneyDelta(pos.eac - current.budget.total)} vs Current` : undefined} />
        <Stat label="Approved changes" value={approved.length} sub={`${moneyDelta(approved.reduce((a, c) => a + c.costImpact, 0))} · each one a new Current`} />
        <Stat label="Snapshots on file" value={snaps.length} sub={`${snaps.filter((x) => x.type === 'period-close').length} period closes · ${snaps.filter((x) => x.type === 'regulatory').length} regulatory`} />
      </div>

      {pendingRebase && (
        <Callout tone="marker" title={`Re-baseline request ${pendingRebase.code} is in review`}>
          {pendingRebase.description} <a className="font-semibold text-link hover:underline" href={projectHref(projectId, 'changes')} onClick={() => setFocusChange(pendingRebase.id)}>Open change</a>
        </Callout>
      )}

      <Panel title="Baseline timeline" subtitle="Every frozen position by date. Click a marker to load it into the comparison."
        actions={<>
          <span className="text-xs text-muted">Clicking sets</span>
          <SegButtons options={[{ id: 'A', label: 'A (from)' }, { id: 'B', label: 'B (to)' }]} value={slot} onChange={setSlot} />
          <Button size="sm" variant="ghost" onClick={() => setShowClose(!showClose)}>{showClose ? 'Hide' : 'Show'} period closes</Button>
          <Button size="sm" onClick={() => setTake(true)} icon={<Icon name="plus" className="h-3.5 w-3.5" />}>Take snapshot</Button>
          <Button size="sm" onClick={() => setRebase(true)} disabled={!current || !!pendingRebase}>Request re-baseline</Button>
        </>}>
        <Timeline s={s} snaps={snaps} changes={approved} aId={aId} bId={bId} onPick={pick} showClose={showClose} asOf={s.settings.asOf} />
      </Panel>

      {A && B && <Compare s={s} a={A} b={B} projectId={projectId} onSwap={() => { setA(B.id); setB(A.id) }} onPickA={setA} onPickB={setB} snaps={snaps} />}

      <Panel title="Snapshot register" subtitle="Immutable records. Open one to see its approving decision and export a signed copy." pad={false}>
        <Table rows={[...snaps].reverse()} rowKey={(x) => x.id} onRowClick={(x) => setOpen(x.id)} maxHeight={460} dense columns={snapColumns(s)} />
      </Panel>

      <SnapshotDrawer s={s} id={open} onClose={() => setOpen(null)} onCompare={(id) => { setB(id); setOpen(null); toast('Loaded into comparison as B', 'info') }} />
      <TakeSnapshotModal open={take} onClose={() => setTake(false)} projectId={projectId} />
      {current && <RebaselineModal open={rebase} onClose={() => setRebase(false)} projectId={projectId} current={current} />}
    </div>
  )
}

function TypeBadge({ type }: { type: SnapshotType }) {
  const m = SNAP_META[type]
  return <Badge tone={m.tone === 'ink' ? 'ink' : m.tone}><Glyph type={type} size={9} />{m.label}</Badge>
}

function Glyph({ type, size = 10 }: { type: SnapshotType; size?: number }) {
  const m = SNAP_META[type]
  const h = size / 2
  return (
    <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} aria-hidden className="shrink-0">
      <g transform={`translate(${h} ${h})`}><Shape shape={m.shape} r={h - 0.5} color={type === 'original' ? 'currentColor' : m.color} /></g>
    </svg>
  )
}

function Shape({ shape, r, color, stroke }: { shape: (typeof SNAP_META)[SnapshotType]['shape']; r: number; color: string; stroke?: string }) {
  const sw = stroke ? 2 : 0
  switch (shape) {
    case 'diamond': return <path d={`M0 ${-r}L${r} 0L0 ${r}L${-r} 0Z`} fill={color} stroke={stroke} strokeWidth={sw} />
    case 'square': return <rect x={-r * 0.85} y={-r * 0.85} width={r * 1.7} height={r * 1.7} fill={color} stroke={stroke} strokeWidth={sw} />
    case 'triangle': return <path d={`M0 ${-r}L${r} ${r * 0.8}L${-r} ${r * 0.8}Z`} fill={color} stroke={stroke} strokeWidth={sw} />
    case 'ring': return <circle r={r * 0.8} fill="rgb(var(--surface))" stroke={color} strokeWidth={2.2} />
    case 'tick': return <rect x={-1} y={-r} width={2} height={r * 2} fill={color} />
    default: return <circle r={r * 0.9} fill={color} stroke={stroke} strokeWidth={sw} />
  }
}

function Timeline({ s, snaps, changes, aId, bId, onPick, showClose, asOf }: { s: AppState; snaps: Snapshot[]; changes: Change[]; aId?: string; bId?: string; onPick: (id: string) => void; showClose: boolean; asOf: string }) {
  const [hover, setHover] = useState<string | null>(null)
  const shown = snaps.filter((x) => showClose || x.type !== 'period-close' || x.id === aId || x.id === bId)
  const dates = [...snaps.map((x) => x.date), ...changes.map((c) => c.decidedAt!), asOf]
  const lo = dates.reduce((a, b) => (a < b ? a : b))
  const hi = asOf
  const span = Math.max(30, daysBetween(lo, hi))
  const W = 760, L = 24, R = 24, AX = 92
  const X = (d: string) => L + (daysBetween(lo, d) / span) * (W - L - R)
  // stack colliding markers upward
  const levels: number[] = []
  const placed = shown.map((x) => {
    const xx = X(x.date)
    let lv = 0
    while (levels[lv] !== undefined && xx - levels[lv] < 15) lv++
    levels[lv] = xx
    return { snap: x, x: xx, y: AX - 12 - lv * 17 }
  })
  const cLevels: number[] = []
  const cPlaced = changes.map((c) => {
    const xx = X(c.decidedAt!)
    let lv = 0
    while (cLevels[lv] !== undefined && xx - cLevels[lv] < 58) lv++
    cLevels[lv] = xx
    return { c, x: xx, lv }
  })
  const H = AX + 34 + Math.max(0, ...cPlaced.map((c) => c.lv)) * 16 + 10
  const y0 = Number(lo.slice(0, 4)), y1 = Number(hi.slice(0, 4))
  const yearTicks = Array.from({ length: y1 - y0 + 1 }, (_, i) => `${y0 + i}-01-01`).filter((d) => d > lo)
  const hv = hover ? snaps.find((x) => x.id === hover) : undefined
  return (
    <div>
      <div className="scroll-x">
        <svg viewBox={`0 0 ${W} ${H}`} className="w-full min-w-[560px]" role="img" aria-label="Snapshot timeline">
          {yearTicks.map((d) => <g key={d}><line x1={X(d)} x2={X(d)} y1={14} y2={AX + 6} stroke="rgb(var(--line))" strokeDasharray="2 3" /><text x={X(d) + 3} y={12} fontSize={10.5} fill="rgb(var(--faint))">{d.slice(0, 4)}</text></g>)}
          <line x1={L} x2={W - R} y1={AX} y2={AX} stroke="rgb(var(--ink))" strokeWidth={2} />
          <g><line x1={X(asOf)} x2={X(asOf)} y1={AX - 6} y2={AX + 6} stroke="rgb(var(--marker))" strokeWidth={3} /><text x={X(asOf)} y={AX + 18} fontSize={10} textAnchor="end" fill="rgb(var(--muted))">Today</text></g>
          {placed.map(({ snap, x, y }) => {
            const m = SNAP_META[snap.type]
            const sel = snap.id === aId ? 'A' : snap.id === bId ? 'B' : null
            return (
              <g key={snap.id} className="cursor-pointer" onClick={() => onPick(snap.id)} onMouseEnter={() => setHover(snap.id)} onMouseLeave={() => setHover(null)}>
                <title>{`${m.label} · ${fmtDate(snap.date)} · ${snap.label} · ${money(snap.budget.total)}`}</title>
                <line x1={x} x2={x} y1={y} y2={AX} stroke={m.color} strokeWidth={1} opacity={0.6} />
                {sel && <circle cx={x} cy={y} r={10} fill="none" stroke="rgb(var(--ink))" strokeWidth={1.5} />}
                <g transform={`translate(${x} ${y})`}><Shape shape={m.shape} r={snap.type === 'period-close' ? 5 : 6.5} color={m.color} /></g>
                {sel && <text x={x} y={y - 13} fontSize={11} fontWeight={700} textAnchor="middle" fill="rgb(var(--ink))">{sel}</text>}
                <rect x={x - 8} y={y - 9} width={16} height={AX - y + 9} fill="transparent" />
              </g>
            )
          })}
          {cPlaced.map(({ c, x, lv }) => (
            <g key={c.id} className="cursor-pointer" onClick={() => { setFocusChange(c.id); go(projectHref(c.projectId, 'changes')) }}>
              <title>{`${c.code} ${c.title} · approved ${fmtDate(c.decidedAt)} · ${moneyDelta(c.costImpact)}${c.scheduleImpactDays ? ` · ${c.scheduleImpactDays} days` : ''}`}</title>
              <line x1={x} x2={x} y1={AX} y2={AX + 22 + lv * 16} stroke={c.costImpact >= 0 ? 'rgb(var(--bad))' : 'rgb(var(--good))'} strokeWidth={1.5} />
              <circle cx={x} cy={AX} r={3} fill={c.costImpact >= 0 ? 'rgb(var(--bad))' : 'rgb(var(--good))'} />
              <text x={x + 3} y={AX + 30 + lv * 16} fontSize={10.5} fontFamily="IBM Plex Mono, monospace" fill="rgb(var(--ink))">{c.code.replace(/^CR-\d+-/, 'CR-')} {moneyDelta(c.costImpact)}</text>
            </g>
          ))}
        </svg>
      </div>
      <div className="mt-2 flex flex-wrap items-center gap-x-4 gap-y-1 text-xs text-muted">
        {(Object.keys(SNAP_META) as SnapshotType[]).map((t) => <span key={t} className="inline-flex items-center gap-1.5"><Glyph type={t} size={10} />{SNAP_META[t].label} ({snaps.filter((x) => x.type === t).length})</span>)}
        <span className="inline-flex items-center gap-1.5"><span className="inline-block h-3 w-0.5 bg-bad" />Approved change</span>
        {hv && <span className="ml-auto text-ink"><b>{SNAP_META[hv.type].label}</b> {fmtDate(hv.date)} · {hv.label} · <span className="tabular-nums">{money(hv.budget.total)}</span></span>}
      </div>
    </div>
  )
}

function Compare({ s, a, b, projectId, onSwap, onPickA, onPickB, snaps }: { s: AppState; a: Snapshot; b: Snapshot; projectId: string; onSwap: () => void; onPickA: (id: string) => void; onPickB: (id: string) => void; snaps: Snapshot[] }) {
  const diff = useMemo(() => compareSnapshots(a, b), [a, b])
  const between = changesBetween(s, projectId, a, b)
  const opt = snaps.map((x) => ({ value: x.id, label: `${fmtDate(x.date)} · ${SNAP_META[x.type].label} · ${x.label}` }))
  const steps = diff.costBridge.sort((x, y) => Math.abs(y.delta) - Math.abs(x.delta)).map((c) => ({ label: categoryName(s, c.categoryId), delta: c.delta }))
  const contStep = (b.budget.total - a.budget.total) - steps.reduce((x, y) => x + y.delta, 0)
  const bridge = Math.abs(contStep) >= 1 ? [...steps, { label: 'Contingency', delta: contStep }] : steps
  return (
    <Panel title="Compare snapshots" subtitle="Bridge from A to B by cost category, with every approved change that explains the movement."
      actions={<Button size="sm" variant="ghost" onClick={onSwap}>Swap A ↔ B</Button>}>
      <div className="grid grid-cols-1 gap-3 md:grid-cols-[1fr_auto_1fr] md:items-end">
        <Field label="A — from"><Select value={a.id} onChange={(e) => onPickA(e.target.value)} options={opt} /></Field>
        <span className="hidden pb-2 text-muted md:block">→</span>
        <Field label="B — to"><Select value={b.id} onChange={(e) => onPickB(e.target.value)} options={opt} /></Field>
      </div>
      <div className="mt-4 grid grid-cols-2 gap-3 md:grid-cols-4">
        <DeltaFig label="Budget (incl. contingency)" a={a.budget.total} b={b.budget.total} />
        <DeltaFig label="Forecast (EAC)" a={a.budget.eac} b={b.budget.eac} />
        <DeltaFig label="Contingency remaining" a={a.budget.contingencyRemaining} b={b.budget.contingencyRemaining} goodUp />
        <DeltaFig label="Actual to date" a={a.budget.actualToDate} b={b.budget.actualToDate} neutral />
      </div>
      <div className="mt-4 grid grid-cols-1 gap-5 xl:grid-cols-[3fr_2fr]">
        <Section title="Cost bridge by category">
          {bridge.length ? <Waterfall start={a.budget.total} end={b.budget.total} steps={bridge} startLabel="A budget" endLabel="B budget" /> : <p className="text-[13px] text-muted">No category movement between these snapshots.</p>}
          {bridge.length > 0 && <p className="text-xs text-faint">Red raises the budget, green lowers it. Contingency is the budget-total change not carried by a category.</p>}
        </Section>
        <Section title={`Approved changes between A and B (${between.length})`}>
          {between.length ? (
            <ul className="flex flex-col divide-y divide-line rounded-md border border-line">
              {between.map((c) => (
                <li key={c.id} className="flex items-start justify-between gap-2 px-3 py-2 text-[13px]">
                  <div className="min-w-0">
                    <a href={projectHref(c.projectId, 'changes')} onClick={() => setFocusChange(c.id)} className="font-mono text-[12px] font-semibold text-link hover:underline">{c.code}</a> <span>{c.title}</span>
                    <div className="text-xs text-muted">{c.rootCause} · {categoryName(s, c.categoryId)} · funded from {c.fundedFrom.replace('-', ' ')} · approved {fmtDate(c.decidedAt)}</div>
                  </div>
                  <span className={cx('tabular-nums font-semibold', c.costImpact > 0 ? 'text-bad' : 'text-good')}>{moneyDelta(c.costImpact)}</span>
                </li>
              ))}
            </ul>
          ) : <p className="text-[13px] text-muted">No approved changes were decided in this window.</p>}
        </Section>
      </div>
      <div className="mt-5 grid grid-cols-1 gap-5 lg:grid-cols-2">
        <Section title={`Milestone movement (${diff.milestoneMoves.length})`}>
          {diff.milestoneMoves.length ? (
            <Table rowKey={(m) => m.name} rows={diff.milestoneMoves} dense columns={[
              { key: 'n', header: 'Milestone', render: (m) => m.name },
              { key: 'f', header: 'A forecast', render: (m) => fmtDate(m.from) },
              { key: 't', header: 'B forecast', render: (m) => fmtDate(m.to) },
              { key: 'd', header: 'Days', num: true, render: (m) => <span className={m.days > 0 ? 'font-semibold text-bad' : 'font-semibold text-good'}>{m.days > 0 ? '+' : ''}{m.days}</span> },
            ]} />
          ) : <p className="text-[13px] text-muted">No milestone moved.</p>}
        </Section>
        <Section title="Scope">
          <div className="flex flex-col gap-2 text-[13px]">
            {diff.scopeAdded.map((x) => <div key={x} className="flex gap-2"><Badge tone="bad">Added</Badge><span>{x}</span></div>)}
            {diff.scopeRemoved.map((x) => <div key={x} className="flex gap-2"><Badge tone="good">Removed</Badge><span>{x}</span></div>)}
            {diff.quantityChanges.map((q) => <div key={q.label} className="flex gap-2"><Badge tone="info">Quantity</Badge><span>{q.label}: {num(q.from)} → {num(q.to)} {q.unit}</span></div>)}
            {!diff.scopeAdded.length && !diff.scopeRemoved.length && !diff.quantityChanges.length && <p className="text-muted">Scope statement and quantities unchanged between A and B.</p>}
            <p className="mt-1 text-xs text-faint">Contingency {money(a.budget.contingencyRemaining)} → {money(b.budget.contingencyRemaining)} ({moneyDelta(diff.contingencyDelta)}). Risk exposure {money(a.riskExposure)} → {money(b.riskExposure)}.</p>
          </div>
        </Section>
      </div>
    </Panel>
  )
}

function DeltaFig({ label, a, b, goodUp, neutral }: { label: string; a: number; b: number; goodUp?: boolean; neutral?: boolean }) {
  const d = b - a
  const tone = neutral || !d ? 'text-muted' : (d > 0) === !!goodUp ? 'text-good' : 'text-bad'
  return (
    <div className="min-w-0 rounded-md border border-line px-3 py-2">
      <div className="eyebrow truncate">{label}</div>
      <div className="mt-1 flex flex-wrap items-baseline gap-x-2 tabular-nums"><span className="text-[13px] text-muted">{money(a)} →</span><span className="font-display text-[19px] font-bold">{money(b)}</span></div>
      <div className={cx('text-xs font-semibold tabular-nums', tone)}>{d ? moneyDelta(d) : 'No change'}</div>
    </div>
  )
}

function SnapshotDrawer({ s, id, onClose, onCompare }: { s: AppState; id: string | null; onClose: () => void; onCompare: (id: string) => void }) {
  const snap = s.snapshots.find((x) => x.id === id)
  if (!snap) return null
  const p = s.projects.find((x) => x.id === snap.projectId)!
  const a = snapshotAuthority(s, snap)
  const cats = Object.entries(snap.budget.byCategory).sort((x, y) => y[1] - x[1])
  const exportHtml = async () => {
    const r = await saveFile(`${p.code}-${snap.type}-${snap.date}.html`, snapshotHtml(s, snap), 'text/html')
    if (r === 'saved') { mutate(() => {}, { entity: 'snapshot', entityId: snap.id, projectId: p.id, action: 'export', summary: `Exported signed record of ${snap.label}` }); toast('Signed record saved') } else if (r === 'unavailable') toast('Download is not available here', 'bad')
  }
  const exportJson = async () => {
    const payload = { schema: 'crosswalk.snapshot/1', exportedAt: s.settings.asOf, exportedBy: personName(s, s.currentUserId), project: { id: p.id, code: p.code, name: p.name }, authority: a, snapshot: snap }
    const r = await saveFile(`${p.code}-${snap.type}-${snap.date}.json`, JSON.stringify({ ...payload, fingerprint: fingerprint(JSON.stringify(snap)) }, null, 2), 'application/json')
    if (r === 'saved') toast('Snapshot JSON saved')
  }
  return (
    <Drawer open onClose={onClose} title={<span className="flex items-center gap-2"><TypeBadge type={snap.type} />{snap.label}</span>} width={640}
      footer={<><Button onClick={() => onCompare(snap.id)}>Compare as B</Button><Button onClick={exportJson} icon={<Icon name="download" className="h-4 w-4" />}>JSON</Button><Button variant="primary" onClick={exportHtml} icon={<Icon name="download" className="h-4 w-4" />}>Signed record (HTML)</Button></>}>
      <div className="flex flex-col gap-5">
        <Callout tone={a.kind === 'Gate decision' || a.kind === 'Approved change' ? 'good' : 'info'} title={a.kind}>
          {a.text}
          {a.signers.length > 0 && <ul className="mt-1.5 text-xs">{a.signers.map((g, i) => <li key={i}>{g.name}{g.at && <> · {fmtDateTime(g.at)}</>}{g.sig && <> · <Mono>sig {g.sig}</Mono></>}</li>)}</ul>}
          {a.changeId && <a className="mt-1 block text-xs font-semibold text-link hover:underline" href={projectHref(p.id, 'changes')} onClick={() => setFocusChange(a.changeId!)}>Open the change record</a>}
        </Callout>
        <KV cols={3} items={[
          ['Date', fmtDate(snap.date)], ['Stage', stageLabel(s, p, snap.stageIndex)], ['Taken by', personName(s, snap.createdById)],
          ['Budget incl. contingency', money(snap.budget.total, 'full')], ['Forecast (EAC)', money(snap.budget.eac, 'full')], ['Contingency remaining', money(snap.budget.contingencyRemaining, 'full')],
          ['Actual to date', money(snap.budget.actualToDate, 'full')], ['Risk exposure', money(snap.riskExposure, 'full')], ['Proceeding', snap.proceeding ?? '—'],
        ]} />
        <Section title="Budget by category (excl. contingency)">
          <Table rowKey={(r) => r[0]} rows={cats} dense columns={[
            { key: 'c', header: 'Category', render: (r) => categoryName(s, r[0]) },
            { key: 'e', header: 'ERP element', render: (r) => <Mono>{s.categories.find((c) => c.id === r[0])?.erpCostElement}</Mono> },
            { key: 'v', header: 'Amount', num: true, render: (r) => money(r[1], 'full') },
          ]} />
        </Section>
        <Section title="Milestones">
          <Table rowKey={(m) => m.id} rows={snap.milestones} dense columns={[
            { key: 'n', header: 'Milestone', render: (m) => m.name },

function TakeSnapshotModal({ open, onClose, projectId }: { open: boolean; onClose: () => void; projectId: string }) {
  const s = useStore()
  const p = s.projects.find((x) => x.id === projectId)!

function RebaselineModal({ open, onClose, projectId, current }: { open: boolean; onClose: () => void; projectId: string; current: Snapshot }) {
  const s = useStore()
  const p = s.projects.find((x) => x.id === projectId)!
  const pos = position(s, projectId)
  const [newTotal, setNewTotal] = useState(Math.max(pos.eac, current.budget.total))
  const [days, setDays] = useState(0)
  const [rootCause, setRootCause] = useState<RootCause>('Scope growth')
  const [categoryId, setCategoryId] = useState('cat-contract')
  const [note, setNote] = useState('')
  const delta = newTotal - current.budget.total
  const doa = doaLabel(s, projectId, delta)
  const submit = () => {
    if (!note.trim()) { toast('Explain why the baseline no longer represents the work.', 'bad'); return }
    const code = nextChangeCode(s, p)
    mutate((d) => {
      const ch: Change = {
        id: uid('chg'), projectId, code, title: `Re-baseline to ${money(newTotal)}`, type: 'scope', rootCause, description: `Re-baseline request against ${current.label} (${fmtDate(current.date)}, ${money(current.budget.total)}). ${note.trim()}`,
        costImpact: delta, scheduleImpactDays: days, scopeImpact: 'Re-baseline of scope, schedule and budget', categoryId, fundedFrom: delta > 0 ? 'new-funding' : 'none', status: 'draft', requestedById: d.currentUserId, requestedAt: d.settings.asOf, baselineSnapshotId: current.id,
      }
      d.changes.push(ch)
      const inst = startWorkflow(d, 'change-request', { title: `${code} Re-baseline ${p.code}`, recordType: 'change', recordId: ch.id, projectId, data: { amount: delta, scheduleImpactDays: days, rootCause, justification: note.trim(), rebaseline: true } })
      const slot = inst?.active.find((a) => a.nodeId === 'prep' && a.assigneeIds.includes(d.currentUserId))
      if (inst && slot) actOnWorkflow(d, inst.id, slot.nodeId, d.currentUserId, 'submit', 'Re-baseline request prepared')
    }, { entity: 'change', entityId: code, projectId, action: 'create', summary: `Requested re-baseline ${code}: ${moneyDelta(delta)}, ${days} days` })
    toast('Re-baseline submitted for approval')
    setNote(''); onClose()
  }
  return (
    <Modal open={open} onClose={onClose} title="Request re-baseline" wide footer={<><Button variant="ghost" onClick={onClose}>Cancel</Button><Button variant="primary" onClick={submit}>Submit for approval</Button></>}>
      <div className="flex flex-col gap-4">

          <Field label="Current baseline"><Input disabled value={`${current.label} · ${money(current.budget.total, 'full')}`} /></Field>
          <Field label="Proposed baseline total" hint={`Today's EAC is ${money(pos.eac, 'full')}`}><MoneyInput value={newTotal} onChange={setNewTotal} /></Field>
          <Field label="Schedule impact (days)"><Input type="number" value={days} onChange={(e) => setDays(Number(e.target.value) || 0)} /></Field>
          <Field label="Root cause"><Select value={rootCause} onChange={(e) => setRootCause(e.target.value as RootCause)} options={[...ROOT_CAUSES]} /></Field>
          <Field label="Primary cost category"><Select value={categoryId} onChange={(e) => setCategoryId(e.target.value)} options={s.categories.filter((c) => c.group !== 'contingency').map((c) => ({ value: c.id, label: `${c.code} ${c.name}` }))} /></Field>
          <Field label="Routes to"><Input disabled value={`${doa.label} (${doa.people.join(', ') || 'no one assigned'}) · ${moneyDelta(delta)}`} /></Field>
        </div>
        <Field label="Why the current baseline no longer represents the work"><TextArea value={note} onChange={(e) => setNote(e.target.value)} placeholder="e.g. City added 2,400 ft of paving restoration as a permit condition; the remaining scope cannot be delivered within the Current baseline." /></Field>
      </div>
    </Modal>
  )
}

const groupSum = <T,>(xs: T[], k: (x: T) => string, v: (x: T) => number = (x: any) => x.costImpact) => xs.reduce<Record<string, number>>((a, x) => { a[k(x)] = (a[k(x)] || 0) + v(x); return a }, {})

function TrendPanel({ s, projectId, changes }: { s: AppState; projectId: string; changes: Change[] }) {
  const approved = changes.filter((c) => c.status === 'approved' && c.decidedAt)
  const periods = s.costPeriods.filter((c) => c.projectId === projectId).map((c) => c.period).sort()
  const first = [approved.map((c) => c.decidedAt!.slice(0, 7)).sort()[0], periods.find((x) => x >= (s.snapshots.find((n) => n.projectId === projectId && n.type === 'original')?.date.slice(0, 7) ?? ''))].filter(Boolean).sort()[0]
  if (!first) return <Panel title="Cumulative approved changes"><p className="text-[13px] text-muted">The trend starts with the first approved change.</p></Panel>
  const months = monthRange(first, s.settings.asOf.slice(0, 7))
  let acc = 0, accC = 0
  const cum = months.map((m) => { acc += sum(approved.filter((c) => c.decidedAt!.slice(0, 7) === m).map((c) => c.costImpact)); return acc })
  const cumC = months.map((m) => { accC += sum(approved.filter((c) => c.decidedAt!.slice(0, 7) === m && c.fundedFrom === 'contingency').map((c) => c.costImpact)); return accC })
  const pending = sum(changes.filter((c) => c.status === 'in-review').map((c) => c.costImpact))
  return (
    <Panel title="Cumulative approved changes" subtitle={`By month of decision · pending ${moneyDelta(pending)} not yet counted`}>
      <LineChart labels={months.map((m) => fmtMonth(m).replace(/ (\d\d)(\d\d)$/, " '$2"))} height={210} series={[
        { name: 'All approved', values: cum, color: 'rgb(var(--bad))', area: true },
        { name: 'From contingency', values: cumC, color: 'rgb(var(--link))' },
        ...(pending ? [{ name: 'If pending approved', values: months.map((_, i) => (i === months.length - 1 ? acc + pending : null)), color: 'rgb(var(--marker))', dashed: true }] : []),
      ]} />
    </Panel>
  )
}

            <Impact label="Contingency after" value={money(live ? contAfter : pos.contingencyRemaining)} sub={live && c.fundedFrom === 'contingency' ? `${money(pos.contingencyRemaining)} − ${money(c.costImpact)}` : 'Not drawn from contingency'} bad={contAfter < 0} />
            <Impact label="Authorized budget after" value={money(budgetAfter)} sub={live && c.fundedFrom === 'new-funding' ? `${money(pos.currentBudget)} + ${money(c.costImpact)}` : 'Unchanged'} />
            <Impact label={live ? 'New EAC (indicative)' : 'Signed EAC'} value={money(eacAfter)} sub={!live ? 'Current signed forecast' : c.fundedFrom === 'contingency' && contAfter >= 0 ? 'Absorbed by contingency already in the forecast' : `${money(pos.eac)} + ${money(eacAfter - pos.eac)} until the next forecast cycle`} bad={eacAfter > budgetAfter} />
          </div>
          {contAfter < 0 && <Callout tone="bad">This change would overdraw contingency by {money(-contAfter)}. Fund the shortfall as new funding or reduce scope.</Callout>}
          <Callout tone="info" title={`Routes to ${doa.label} under the DOA matrix`}>
            {money(Math.abs(c.costImpact), 'full')} falls in the {doa.label} band of {doa.matrix}{doa.people.length ? ` — ${doa.people.join(', ')}` : ''}. {c.scheduleImpactDays > 30 ? 'Schedule impact over 30 days adds a scheduler review.' : 'No scheduler review (30 days or less).'}
          </Callout>
        </Section>

        <CommitmentsSection c={c} />
        {c.scheduleImpactDays !== 0 && <ScheduleSection c={c} />}

        <Section title="Traceability">
          <ul className="flex flex-col divide-y divide-line rounded-md border border-line text-[13px]">
            <TraceRow label="Baseline it changed" value={base ? `${base.label} · ${fmtDate(base.date)} · ${money(base.budget.total)}` : '—'} href={base ? projectHref(c.projectId, 'baselines') : undefined} onClick={() => base && setFocusSnapshot(base.id)} />
            <TraceRow label="Resulting snapshot" value={res ? `${res.label} · ${fmtDate(res.date)} · ${money(res.budget.total)}` : live ? 'Created automatically on approval' : '—'} href={res ? projectHref(c.projectId, 'baselines') : undefined} onClick={() => res && setFocusSnapshot(res.id)} />
            <TraceRow label="Linked risk" value={risk ? `${risk.code} ${risk.title} (${risk.status})` : 'None'} href={risk ? projectHref(c.projectId, 'risk') : undefined} />
            <TraceRow label="Variance explanations" value={variance.length ? variance.map((v) => `${v.description} (${v.status})`).join('; ') : 'Not cited in a variance explanation'} href={variance.length ? projectHref(c.projectId, 'cost') : undefined} />
          </ul>
        </Section>

        {chain.length > 0 && (
          <Section title="Approval chain">
            <ol className="flex flex-col gap-1.5">
              {chain.map((x, i) => (
                <li key={i} className="flex flex-wrap items-center gap-2 rounded-md border border-line px-2.5 py-1.5 text-[13px]">
                  <Avatar name={x.actor} size={20} /><b>{x.actor}</b><span className="text-muted">{x.title}</span><span className="text-muted">· {x.step}</span>
                  {x.sig && <Badge tone="good" title={`Signature ${x.sig}`}>e-signed <Mono className="text-[10.5px]">{x.sig.slice(0, 8)}</Mono></Badge>}
                  <span className="ml-auto text-xs text-faint">{fmtDateTime(x.at)}</span>
                </li>
              ))}
            </ol>
          </Section>
        )}
        <p className="text-xs text-faint">Project {p.code} · change id <Mono>{c.id}</Mono>. Approved changes are immutable; corrections are raised as a new change.</p>
      </div>
    </Drawer>
  )
}

function Impact({ label, value, sub, bad }: { label: string; value: string; sub: string; bad?: boolean }) {
  return <div className="min-w-0 rounded-md border border-line px-3 py-2"><div className="eyebrow">{label}</div><div className={cx('font-display text-[19px] font-bold tabular-nums', bad && 'text-bad')}>{value}</div><div className="text-xs text-muted">{sub}</div></div>
}

function TraceRow({ label, value, href, onClick }: { label: string; value: string; href?: string; onClick?: () => void }) {
  return (
    <li className="flex flex-wrap items-baseline justify-between gap-2 px-3 py-2">
      <span className="text-xs font-semibold uppercase tracking-wide text-faint">{label}</span>
      {href ? <a href={href} onClick={onClick} className="min-w-0 text-right font-semibold text-link hover:underline">{value}</a> : <span className="min-w-0 text-right">{value}</span>}
    </li>
  )
}

function NewChangeModal({ open, onClose, projectId, onCreated }: { open: boolean; onClose: () => void; projectId: string; onCreated: (id: string) => void }) {
  const s = useStore()
  const p = s.projects.find((x) => x.id === projectId)!
  const pos = position(s, projectId)
  const blank = { type: 'cost' as Change['type'], title: '', description: '', rootCause: 'Third-party / substructures' as RootCause, categoryId: 'cat-contract', cost: 0, days: 0, scope: '', funded: 'contingency' as Change['fundedFrom'], riskId: '', contractIds: [] as string[] }
  const [f, setF] = useState(blank)
  const set = <K extends keyof typeof blank>(k: K, v: (typeof blank)[K]) => setF((x) => ({ ...x, [k]: v }))
  const doa = doaLabel(s, projectId, f.cost)
  const def = latestDef(s, 'change-request')
  const route = useMemo(() => (def ? simulate(s, def, { amount: f.cost, scheduleImpactDays: f.days, rootCause: f.rootCause }, projectId) : []), [s, def, f.cost, f.days, f.rootCause, projectId])
  const risks = s.risks.filter((r) => r.projectId === projectId && r.status !== 'closed')
  const commitments = s.contracts.filter((c) => c.projectId === projectId && c.status !== 'closed')
  const contAfter = pos.contingencyRemaining - (f.funded === 'contingency' ? f.cost : 0)
  const submit = () => {
    if (!f.title.trim() || !f.description.trim()) { toast('Give the change a title and a description.', 'bad'); return }
    if (f.type !== 'schedule' && !f.cost && f.type !== 'scope') { toast('Enter the cost impact.', 'bad'); return }
    const code = nextChangeCode(s, p)
    const id = uid('chg')
    mutate((d) => {
      const base = [...d.snapshots].filter((x) => x.projectId === projectId && (x.type === 'current' || x.type === 'original')).sort((a, b) => a.date.localeCompare(b.date) || (a.type === 'current' ? 1 : -1)).pop()
      d.changes.push({ id, projectId, code, title: f.title.trim(), type: f.type, rootCause: f.rootCause, description: f.description.trim(), costImpact: f.cost, scheduleImpactDays: f.days, scopeImpact: f.scope.trim() || 'None', categoryId: f.categoryId, fundedFrom: f.cost ? f.funded : 'none', status: 'draft', requestedById: d.currentUserId, requestedAt: d.settings.asOf, baselineSnapshotId: base?.id, riskId: f.riskId || undefined, contractIds: f.contractIds.length ? f.contractIds : undefined })
      const inst = startWorkflow(d, 'change-request', { title: `${code} ${f.title.trim()}`, recordType: 'change', recordId: id, projectId, data: { amount: f.cost, scheduleImpactDays: f.days, rootCause: f.rootCause, justification: f.description.trim() } })
      const slot = inst?.active.find((a) => a.nodeId === 'prep' && a.assigneeIds.includes(d.currentUserId))
      if (inst && slot) actOnWorkflow(d, inst.id, slot.nodeId, d.currentUserId, 'submit', 'Change request prepared')
    }, { entity: 'change', entityId: id, projectId, action: 'create', summary: `Raised ${code} ${f.title.trim()} (${moneyDelta(f.cost)}, ${f.days} days)` })
    toast('Submitted for approval')
    setF(blank)
    onCreated(id)
  }
  return (
    <Modal open={open} onClose={onClose} wide title={`New change request · ${nextChangeCode(s, p)}`} footer={<><Button variant="ghost" onClick={onClose}>Cancel</Button><Button variant="primary" onClick={submit}>Create & route</Button></>}>
      <div className="grid grid-cols-1 gap-4 lg:grid-cols-[3fr_2fr]">
        <div className="flex flex-col gap-3">
          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
            <Field label="Type"><Select value={f.type} onChange={(e) => set('type', e.target.value as Change['type'])} options={[{ value: 'cost', label: 'Cost' }, { value: 'scope', label: 'Scope' }, { value: 'schedule', label: 'Schedule' }, { value: 'trend', label: 'Trend' }]} /></Field>
            <Field label="Root cause"><Select value={f.rootCause} onChange={(e) => set('rootCause', e.target.value as RootCause)} options={[...ROOT_CAUSES]} /></Field>
          </div>
          <Field label="Title"><Input value={f.title} onChange={(e) => set('title', e.target.value)} placeholder="e.g. Relocate conflicting 8-inch water main at Sta. 42+10" /></Field>
          <Field label="Description and justification"><TextArea value={f.description} onChange={(e) => set('description', e.target.value)} placeholder="What was found, why the change is needed, alternatives considered, and the evidence attached (photos, contractor quote, RFI)." /></Field>
          <div className="grid grid-cols-1 gap-3 sm:grid-cols-3">
            <Field label="Cost impact" hint="Negative for a saving"><MoneyInput value={f.cost} onChange={(v) => set('cost', v)} /></Field>
            <Field label="Schedule impact (days)"><Input type="number" value={f.days} onChange={(e) => set('days', Number(e.target.value) || 0)} /></Field>
            <Field label="Cost category"><Select value={f.categoryId} onChange={(e) => set('categoryId', e.target.value)} options={s.categories.filter((c) => c.group === 'direct' || c.group === 'indirect').map((c) => ({ value: c.id, label: `${c.code} ${c.name}` }))} /></Field>
          </div>
          <Field label="Scope impact"><Input value={f.scope} onChange={(e) => set('scope', e.target.value)} placeholder="None, or what is added or removed" /></Field>
          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
            <Field label="Funding source"><SegButtons options={[{ id: 'contingency', label: 'Contingency' }, { id: 'new-funding', label: 'New funding' }]} value={f.funded === 'none' ? 'contingency' : f.funded} onChange={(v) => set('funded', v)} /></Field>
            <Field label="Linked risk (optional)"><Select value={f.riskId} onChange={(e) => set('riskId', e.target.value)} options={[{ value: '', label: 'None' }, ...risks.map((r) => ({ value: r.id, label: `${r.code} ${r.title}` }))]} /></Field>
          </div>
          <div>
            <div className="mb-1 text-[12px] font-semibold text-muted">Commitments affected</div>
            {commitments.length ? (
              <ul className="flex flex-col divide-y divide-line rounded-md border border-line">
                {commitments.map((k) => (
                  <li key={k.id} className="flex items-center justify-between gap-2 px-2.5 py-1.5">
                    <Checkbox checked={f.contractIds.includes(k.id)} onChange={(v) => set('contractIds', v ? [...f.contractIds, k.id] : f.contractIds.filter((x) => x !== k.id))} label={<span><b>{k.vendor}</b> <span className="text-muted">· {k.type} · PO {k.poNumber ?? '—'}</span></span>} />
                    <span className="shrink-0 text-[12px] tabular-nums text-muted">{money(k.value + k.changeOrders)}</span>
                  </li>
                ))}
              </ul>
            ) : <p className="text-[12px] text-faint">No commitments on this project yet.</p>}
            <p className="mt-1 text-[11.5px] text-faint">After the change is created, raise an RFI / change order on each affected commitment — that is how the vendor's commitment moves.</p>
          </div>
        </div>
        <div className="flex flex-col gap-3 rounded-md border border-line bg-sunken/40 p-3">
          <div className="eyebrow">Impact preview</div>
          <KV cols={1} items={[
            ['Contingency after', <span className={cx('font-semibold tabular-nums', contAfter < 0 && 'text-bad')}>{money(contAfter, 'full')}{f.funded === 'contingency' && f.cost ? <span className="font-normal text-muted"> ({money(pos.contingencyRemaining)} − {money(f.cost)})</span> : null}</span>],
            ['Authorized budget after', <span className="tabular-nums">{money(pos.currentBudget + (f.funded === 'new-funding' ? f.cost : 0), 'full')}</span>],
            ['Signed EAC today', <span className="tabular-nums">{money(pos.eac, 'full')}</span>],
            ['Dollar authority', <b>{doa.label}</b>],
          ]} />
          <div className="eyebrow mt-1">Routing (live from the workflow)</div>
          <ol className="flex flex-col gap-1 text-[12.5px]">
            {route.map((r, i) => <li key={i} className="flex gap-2"><span className="w-4 text-right tabular-nums text-faint">{i + 1}</span><span className="min-w-0"><b>{r.node.label}</b><span className="block text-muted">{r.people.map((x) => x.name).join(', ') || 'Unassigned'}</span></span></li>)}
          </ol>
          {contAfter < 0 && <Callout tone="bad">Overdraws contingency by {money(-contAfter)}.</Callout>}
        </div>
      </div>
    </Modal>
  )
}

// ───────────────────────────── Commitments the change affects → RFI / CCOs
function CommitmentsSection({ c }: { c: Change }) {
  const s = useStore()
  const me = useMe()
  const p = s.projects.find((x) => x.id === c.projectId)!
  const [editing, setEditing] = useState(false)
  const [sel, setSel] = useState<string[]>([])
  const [cco, setCco] = useState<CcoSeed | null>(null)
  const canEdit = me.id === p.team.pm || me.id === p.team.projectControls || me.roles.includes('admin') || me.roles.includes('pmo')
  const affected = changeCommitments(s, c)
  const ccos = ccosOfChange(s, c.id)
  const all = s.contracts.filter((k) => k.projectId === c.projectId)
  const raised = sum(ccos.filter((x) => x.status !== 'rejected').map((x) => x.amount))
  const save = () => {
    const names = all.filter((k) => sel.includes(k.id)).map((k) => k.vendor)
    mutate((d) => { const x = d.changes.find((y) => y.id === c.id)!; x.contractIds = sel.length ? sel : undefined }, { entity: 'change', entityId: c.id, projectId: c.projectId, action: 'commitments', summary: `${c.code}: commitments affected → ${names.join(', ') || 'none'}` })
    toast('Commitments affected saved'); setEditing(false)
  }
  return (
    <Section title="Commitments affected" actions={canEdit && !editing ? <Button size="sm" variant="ghost" onClick={() => { setSel(c.contractIds ?? []); setEditing(true) }}>Edit</Button> : undefined}>
      {editing ? (
        <div className="flex flex-col gap-2 rounded-md border border-line p-2.5">
          {all.length ? all.map((k) => <Checkbox key={k.id} checked={sel.includes(k.id)} onChange={(v) => setSel(v ? [...sel, k.id] : sel.filter((x) => x !== k.id))} label={<span><b>{k.vendor}</b> <span className="text-muted">· {k.type} · PO {k.poNumber ?? '—'}</span></span>} />) : <span className="text-[13px] text-faint">No commitments on this project.</span>}
          <div className="flex justify-end gap-2"><Button size="sm" variant="ghost" onClick={() => setEditing(false)}>Cancel</Button><Button size="sm" variant="primary" onClick={save}>Save</Button></div>
        </div>
      ) : affected.length === 0 ? (
        <p className="text-[13px] text-muted">No commitments linked. If a vendor will be paid differently because of this change, link the commitment and raise an RFI / change order on it.</p>
      ) : (
        <ul className="flex flex-col divide-y divide-line rounded-md border border-line text-[13px]">
          {affected.map((k) => {
            const r = commitmentRow(s, k)
            const mine = ccos.filter((x) => x.contractId === k.id)
            const live = k.status !== 'rfp' && k.status !== 'closed'
            return (
              <li key={k.id} className="flex flex-col gap-1.5 px-3 py-2">
                <div className="flex flex-wrap items-center gap-2">
                  <b>{k.vendor}</b><span className="text-muted">{k.type} · PO <Mono>{k.poNumber ?? '—'}</Mono></span>
                  <span className="ml-auto tabular-nums text-muted">current {money(r.current)}</span>
                </div>
                {mine.length > 0 && <div className="flex flex-wrap gap-1.5">{mine.map((x) => <a key={x.id} href={projectHref(c.projectId, 'finance')} onClick={() => setFocusCommitment(k.id, x.id)} className="inline-flex items-center gap-1 rounded border border-line px-1.5 py-0.5 text-[12px] hover:border-ink/40"><Mono className="font-semibold">{x.code}</Mono><span className="tabular-nums">{moneyDelta(x.amount)}</span><StatusBadge status={x.status} /></a>)}</div>}
                <div className="flex flex-wrap gap-2">
                  {canEdit && live && c.status !== 'rejected' && <Button size="sm" onClick={() => setCco({ contractId: k.id, changeId: c.id })}>Raise RFI / CCO</Button>}
                  <a href={projectHref(c.projectId, 'finance')} onClick={() => setFocusCommitment(k.id)} className="self-center text-[12px] font-semibold text-link hover:underline">Open commitment →</a>
                </div>
              </li>
            )
          })}
        </ul>
      )}
      {ccos.length > 0 && (
        <p className={cx('text-[12px]', raised !== c.costImpact ? 'text-warn' : 'text-muted')}>
          RFI / CCOs raised under this change: <b className="tabular-nums">{moneyDelta(raised)}</b> of <b className="tabular-nums">{moneyDelta(c.costImpact)}</b>{raised !== c.costImpact ? ` — ${money(Math.abs(c.costImpact - raised))} ${raised < c.costImpact ? 'not yet placed with a vendor (company cost or still to raise)' : 'more than the change'}` : ' — fully placed with vendors'}.
        </p>
      )}
      <CcoModal projectId={c.projectId} seed={cco} onClose={() => setCco(null)} />
    </Section>
  )
}

// ───────────────────────────── Schedule impact → scheduler task (P6 is not written back)
function ScheduleSection({ c }: { c: Change }) {
  const s = useStore()
  const me = useMe()
  const p = s.projects.find((x) => x.id === c.projectId)!
  const n = c.scheduleNotice
  const schedulerId = n?.schedulerId ?? p.team.scheduler
  const action = n?.actionId ? s.actions.find((a) => a.id === n.actionId) : undefined
  const isScheduler = me.id === schedulerId || me.roles.includes('admin')
  const canNotify = me.id === p.team.pm || me.id === p.team.projectControls || me.roles.includes('admin') || me.roles.includes('pmo')
  const notify = () => {
    let res: { ok: boolean; reason?: string } = { ok: false }
    mutate((d) => { res = notifyScheduler(d, c.id) })
    toast(res.ok ? `Task sent to ${personName(s, schedulerId)}` : res.reason ?? 'Could not notify', res.ok ? 'good' : 'bad')
  }
  const done = () => { mutate((d) => completeScheduleNotice(d, c.id)); toast('Marked P6 updated') }
  return (
    <Section title="Schedule">
      <div className="flex flex-col gap-2 rounded-md border border-line px-3 py-2.5 text-[13px]">
        <div className="flex flex-wrap items-center gap-2">
          <b className="tabular-nums">{c.scheduleImpactDays > 0 ? '+' : ''}{c.scheduleImpactDays} days</b>
          <span className="text-muted">P6 schedule <Mono>{p.scheduleId}</Mono> · scheduler {personName(s, schedulerId)}</span>
          {n ? <Badge tone={n.status === 'done' ? 'good' : 'info'}>{n.status === 'done' ? 'P6 updated' : `Task sent ${fmtDateTime(n.sentAt)}`}</Badge> : <Badge tone="marker">Scheduler not notified</Badge>}
        </div>
        {action && <div className="text-[12px] text-muted">Task: {action.title} · due {fmtDate(action.due)} · {action.status}</div>}
        <div className="flex flex-wrap gap-2">
          {!n && canNotify && <Button size="sm" variant="marker" onClick={notify}>Notify scheduler</Button>}
          {n?.status === 'sent' && isScheduler && <Button size="sm" variant="primary" onClick={done}>Mark P6 updated</Button>}
        </div>
        <p className="text-[11.5px] text-faint">Crosswalk does not write to P6. The scheduler gets a task and a notification; approved changes move the milestone forecasts here, and the scheduler reflects them in the P6 schedule.</p>
      </div>
    </Section>
  )
}
