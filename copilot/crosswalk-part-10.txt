// Project register: filterable/sortable table with quick filters, CSV export and a territory map toggle.
import { prefs } from '@/core/data'
import { useMemo, useState } from 'react'
import { useMe, useStore } from '@/core/store'
import { go, projectHref } from '@/core/router'
import { daysBetween, fmtDate, money, moneyDelta, cx, sum } from '@/core/util'
import { saveFile, toCSV } from '@/core/download'
import { Badge, Button, HealthDot, Input, Mono, PageHeader, Panel, Progress, SegButtons, Select, StageTrack, StatusBadge, Table, Empty, type Column } from '@/ui/kit'
import { Icon } from '@/ui/icons'
import { toast } from '@/ui/toast'
import { ProgramTag } from '../shared/pickers'
import { PHASES, projectRow, type ProjectRow } from '../portfolio/lib'
import { TerritoryMap } from './TerritoryMap'

type Quick = 'all' | 'mine' | 'over' | 'gate60' | 'construction' | 'red'
const QUICK: { id: Quick; label: string }[] = [
  { id: 'all', label: 'All' },
  { id: 'mine', label: 'My projects' },
  { id: 'over', label: 'Over budget' },
  { id: 'gate60', label: 'Gate due ≤ 60d' },
  { id: 'construction', label: 'Construction' },
  { id: 'red', label: 'Red health' },
]

const VIEW_KEY = 'crosswalk:projects:view'
function loadPref(): { view: 'table' | 'map'; quick: Quick } {
  try { const v = JSON.parse(prefs.getRaw(VIEW_KEY) || '{}'); return { view: v.view === 'map' ? 'map' : 'table', quick: QUICK.some((q) => q.id === v.quick) ? v.quick : 'all' } } catch { return { view: 'table', quick: 'all' } }
}
function savePref(v: { view: string; quick: string }) { try { prefs.setRaw(VIEW_KEY, JSON.stringify(v)) } catch { /* storage blocked */ } }

export default function Projects(_props: { projectId?: string; id?: string }) {
  const s = useStore()
  const me = useMe()
  const pref = useMemo(loadPref, [])
  const [view, setViewRaw] = useState<'table' | 'map'>(pref.view)
  const [quick, setQuickRaw] = useState<Quick>(pref.quick)
  const [q, setQ] = useState('')
  const [program, setProgram] = useState('')
  const [phase, setPhase] = useState('')
  const [status, setStatus] = useState('open')
  const [pm, setPm] = useState('')
  const setView = (v: 'table' | 'map') => { setViewRaw(v); savePref({ view: v, quick }) }
  const setQuick = (v: Quick) => { setQuickRaw(v); savePref({ view, quick: v }) }

  const all = useMemo(() => s.projects.map((p) => projectRow(s, p)), [s])
  const asOf = s.settings.asOf
  const isMine = (r: ProjectRow) => Object.values(r.p.team).includes(me.id)
  const rows = all.filter((r) => {
    if (quick === 'mine' && !isMine(r)) return false
    if (quick === 'over' && !(r.variance > 0)) return false
    if (quick === 'gate60' && !(r.nextReview && daysBetween(asOf, r.nextReview.scheduled) <= 60)) return false
    if (quick === 'construction' && r.phase !== 3) return false
    if (quick === 'red' && r.overall !== 'red') return false
    if (program && r.p.programId !== program) return false
    if (phase !== '' && String(r.phase) !== phase) return false
    if (status === 'open' && !(r.p.status === 'active' || r.p.status === 'on-hold')) return false
    if (status !== 'open' && status !== '' && r.p.status !== status) return false
    if (pm && r.p.team.pm !== pm) return false
    if (q.trim()) {
      const hay = `${r.p.code} ${r.p.name} ${r.p.location.city} ${r.p.location.district} ${r.pmName} ${r.p.asset.lineId}`.toLowerCase()
      if (!hay.includes(q.toLowerCase())) return false
    }
    return true
  })
  const mineCount = all.filter(isMine).length
  const pms = s.people.filter((p) => p.roles.includes('pm'))

  const columns: Column<ProjectRow>[] = [
    { key: 'code', header: 'Project', sort: (r) => r.p.code, render: (r) => (
      <a href={projectHref(r.p.id)} onClick={(e) => e.stopPropagation()} className="flex min-w-[240px] max-w-[360px] flex-col">
        <span className="flex items-center gap-1.5"><ProgramTag programId={r.p.programId} /><Mono className="font-semibold">{r.p.code}</Mono>{r.p.status !== 'active' && <StatusBadge status={r.p.status} />}</span>
        <span className="truncate font-semibold text-ink" title={r.p.name}>{r.p.name}</span>
        <span className="text-[11px] text-faint">{r.p.location.city} · {r.p.location.district} · Tier {r.p.sizeTier}</span>
      </a>
    ) },
    { key: 'stage', header: 'Stage', sort: (r) => r.phase * 10 + r.p.stageIndex, render: (r) => (
      <div className="w-[150px]"><StageTrack stages={r.stages} current={r.p.stageIndex} status={r.p.status} /><div className="mt-1 truncate text-[11px] text-muted" title={r.stageName}>{r.p.stageIndex + 1}/{r.stages.length} · {r.stageName}</div></div>
    ) },
    { key: 'health', header: 'Health', sort: (r) => ({ red: 0, amber: 1, green: 2 })[r.overall], render: (r) => (
      <span className="flex gap-1" title={`Scope ${r.p.health.scope} · Schedule ${r.p.health.schedule} · Cost ${r.p.health.cost} · Risk ${r.p.health.risk} · Safety ${r.p.health.safety}`}>
        {(['scope', 'schedule', 'cost', 'risk', 'safety'] as const).map((k) => <HealthDot key={k} rag={r.p.health[k]} />)}
      </span>
    ) },
    { key: 'pm', header: 'PM', sort: (r) => r.pmName, render: (r) => <span className="whitespace-nowrap">{r.pmName}</span> },
    { key: 'budget', header: 'Budget', num: true, sort: (r) => r.budget, render: (r) => <span title={r.budgetBasis === 'estimate' ? 'No budget yet — latest estimate shown' : `${r.budgetBasis} budget`}>{money(r.budget)}{r.budgetBasis === 'estimate' && <span className="ml-1 text-[10px] text-faint">est</span>}{r.budgetBasis === 'planning' && <span className="ml-1 text-[10px] text-faint">plan</span>}</span> },
    { key: 'eac', header: 'EAC', num: true, sort: (r) => r.eac, render: (r) => money(r.eac) },
    { key: 'var', header: 'Variance', num: true, sort: (r) => r.variance, render: (r) => r.budgetBasis === 'estimate' || r.budgetBasis === 'none' ? <span className="text-faint">—</span> : <span className={cx('font-semibold', r.variance > 0 ? 'text-bad' : 'text-good')}>{moneyDelta(r.variance)}</span> },
    { key: 'next', header: 'Next milestone', sort: (r) => r.next?.forecast ?? '9', render: (r) => r.next ? (
      <div className="min-w-[150px]"><div className="truncate text-[13px]">{r.next.name}</div><div className="text-[11px] text-muted">{fmtDate(r.next.forecast)}{r.nextSlip > 0 && <span className={cx('ml-1 font-semibold', r.nextSlip > 30 ? 'text-bad' : 'text-warn')}>+{r.nextSlip}d</span>}</div></div>
    ) : <span className="text-faint">—</span> },
    { key: 'ready', header: 'Gate readiness', sort: (r) => r.readiness, render: (r) => r.p.status === 'active' ? (
      <div className="w-[120px]" title={r.blockers.join('\n') || 'Ready for gate'}>
        <div className="flex items-center justify-between text-[11px]"><span className="tabular-nums font-semibold">{r.readiness}%</span>{r.nextReview && <span className="text-faint">{fmtDate(r.nextReview.scheduled).replace(/, \d{4}/, '')}</span>}</div>
        <Progress value={r.readiness} tone={r.readiness >= 80 ? 'good' : r.readiness >= 50 ? 'warn' : 'bad'} />
      </div>
    ) : <span className="text-faint">—</span> },
  ]

  const exportCsv = async () => {
    const res = await saveFile(`project-register-${asOf}.csv`, toCSV(rows.map((r) => ({
      Project: r.p.code, Name: r.p.name, Program: r.programCode, District: r.p.location.district, City: r.p.location.city, Stage: `${r.p.stageIndex + 1}/${r.stages.length} ${r.stageName}`, Status: r.p.status,
      PM: r.pmName, Sponsor: s.people.find((x) => x.id === r.p.team.sponsor)?.name ?? '', Budget: r.budget, BudgetBasis: r.budgetBasis, EAC: r.eac, Variance: r.variance, ActualToDate: r.pos.actualToDate,
      ContingencyRemaining: r.pos.contingencyRemaining, Health: r.overall, Scope: r.p.health.scope, Schedule: r.p.health.schedule, Cost: r.p.health.cost, Risk: r.p.health.risk, Safety: r.p.health.safety,

    }))), 'text/csv')
    if (res === 'saved') toast(`Exported ${rows.length} projects`)
    else if (res === 'unavailable') toast('Download is not available here', 'bad')
  }

  const totBudget = sum(rows.map((r) => r.budget)), totEac = sum(rows.map((r) => r.eac))
  return (
    <div className="flex flex-col gap-4">
      <PageHeader
        eyebrow="Capital delivery"
        title="Project register"
        subtitle={`${all.filter((r) => r.p.status === 'active').length} active projects across ${s.programs.length} programs. Budget shows the control budget once authorized, else the planning budget or latest estimate.`}
        actions={<>
          <SegButtons value={view} onChange={setView} options={[{ id: 'table', label: 'Table' }, { id: 'map', label: 'Map' }]} />
          <Button icon={<Icon name="download" className="h-4 w-4" />} onClick={exportCsv}>Export CSV</Button>

      <Panel pad={false}>

          <Table rows={rows} columns={columns} rowKey={(r) => r.p.id} onRowClick={(r) => go(projectHref(r.p.id))} maxHeight={680}

export function TerritoryMap({ rows }: { rows: ProjectRow[] }) {
  const [hover, setHover] = useState<string | null>(null)
  const [pinned, setPinned] = useState<string | null>(null)
  const maxB = Math.max(...rows.map((r) => r.budget), 1)
  const radius = (b: number) => 6 + Math.sqrt(Math.max(b, 0) / maxB) * 22
  const sel = rows.find((r) => r.p.id === (pinned ?? hover))
  // a few stations along the lines for context
  const stations = useMemo(() => rows.filter((r) => r.p.asset.type === 'compressor-station'), [rows])
  const ordered = rows.slice().sort((a, b) => b.budget - a.budget)
  return (
    <div className="relative">

        </defs>
        <rect width={W} height={H} fill="url(#tm-grid)" />

        <path d="M0 330 C 30 360, 20 420, 40 470 S 30 560, 60 600 L0 600 Z" fill="rgb(var(--info))" opacity="0.12" />

            <text x={d.label[0]} y={d.label[1]} textAnchor="middle" fontSize={12} fontWeight={700} letterSpacing="0.14em" fill="rgb(var(--faint))">{d.name.toUpperCase()}</text>
          </g>
        ))}
        {LINES.map((l) => (

            </g>

      <div className="mt-2 flex flex-wrap items-center gap-x-4 gap-y-1 text-[11px] text-muted">
        <span className="inline-flex items-center gap-1.5"><HealthDot rag="green" /> On track</span>

  )
}

// readiness across projects, large-project annual report, program evidence package, regulatory snapshots (RC-02).
import { useMemo, useState } from 'react'
import { mutate, useStore, personName } from '@/core/store'
import type { AppState } from '@/core/types'
import { activeBudget, position, snapshotsFor, takeSnapshot } from '@/core/engine/position'
import { stageName } from '@/core/engine/gates'
import { saveFile, toCSV } from '@/core/download'
import { go, projectHref } from '@/core/router'
import { fmtDate, money, moneyDelta, sum, cx } from '@/core/util'
import { Badge, Button, Callout, Checkbox, Field, Input, Modal, PageHeader, Panel, Progress, SegButtons, Stat, StatusBadge, Table, Mono } from '@/ui/kit'
import { BarChart } from '@/ui/charts'
import { ProgramSelect, ProgramTag } from '../shared/pickers'
import { toast } from '@/ui/toast'
import { Icon } from '@/ui/icons'
import { assessEvidence, forecastByYear, actualsByYear, programYears } from '../project/tabs/controls/trace'
import { evidenceRows, programBinderHtml } from '../project/tabs/controls/binder'

const LARGE = 50_000_000
const CPCN = 75_000_000

export default function Regulatory() {
  const s = useStore()
  const [programId, setProgramId] = useState('')
  const [snapOpen, setSnapOpen] = useState(false)
  const year = Number(s.settings.asOf.slice(0, 4))
  const years = useMemo(() => programYears(s, programId || undefined), [s, programId])
  const projects = s.projects.filter((p) => (!programId || p.programId === programId) && p.status !== 'cancelled')
  const assessed = useMemo(() => projects.filter((p) => s.budgets.some((b) => b.projectId === p.id)).map((p) => ({ p, a: assessEvidence(s, p.id), pos: position(s, p.id) })), [s, programId]) // eslint-disable-line react-hooks/exhaustive-deps
  const cur = years.find((y) => y.year === year)
  const ready = assessed.filter((x) => x.a.score >= 90).length
  const avg = assessed.length ? Math.round(sum(assessed.map((x) => x.a.score * Math.max(1, x.pos.currentBudget))) / sum(assessed.map((x) => Math.max(1, x.pos.currentBudget)))) : 0
  const prog = s.programs.find((x) => x.id === programId)
  const regSnaps = s.snapshots.filter((x) => x.type === 'regulatory' && projects.some((p) => p.id === x.projectId)).sort((a, b) => b.date.localeCompare(a.date))

  const downloadProgram = async (kind: 'html' | 'csv') => {
    if (!prog) return
    const pids = assessed.map((x) => x.p.id)
    const r = kind === 'html'
      ? await saveFile(`${prog.code}-program-evidence-package-${s.settings.asOf}.html`, programBinderHtml(s, prog.id, pids), 'text/html')
      : await saveFile(`${prog.code}-program-evidence-extract-${s.settings.asOf}.csv`, toCSV(pids.flatMap((id) => evidenceRows(s, id))), 'text/csv')
    if (r === 'saved') { mutate(() => {}, { entity: 'evidence', entityId: prog.id, action: 'export', summary: `Downloaded ${prog.code} program evidence ${kind === 'html' ? 'binder' : 'CSV'} (${pids.length} projects)` }); toast(kind === 'html' ? 'Program binder saved' : 'Program CSV saved') }
    else if (r === 'unavailable') toast('Download is not available here', 'bad')
  }

  return (
    <div className="flex flex-col gap-4">
      <PageHeader eyebrow="Controls" title="Rate case & evidence" subtitle="Authorized funding against recorded and forecast cost, and how ready each project is for a reasonableness review. Every figure is computed from budgets, ERP cost periods, signed forecasts and approved decisions."
        actions={<>
          <div className="w-[260px]"><ProgramSelect value={programId} onChange={setProgramId} /></div>
          <Button onClick={() => setSnapOpen(true)} icon={<Icon name="plus" className="h-4 w-4" />}>Regulatory snapshot</Button>
        </>} />

      <div className="grid grid-cols-2 gap-3 lg:grid-cols-5">
        <Stat label={`Authorized ${year}`} value={money(cur?.authorized)} sub={cur?.proceeding || '—'} />
        <Stat label={`Recorded ${year} to date`} value={money(cur?.recorded)} sub={cur?.authorized ? `${((cur.recorded / cur.authorized) * 100).toFixed(0)}% of authorized` : undefined} />
        <Stat label={`Forecast ${year}`} value={money(cur?.forecast)} tone={cur && cur.forecast > cur.authorized ? 'bad' : 'good'} sub={cur ? `${moneyDelta(cur.forecast - cur.authorized)} vs authorized` : undefined} />
        <Stat label="Review readiness" value={`${avg}%`} tone={avg >= 90 ? 'good' : avg >= 75 ? 'warn' : 'bad'} sub={`Dollar-weighted · ${ready} of ${assessed.length} projects ≥ 90%`} />
        <Stat label="Large projects" value={assessed.filter((x) => Math.max(x.pos.currentBudget, x.pos.eac) >= LARGE).length} sub={`≥ ${money(LARGE)} · ${assessed.filter((x) => Math.max(x.pos.currentBudget, x.pos.eac) > CPCN).length} over CPCN threshold`} />
      </div>

      <FundingPanel s={s} years={years} programId={programId} year={year} />
      <ReadinessPanel assessed={assessed} s={s} />
      <LargeProjectsPanel s={s} assessed={assessed} year={year} />

      <div className="grid grid-cols-1 gap-4 xl:grid-cols-2">
        <Panel title="Program evidence package" subtitle="One binder covering every budgeted project in the program — cover, estimates, baselines, change approval chains, variances, contingency, safety, records and gates.">
          {prog ? (
            <div className="flex flex-col gap-3">
              <div className="text-[13px]"><ProgramTag programId={prog.id} /> <b>{prog.name}</b> · {prog.regulatoryDriver} · recovery via {prog.recovery}</div>
              <div className="grid grid-cols-3 gap-2 text-[13px]">
                <MiniFig k="Projects" v={String(assessed.length)} />
                <MiniFig k="Gaps to close" v={String(sum(assessed.map((x) => x.a.gaps.length)))} />
                <MiniFig k="Recorded" v={money(sum(assessed.map((x) => x.pos.actualToDate)))} />
              </div>
              <div className="flex flex-wrap gap-2"><Button variant="primary" onClick={() => downloadProgram('html')} icon={<Icon name="download" className="h-4 w-4" />}>Download HTML binder</Button><Button onClick={() => downloadProgram('csv')} icon={<Icon name="download" className="h-4 w-4" />}>CSV extract</Button></div>
            </div>
          ) : <Callout tone="info">Choose a program above to assemble its evidence package.<div className="mt-2 flex flex-wrap gap-1.5">{s.programs.map((p) => <Button key={p.id} size="sm" onClick={() => setProgramId(p.id)}>{p.code}</Button>)}</div></Callout>}
        </Panel>
        <Panel title="Regulatory snapshots" subtitle="Positions frozen for a filing or data request" pad={false}>
          <Table rows={regSnaps} rowKey={(x) => x.id} dense maxHeight={260} empty={<p className="p-4 text-[13px] text-muted">No regulatory snapshots yet. Take one to freeze every active project's position for a proceeding.</p>} columns={[
            { key: 'd', header: 'Date', render: (x) => fmtDate(x.date) },
            { key: 'p', header: 'Project', render: (x) => <a className="font-mono text-[12px] font-semibold text-link hover:underline" href={projectHref(x.projectId, 'baselines')}>{s.projects.find((p) => p.id === x.projectId)?.code}</a> },
            { key: 'g', header: 'Proceeding', render: (x) => <span className="text-[12.5px]">{x.proceeding}</span> },
            { key: 'b', header: 'Budget', num: true, render: (x) => money(x.budget.total) },
            { key: 'e', header: 'EAC', num: true, render: (x) => money(x.budget.eac) },
            { key: 'by', header: 'By', render: (x) => <span className="text-xs text-muted">{personName(s, x.createdById)}</span> },
          ]} />
        </Panel>
      </div>
      <RegSnapshotModal open={snapOpen} onClose={() => setSnapOpen(false)} initialProgram={programId} />
    </div>
  )
}

function MiniFig({ k, v }: { k: string; v: string }) {
  return <div className="rounded-md border border-line px-2.5 py-2"><div className="text-[11px] font-semibold uppercase tracking-wide text-faint">{k}</div><div className="font-display text-[17px] font-bold tabular-nums">{v}</div></div>
}

// ───────────────────────────── BL-08
function FundingPanel({ s, years, programId, year }: { s: AppState; years: ReturnType<typeof programYears>; programId: string; year: number }) {
  const [view, setView] = useState<'years' | 'programs'>('years')
  const shown = years.filter((y) => y.authorized || y.recorded || y.forecast)
  const progRows = s.programs.map((pr) => { const y = programYears(s, pr.id).find((x) => x.year === year); return { pr, y } })
  return (
    <Panel title="Authorized vs recorded vs forecast" subtitle={view === 'years' ? `By year${programId ? '' : ', all programs'} · recorded = ERP actuals; forecast = actuals for past years plus the signed EAC spread for the rest` : `By program for ${year}`}
      actions={<SegButtons options={[{ id: 'years', label: 'By year' }, { id: 'programs', label: `By program ${year}` }]} value={view} onChange={setView} />}>
      {view === 'years' ? (
        <>
          <BarChart labels={shown.map((y) => String(y.year))} highlight={shown.findIndex((y) => y.year === year)} series={[
            { name: 'Authorized', values: shown.map((y) => y.authorized), color: 'rgb(var(--ink))' },
            { name: 'Planned (budgets)', values: shown.map((y) => y.planned), color: 'rgb(var(--muted))' },
            { name: 'Recorded', values: shown.map((y) => y.recorded), color: 'rgb(var(--link))' },
            { name: 'Forecast', values: shown.map((y) => y.forecast), color: 'rgb(var(--marker))' },
          ]} />
          <div className="mt-3">
            <Table rows={shown} rowKey={(y) => String(y.year)} dense columns={[
              { key: 'y', header: 'Year', render: (y) => <b className={y.year === year ? 'text-ink' : ''}>{y.year}{y.year === year && <span className="ml-1 text-xs font-normal text-muted">(current)</span>}</b> },
              { key: 'pr', header: 'Proceeding', render: (y) => <span className="text-[12px] text-muted">{y.proceeding || '—'}</span> },
              { key: 'a', header: 'Authorized', num: true, render: (y) => money(y.authorized) },
              { key: 'p', header: 'Planned', num: true, render: (y) => money(y.planned) },
              { key: 'r', header: 'Recorded', num: true, render: (y) => money(y.recorded) },
              { key: 'f', header: 'Forecast', num: true, render: (y) => money(y.forecast) },
              { key: 'v', header: 'Forecast − authorized', num: true, render: (y) => <span className={cx('font-semibold', y.forecast > y.authorized ? 'text-bad' : 'text-good')}>{moneyDelta(y.forecast - y.authorized)}</span> },
              { key: 'vp', header: '%', num: true, render: (y) => (y.authorized ? `${(((y.forecast - y.authorized) / y.authorized) * 100).toFixed(1)}%` : '—') },
            ]} />
          </div>
        </>
      ) : (
        <>
          <BarChart labels={progRows.map((x) => x.pr.code)} series={[
            { name: 'Authorized', values: progRows.map((x) => x.y?.authorized ?? 0), color: 'rgb(var(--ink))' },
            { name: 'Recorded', values: progRows.map((x) => x.y?.recorded ?? 0), color: 'rgb(var(--link))' },
            { name: 'Forecast', values: progRows.map((x) => x.y?.forecast ?? 0), color: 'rgb(var(--marker))' },
          ]} />
          <div className="mt-3">
            <Table rows={progRows} rowKey={(x) => x.pr.id} dense columns={[
              { key: 'p', header: 'Program', render: (x) => <span className="flex items-center gap-2"><ProgramTag programId={x.pr.id} />{x.pr.name}</span> },
              { key: 'm', header: 'Recovery', render: (x) => <span className="text-[12px] text-muted">{x.pr.recovery}</span> },
              { key: 'a', header: 'Authorized', num: true, render: (x) => money(x.y?.authorized) },
              { key: 'r', header: 'Recorded', num: true, render: (x) => money(x.y?.recorded) },
              { key: 'f', header: 'Forecast', num: true, render: (x) => money(x.y?.forecast) },
              { key: 'v', header: 'Variance', num: true, render: (x) => { const v = (x.y?.forecast ?? 0) - (x.y?.authorized ?? 0); return <span className={cx('font-semibold', v > 0 ? 'text-bad' : 'text-good')}>{moneyDelta(v)}</span> } },
            ]} />
          </div>
        </>
      )}
    </Panel>
  )
}

// ───────────────────────────── Readiness across projects (AI-06 style, deterministic)
function ReadinessPanel({ s, assessed }: { s: AppState; assessed: { p: AppState['projects'][number]; a: ReturnType<typeof assessEvidence>; pos: ReturnType<typeof position> }[] }) {
  const rows = [...assessed].sort((x, y) => x.a.score - y.a.score || y.pos.currentBudget - x.pos.currentBudget)
  const allGaps = assessed.flatMap((x) => x.a.gaps)
  const themes = Object.entries(allGaps.reduce<Record<string, number>>((a, g) => { a[g.area] = (a[g.area] || 0) + 1; return a }, {})).sort((a, b) => b[1] - a[1])
  return (
    <Panel title="Reasonableness-review readiness" subtitle="Deterministic checks over each project's records: approvals, signatures, baselines, variances, contingency, records, safety and gates. Lowest first." pad={false}>
      <div className="flex flex-wrap gap-2 border-b border-line px-4 py-2.5 text-xs text-muted">
        <span className="font-semibold text-ink">{allGaps.length} gaps across {assessed.length} projects:</span>
        {themes.map(([k, v]) => <Badge key={k}>{k} {v}</Badge>)}
      </div>
      <Table rows={rows} rowKey={(x) => x.p.id} maxHeight={480} onRowClick={(x) => go(projectHref(x.p.id, 'evidence'))} columns={[
        { key: 'p', header: 'Project', render: (x) => <span className="block min-w-[200px]"><span className="flex items-center gap-1.5"><ProgramTag programId={x.p.programId} /><Mono className="font-semibold">{x.p.code}</Mono></span><span className="block text-[12.5px]">{x.p.name}</span></span>, sort: (x) => x.p.code },
        { key: 'st', header: 'Stage', render: (x) => <span className="text-[12px] text-muted">{stageName(s, x.p)}{x.p.status !== 'active' && <> · <StatusBadge status={x.p.status} /></>}</span> },
        { key: 'b', header: 'Budget', num: true, render: (x) => money(x.pos.currentBudget), sort: (x) => x.pos.currentBudget },
        { key: 'sc', header: 'Score', render: (x) => <span className="flex min-w-[110px] items-center gap-2"><span className={cx('w-9 text-right font-bold tabular-nums', x.a.score >= 90 ? 'text-good' : x.a.score >= 75 ? 'text-warn' : 'text-bad')}>{x.a.score}%</span><Progress className="w-16" value={x.a.score} tone={x.a.score >= 90 ? 'good' : x.a.score >= 75 ? 'warn' : 'bad'} /></span>, sort: (x) => x.a.score },
        { key: 'g', header: 'Top gaps', render: (x) => x.a.gaps.length ? <ul className="min-w-[260px] text-[12.5px]">{x.a.gaps.slice(0, 2).map((g) => <li key={g.id} className={g.severity === 'major' ? 'text-ink' : 'text-muted'}>{g.gap}</li>)}{x.a.gaps.length > 2 && <li className="text-faint">+{x.a.gaps.length - 2} more</li>}</ul> : <span className="text-[12.5px] text-good">Ready</span> },
      ]} />
    </Panel>
  )
}

// ───────────────────────────── Large-project report (GO 177-style)
function LargeProjectsPanel({ s, assessed, year }: { s: AppState; assessed: { p: AppState['projects'][number]; a: ReturnType<typeof assessEvidence>; pos: ReturnType<typeof position> }[]; year: number }) {
  const rows = assessed.filter((x) => Math.max(x.pos.currentBudget, x.pos.eac) >= LARGE).map((x) => {
    const p = x.p
    const act = actualsByYear(s, p.id)
    const fy = forecastByYear(s, p.id)
    const ins = p.milestones.find((m) => /in.service/i.test(m.name))
    const authorized = x.pos.currentBudget
    const latestVar = s.variances.filter((v) => v.projectId === p.id).sort((a, b) => b.date.localeCompare(a.date))[0]
    const approvedChanges = s.changes.filter((c) => c.projectId === p.id && c.status === 'approved')
    return {
      x, p, authorized, original: x.pos.originalBudget, recordedYear: act[String(year)] ?? 0, recordedCum: x.pos.actualToDate, forecastYear: fy[String(year)] ?? 0, eac: x.pos.eac,
      cpcn: Math.max(authorized, x.pos.eac) > CPCN, ins, budgetKind: activeBudget(s, p.id)?.kind,
      explanation: latestVar?.narrative || (approvedChanges.length ? `${approvedChanges.length} approved change(s): ${approvedChanges.map((c) => c.code).join(', ')}` : 'Within authorization'),
    }
  }).sort((a, b) => b.authorized - a.authorized)
  const csv = async () => {
    const r = await saveFile(`large-project-report-${year}.csv`, toCSV(rows.map((r) => ({ project: r.p.code, name: r.p.name, program: s.programs.find((x) => x.id === r.p.programId)?.code, driver: r.p.regulatoryCitation, proceeding: r.p.fundingProceeding, stage: stageName(s, r.p), budget_type: r.budgetKind ?? '', cpcn_required: r.cpcn ? 'Y' : 'N', original_authorized: r.original, current_authorized: r.authorized, [`recorded_${year}`]: r.recordedYear, recorded_cumulative: r.recordedCum, [`forecast_${year}`]: r.forecastYear, eac: r.eac, variance: r.eac - r.authorized, variance_pct: r.authorized ? +(((r.eac - r.authorized) / r.authorized) * 100).toFixed(1) : '', in_service_baseline: r.ins?.baseline ?? '', in_service_forecast: r.ins?.actual ?? r.ins?.forecast ?? '', evidence_score: r.x.a.score, explanation: r.explanation }))), 'text/csv')
    if (r === 'saved') { mutate(() => {}, { entity: 'report', entityId: `large-${year}`, action: 'export', summary: `Downloaded ${year} large-project report (${rows.length} projects)` }); toast('Report saved') }
  }
  return (
    <Panel title={`Large-project annual report — ${year}`} subtitle={`GO 177-style report of projects with planned investment of ${money(LARGE)} or more · CPCN flag above ${money(CPCN)}`} pad={false}
      actions={<Button size="sm" onClick={csv} disabled={!rows.length} icon={<Icon name="download" className="h-3.5 w-3.5" />}>CSV</Button>}>
      <Table rows={rows} rowKey={(r) => r.p.id} onRowClick={(r) => go(projectHref(r.p.id, 'evidence'))} empty={<p className="p-4 text-[13px] text-muted">No projects at or above {money(LARGE)} in this selection.</p>} columns={[
        { key: 'p', header: 'Project', render: (r) => <span className="block min-w-[200px]"><span className="flex items-center gap-1.5"><ProgramTag programId={r.p.programId} /><Mono className="font-semibold">{r.p.code}</Mono>{r.cpcn && <Badge tone="warn" title={`Planned investment above ${money(CPCN)}`}>CPCN</Badge>}</span><span className="block text-[12.5px]">{r.p.name}</span><span className="block text-[11.5px] text-faint">{r.p.regulatoryCitation}</span></span> },
        { key: 'st', header: 'Stage', render: (r) => <span className="text-[12px] text-muted">{stageName(s, r.p)}<span className="block">{r.budgetKind ? `${r.budgetKind} budget` : 'no budget'}</span></span> },
        { key: 'a', header: 'Authorized', num: true, render: (r) => <span>{money(r.authorized)}{r.original !== r.authorized && <span className="block text-[11px] text-faint">orig. {money(r.original)}</span>}</span>, sort: (r) => r.authorized },
        { key: 'ry', header: `Recorded ${year}`, num: true, render: (r) => money(r.recordedYear) },
        { key: 'rc', header: 'Recorded cum.', num: true, render: (r) => money(r.recordedCum) },
        { key: 'fy', header: `Forecast ${year}`, num: true, render: (r) => money(r.forecastYear) },
        { key: 'e', header: 'EAC', num: true, render: (r) => money(r.eac) },
        { key: 'v', header: 'Variance', num: true, render: (r) => <span className={cx('font-semibold', r.eac > r.authorized ? 'text-bad' : 'text-good')}>{moneyDelta(r.eac - r.authorized)}<span className="block text-[11px] font-normal text-muted">{r.authorized ? `${(((r.eac - r.authorized) / r.authorized) * 100).toFixed(1)}%` : ''}</span></span>, sort: (r) => r.eac - r.authorized },
        { key: 'i', header: 'In service', render: (r) => <span className="text-[12px]">{fmtDate(r.ins?.actual ?? r.ins?.forecast)}{r.ins && !r.ins.actual && r.ins.forecast !== r.ins.baseline && <span className="block text-faint">baseline {fmtDate(r.ins.baseline)}</span>}</span> },
        { key: 'x', header: 'Explanation', render: (r) => <span className="block min-w-[220px] text-[12px] text-muted">{r.explanation}</span> },
      ]} />
    </Panel>
  )
}

// ───────────────────────────── RC-02 regulatory snapshot
function RegSnapshotModal({ open, onClose, initialProgram }: { open: boolean; onClose: () => void; initialProgram: string }) {
  const s = useStore()
  const [programId, setProgramId] = useState(initialProgram || s.programs[0]?.id)
  const progFunding = s.funding.filter((f) => f.programId === programId)
  const proceedings = Array.from(new Set(progFunding.map((f) => f.proceeding))).reverse()
  const [proceeding, setProceeding] = useState('')
  const [label, setLabel] = useState('')
  const [excluded, setExcluded] = useState<string[]>([])
  const proc = proceeding || proceedings[0] || ''
  const eligible = s.projects.filter((p) => p.programId === programId && p.status === 'active' && s.budgets.some((b) => b.projectId === p.id))
  const chosen = eligible.filter((p) => !excluded.includes(p.id))
  const submit = () => {
    if (!proc.trim()) { toast('Name the proceeding.', 'bad'); return }
    if (!chosen.length) { toast('Select at least one project.', 'bad'); return }
    const prog = s.programs.find((x) => x.id === programId)!
    const lab = label.trim() || `Regulatory snapshot — ${proc.trim()}`
    mutate((d) => { for (const p of chosen) takeSnapshot(d, p.id, 'regulatory', lab, { proceeding: proc.trim() }) }, { entity: 'snapshot', entityId: prog.id, action: 'create', summary: `Took ${proc.trim()} regulatory snapshot for ${chosen.length} ${prog.code} projects` })
    toast(`Froze ${chosen.length} project positions`)
    setExcluded([]); setLabel(''); onClose()
  }
  return (
    <Modal open={open} onClose={onClose} title="Take regulatory snapshot" wide footer={<><Button variant="ghost" onClick={onClose}>Cancel</Button><Button variant="primary" onClick={submit}>Freeze {chosen.length} project{chosen.length === 1 ? '' : 's'}</Button></>}>
      <div className="flex flex-col gap-3">
        <p className="text-[13px] text-muted">Freezes scope, schedule, budget by category, EAC, actuals and contingency for every active budgeted project in the program, tagged with the proceeding. Snapshots are immutable and appear on each project's baseline timeline.</p>
        <div className="grid grid-cols-1 gap-3 md:grid-cols-2">
          <Field label="Program"><ProgramSelect includeAll={false} value={programId} onChange={(v) => { setProgramId(v); setProceeding(''); setExcluded([]) }} /></Field>
          <Field label="Proceeding">
            <Input list="reg-proceedings" value={proc} onChange={(e) => setProceeding(e.target.value)} />
            <datalist id="reg-proceedings">{proceedings.map((x) => <option key={x} value={x} />)}</datalist>
          </Field>
        </div>
        <Field label="Label (optional)"><Input value={label} onChange={(e) => setLabel(e.target.value)} placeholder={`Regulatory snapshot — ${proc}`} /></Field>
        <div className="rounded-md border border-line">
          <div className="border-b border-line bg-sunken/50 px-3 py-1.5 text-xs font-semibold text-muted">{eligible.length} eligible projects</div>
          <div className="flex max-h-[260px] flex-col gap-1.5 overflow-y-auto p-3">
            {eligible.map((p) => { const pos = position(s, p.id); const last = snapshotsFor(s, p.id).filter((x) => x.type === 'regulatory').pop(); return <Checkbox key={p.id} checked={!excluded.includes(p.id)} onChange={(v) => setExcluded(v ? excluded.filter((x) => x !== p.id) : [...excluded, p.id])} label={<span><Mono className="font-semibold">{p.code}</Mono> {p.name} <span className="text-muted">· {money(pos.currentBudget)} · EAC {money(pos.eac)}{last ? ` · last regulatory ${fmtDate(last.date)}` : ''}</span></span>} /> })}

/** Where the project's contingency number comes from: the budget in force, else the latest live estimate. */
export interface ContingencyBasis { base: number; contingency: number; original: number; source: string }
export function contingencyBasis(s: AppState, pid: string): ContingencyBasis | null {
  const pos = position(s, pid)
  if (pos.budgetKind) return { base: pos.baseBudget, contingency: pos.contingencyRemaining, original: pos.contingencyOriginal, source: pos.budgetKind === 'control' ? 'Control budget' : 'Planning budget' }
  const est = s.estimates.filter((e) => e.projectId === pid && e.status !== 'rejected' && e.status !== 'superseded').sort((a, b) => b.version - a.version)[0]
  if (!est) return null
  const t = estimateTotals(est)
  return { base: t.base, contingency: t.contingency, original: t.contingency, source: `Class ${est.cls} estimate v${est.version} (${est.status.replace('-', ' ')})` }
}

/**
 * Run the engine Monte Carlo for a project. Base cost = budget (or estimate) excluding contingency;
 * schedule percentiles are risk-driven delay in days (base 0), applied to the in-service forecast.
 */
export function runMonteCarlo(s: AppState, pid: string, iterations = 5000, seed = 7): MonteCarloResult | null {
  const basis = contingencyBasis(s, pid)
  if (!basis) return null
  return monteCarlo(pid, s.risks.filter((r) => r.projectId === pid), basis.base, 0, s.settings.asOf, iterations, seed)
}

export type Verdict = 'adequate' | 'p50-only' | 'short' | 'no-risk'
export interface Adequacy { remaining: number; need50: number; need80: number; gap80: number; verdict: Verdict; coverage: number; label: string; tone: 'good' | 'warn' | 'bad' }
/** Contingency adequacy (RK-06): contingency remaining vs the P80 − base requirement from Monte Carlo. */
export function adequacy(remaining: number, mc: Pick<MonteCarloResult, 'costP' | 'baseCost'>): Adequacy {
  const need50 = Math.max(0, mc.costP.P50 - mc.baseCost)
  const need80 = Math.max(0, mc.costP.P80 - mc.baseCost)
  const gap80 = remaining - need80
  const coverage = need80 ? remaining / need80 : remaining > 0 ? 9.99 : 1
  if (need80 === 0) return { remaining, need50, need80, gap80, coverage, verdict: 'no-risk', label: 'No quantified risk to cover', tone: 'good' }
  if (remaining >= need80) return { remaining, need50, need80, gap80, coverage, verdict: 'adequate', label: 'Adequate at P80', tone: 'good' }
  if (remaining >= need50) return { remaining, need50, need80, gap80, coverage, verdict: 'p50-only', label: 'Covers P50 only', tone: 'warn' }
  return { remaining, need50, need80, gap80, coverage, verdict: 'short', label: 'Short of P50', tone: 'bad' }
}

/** Contingency burn-down (RK-08): remaining contingency each month vs a draw-down proportional to % complete. */
export interface BurnPoint { month: string; remaining: number; ideal: number; pctComplete: number }
export function burndown(s: AppState, pid: string): BurnPoint[] | null {
  const b = activeBudget(s, pid)
  if (!b || b.kind !== 'control') return null
  const pos = position(s, pid)
  const start = b.createdAt.slice(0, 7)
  const end = s.settings.asOf.slice(0, 7)
  if (start > end) return null
  const periods = s.costPeriods.filter((c) => c.projectId === pid)
  const snaps = s.snapshots.filter((x) => x.projectId === pid && x.date >= b.createdAt).sort((a, c) => a.date.localeCompare(c.date))
  const bac = Math.max(1, pos.baseBudget)
  let cumEv = sum(periods.filter((p) => p.period < start).map((p) => p.earned))
  return monthRange(start, end).map((m) => {
    cumEv += sum(periods.filter((p) => p.period === m).map((p) => p.earned))
    const pct = Math.min(1, cumEv / bac)
    const snap = [...snaps].reverse().find((x) => x.date.slice(0, 7) <= m)
    const remaining = m === end ? pos.contingencyRemaining : snap?.budget.contingencyRemaining ?? pos.contingencyOriginal
    return { month: m, remaining, ideal: Math.round(pos.contingencyOriginal * (1 - pct)), pctComplete: pct }
  })
}

export function exposureSummary(risks: Risk[], asOf: string) {
  const act = risks.filter(isActive)
  const threats = act.filter((r) => r.kind === 'threat')
  const opps = act.filter((r) => r.kind === 'opportunity')
  return {
    active: act.length,
    high: act.filter((r) => riskScore(r) >= 15).length,
    highNoResponse: act.filter((r) => riskScore(r) >= 15 && r.responses.length === 0).length,
    threat: Math.round(sum(threats.map(expectedCost))),
    opportunity: Math.round(-sum(opps.map(expectedCost))),
    net: Math.round(sum(act.map(expectedCost))),
    overdueResponses: act.flatMap((r) => r.responses).filter((x) => x.status === 'open' && x.due < asOf).length,
    scheduleDays: Math.round(sum(threats.map((r) => r.probability * r.scheduleDays.ml))),
  }
}

// Small workflow helpers shared by the risk and safety modules.
import type { AppState } from '@/core/types'
import { actOnWorkflow } from '@/core/engine/workflow'

/**
 * The person who fills in and submits a record has completed the process's first "form" step.
 * If the current user is an assignee on that step (or an administrator), record their submit.
 */
export function submitFirstForm(d: AppState, instId: string | undefined, comment: string) {
  if (!instId) return
  const inst = d.workflows.find((w) => w.id === instId)
  const slot = inst?.active[0]
  if (!inst || !slot) return
  const def = d.workflowDefs.find((x) => x.id === inst.defId)
  const node = def?.nodes.find((n) => n.id === slot.nodeId)
  if (node?.type !== 'form') return
  const me = d.people.find((p) => p.id === d.currentUserId)
  if (slot.assigneeIds.includes(d.currentUserId) || me?.roles.includes('admin')) {
    try { actOnWorkflow(d, inst.id, slot.nodeId, d.currentUserId, 'submit', comment) } catch { /* leave at the form step */ }
  }
}

const QKEY = 'crosswalk:field-queue:v1'
const PKEY = 'crosswalk:field-plan:v1'
interface Queued { tailboards: Tailboard[]; events: Omit<SafetyEvent, 'id'>[] }
const readQueue = (): Queued => { try { const r = prefs.getRaw(QKEY); if (r) return JSON.parse(r) } catch { /* storage blocked */ } return { tailboards: [], events: [] } }
const writeQueue = (q: Queued) => { try { prefs.setRaw(QKEY, JSON.stringify(q)) } catch { /* in-memory only */ } }

// Safety helpers (SF-*): ticket expiry, OQ coverage, leading indicators, printable JSSP.
import type { AppState, HazardItem, SafetyPlan } from '@/core/types'

export type Level = 'ok' | 'warn' | 'bad'
export const levelTone = (l: Level) => (l === 'bad' ? 'bad' : l === 'warn' ? 'warn' : 'good') as 'bad' | 'warn' | 'good'

/** Ticket status (SF-04): expired = bad, expiring within 3 days = warn. */
export function ticketLevel(expires: string, asOf: string): { level: Level; days: number; label: string } {
  const days = daysBetween(asOf, expires)
  if (days < 0) return { level: 'bad', days, label: `Expired ${-days} day${days === -1 ? '' : 's'} ago` }
  if (days === 0) return { level: 'warn', days, label: 'Expires today' }
  if (days <= 3) return { level: 'warn', days, label: `Expires in ${days} day${days === 1 ? '' : 's'}` }
  return { level: 'ok', days, label: `${days} days left` }
}

/** Crew member OQ status (SF-06). */
export function oqLevel(expires: string, asOf: string): { level: Level; label: string } {
  const d = daysBetween(asOf, expires)
  if (d < 0) return { level: 'bad', label: `OQ expired ${fmtDate(expires)}` }
  if (d <= 30) return { level: 'warn', label: `OQ expires ${fmtDate(expires)}` }
  return { level: 'ok', label: `OQ valid to ${fmtDate(expires)}` }
}

/** Operator-qualification records (an operator-qualification provider feed) are a phase-2 module, off for the pilot. */
export const opQualOn = (s: Pick<AppState, 'settings'>) => !!s.settings.features?.opQual

export const planHazards = (s: Pick<AppState, 'hazards'>, plan: Pick<SafetyPlan, 'hazardIds'>): HazardItem[] => plan.hazardIds.map((id) => s.hazards.find((h) => h.id === id)).filter((h): h is HazardItem => !!h)

/**
 * Operator-qualification coverage (SF-06): for each hazard that involves covered tasks, at least one crew
 * member must hold a current OQ for one of those tasks (e.g. Hot tapping OR Tapping & stopping).
 */
export function oqCoverage(s: Pick<AppState, 'hazards'>, plan: SafetyPlan, asOf: string) {
  const groups = new Map<string, string[]>()
  for (const h of planHazards(s, plan)) if (h.oqTasks.length) groups.set(h.oqTasks.join(' / '), h.oqTasks)
  return Array.from(groups.entries()).map(([task, tasks]) => {
    const holders = plan.crew.filter((c) => c.oqTasks.some((t) => tasks.includes(t)))
    const valid = holders.filter((c) => c.oqExpires >= asOf)
    return { task, tasks, holders, valid, level: (valid.length ? 'ok' : 'bad') as Level }
  })
}

export const requiredPermits = (s: Pick<AppState, 'hazards'>, plan: Pick<SafetyPlan, 'hazardIds'>) => Array.from(new Set(planHazards(s, plan).flatMap((h) => h.permits)))

/** Pre-mobilization checks a reviewer looks at before approving the plan. */
export function planIssues(s: AppState, plan: SafetyPlan): { level: Level; text: string }[] {
  const out: { level: Level; text: string }[] = []
  const asOf = s.settings.asOf
  for (const t of plan.tickets) {
    const st = ticketLevel(t.expires, asOf)
    if (st.level !== 'ok') out.push({ level: st.level, text: `${t.type} ${t.number}: ${st.label.toLowerCase()}` })
  }
  const have = new Set(plan.tickets.map((t) => t.type as string))
  for (const p of requiredPermits(s, plan)) {
    const covered = have.has(p) || (p === 'Excavation permit' && have.has('Encroachment permit')) || (p === 'Discharge permit' && s.permits.some((x) => x.projectId === plan.projectId && /discharge/i.test(x.type) && x.status === 'issued'))
    if (!covered) out.push({ level: p === '811 locate' ? 'bad' : 'warn', text: `${p} required by the hazards but not listed in tickets & permits` })
  }
  const oqOn = opQualOn(s)
  if (oqOn) for (const c of oqCoverage(s, plan, asOf)) if (!c.valid.length) out.push({ level: 'bad', text: `No crew member with a current OQ for “${c.task}”` })
  for (const c of plan.crew) {
    const oq = oqLevel(c.oqExpires, asOf)
    if (oqOn && oq.level === 'bad' && c.oqTasks.length) out.push({ level: 'bad', text: `${c.name}: ${oq.label} — may not perform ${c.oqTasks.join(', ')}` })
    if (c.prequal === 'Non-compliant') out.push({ level: 'bad', text: `${c.company} prequalification non-compliant (${c.name})` })
  }
  const prequal = Array.from(new Set(plan.crew.filter((c) => c.prequal === 'Conditional').map((c) => c.company)))
  for (const co of prequal) out.push({ level: 'warn', text: `${co}: ISNetworld-style prequalification is Conditional` })
  if (!plan.site.hospital || !plan.site.musterPoint) out.push({ level: 'bad', text: 'Emergency information incomplete (hospital or muster point)' })
  if (planHazards(s, plan).some((h) => h.activity === 'Traffic control') && !plan.trafficPlan) out.push({ level: 'warn', text: 'Traffic control activity without a traffic plan flag' })
  return out
}

export function workingDays(from: string, to: string): string[] {
  const out: string[] = []
  for (let d = from; d <= to; d = addDays(d, 1)) { const w = new Date(d + 'T12:00:00Z').getUTCDay(); if (w !== 0 && w !== 6) out.push(d) }
  return out
}

/** Leading indicators (SF-10) for a set of projects over the last `days` calendar days. */
export function indicators(s: AppState, projectIds: string[], days = 30, contractor?: string) {
  const asOf = s.settings.asOf
  const from = addDays(asOf, -days)
  const ids = new Set(projectIds)
  const plans = s.safetyPlans.filter((p) => ids.has(p.projectId) && p.status === 'approved' && (!contractor || p.contractor === contractor))
  // a project is "working" from its first tailboard in the window (or plan approval) until yesterday
  const recent = workingDays(addDays(asOf, -14), addDays(asOf, -1))
  let expected = 0, held = 0
  for (const pl of plans) {
    const tbs = s.tailboards.filter((t) => t.planId === pl.id || (t.projectId === pl.projectId && !plans.some((x) => x.id === t.planId)))
    const days0 = recent.filter((d) => !pl.approvedAt || d >= pl.approvedAt)
    expected += days0.length
    held += days0.filter((d) => tbs.some((t) => t.date === d)).length
  }
  const events = s.safetyEvents.filter((e) => ids.has(e.projectId) && (!contractor || e.contractor === contractor))
  const win = events.filter((e) => e.date >= from)
  const tickets = plans.flatMap((p) => p.tickets)
  const lapsed = tickets.filter((t) => t.expires < asOf).length
  const expiring = tickets.filter((t) => t.expires >= asOf && daysBetween(asOf, t.expires) <= 3).length
  const wd = Math.max(1, workingDays(from, addDays(asOf, -1)).length)
  return {
    plans: plans.length,
    tailboardRate: expected ? held / expected : null,
    tailboardsHeld: held, tailboardsExpected: expected,
    observations: win.filter((e) => e.type === 'observation').length,
    nearMisses: win.filter((e) => e.type === 'near-miss').length,
    incidents: win.filter((e) => e.type === 'incident').length,
    obsPerWeek: (win.filter((e) => e.type !== 'incident').length / wd) * 5 / Math.max(1, plans.length),
    lapsed, expiring,
    openEvents: events.filter((e) => e.status === 'open').length,
    sif: events.filter((e) => e.sifPotential && e.status === 'open').length,
    sifAll: win.filter((e) => e.sifPotential).length,
  }
}

          <div><Button size="sm" onClick={() => set('crew', [...d.crew, { id: uid('crw'), name: '', company: d.contractor, role: 'Laborer', oqTasks: [], oqExpires: addDays(asOf, 365), prequal: 'Compliant' }])}>Add crew member</Button></div>

}

function HazardDrawer({ hazard, onClose }: { hazard: HazardItem | null; onClose: () => void }) {
  const s = useStore()
  const [h, setH] = useState<HazardItem | null>(null)
  const [permits, setPermits] = useState('')
  const [oq, setOq] = useState('')
  useEffect(() => {
    setH(hazard ? JSON.parse(JSON.stringify(hazard)) : null)
    setPermits(hazard?.permits.join(', ') ?? '')
    setOq(hazard?.oqTasks.join(', ') ?? '')
  }, [hazard])
  if (!hazard || !h) return null
  const isNew = !s.hazards.some((x) => x.id === h.id)
  const used = s.safetyPlans.filter((x) => x.status !== 'superseded' && x.hazardIds.includes(h.id))
  const save = () => {
    if (!h.activity.trim() || !h.hazard.trim()) { toast('Activity and hazard are required.', 'bad'); return }
    const next: HazardItem = { ...h, controls: h.controls.filter((c) => c.text.trim()), permits: permits.split(',').map((x) => x.trim()).filter(Boolean), oqTasks: oq.split(',').map((x) => x.trim()).filter(Boolean) }
    if (!next.controls.length) { toast('Add at least one control.', 'bad'); return }

    onClose()
  }
  return (

        {!isNew && used.length > 0 && <Callout tone="info">Used by {used.length} live plan{used.length > 1 ? 's' : ''}. Approved plans keep the controls they were approved with; new and revised plans pick up this change from the effective date.</Callout>}

const RULES = `You are the portfolio assistant inside Crosswalk, a capital-project platform for a gas utility (Arroyo Valley Gas; all data is a fictional demo).
Answer ONLY from the tool results. Call tools as needed. Cite project codes like PRJ-1042. Use $M with one decimal for money. Keep answers short: a direct answer first, then up to 6 bullets. If the data cannot answer, say what is missing.`

function tools() {
  const s = getState()
  const row = (p: (typeof s.projects)[number]) => {
    const pos = position(s, p.id)
    return { code: p.code, name: p.name, program: s.programs.find((x) => x.id === p.programId)?.code, stage: stageName(s, p), status: p.status, pm: personName(s, p.team.pm), budget: money(pos.currentBudget), eac: money(pos.eac), actual: money(pos.actualToDate), contingencyLeft: money(pos.contingencyRemaining), riskExposure: money(pos.riskExposure), health: p.health }
  }
  return [
    {
      name: 'list_projects', description: 'List projects with stage, budget, forecast (EAC), actuals, contingency, risk exposure and RAG health. Optional filters: program code (PSE, TIM, DRP, CSM, VAP), status, health dimension "cost"|"schedule"|"risk" equal to "red"|"amber".',
      inputSchema: { type: 'object', properties: { program: { type: 'string' }, status: { type: 'string' }, healthDim: { type: 'string' }, healthIs: { type: 'string' } } },
      execute: (i: Record<string, unknown>) => s.projects.filter((p) => (!i.program || s.programs.find((x) => x.id === p.programId)?.code === String(i.program).toUpperCase()) && (!i.status || p.status === i.status) && (!i.healthDim || (p.health as any)[String(i.healthDim)] === i.healthIs)).map(row),
    },
    {
      name: 'get_project', description: 'Full detail for one project by code (e.g. PRJ-1042): scope, milestones, gate readiness, top risks, changes, variances, safety summary.',
      inputSchema: { type: 'object', properties: { code: { type: 'string' } }, required: ['code'] },
      execute: (i: Record<string, unknown>) => {
        const p = s.projects.find((x) => x.code.toLowerCase() === String(i.code).toLowerCase())
        if (!p) throw new Error('No project with that code')
        const r = readiness(s, p)
        return {
          ...row(p), scope: p.scope.summary, milestones: p.milestones.map((m) => ({ name: m.name, baseline: m.baseline, forecast: m.forecast, actual: m.actual })),
          gateReadiness: { pct: r.pct, blockers: r.blockers },
          topRisks: s.risks.filter((x) => x.projectId === p.id && isActive(x)).sort((a, b) => riskScore(b) - riskScore(a)).slice(0, 6).map((x) => ({ code: x.code, title: x.title, score: riskScore(x), ml: money(x.cost.ml), status: x.status })),
          changes: s.changes.filter((c) => c.projectId === p.id).map((c) => ({ code: c.code, title: c.title, cost: money(c.costImpact), status: c.status, rootCause: c.rootCause })),
          variances: s.variances.filter((v) => v.projectId === p.id).map((v) => ({ description: v.description, status: v.status, narrative: v.narrative })),
          safety: { plans: s.safetyPlans.filter((x) => x.projectId === p.id).map((x) => `${x.workPackage} v${x.version} ${x.status}`), openEvents: s.safetyEvents.filter((e) => e.projectId === p.id && e.status === 'open').length },
        }
      },
    },
    {
      name: 'portfolio_totals', description: 'Totals by program: count of projects, budget, EAC, actual to date, and authorized funding for the current year.',
      execute: () => s.programs.map((pr) => {

export function AskPanel({ open, onClose }: { open: boolean; onClose: () => void }) {

  const [turns, setTurns] = useState<{ role: 'user' | 'assistant'; content: string }[]>([])

    if (!text.trim() || busy) return
    const next = [...turns, { role: 'user' as const, content: text }]
    setTurns(next); setQ(''); setBusy(true); setDraft('Thinking…')
    ctl.current = new AbortController()

      setTurns([...next, { role: 'assistant', content: aiMessage(e) }])

  }
  const linkify = (t: string) => t.split(/(PRJ-\d{4})/g).map((part, i) => (/^PRJ-\d{4}$/.test(part) ? <a key={i} className="font-mono font-semibold text-link underline-offset-2 hover:underline" href={projectHref(part)} onClick={onClose}>{part}</a> : <span key={i}>{part}</span>))

/** While the data is shared, refuse file pickers and dropped files anywhere in the app. */
export function useUploadGuard() {
  const locked = useUploadsLocked()
  useEffect(() => {
    if (!locked) return
    const msg = 'File uploads are off on the shared public demo (fictional data only).'
    const onClick = (e: MouseEvent) => {
      const t = e.target as HTMLElement | null
      const input = t?.closest?.('input[type="file"]') ?? (t instanceof HTMLLabelElement ? (t.control as HTMLElement | null) : t?.closest?.('label')?.querySelector?.('input[type="file"]'))
      if (input && (input as HTMLInputElement).type === 'file') { e.preventDefault(); e.stopPropagation(); toast(msg, 'info') }
    }
    const onDrop = (e: DragEvent) => { if (e.dataTransfer?.files?.length) { e.preventDefault(); e.stopPropagation(); toast(msg, 'info') } }
    const onDragOver = (e: DragEvent) => { if (e.dataTransfer?.types?.includes('Files')) e.preventDefault() }
    document.addEventListener('click', onClick, true)
    document.addEventListener('drop', onDrop, true)
    document.addEventListener('dragover', onDragOver, true)
    return () => { document.removeEventListener('click', onClick, true); document.removeEventListener('drop', onDrop, true); document.removeEventListener('dragover', onDragOver, true) }
  }, [locked])
}

export function wrap(text: string, max: number, lines = 2): string[] {
  const words = text.split(/\s+/)
  const out: string[] = []
  let cur = ''
  for (const w of words) {
    if ((cur + ' ' + w).trim().length > max && cur) { out.push(cur); cur = w } else cur = (cur + ' ' + w).trim()
  }
  if (cur) out.push(cur)
  if (out.length > lines) { const keep = out.slice(0, lines); keep[lines - 1] = keep[lines - 1].slice(0, max - 1) + '…'; return keep }
  return out.map((l) => (l.length > max + 2 ? l.slice(0, max) + '…' : l))
}

export interface EdgeGeom { d: string; mid: { x: number; y: number }; back: boolean }
export function edgeGeom(def: WorkflowDef, e: WfEdge): EdgeGeom | null {
  const S = def.nodes.find((n) => n.id === e.from), T = def.nodes.find((n) => n.id === e.to)
  if (!S || !T) return null
  return geomBetween(S, T, e.outcome === 'return' || e.outcome === 'reject')
}
export function geomBetween(S: Pick<WfNode, 'x' | 'y' | 'type'>, T: Pick<WfNode, 'x' | 'y' | 'type'>, isReturn: boolean): EdgeGeom {
  const s = nodeSize(S.type), t = nodeSize(T.type)
  const dx = T.x - S.x, dy = T.y - S.y
  let p0: [number, number], p1: [number, number], p2: [number, number], p3: [number, number]
  const back = isReturn || dx < -10
  if (back) {
    const depth = 46 + Math.abs(dx) * 0.12 + Math.max(0, Math.abs(dy) * 0.2)
    const base = Math.max(S.y + s.h / 2, T.y + t.h / 2)
    p0 = [S.x - 8, S.y + s.h / 2]
    p3 = [T.x + 8, T.y + t.h / 2]
    p1 = [p0[0], base + depth]
    p2 = [p3[0], base + depth]
  } else if (dx > (s.w + t.w) / 2 * 0.7) {
    p0 = [S.x + s.w / 2, S.y]
    p3 = [T.x - t.w / 2, T.y]
    const c = Math.max(28, (p3[0] - p0[0]) * 0.45)
    p1 = [p0[0] + c, p0[1]]
    p2 = [p3[0] - c, p3[1]]
  } else {
    const down = dy >= 0
    p0 = [S.x, S.y + (down ? s.h / 2 : -s.h / 2)]
    p3 = [T.x, T.y + (down ? -t.h / 2 : t.h / 2)]
    const c = Math.max(24, Math.abs(p3[1] - p0[1]) * 0.45) * (down ? 1 : -1)
    p1 = [p0[0], p0[1] + c]
    p2 = [p3[0], p3[1] - c]
  }
  const mid = { x: 0.125 * p0[0] + 0.375 * p1[0] + 0.375 * p2[0] + 0.125 * p3[0], y: 0.125 * p0[1] + 0.375 * p1[1] + 0.375 * p2[1] + 0.125 * p3[1] }
  const f = (p: [number, number]) => `${p[0].toFixed(1)} ${p[1].toFixed(1)}`
  return { d: `M${f(p0)} C${f(p1)} ${f(p2)} ${f(p3)}`, mid, back }
}

        const on = edgeOn(e)
        return (
          <g key={e.id} opacity={on ? 1 : 0.22} onPointerDown={(ev) => { ev.stopPropagation(); onSelect?.({ kind: 'edge', id: e.id }) }} style={{ cursor: readOnly ? 'default' : 'pointer' }}>
            <path d={g.d} fill="none" stroke="transparent" strokeWidth={14} />
            <path d={g.d} fill="none" stroke={tok(c)} strokeWidth={sel ? 2.5 : highlight && on ? 2.2 : 1.6} strokeDasharray={g.back || e.outcome === 'return' || e.outcome === 'reject' ? '6 4' : undefined} markerEnd={`url(#wf-arrow-${c})`} />
            {label && (
              <g transform={`translate(${g.mid.x},${g.mid.y})`}>
                <rect x={-(Math.min(label.length, 26) * 3.1 + 7)} y={-9} width={Math.min(label.length, 26) * 6.2 + 14} height={18} rx={9} fill={tok('surface')} stroke={tok(sel ? 'link' : e.outcome === 'return' ? 'warn' : e.outcome === 'reject' ? 'bad' : 'line')} />
                <text textAnchor="middle" y={3.5} fontSize={10.5} fontWeight={600} fill={tok(e.outcome === 'return' ? 'warn' : e.outcome === 'reject' ? 'bad' : 'muted')}>{trunc(label, 26)}</text>
              </g>
            )}
          </g>
        )
      })}
      {/* in-progress connection */}
      {link && linkFrom && <path d={`M${linkFrom.x + nodeSize(linkFrom.type).w / 2} ${linkFrom.y} L${link.x} ${link.y}`} stroke={tok('link')} strokeWidth={2} strokeDasharray="4 3" markerEnd="url(#wf-arrow-link)" fill="none" />}
      {/* nodes */}
      {def.nodes.map((n) => (
        <g key={n.id} opacity={dim(n.id)} transform={`translate(${n.x},${n.y})`} onPointerDown={(e) => onNodeDown(e, n)} style={{ cursor: readOnly ? 'pointer' : connectMode ? 'crosshair' : 'grab' }}>
          <NodeShape s={s} def={def} n={n} selected={selected?.kind === 'node' && selected.id === n.id} issue={issues?.has(n.id)} pending={pending === n.id} target={hoverNode === n.id} active={active?.has(n.id)} done={done?.has(n.id)} />
          {!readOnly && selected?.kind === 'node' && selected.id === n.id && n.type !== 'end' && (
            <g transform={`translate(${nodeSize(n.type).w / 2 + 12},0)`} onPointerDown={(e) => onHandleDown(e, n)} style={{ cursor: 'crosshair' }}>
              <title>Drag to another step to connect</title>
              <circle r={8} fill={tok('link')} stroke={tok('surface')} strokeWidth={2} />
              <path d="M-3 0 H3 M0 -3 V3" stroke={tok('surface')} strokeWidth={1.6} />
            </g>
          )}
        </g>
      ))}
    </svg>
  )
}

export function NodeShape({ s, def, n, selected, issue, pending, target, active, done }: { s: AppState; def: WorkflowDef; n: WfNode; selected?: boolean; issue?: boolean; pending?: boolean; target?: boolean; active?: boolean; done?: boolean }) {
  const z = nodeSize(n.type)
  const col = TYPE_COLOR[n.type]
  const ring = selected || pending || target ? tok('link') : issue ? tok('bad') : active ? tok('marker') : tok('line')
  const rw = selected || pending || target || issue || active ? 2.2 : 1.2
  const label = (y: number, max: number, lines = 2) => wrap(n.label, max, lines).map((l, i) => <text key={i} x={0} y={y + i * 13} textAnchor="middle" fontSize={11} fontWeight={600} fill={tok('ink')}>{l}</text>)
  const status = active ? <circle cx={z.w / 2 - 2} cy={-z.h / 2 + 2} r={6} fill={tok('marker')} stroke={tok('surface')} strokeWidth={2} /> : done ? <g transform={`translate(${z.w / 2 - 2},${-z.h / 2 + 2})`}><circle r={7} fill={tok('good')} stroke={tok('surface')} strokeWidth={2} /><path d="M-3 0 L-1 2.5 L3 -2.5" stroke={tok('surface')} strokeWidth={1.6} fill="none" /></g> : null
  const issueDot = issue ? <g transform={`translate(${-z.w / 2 + 2},${-z.h / 2 + 2})`}><circle r={7} fill={tok('bad')} /><text y={3.5} textAnchor="middle" fontSize={10} fontWeight={700} fill={tok('surface')}>!</text></g> : null
  switch (n.type) {
    case 'start':
    case 'end':
      return (
        <g>
          <circle r={18} fill={n.type === 'start' ? tok('good', 0.14) : tok('surface')} stroke={selected ? ring : n.type === 'start' ? tok('good') : tok('ink')} strokeWidth={n.type === 'end' ? 3 : rw + 0.6} />
          {n.type === 'end' && <circle r={11} fill={tok('ink')} />}
          {n.type === 'start' && <path d="M-4 -6 L6 0 L-4 6 z" fill={tok('good')} />}
          {label(34, 16, 1)}
          {issueDot}
        </g>
      )
    case 'condition':
      return (
        <g>
          <polygon points={`0,${-z.h / 2} ${z.w / 2},0 0,${z.h / 2} ${-z.w / 2},0`} fill={tok('warn', 0.1)} stroke={selected || pending || target ? ring : issue ? tok('bad') : tok('warn')} strokeWidth={rw + 0.3} />
          <text y={5} textAnchor="middle" fontSize={15} fontWeight={700} fill={tok('warn')}>?</text>
          {label(z.h / 2 + 14, 24)}
          {issueDot}
        </g>
      )
    case 'parallel-split':
    case 'parallel-join':
      return (
        <g>
          <rect x={-z.w / 2} y={-z.h / 2} width={z.w} height={z.h} rx={2} fill={tok('ink')} stroke={selected || pending || target ? ring : 'none'} strokeWidth={3} />
          {label(z.h / 2 + 14, 18)}
          {issueDot}
        </g>
      )
    case 'notify':
    case 'system': {
      const sub = n.type === 'system' ? 'SYSTEM' : `NOTIFY · ${describeAssignee(s, n.assignee, def.fields)}`
      return (
        <g>
          <rect x={-z.w / 2} y={-z.h / 2} width={z.w} height={z.h} rx={z.h / 2} fill={n.type === 'system' ? tok('sunken') : tok('info', 0.08)} stroke={ring} strokeWidth={rw} strokeDasharray={n.type === 'system' ? '4 3' : undefined} />
          <text x={0} y={-8} textAnchor="middle" fontSize={8.5} fontWeight={700} letterSpacing={0.6} fill={tok('faint')}>{trunc(sub, 26)}</text>
          {wrap(n.label, 22, 2).map((l, i, arr) => <text key={i} x={0} y={arr.length > 1 ? 6 + i * 12 : 10} textAnchor="middle" fontSize={arr.length > 1 ? 10.5 : 11} fontWeight={600} fill={tok('ink')}>{l}</text>)}
          {status}{issueDot}
        </g>
      )
    }
    default: {
      const lane = laneFor(s, def, n)
      const lines = wrap(n.label, 22)
      return (
        <g>
          <rect x={-z.w / 2} y={-z.h / 2} width={z.w} height={z.h} rx={5} fill={tok('surface')} stroke={ring} strokeWidth={rw} />
          <rect x={-z.w / 2} y={-z.h / 2} width={4} height={z.h} rx={2} fill={tok(col)} />
          <text x={-z.w / 2 + 11} y={-z.h / 2 + 13} fontSize={8.5} fontWeight={700} letterSpacing={0.6} fill={tok(n.type === 'approval' ? 'warn' : col)}>{nodeTypeLabel(n.type).toUpperCase()}{n.rule === 'all' ? ' · ALL' : ''}</text>
          {n.slaDays ? <text x={z.w / 2 - 8} y={-z.h / 2 + 13} textAnchor="end" fontSize={9} fontWeight={600} fill={tok('muted')}>{n.slaDays}d SLA</text> : null}
          {lines.map((l, i) => <text key={i} x={-z.w / 2 + 11} y={-z.h / 2 + 29 + i * 13} fontSize={11.5} fontWeight={600} fill={tok('ink')}>{l}</text>)}
          <text x={-z.w / 2 + 11} y={z.h / 2 - 7} fontSize={10} fill={tok(n.assignee ? 'muted' : 'bad')}>{trunc(n.assignee ? lane : 'No assignee', 25)}</text>
          {n.type === 'approval' && <path d={`M${z.w / 2 - 16} ${z.h / 2 - 16} l3 3 l6 -6`} stroke={tok('warn')} strokeWidth={1.6} fill="none" />}
          {status}{issueDot}
        </g>
      )
    }
  }
}

export const conditionText = (e: WfEdge) => (e.condition ? describeCondition(e.condition) : 'Default (no rule matched)')
