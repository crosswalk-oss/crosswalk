// Process catalog (WF-01, WF-03, WF-12): every business process with its approvers and routing, the DOA matrix, and process analytics.

import type { AppState, DoaMatrix, RecordType, RoleId, WorkflowDef } from '@/core/types'

type Tab = 'catalog' | 'doa' | 'analytics'

export default function Catalog(_props: { projectId?: string; id?: string }) {
  const s = useStore()
  const [tab, setTab] = useState<Tab>('catalog')
  const [newOpen, setNewOpen] = useState(false)
  const keys = useMemo(() => Array.from(new Set(s.workflowDefs.map((d) => d.key))), [s.workflowDefs])
  const stats = useMemo(() => Object.fromEntries(keys.map((k) => [k, processStats(s, k)])) as Record<string, ProcessStat>, [s.workflows, s.workflowDefs, keys]) // eslint-disable-line react-hooks/exhaustive-deps
  const current = (k: string) => { const v = versionsOf(s, k); return v.find((d) => d.status === 'published') ?? v[0] }
  const published = keys.map(current).filter((d) => d.status === 'published')
  const running = s.workflows.filter((w) => w.status === 'running')
  const overdue = running.reduce((a, w) => a + w.active.filter((x) => x.due && x.due < s.settings.asOf).length, 0)
  const year = s.workflows.filter((w) => w.completedAt && w.completedAt.slice(0, 10) >= `${Number(s.settings.asOf.slice(0, 4)) - 1}${s.settings.asOf.slice(4)}`)
  const allCycles = Object.values(stats).flatMap((x) => x.cycles)
  const pendingNotices = useMemo(() => planReminders(s).length, [s.workflows, s.settings.workflow]) // eslint-disable-line react-hooks/exhaustive-deps
  const manual = async () => {
    const r = await saveFile(`process-manual-${s.settings.asOf}.html`, manualHtml(getState(), published), 'text/html')
    toast(r === 'saved' ? `Process manual downloaded — ${published.length} SOPs` : r === 'declined' ? 'Download cancelled' : 'Downloads are not available here', r === 'saved' ? 'good' : 'info')
  }
  return (
    <div className="flex flex-col gap-4">
      <PageHeader eyebrow="Configure" title="Workflows"
        subtitle="Every business process documented with its approvers, routing rules and approval authority. Edit a process to change how work routes — the SOP regenerates from the same definition."
        actions={<><Button icon={<Icon name="bell" className="h-4 w-4" />} onClick={sendRemindersNow} title={`Reminders ${s.settings.workflow.reminderDays} day(s) before due; escalation to the manager ${s.settings.workflow.escalateAfterDays ? `${s.settings.workflow.escalateAfterDays} day(s) overdue` : 'off'}`}>Send reminders now{pendingNotices > 0 && <span className="rounded-full bg-marker px-1.5 text-[11px] font-bold tabular-nums text-markerink">{pendingNotices}</span>}</Button><Button icon={<Icon name="doc" className="h-4 w-4" />} onClick={manual}>Download process manual</Button><Button variant="primary" icon={<Icon name="plus" className="h-4 w-4" />} onClick={() => setNewOpen(true)}>New process</Button></>} />
      <div className="grid grid-cols-2 gap-3 md:grid-cols-5">
        <Stat label="Published processes" value={published.length} sub={`${keys.length - published.length ? `${keys.length - published.length} in draft · ` : ''}${s.workflowDefs.filter((d) => d.status === 'retired').length} retired versions`} />
        <Stat label="Requests in flight" value={running.length} sub={`${new Set(running.map((w) => w.projectId).filter(Boolean)).size} projects`} />
        <Stat label="Completed (12 mo)" value={year.length} sub={`${year.filter((w) => w.status === 'approved').length} approved`} />
        <Stat label="Median cycle time" value={fmtDur(allCycles.length ? allCycles.slice().sort((a, b) => a - b)[Math.floor(allCycles.length / 2)] : undefined)} sub="start to completion" />
        <Stat label="Overdue steps" value={overdue} tone={overdue ? 'bad' : 'good'} sub={<>past their SLA · <a href="#admin-notify" className="text-link">reminder rules</a></>} />
      </div>
      <Tabs<Tab> value={tab} onChange={setTab} tabs={[{ id: 'catalog', label: 'Process catalog', count: keys.length }, { id: 'doa', label: 'Delegation of authority' }, { id: 'analytics', label: 'Process analytics' }]} />
      {tab === 'catalog' && <CatalogTable s={s} keys={keys} stats={stats} current={current} />}
      {tab === 'doa' && <DoaEditor s={s} />}
      {tab === 'analytics' && <Analytics s={s} keys={keys} stats={stats} />}
      <NewProcessModal open={newOpen} onClose={() => setNewOpen(false)} s={s} keys={keys} current={current} />
    </div>
  )
}

function approversSummary(s: AppState, d: WorkflowDef): string {
  const out: string[] = []
  for (const n of d.nodes.filter((x) => x.type === 'approval')) { const t = describeAssignee(s, n.assignee, d.fields); if (!out.includes(t)) out.push(t) }
  return out.join(' → ') || '—'
}

// ───────────────────────────── New process
function NewProcessModal({ open, onClose, s, keys, current }: { open: boolean; onClose: () => void; s: AppState; keys: string[]; current: (k: string) => WorkflowDef }) {
  const [name, setName] = useState('')
  const [key, setKey] = useState('')
  const [keyTouched, setKeyTouched] = useState(false)
  const [from, setFrom] = useState<string>('')
  const [rt, setRt] = useState<RecordType>('change')
  const [owner, setOwner] = useState<RoleId>('pmo')
  const k = keyTouched ? key : slugify(name)
  const clash = keys.includes(k)
  const create = () => {
    if (!name.trim() || !k || clash) return
    const src = from ? current(from) : undefined
    const def: WorkflowDef = src
      ? { ...JSON.parse(JSON.stringify(src)), id: `wfd-${k}`, key: k, name: name.trim(), version: 1, status: 'draft', effective: s.settings.asOf }
      : blankDef(k, name.trim(), rt, owner, s.settings.asOf)
    if (!src) { def.description = `Routes ${recordTypeLabel(rt).toLowerCase()} records for review and approval.`; def.purpose = '' }
    mutate((d) => { d.workflowDefs.push(def) }, { entity: 'workflow-def', entityId: k, action: 'create', summary: `Created process ${def.name} (${k}) as draft v1${src ? ` from ${src.name}` : ''}` })
    toast(`Draft process created — design it, then publish`)
    onClose()
    setName(''); setKey(''); setKeyTouched(false); setFrom('')
    go(workflowHref(k))
  }
  return (
    <Modal open={open} onClose={onClose} title="New process" footer={<><Button variant="ghost" onClick={onClose}>Cancel</Button><Button variant="primary" disabled={!name.trim() || !k || clash} onClick={create}>Create draft</Button></>}>
      <div className="flex flex-col gap-3">
        <Field label="Process name"><Input id="new-wf-name" value={name} onChange={(e) => setName(e.target.value)} placeholder="e.g. Emergency work authorization" /></Field>
        <Field label="Process key" hint={clash ? <span className="text-bad">That key is already used.</span> : 'Stable identifier modules use to start this process. Lowercase and hyphens.'}>
          <Input className="font-mono" value={k} onChange={(e) => { setKeyTouched(true); setKey(slugify(e.target.value)) }} />
        </Field>
        <Field label="Start from">
          <Select value={from} onChange={(e) => setFrom(e.target.value)} options={[{ value: '', label: 'Blank process (request → manager review → dollar-authority approval)' }, ...keys.map((x) => ({ value: x, label: `Duplicate: ${current(x).name} v${current(x).version}` }))]} />
        </Field>
        {!from && (
          <div className="grid grid-cols-2 gap-3">
            <Field label="Record type"><Select value={rt} onChange={(e) => setRt(e.target.value as RecordType)} options={RECORD_TYPES.map((r) => ({ value: r.id, label: r.label }))} /></Field>
            <Field label="Process owner"><Select value={owner} onChange={(e) => setOwner(e.target.value as RoleId)} options={s.roles.map((r) => ({ value: r.id, label: r.name }))} /></Field>
          </div>
        )}
        <Callout>New processes start as a draft. Nothing routes on them until you publish v1.</Callout>
      </div>
    </Modal>
  )
}

// ───────────────────────────── Delegation of authority (WF-03)
function DoaEditor({ s }: { s: AppState }) {
  const [mid, setMid] = useState(s.doaMatrices[0]?.id)
  const m = s.doaMatrices.find((x) => x.id === mid) ?? s.doaMatrices[0]
  const me = s.people.find((p) => p.id === s.currentUserId)!
  const canEdit = me.roles.includes('finance') || me.roles.includes('admin')
  const [rows, setRows] = useState<DoaMatrix['rows'] | null>(null)
  const [effective, setEffective] = useState(s.settings.asOf)
  const [note, setNote] = useState('')
  const [newOpen, setNewOpen] = useState(false)
  const [newName, setNewName] = useState('')
  if (!m) return <Empty title="No DOA matrix">Create one to route approvals by dollar amount.</Empty>
  const work = rows ?? m.rows
  const sorted = work.slice().sort((a, b) => a.upTo - b.upTo)
  const dirty = !!rows && JSON.stringify(rows) !== JSON.stringify(m.rows)
  const programs = s.programs.filter((p) => p.doaMatrixId === m.id)
  const history = s.audit.filter((a) => a.entity === 'doa' && a.entityId === m.id).slice().reverse()
  const doaProcesses = Array.from(new Set(s.workflowDefs.filter((d) => d.status === 'published' && d.nodes.some((n) => n.assignee?.kind === 'doa')).map((d) => d.name)))
  // impact on in-flight requests that will reach (or sit at) a DOA step
  const impact = s.workflows.filter((w) => w.status === 'running').map((w) => {
    const def = s.workflowDefs.find((d) => d.id === w.defId)
    const doaNode = def?.nodes.find((n) => n.assignee?.kind === 'doa')
    if (!doaNode || doaNode.assignee?.kind !== 'doa') return null
    const amt = Math.abs(Number(w.data[doaNode.assignee.field] ?? 0))
    const before = doaRoleFor(s, w, amt)
    const after = (sorted.find((r) => amt <= r.upTo) ?? sorted[sorted.length - 1])?.roleId
    const at = w.active.some((a) => a.nodeId === doaNode.id)
    return { w, amt, before, after, at }
  }).filter(Boolean) as { w: (typeof s.workflows)[number]; amt: number; before?: RoleId; after?: RoleId; at: boolean }[]
  const changed = impact.filter((x) => x.before !== x.after)
  const set = (i: number, patch: Partial<DoaMatrix['rows'][number]>) => setRows(sorted.map((r, j) => (j === i ? { ...r, ...patch } : r)))
  const issues: string[] = []
  if (!sorted.length) issues.push('Add at least one tier.')
  if (sorted.length && sorted[sorted.length - 1].upTo < 1e12) issues.push('The last tier should have no upper limit so every amount has an approver.')
  if (new Set(sorted.map((r) => r.upTo)).size !== sorted.length) issues.push('Two tiers have the same limit.')
  const publish = () => {
    const nv = m.version + 1
    const summary = `Published ${m.name} v${nv} effective ${effective}: ${sorted.map((r) => `${r.upTo >= 1e12 ? 'above' : '≤' + money(r.upTo)} ${r.label}`).join(', ')}${note.trim() ? ` — ${note.trim()}` : ''}`
    mutate((d) => { const x = d.doaMatrices.find((y) => y.id === m.id)!; x.rows = sorted; x.version = nv; x.effective = effective }, { entity: 'doa', entityId: m.id, action: 'publish', summary })
    setRows(null); setNote('')
    toast(`${m.name} v${nv} published — new dollar-authority steps route on it`)
  }
  const createMatrix = () => {
    const id = `doa-${slugify(newName) || 'matrix'}`
    if (s.doaMatrices.some((x) => x.id === id)) { toast('A matrix with that name exists', 'bad'); return }
    mutate((d) => { d.doaMatrices.push({ id, name: newName.trim(), version: 1, effective: d.settings.asOf, rows: JSON.parse(JSON.stringify(m.rows)) }) }, { entity: 'doa', entityId: id, action: 'create', summary: `Created ${newName.trim()} v1 from ${m.name} v${m.version}` })
    setMid(id); setNewOpen(false); setNewName('')
    toast('Matrix created — assign it to programs in Configuration')
  }
  return (
    <div className="grid grid-cols-1 gap-4 xl:grid-cols-[minmax(0,1fr)_380px]">
      <div className="flex min-w-0 flex-col gap-4">
        <Panel title={<span className="flex flex-wrap items-center gap-2">{s.doaMatrices.length > 1 ? <Select aria-label="Matrix" className="h-7 w-auto py-0.5" value={m.id} onChange={(e) => { setMid(e.target.value); setRows(null) }} options={s.doaMatrices.map((x) => ({ value: x.id, label: x.name }))} /> : m.name}<Badge tone="good">v{m.version}</Badge><span className="text-[12px] font-normal text-muted">effective {fmtDate(m.effective)}</span></span>}
          subtitle={`Owned by Capital Finance. Used by ${programs.length ? programs.map((p) => p.code).join(', ') : 'no programs'}. A dollar-authority step routes to the first tier whose limit covers the amount.`}
          actions={canEdit ? <><Button size="sm" variant="ghost" onClick={() => setNewOpen(true)}>New matrix</Button><Button size="sm" onClick={() => setRows([...sorted.slice(0, -1), { upTo: Math.max(1000, (sorted[sorted.length - 2]?.upTo ?? 0) * 2), roleId: 'exec', label: 'New tier' }, ...sorted.slice(-1)])}>Add tier</Button></> : undefined} pad={false}>
          {!canEdit && <div className="px-4 pt-3"><Callout tone="info">Only Capital Finance or an administrator can change approval authority. You are viewing the matrix in force.</Callout></div>}
          <div className="scroll-x">
            <table className="tbl">
              <thead><tr><th className="num">From</th><th className="num">Up to</th><th>Approver role</th><th>Label</th><th>People holding the role</th><th /></tr></thead>
              <tbody>
                {sorted.map((r, i) => {
                  const holders = s.people.filter((p) => p.roles.includes(r.roleId) && !p.company)
                  const last = i === sorted.length - 1
                  return (
                    <tr key={i}>
                      <td className="num text-muted">{i === 0 ? '$0' : money(sorted[i - 1].upTo + 1)}</td>
                      <td className="num w-[170px]">{last && r.upTo >= 1e12 ? <span className="text-muted">No limit</span> : canEdit ? <MoneyInput aria-label="Up to" className="h-7 py-0.5" value={r.upTo} onChange={(v) => set(i, { upTo: v })} /> : money(r.upTo, 'full')}</td>
                      <td className="min-w-[170px]">{canEdit ? <Select aria-label="Role" className="h-7 py-0.5" value={r.roleId} onChange={(e) => set(i, { roleId: e.target.value as RoleId })} options={s.roles.map((x) => ({ value: x.id, label: x.name }))} /> : s.roles.find((x) => x.id === r.roleId)?.name}</td>
                      <td className="min-w-[150px]">{canEdit ? <Input aria-label="Label" className="h-7 py-0.5" value={r.label} onChange={(e) => set(i, { label: e.target.value })} /> : r.label}</td>
                      <td className="text-[12px] text-muted">{holders.length ? holders.map((p) => p.name).join(', ') : <span className="text-bad">Nobody holds this role</span>}</td>
                      <td>{canEdit && !last && <button className="text-[12px] font-semibold text-bad hover:underline" onClick={() => setRows(sorted.filter((_, j) => j !== i))}>Remove</button>}</td>
                    </tr>
                  )
                })}
              </tbody>
            </table>
          </div>
          <DoaBand rows={sorted} />
        </Panel>
        {dirty && (
          <Panel title={`Publish v${m.version + 1}`} subtitle="Publishing replaces the matrix in force from its effective date. Steps already assigned keep their approver; new dollar-authority steps route on the new tiers.">
            <div className="flex flex-col gap-3">
              {issues.map((x) => <Callout key={x} tone="bad">{x}</Callout>)}
              {changed.length > 0 ? (
                <Callout tone="warn" title={`${changed.length} in-flight request${changed.length > 1 ? 's' : ''} would route to a different tier`}>
                  <ul className="mt-1 list-disc pl-4">{changed.slice(0, 6).map((x) => <li key={x.w.id}>{x.w.title} ({money(x.amt)}): {s.roles.find((r) => r.id === x.before)?.name} → {s.roles.find((r) => r.id === x.after)?.name}{x.at ? ' — already at the approval step, keeps its current approver' : ''}</li>)}</ul>
                </Callout>
              ) : <Callout tone="good">No in-flight request changes tier.</Callout>}
              <div className="grid grid-cols-1 gap-3 sm:grid-cols-[180px_minmax(0,1fr)]">
                <Field label="Effective date"><Input type="date" value={effective} onChange={(e) => setEffective(e.target.value)} /></Field>
                <Field label="Reason for change"><TextArea className="min-h-[38px]" value={note} onChange={(e) => setNote(e.target.value)} placeholder="e.g. Board resolution 2026-14 raised director authority" /></Field>
              </div>
              <div className="flex justify-end gap-2"><Button variant="ghost" onClick={() => setRows(null)}>Discard</Button><Button variant="primary" disabled={!!issues.length} onClick={publish}>Publish v{m.version + 1}</Button></div>
            </div>
          </Panel>
        )}
      </div>
      <div className="flex flex-col gap-4">
        <Panel title="Where it applies">
          <div className="flex flex-col gap-2 text-[13px]">
            <div><div className="eyebrow mb-1">Programs</div><div className="flex flex-wrap gap-1">{programs.map((p) => <Badge key={p.id} tone="ink">{p.code}</Badge>)}{!programs.length && <span className="text-muted">None</span>}</div></div>
            <div><div className="eyebrow mb-1">Processes with a dollar-authority step</div><ul className="list-disc pl-4 text-[12.5px]">{doaProcesses.map((n) => <li key={n}>{n}</li>)}</ul></div>
            <div><div className="eyebrow mb-1">In flight</div><span className="text-[12.5px]">{impact.length} running request{impact.length === 1 ? '' : 's'} route through dollar authority ({impact.filter((x) => x.at).length} at the approval step now).</span></div>
          </div>
        </Panel>
        <Panel title="Revision history" pad={false}>
          {history.length ? (

/** Log-scale band showing how authority steps up by amount. */
function DoaBand({ rows }: { rows: DoaMatrix['rows'] }) {
  if (!rows.length) return null
  const lo = Math.log10(10_000), hi = Math.log10(200_000_000)
  const x = (v: number) => ((Math.log10(Math.max(10_000, Math.min(v, 200_000_000))) - lo) / (hi - lo)) * 100
  let prev = 0
  return (
    <div className="border-t border-line px-4 py-3">
      <div className="relative h-7 overflow-hidden rounded bg-sunken">
        {rows.map((r, i) => { const a = x(prev || 10_000), b = x(r.upTo); prev = r.upTo; return <div key={i} className="absolute top-0 flex h-7 items-center overflow-hidden border-r-2 border-surface px-1.5 text-[11px] font-semibold text-white" style={{ left: `${a}%`, width: `${Math.max(0, b - a)}%`, background: SERIES[i % SERIES.length] }} title={r.label}><span className="truncate">{r.label}</span></div> })}
      </div>
      <div className="mt-1 flex justify-between text-[10.5px] text-faint tabular-nums">{[10_000, 100_000, 1_000_000, 10_000_000, 100_000_000].map((v) => <span key={v}>{money(v)}</span>)}</div>
    </div>
  )
}

      case 'person': return onChange({ kind: 'person', personId: s.people.find((p) => !p.company)?.id ?? s.people[0].id })

      default: return onChange(undefined)
    }
  }
  const project = s.projects.find((p) => p.id === preview.projectId)
  const resolved = value ? resolveAssignees(s, previewInstance({ id: 'p', key: 'p', version: 0 } as WorkflowDef, preview.projectId, preview.requesterId, preview.data), value) : []

        </Field>
      )}
      {value?.kind === 'person' && (
        <Field label="Person"><Select value={value.personId} onChange={(e) => onChange({ kind: 'person', personId: e.target.value })} options={s.people.map((p) => ({ value: p.id, label: `${p.name} · ${p.title}` }))} /></Field>
      )}
      {value?.kind === 'doa' && (
        <Field label="Amount field" hint="The DOA matrix picks the approver whose limit first covers this amount.">
          <Select value={value.field} onChange={(e) => onChange({ kind: 'doa', field: e.target.value })} options={moneyFields.length ? moneyFields.map((f) => ({ value: f.key, label: `${f.label} (${f.key})` })) : [{ value: value.field, label: `${value.field} — add a money field` }]} />

import { useEffect, useState } from 'react'

/** True when the viewport is narrower than `px` (phones / small tablets). */
export function useNarrow(px = 768): boolean {
  const q = `(max-width: ${px - 1}px)`
  const get = () => (typeof window !== 'undefined' && window.matchMedia ? window.matchMedia(q).matches : false)
  const [v, setV] = useState(get)
  useEffect(() => {
    if (!window.matchMedia) return
    const m = window.matchMedia(q)
    const f = () => setV(m.matches)
    f()
    m.addEventListener?.('change', f)
    return () => m.removeEventListener?.('change', f)
  }, [q])
  return v
}

/** Swimlane label with sensible placement for non-human nodes. */
export function laneFor(s: AppState, def: WorkflowDef, n: WfNode): string {
  if (n.lane) return n.lane
  if (n.assignee) return laneOf(s, n)
  if (n.type === 'start') {
    const first = def.edges.find((e) => e.from === n.id)
    const t = first && def.nodes.find((x) => x.id === first.to)
    if (t?.assignee) return laneOf(s, t)
  }
  if (n.type === 'condition' || n.type === 'parallel-split' || n.type === 'parallel-join') {
    const into = def.edges.find((e) => e.to === n.id && !e.outcome?.match(/return|reject/))
    const p = into && def.nodes.find((x) => x.id === into.from)
    if (p && p.id !== n.id) return laneFor(s, def, p)
  }
  return 'System'
}

export function lanesOf(s: AppState, def: WorkflowDef): string[] {
  const out: string[] = []
  for (const n of def.nodes.slice().sort((a, b) => a.x - b.x || a.y - b.y)) {

    if (l !== 'System' && !out.includes(l)) out.push(l)
  }
  if (def.nodes.some((n) => laneFor(s, def, n) === 'System')) out.push('System')
  return out
}

export function outcomePaths(def: WorkflowDef, n: WfNode): string {
  const name = (id: string) => def.nodes.find((x) => x.id === id)?.label ?? '?'
  const out = def.edges.filter((e) => e.from === n.id)
  if (n.type === 'end') return 'Process completes; the requester is notified of the outcome.'
  if (!out.length) return '—'
  if (n.type === 'condition') {
    const conds = out.filter((e) => e.condition).map((e) => `If ${describeCondition(e.condition)} → ${name(e.to)}`)
    const def0 = out.find((e) => !e.condition)
    return [...conds, def0 ? `Otherwise → ${name(def0.to)}` : ''].filter(Boolean).join('; ')
  }
  if (n.type === 'parallel-split') return `In parallel → ${out.map((e) => name(e.to)).join(' + ')}`
  const parts = out.map((e) => `${e.outcome === 'return' ? 'Return' : e.outcome === 'reject' ? 'Reject' : n.type === 'form' ? 'Submit' : isHuman(n.type) ? 'Approve' : 'Then'} → ${name(e.to)}`)
  if (n.type === 'approval' && !out.some((e) => e.outcome === 'reject')) parts.push('Reject → process ends as rejected')
  if (isHuman(n.type) && n.type !== 'form' && !out.some((e) => e.outcome === 'return')) parts.push('Return → process ends as returned')
  return parts.join('; ')
}

/** Nodes in reading order (breadth-first from start, forward edges only, then by x). */
export function orderedNodes(def: WorkflowDef): WfNode[] {
  const start = def.nodes.find((n) => n.type === 'start')
  const seen = new Set<string>()
  const out: WfNode[] = []
  if (start) {
    const q = [start.id]
    while (q.length) {
      const id = q.shift()!
      if (seen.has(id)) continue
      seen.add(id)
      const n = def.nodes.find((x) => x.id === id)
      if (!n) continue
      out.push(n)
      def.edges.filter((e) => e.from === id && e.outcome !== 'return' && e.outcome !== 'reject').map((e) => def.nodes.find((x) => x.id === e.to)).filter(Boolean).sort((a, b) => a!.x - b!.x || a!.y - b!.y).forEach((t) => q.push(t!.id))
    }
  }
  const rest = def.nodes.filter((n) => !seen.has(n.id)).sort((a, b) => a.x - b.x)
  const all = [...out, ...rest]
  // keep end nodes last
  return [...all.filter((n) => n.type !== 'end'), ...all.filter((n) => n.type === 'end')]
}

// ───────────────────────────── Validation (WF-08 publish checklist)
export interface Check { id: string; ok: boolean; level: 'error' | 'warn'; label: string; detail?: string; nodeIds?: string[]; edgeIds?: string[] }

export function reachable(def: WorkflowDef, fromId: string, dir: 'fwd' | 'back' = 'fwd'): Set<string> {
  const seen = new Set<string>()
  const q = [fromId]
  while (q.length) {
    const id = q.pop()!
    if (seen.has(id)) continue
    seen.add(id)
    for (const e of def.edges) {
      if (dir === 'fwd' && e.from === id) q.push(e.to)
      if (dir === 'back' && e.to === id) q.push(e.from)
    }
  }
  return seen
}

export function validate(def: WorkflowDef): Check[] {
  const out: Check[] = []
  const starts = def.nodes.filter((n) => n.type === 'start')
  const ends = def.nodes.filter((n) => n.type === 'end')
  out.push({ id: 'start', ok: starts.length === 1, level: 'error', label: 'Exactly one start', detail: starts.length === 0 ? 'Add a Start node.' : starts.length > 1 ? `${starts.length} start nodes — keep one.` : undefined, nodeIds: starts.length > 1 ? starts.map((n) => n.id) : undefined })
  out.push({ id: 'end', ok: ends.length >= 1, level: 'error', label: 'At least one end', detail: ends.length ? undefined : 'Add an End node so the process can complete.' })
  const ids = new Set(def.nodes.map((n) => n.id))
  const dangling = def.edges.filter((e) => !ids.has(e.from) || !ids.has(e.to))
  out.push({ id: 'edges', ok: !dangling.length, level: 'error', label: 'Every connection has both ends', detail: dangling.length ? `${dangling.length} broken connection(s).` : undefined, edgeIds: dangling.map((e) => e.id) })
  if (starts[0]) {
    const r = reachable(def, starts[0].id)
    const unreachable = def.nodes.filter((n) => !r.has(n.id))
    out.push({ id: 'reach', ok: !unreachable.length, level: 'error', label: 'Every step is reachable from start', detail: unreachable.length ? `Not reachable: ${unreachable.map((n) => n.label).join(', ')}` : undefined, nodeIds: unreachable.map((n) => n.id) })
    const deadEnds = def.nodes.filter((n) => n.type !== 'end' && r.has(n.id) && !ends.some((e) => reachable(def, n.id).has(e.id)))
    out.push({ id: 'finish', ok: !deadEnds.length, level: 'error', label: 'Every step can reach an end', detail: deadEnds.length ? `Dead ends: ${deadEnds.map((n) => n.label).join(', ')}` : undefined, nodeIds: deadEnds.map((n) => n.id) })
  }
  const noAssignee = def.nodes.filter((n) => isHuman(n.type) && !n.assignee)
  out.push({ id: 'assignee', ok: !noAssignee.length, level: 'error', label: 'Forms, reviews and approvals have assignees', detail: noAssignee.length ? `Missing: ${noAssignee.map((n) => n.label).join(', ')}` : undefined, nodeIds: noAssignee.map((n) => n.id) })
  const badDoa = def.nodes.filter((n) => n.assignee?.kind === 'doa' && !def.fields.some((f) => f.key === (n.assignee as { field: string }).field && (f.type === 'money' || f.type === 'number')))
  out.push({ id: 'doa', ok: !badDoa.length, level: 'error', label: 'Dollar-authority steps read a money field', detail: badDoa.length ? `Add the amount field or pick another: ${badDoa.map((n) => n.label).join(', ')}` : undefined, nodeIds: badDoa.map((n) => n.id) })
  const conds = def.nodes.filter((n) => n.type === 'condition')
  const badCond = conds.filter((n) => { const o = def.edges.filter((e) => e.from === n.id); return o.length < 2 || o.filter((e) => !e.condition).length !== 1 })
  out.push({ id: 'cond', ok: !badCond.length, level: 'error', label: 'Decisions have a rule and exactly one default path', detail: badCond.length ? `Fix: ${badCond.map((n) => n.label).join(', ')}` : undefined, nodeIds: badCond.map((n) => n.id) })
  const noForward = def.nodes.filter((n) => n.type !== 'end' && !def.edges.some((e) => e.from === n.id && e.outcome !== 'return' && e.outcome !== 'reject'))
  out.push({ id: 'forward', ok: !noForward.length, level: 'error', label: 'Every step has a way forward', detail: noForward.length ? `No forward path: ${noForward.map((n) => n.label).join(', ')}` : undefined, nodeIds: noForward.map((n) => n.id) })
  const badSplit = def.nodes.filter((n) => (n.type === 'parallel-split' && def.edges.filter((e) => e.from === n.id).length < 2) || (n.type === 'parallel-join' && def.edges.filter((e) => e.to === n.id).length < 2))
  out.push({ id: 'parallel', ok: !badSplit.length, level: 'error', label: 'Parallel splits and joins have two or more branches', detail: badSplit.length ? `Fix: ${badSplit.map((n) => n.label).join(', ')}` : undefined, nodeIds: badSplit.map((n) => n.id) })
  const noSla = def.nodes.filter((n) => isHuman(n.type) && !n.slaDays)
  out.push({ id: 'sla', ok: !noSla.length, level: 'warn', label: 'Human steps have an SLA', detail: noSla.length ? `No SLA: ${noSla.map((n) => n.label).join(', ')}` : undefined, nodeIds: noSla.map((n) => n.id) })
  const approvals = def.nodes.filter((n) => n.type === 'approval')
  out.push({ id: 'approval', ok: approvals.length > 0, level: 'warn', label: 'At least one e-signed approval', detail: approvals.length ? undefined : 'Without an approval step nobody signs the outcome.' })
  out.push({ id: 'purpose', ok: !!def.purpose.trim() && !!def.name.trim(), level: 'warn', label: 'Name and purpose filled in (used in the SOP)' })
  return out
}

export interface TraceStep { node: WfNode; kind: 'human' | 'auto'; via?: string; people: Person[]; delegatedFrom: { from: Person; to: Person }[]; note?: string }

/** Walk the route a request would take, resolving real people (with DOA tiers and out-of-office delegation). */
export function trace(s: AppState, def: WorkflowDef, data: Record<string, unknown>, projectId?: string, startedById?: string): TraceStep[] {
  const fake: WorkflowInstance = { id: 'sim', defId: def.id, defKey: def.key, defVersion: def.version, title: 'Simulation', recordType: def.recordType, recordId: 'sim', projectId, status: 'running', active: [], startedAt: '', startedById: startedById ?? s.currentUserId, data, history: [] }
  const noOoo: AppState = { ...s, people: s.people.map((p) => ({ ...p, outOfOffice: undefined })) }
  const project = projectId ? s.projects.find((p) => p.id === projectId) : undefined
  const ctx = { ...data, project }
  const out: TraceStep[] = []
  const seen = new Set<string>()
  const person = (id: string) => s.people.find((p) => p.id === id)
  const walk = (id: string, via: string | undefined, depth: number) => {
    if (depth > 60) return
    const n = def.nodes.find((x) => x.id === id)
    if (!n || seen.has(id)) return
    seen.add(id)
    const edges = def.edges.filter((e) => e.from === id)
    if (isHuman(n.type) || n.type === 'notify') {
      const ids = resolveAssignees(s, fake, n.assignee)
      const raw = resolveAssignees(noOoo, fake, n.assignee)
      const delegatedFrom = raw.filter((r) => !ids.includes(r)).map((r) => { const p = person(r)!; const to = person(p.outOfOffice?.delegateId ?? '')!; return { from: p, to } }).filter((x) => x.from && x.to)
      let note: string | undefined
      if (n.assignee?.kind === 'doa') {
        const field = (n.assignee as { field: string }).field
        const amt = Math.abs(Number(data[field] ?? 0))
        const role = doaRoleFor(s, fake, amt)
        const m = doaMatrixFor(s, projectId)
        const row = m?.rows.slice().sort((a, b) => a.upTo - b.upTo).find((r) => amt <= r.upTo)
        note = `${money(amt)} falls in the “${row?.label ?? role}” tier${row && row.upTo < 1e12 ? ` (up to ${money(row.upTo)})` : ''} of ${m?.name ?? 'the DOA matrix'} v${m?.version ?? '?'} → ${s.roles.find((r) => r.id === role)?.name ?? role}`
      }
      if (!ids.length) note = (note ? note + '. ' : '') + 'No one resolves — the engine would skip this step. Pick a project or fix the assignee.'
      out.push({ node: n, kind: 'human', via, people: ids.map(person).filter(Boolean) as Person[], delegatedFrom, note: note ?? (n.rule === 'all' && ids.length > 1 ? 'All must approve' : ids.length > 1 ? 'Any one may act' : undefined) })
    } else out.push({ node: n, kind: 'auto', via, people: [], delegatedFrom: [] })
    if (n.type === 'condition') {
      const hit = edges.find((e) => e.condition && evalCondition(e.condition, ctx)) ?? edges.find((e) => !e.condition)
      if (hit) { out[out.length - 1].note = `${hit.condition ? describeCondition(hit.condition) + ' → true' : 'No rule matched → default path'}: ${hit.label || 'continue'}`; walk(hit.to, hit.label, depth + 1) }
      else out[out.length - 1].note = 'No path matches and there is no default.'
    } else if (n.type === 'parallel-split') edges.forEach((e) => walk(e.to, 'parallel', depth + 1))
    else if (n.type === 'parallel-join') {
      const e = edges[0]
      if (e) { seen.delete(n.id); if (def.edges.filter((x) => x.to === n.id).every((x) => seen.has(x.from))) { seen.add(n.id); walk(e.to, undefined, depth + 1) } else { out.pop() } }
    } else {
      const e = edges.find((e) => !e.outcome || e.outcome === 'approve' || e.outcome === 'default')
      if (e) walk(e.to, e.label, depth + 1)
    }
  }
  const st = def.nodes.find((n) => n.type === 'start')
  if (st) walk(st.id, undefined, 0)
  return out
}

export interface StepStat { nodeId: string; label: string; type: WfNodeType; n: number; avgH: number; maxH: number; slaDays?: number; breaches: number; runningNow: number; returns: number; oldestRunningH: number }
export interface ProcessStat {
  key: string; name: string; total: number; completed: number; running: number; approved: number; rejected: number; returnedOut: number
  avgCycleH?: number; medianCycleH?: number; p90CycleH?: number; returnRate: number; slaBreaches: number; overdueNow: number
  steps: StepStat[]; bottleneck?: StepStat; cycles: number[]
}

const ACTIONS = new Set(['approve', 'reject', 'return', 'submit'])

/** Per-step elapsed hours for one instance: each actor decision measured from when its step was entered. */
export function stepDurations(def: WorkflowDef | undefined, inst: WorkflowInstance): { nodeId: string; hours: number; action: string }[] {
  const hist = inst.history.slice().sort((a, b) => a.at.localeCompare(b.at))
  const out: { nodeId: string; hours: number; action: string }[] = []
  const preds = (id: string, depth = 0): string[] => {
    if (!def || depth > 6) return []
    const direct = def.edges.filter((e) => e.to === id).map((e) => e.from)
    return direct.flatMap((p) => { const n = def.nodes.find((x) => x.id === p); return n && (n.type === 'parallel-split' || n.type === 'parallel-join') ? [p, ...preds(p, depth + 1)] : [p] })
  }
  const acted = new Set<string>()
  hist.forEach((h, i) => {
    if (!ACTIONS.has(h.action)) return
    const key = h.nodeId + '|' + h.actorId
    if (acted.has(key) && h.action === 'approve') return
    acted.add(key)
    const p = new Set(preds(h.nodeId))
    let enter: string | undefined
    for (let j = i - 1; j >= 0; j--) {
      const e = hist[j]
      if (p.has(e.nodeId) && (ACTIONS.has(e.action) || e.action === 'system' || e.action === 'start')) { enter = e.at; break }
    }
    enter ??= hist[i - 1]?.at ?? inst.startedAt
    out.push({ nodeId: h.nodeId, hours: Math.max(0, (Date.parse(h.at) - Date.parse(enter)) / H), action: h.action })
  })
  return out
}

function quantile(xs: number[], q: number) { if (!xs.length) return undefined; const a = xs.slice().sort((x, y) => x - y); return a[Math.min(a.length - 1, Math.floor(q * (a.length - 1) + 0.5))] }

export function processStats(s: AppState, key: string): ProcessStat {
  const versions = versionsOf(s, key)
  const latest = versions.find((d) => d.status === 'published') ?? versions[0]
  const insts = s.workflows.filter((w) => w.defKey === key)
  const done = insts.filter((w) => w.status !== 'running' && w.completedAt)
  const cycles = done.map((w) => (Date.parse(w.completedAt!) - Date.parse(w.startedAt)) / H)
  const now = nowMs(s)
  const steps = new Map<string, StepStat>()
  const stat = (def: WorkflowDef | undefined, nodeId: string) => {
    let x = steps.get(nodeId)
    if (!x) {
      const n = latest?.nodes.find((y) => y.id === nodeId) ?? def?.nodes.find((y) => y.id === nodeId)
      x = { nodeId, label: n?.label ?? nodeId, type: n?.type ?? 'review', n: 0, avgH: 0, maxH: 0, slaDays: n?.slaDays, breaches: 0, runningNow: 0, returns: 0, oldestRunningH: 0 }
      steps.set(nodeId, x)
    }
    return x
  }
  let slaBreaches = 0
  for (const w of insts) {
    const def = s.workflowDefs.find((d) => d.id === w.defId)
    for (const d of stepDurations(def, w)) {
      const node = def?.nodes.find((n) => n.id === d.nodeId)
      if (!node || !isHuman(node.type)) continue
      const x = stat(def, d.nodeId)
      x.avgH = (x.avgH * x.n + d.hours) / (x.n + 1)
      x.n++
      x.maxH = Math.max(x.maxH, d.hours)
      if (d.action === 'return') x.returns++
      if (node.slaDays && d.hours > node.slaDays * 24) { x.breaches++; slaBreaches++ }
    }
    for (const a of w.status === 'running' ? w.active : []) {
      const x = stat(def, a.nodeId)
      x.runningNow++
      x.oldestRunningH = Math.max(x.oldestRunningH, (now - Date.parse(a.since)) / H)
    }
  }
  const overdueNow = insts.filter((w) => w.status === 'running').reduce((a, w) => a + w.active.filter((x) => x.due && x.due < s.settings.asOf).length, 0)
  const order = latest ? orderedNodes(latest).map((n) => n.id) : []
  const list = [...steps.values()].sort((a, b) => (order.indexOf(a.nodeId) + 1 || 99) - (order.indexOf(b.nodeId) + 1 || 99))
  const bottleneck = list.filter((x) => x.n).sort((a, b) => b.avgH - a.avgH)[0]
  return {
    key, name: latest?.name ?? key, total: insts.length, completed: done.length, running: insts.length - done.length,
    approved: done.filter((w) => w.status === 'approved').length, rejected: done.filter((w) => w.status === 'rejected').length, returnedOut: done.filter((w) => w.status === 'returned').length,
    avgCycleH: cycles.length ? cycles.reduce((a, b) => a + b, 0) / cycles.length : undefined, medianCycleH: quantile(cycles, 0.5), p90CycleH: quantile(cycles, 0.9),
    returnRate: insts.length ? insts.filter((w) => w.history.some((h) => h.action === 'return')).length / insts.length : 0,
    slaBreaches, overdueNow, steps: list, bottleneck, cycles,
  }
}

/** Low-risk items eligible for bulk approval: routine reviews and small-dollar decisions, never gates, estimates or exceptions. */
export function isLowRisk(t: Task): boolean {
  if (t.node.type === 'form') return false
  if (['gate', 'estimate', 'waiver', 'budget', 'contract', 'closeout'].includes(t.inst.recordType)) return false
  const amt = taskAmount(t)

  return t.node.type === 'review' || ['deliverable', 'variance', 'contingency', 'change', 'initiation', 'safety-plan'].includes(t.inst.recordType)
}

/** A short, deterministic e-signature fingerprint (the password itself is never stored). */
export function signatureHash(parts: string[]): string {
  let h1 = 0x811c9dc5, h2 = 0x01000193
  for (const ch of parts.join('|')) { h1 = Math.imul(h1 ^ ch.charCodeAt(0), 16777619) >>> 0; h2 = Math.imul(h2 ^ ch.charCodeAt(0), 2246822519) >>> 0 }
  return (h1.toString(16).padStart(8, '0') + h2.toString(16).padStart(8, '0')).slice(0, 16)
}

// ───────────────────────────── Version diff (feeds the publish note and revision history)
export function diffDefs(s: Pick<AppState, 'roles' | 'people'>, a: WorkflowDef | undefined, b: WorkflowDef): string {
  if (!a) return 'Initial release.'
  const out: string[] = []
  const an = new Map(a.nodes.map((n) => [n.id, n])), bn = new Map(b.nodes.map((n) => [n.id, n]))
  const added = b.nodes.filter((n) => !an.has(n.id)), removed = a.nodes.filter((n) => !bn.has(n.id))
  if (added.length) out.push(`Added ${added.map((n) => `“${n.label}”`).join(', ')}`)
  if (removed.length) out.push(`Removed ${removed.map((n) => `“${n.label}”`).join(', ')}`)
  for (const n of b.nodes) {
    const o = an.get(n.id)
    if (!o) continue
    const ch: string[] = []
    if (o.label !== n.label) ch.push(`renamed from “${o.label}”`)
    if (JSON.stringify(o.assignee) !== JSON.stringify(n.assignee)) ch.push(`assignee ${describeAssignee(s, o.assignee, a.fields)} → ${describeAssignee(s, n.assignee, b.fields)}`)
    if ((o.slaDays ?? 0) !== (n.slaDays ?? 0)) ch.push(`SLA ${o.slaDays ?? 0}→${n.slaDays ?? 0} days`)
    if ((o.rule ?? 'any') !== (n.rule ?? 'any')) ch.push(`rule ${o.rule ?? 'any'}→${n.rule ?? 'any'}`)
    if (o.type !== n.type) ch.push(`type ${o.type}→${n.type}`)
    if (ch.length) out.push(`“${n.label}”: ${ch.join(', ')}`)
  }
  const ek = (e: WfEdge) => `${e.from}>${e.to}:${e.outcome ?? ''}:${JSON.stringify(e.condition ?? null)}`
  const ae = new Set(a.edges.map(ek)), be = new Set(b.edges.map(ek))
  const eAdd = b.edges.filter((e) => !ae.has(ek(e))).length, eDel = a.edges.filter((e) => !be.has(ek(e))).length
  if (eAdd || eDel) out.push(`Routing: ${eAdd} path(s) added, ${eDel} removed`)
  const fa = new Set(a.fields.map((f) => f.key)), fb = new Set(b.fields.map((f) => f.key))
  const fAdd = b.fields.filter((f) => !fa.has(f.key)), fDel = a.fields.filter((f) => !fb.has(f.key))
  if (fAdd.length) out.push(`New field(s): ${fAdd.map((f) => f.label).join(', ')}`)
  if (fDel.length) out.push(`Removed field(s): ${fDel.map((f) => f.label).join(', ')}`)
  if (a.name !== b.name) out.push(`Renamed to “${b.name}”`)
  if (a.purpose !== b.purpose || a.description !== b.description) out.push('Updated description/purpose')
  return out.length ? out.join('; ') + '.' : 'No functional changes.'
}

  let cur = ''
  for (const w of t.split(/\s+/)) {
    if ((cur + ' ' + w).trim().length > max && cur) { out.push(cur); cur = w } else cur = (cur + ' ' + w).trim()
  }
  if (cur) out.push(cur)
  if (out.length > lines) { const k = out.slice(0, lines); k[lines - 1] = k[lines - 1].slice(0, max - 1) + '…'; return k }
  return out
}

export interface SwimlaneLayout { lanes: string[]; cols: number; pos: Record<string, { col: number; lane: number; x: number; y: number }>; width: number; height: number }

const LANE_H = 92, HEAD_W = 132, COL_W = 158, TOP = 8

export function swimlaneLayout(s: AppState, def: WorkflowDef): SwimlaneLayout {
  const lanes = lanesOf(s, def)
  const sorted = def.nodes.slice().sort((a, b) => a.x - b.x || a.y - b.y)
  // cluster x positions into columns
  const colOf: Record<string, number> = {}
  let col = -1, lastX = -Infinity
  for (const n of sorted) { if (n.x - lastX > 45) { col++; lastX = n.x } colOf[n.id] = col }
  // avoid two nodes in the same cell: push later ones right
  const taken = new Set<string>()
  const pos: SwimlaneLayout['pos'] = {}
  let shift = 0
  const colShift: Record<number, number> = {}
  for (const n of sorted) {
    const lane = Math.max(0, lanes.indexOf(laneFor(s, def, n)))
    let c = colOf[n.id] + (colShift[colOf[n.id]] ?? shift)
    while (taken.has(`${c}:${lane}`)) { c++; shift++; colShift[colOf[n.id]] = shift }
    taken.add(`${c}:${lane}`)
    pos[n.id] = { col: c, lane, x: HEAD_W + c * COL_W + COL_W / 2, y: TOP + lane * LANE_H + LANE_H / 2 }
  }
  const cols = Math.max(1, ...Object.values(pos).map((p) => p.col + 1))
  return { lanes, cols, pos, width: HEAD_W + cols * COL_W + 20, height: TOP + lanes.length * LANE_H + 40 }
}

const BOX = { w: 128, h: 50 }
function half(n: WfNode) {
  if (n.type === 'start' || n.type === 'end') return { w: 15, h: 15 }
  if (n.type === 'condition') return { w: 30, h: 30 }
  if (n.type === 'parallel-split' || n.type === 'parallel-join') return { w: 5, h: 30 }
  return { w: BOX.w / 2, h: BOX.h / 2 }
}

function edgePath(L: SwimlaneLayout, def: WorkflowDef, e: WfEdge): { d: string; mx: number; my: number; back: boolean } | null {
  const S = def.nodes.find((n) => n.id === e.from), T = def.nodes.find((n) => n.id === e.to)
  if (!S || !T) return null
  const a = L.pos[S.id], b = L.pos[T.id]
  const hs = half(S), ht = half(T)
  const back = e.outcome === 'return' || e.outcome === 'reject' || b.col < a.col
  if (back) {
    const yb = Math.max(a.y, b.y) + LANE_H / 2 - 6
    const x0 = a.x - 10, y0 = a.y + hs.h, x1 = b.x + 10, y1 = b.y + ht.h
    return { d: `M${x0} ${y0} C${x0} ${yb} ${x1} ${yb} ${x1} ${y1}`, mx: (x0 + x1) / 2, my: yb - 6, back: true }
  }
  if (b.col === a.col) {
    const down = b.y > a.y
    const x0 = a.x, y0 = a.y + (down ? hs.h : -hs.h), x1 = b.x, y1 = b.y + (down ? -ht.h : ht.h)
    return { d: `M${x0} ${y0} L${x1} ${y1}`, mx: x0, my: (y0 + y1) / 2, back: false }
  }
  const x0 = a.x + hs.w, y0 = a.y, x1 = b.x - ht.w, y1 = b.y
  const c = Math.max(30, (x1 - x0) * 0.5)
  // labels from decisions sit near the diamond so they never hide behind a later box
  const f = S.type === 'condition' ? 0.18 : 0.5
  const bez = (p0: number, p1: number, p2: number, p3: number) => (1 - f) ** 3 * p0 + 3 * (1 - f) ** 2 * f * p1 + 3 * (1 - f) * f * f * p2 + f ** 3 * p3
  return { d: `M${x0} ${y0} C${x0 + c} ${y0} ${x1 - c} ${y1} ${x1} ${y1}`, mx: bez(x0, x0 + c, x1 - c, x1), my: bez(y0, y0, y1, y1), back: false }
}

/** Swimlane diagram as an SVG string (lanes from laneOf, nodes ordered by x). */
export function swimlaneSvg(s: AppState, def: WorkflowDef, P: Palette, opts: { idPrefix?: string } = {}): string {
  const L = swimlaneLayout(s, def)
  const id = opts.idPrefix ?? 'sl'
  const parts: string[] = []
  parts.push(`<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 ${L.width} ${L.height}" width="${L.width}" height="${L.height}" font-family="Public Sans, system-ui, sans-serif" role="img" aria-label="Swimlane diagram: ${esc(def.name)}">`)
  parts.push(`<defs>${(['faint', 'warn', 'bad'] as const).map((c) => `<marker id="${id}-a-${c}" viewBox="0 0 10 10" refX="9" refY="5" markerWidth="7" markerHeight="7" orient="auto-start-reverse" markerUnits="userSpaceOnUse"><path d="M0 0 L10 5 L0 10 z" fill="${P[c]}"/></marker>`).join('')}</defs>`)
  L.lanes.forEach((lane, i) => {
    const y = TOP + i * LANE_H
    parts.push(`<rect x="0" y="${y}" width="${L.width - 10}" height="${LANE_H}" fill="${i % 2 ? P.surface : P.sunken}" stroke="${P.line}"/>`)
    parts.push(`<rect x="0" y="${y}" width="${HEAD_W - 8}" height="${LANE_H}" fill="${P.sunken}" stroke="${P.line}"/>`)
    wrapText(lane, 16, 3).forEach((l, j, arr) => parts.push(`<text x="10" y="${y + LANE_H / 2 - (arr.length - 1) * 7 + j * 14 + 4}" font-size="11.5" font-weight="700" fill="${P.ink}">${esc(l)}</text>`))
  })
  for (const e of def.edges) {
    const g = edgePath(L, def, e)
    if (!g) continue
    const c = e.outcome === 'return' ? 'warn' : e.outcome === 'reject' ? 'bad' : 'faint'
    parts.push(`<path d="${g.d}" fill="none" stroke="${P[c]}" stroke-width="1.5" ${g.back ? 'stroke-dasharray="5 4"' : ''} marker-end="url(#${id}-a-${c})"/>`)
    const label = edgeOutcomeLabel(e)
    if (label && (e.condition || g.back || e.label)) {
      const t = label.length > 24 ? label.slice(0, 23) + '…' : label
      parts.push(`<rect x="${g.mx - t.length * 2.9 - 5}" y="${g.my - 8}" width="${t.length * 5.8 + 10}" height="15" rx="7" fill="${P.surface}" stroke="${P[c === 'faint' ? 'line' : c]}"/><text x="${g.mx}" y="${g.my + 3}" text-anchor="middle" font-size="9.5" font-weight="600" fill="${P[c === 'faint' ? 'muted' : c]}">${esc(t)}</text>`)
    }
  }
  const order = orderedNodes(def)
  for (const n of def.nodes) {
    const p = L.pos[n.id]
    const num = order.filter((x) => isHuman(x.type) || x.type === 'system' || x.type === 'notify').indexOf(n) + 1
    if (n.type === 'start' || n.type === 'end') {
      parts.push(`<circle cx="${p.x}" cy="${p.y}" r="15" fill="${n.type === 'start' ? P.surface : P.ink}" stroke="${n.type === 'start' ? P.good : P.ink}" stroke-width="2.5"/>`)
      parts.push(`<text x="${p.x}" y="${p.y + 30}" text-anchor="middle" font-size="10" font-weight="600" fill="${P.muted}">${esc(n.label)}</text>`)
    } else if (n.type === 'condition') {
      parts.push(`<polygon points="${p.x},${p.y - 30} ${p.x + 30},${p.y} ${p.x},${p.y + 30} ${p.x - 30},${p.y}" fill="${P.surface}" stroke="${P.warn}" stroke-width="1.8"/><text x="${p.x}" y="${p.y + 5}" text-anchor="middle" font-size="14" font-weight="700" fill="${P.warn}">?</text>`)
      wrapText(n.label, 22).forEach((l, j) => parts.push(`<text x="${p.x}" y="${p.y + 42 + j * 11}" text-anchor="middle" font-size="9.5" font-weight="600" fill="${P.ink}">${esc(l)}</text>`))
    } else if (n.type === 'parallel-split' || n.type === 'parallel-join') {
      parts.push(`<rect x="${p.x - 5}" y="${p.y - 30}" width="10" height="60" rx="2" fill="${P.ink}"/>`)
    } else {
      const sys = n.type === 'system' || n.type === 'notify'
      const color = n.type === 'approval' ? P.marker : n.type === 'form' ? P.link : n.type === 'review' ? P.info : P.faint
      parts.push(`<rect x="${p.x - BOX.w / 2}" y="${p.y - BOX.h / 2}" width="${BOX.w}" height="${BOX.h}" rx="${sys ? 25 : 5}" fill="${P.surface}" stroke="${n.type === 'approval' ? P.warn : P.line}" stroke-width="${n.type === 'approval' ? 1.6 : 1.2}" ${sys ? 'stroke-dasharray="4 3"' : ''}/>`)
      if (!sys) parts.push(`<rect x="${p.x - BOX.w / 2}" y="${p.y - BOX.h / 2}" width="4" height="${BOX.h}" rx="2" fill="${color}"/>`)
      parts.push(`<text x="${p.x - BOX.w / 2 + (sys ? 14 : 10)}" y="${p.y - BOX.h / 2 + 12}" font-size="8" font-weight="700" letter-spacing="0.5" fill="${P.faint}">${num ? num + ' · ' : ''}${esc(nodeTypeLabel(n.type).toUpperCase())}${n.slaDays ? ' · ' + n.slaDays + 'D' : ''}</text>`)
      wrapText(n.label, 21).forEach((l, j) => parts.push(`<text x="${p.x - BOX.w / 2 + (sys ? 14 : 10)}" y="${p.y - BOX.h / 2 + 26 + j * 12}" font-size="10.5" font-weight="600" fill="${P.ink}">${esc(l)}</text>`))
    }
  }
  parts.push('</svg>')
  return parts.join('')
}

export interface StepRow { num: number; node: WfNode; who: string; sla: string; duty: string; outcomes: string }
export function stepRows(s: AppState, def: WorkflowDef): StepRow[] {
  let i = 0
  return orderedNodes(def).filter((n) => n.type !== 'parallel-join').map((n) => ({
    num: n.type === 'start' || n.type === 'end' ? 0 : ++i,
    node: n,
    who: isHuman(n.type) || n.type === 'notify' ? `${describeAssignee(s, n.assignee, def.fields)}${n.rule === 'all' ? ' (all must act)' : n.rule === 'any' ? ' (any one)' : ''}` : n.type === 'start' ? 'Requester' : 'Mainline (automatic)',
    sla: n.slaDays ? `${n.slaDays} business day${n.slaDays > 1 ? 's' : ''}` : '—',
    duty: stepDuty(def, n),
    outcomes: outcomePaths(def, n),
  }))
}

export type RaciCode = 'R' | 'A' | 'C' | 'I' | 'A/R'
export function raci(s: AppState, def: WorkflowDef): { cols: string[]; rows: { label: string; cells: Record<string, RaciCode | ''> }[] } {
  const lanes = lanesOf(s, def).filter((l) => l !== 'System')
  const owner = s.roles.find((r) => r.id === def.ownerRole)?.name ?? def.ownerRole
  const cols = [...lanes]
  if (!cols.includes(owner)) cols.push(owner)
  if (!cols.includes('Requester')) cols.push('Requester')
  const rows = orderedNodes(def).filter((n) => isHuman(n.type) || n.type === 'system' || n.type === 'notify').map((n) => {
    const cells: Record<string, RaciCode | ''> = Object.fromEntries(cols.map((c) => [c, ''])) as Record<string, RaciCode | ''>
    const lane = laneFor(s, def, n)
    if (n.type === 'approval') cells[lane] = 'A/R'
    else if (n.type === 'notify') cells[lane] = 'I'
    else if (lane !== 'System') cells[lane] = 'R'
    if (n.type !== 'approval' && !cells[owner]) cells[owner] = 'A'
    if (n.type === 'approval') {
      for (const e of def.edges.filter((x) => x.to === n.id)) {
        const p = def.nodes.find((x) => x.id === e.from)
        if (p && p.type === 'review') { const l = laneFor(s, def, p); if (!cells[l]) cells[l] = 'C' }
      }
      if (!cells.Requester) cells.Requester = 'I'
    }
    if (n.type === 'system' && !cells.Requester) cells.Requester = 'I'
    return { label: n.label, cells }
  })
  return { cols, rows }
}

export interface Revision { version: number; status: string; effective: string; summary: string; at?: string; by?: string }
export function revisions(s: AppState, def: WorkflowDef): Revision[] {
  const vs = versionsOf(s, def.key)
  const log = s.audit.filter((a) => a.entity === 'workflow-def' && a.entityId === def.key)
  return vs.map((v) => {
    const hit = log.filter((a) => a.summary.includes(`v${v.version}`) && a.action === 'publish').pop()
    return { version: v.version, status: v.status, effective: v.effective, summary: hit?.summary.replace(/^Published [^:]+: ?/, '') || (v.version === vs[vs.length - 1].version ? (v.version === 1 ? 'Initial release.' : 'Current release migrated into Crosswalk; earlier versions archived in the legacy procedure library.') : 'Revision.'), at: hit?.at, by: hit?.actorId }
  })
}

export function rolesInvolved(s: AppState, def: WorkflowDef): { lane: string; duties: string[] }[] {
  return lanesOf(s, def).map((lane) => ({ lane, duties: orderedNodes(def).filter((n) => laneFor(s, def, n) === lane && n.type !== 'start' && n.type !== 'end' && n.type !== 'parallel-split' && n.type !== 'parallel-join').map((n) => n.label) })).filter((x) => x.duties.length)
}

===== FILE: migrations/001_shared_state.sql =====
-- Shared state for the hosted demo (AGENT_RULES §5: plain SQL migrations, Postgres only).
--
-- A "generation" is one run of the shared dataset. It starts from a full snapshot (seq 0) and grows by an
-- ordered log of changes (immer patches). "Reset demo data" starts a new generation; nothing is ever
-- updated or deleted (rule 21: append-only), so every past edit stays on record.

create table if not exists generations (
  id             bigserial primary key,
  schema_version integer     not null,
  started_at     timestamptz not null default now(),
  started_by     text        not null,
  reason         text        not null default ''
);

create table if not exists changes (
  generation_id bigint      not null references generations (id),
  seq           integer     not null check (seq > 0),
  at            timestamptz not null default now(),
  actor         text        not null,
  summary       text        not null default '',
  patches       text        not null, -- JSON array of immer patches, stored as text and passed through untouched
  primary key (generation_id, seq)    -- two writers racing for the same seq: the second gets a conflict
);

create table if not exists snapshots (
  generation_id bigint      not null references generations (id),
  seq           integer     not null check (seq >= 0),
  at            timestamptz not null default now(),
  state_gz      text        not null, -- base64 of the gzipped JSON state after change `seq`
  primary key (generation_id, seq)
);

create index if not exists generations_schema_idx on generations (schema_version, id desc);

-- Append-only: refuse every UPDATE and DELETE (and TRUNCATE) on the three tables.
create or replace function crosswalk_append_only() returns trigger language plpgsql as $$
begin
  raise exception 'crosswalk: % is append-only; % is not allowed', tg_table_name, tg_op;
end $$;

drop trigger if exists generations_append_only on generations;
create trigger generations_append_only before update or delete on generations for each row execute function crosswalk_append_only();
drop trigger if exists generations_no_truncate on generations;
create trigger generations_no_truncate before truncate on generations for each statement execute function crosswalk_append_only();

drop trigger if exists changes_append_only on changes;
create trigger changes_append_only before update or delete on changes for each row execute function crosswalk_append_only();
drop trigger if exists changes_no_truncate on changes;
create trigger changes_no_truncate before truncate on changes for each statement execute function crosswalk_append_only();

drop trigger if exists snapshots_append_only on snapshots;
create trigger snapshots_append_only before update or delete on snapshots for each row execute function crosswalk_append_only();
drop trigger if exists snapshots_no_truncate on snapshots;
create trigger snapshots_no_truncate before truncate on snapshots for each statement execute function crosswalk_append_only();

// License check (AGENT_RULES rule 8): every installed package must carry a permissive license
// (MIT, Apache-2.0, BSD, ISC and close relatives). Exits non-zero and lists offenders otherwise.
// Usage: npm run licenses            (all installed packages, dev tools included)
import fs from 'node:fs'
import path from 'node:path'

const ALLOWED = ['MIT', 'MIT-0', 'Apache-2.0', 'BSD-2-Clause', 'BSD-3-Clause', '0BSD', 'ISC']
// Reviewed exceptions: build-time data, never shipped in the app bundle.
const EXCEPTIONS = { 'caniuse-lite': 'CC-BY-4.0 browser-support data read by autoprefixer at build time only' }
const ok = (expr) => {
  if (!expr) return false
  const e = String(expr).replace(/[()]/g, ' ')
  if (/\bOR\b/.test(e)) return e.split(/\bOR\b/).some((x) => ok(x.trim()))
  if (/\bAND\b/.test(e)) return e.split(/\bAND\b/).every((x) => ok(x.trim()))
  return ALLOWED.includes(e.trim())
}
const seen = new Map()
function walk(dir) {
  if (!fs.existsSync(dir)) return
  for (const name of fs.readdirSync(dir)) {
    if (name.startsWith('.')) continue
    const p = path.join(dir, name)
    if (name.startsWith('@')) { walk(p); continue }
    const pj = path.join(p, 'package.json')
    if (fs.existsSync(pj)) {
      const j = JSON.parse(fs.readFileSync(pj, 'utf8'))
      const lic = typeof j.license === 'object' ? j.license?.type : j.license ?? (Array.isArray(j.licenses) ? j.licenses.map((l) => l.type).join(' OR ') : undefined)
      seen.set(`${j.name}@${j.version}`, lic)
    }
    walk(path.join(p, 'node_modules'))
  }
}

const bad = [...seen].filter(([n, l]) => !ok(l) && !EXCEPTIONS[n.slice(0, n.lastIndexOf('@'))])
for (const [n, why] of Object.entries(EXCEPTIONS)) console.log(`exception: ${n} — ${why}`)
const counts = {}
for (const [, l] of seen) counts[l ?? 'UNKNOWN'] = (counts[l ?? 'UNKNOWN'] || 0) + 1

===== FILE: src/server/api.ts =====
// The app's own API (AGENT_RULES §5 rule 18): one plain `(request) => response` function with its
// dependencies passed in, so it runs unchanged on Cloudflare Pages Functions now and AWS Lambda later.
//
//   GET  /api/health                          → { ok, editable }
//   POST /api/auth                            → 204 when the passcode is right
//   GET  /api/state?schema=V                  → { generation, snapshot, changes, head } (generation null when none yet)
//   GET  /api/changes?generation=G&after=N    → { generation, changes, head } or { reset: true }
//   POST /api/changes    { generation, seq, actor, summary, patches }   → { seq } | 409 { head }
//   POST /api/snapshot   { generation, seq, stateGz }                    → 204
//   POST /api/generation { schema, reason, actor, stateGz }              → { generation }  (first edit, or "Reset demo data")
//
// Writes need the edit passcode in the `x-crosswalk-passcode` header. Reads are open: the demo is fictional data.
import { Conflict, type Repo } from './repo'

export interface ApiEnv { repo: Repo; passcode: string }

const MAX_CHANGE_BYTES = 2_000_000
const MAX_SNAPSHOT_BYTES = 8_000_000

const json = (body: unknown, status = 200) =>
  new Response(JSON.stringify(body), { status, headers: { 'content-type': 'application/json; charset=utf-8', 'cache-control': 'no-store' } })
const empty = (status = 204) => new Response(null, { status, headers: { 'cache-control': 'no-store' } })
const fail = (status: number, error: string) => json({ error }, status)

/** Constant-time string comparison, so response timing does not leak the passcode. */
export function sameSecret(a: string, b: string): boolean {
  const ea = new TextEncoder().encode(a), eb = new TextEncoder().encode(b)
  let diff = ea.length ^ eb.length
  for (let i = 0; i < Math.max(ea.length, eb.length); i++) diff |= (ea[i] ?? 0) ^ (eb[i] ?? 0)
  return diff === 0
}

async function body<T>(req: Request, max: number): Promise<T | null> {
  const text = await req.text()
  if (text.length > max) return null
  try { return JSON.parse(text) as T } catch { return null }
}

const int = (v: unknown) => (typeof v === 'number' && Number.isInteger(v) ? v : typeof v === 'string' && /^\d+$/.test(v) ? Number(v) : NaN)
const str = (v: unknown, max = 200) => (typeof v === 'string' ? v.slice(0, max) : '')

export async function handle(req: Request, env: ApiEnv): Promise<Response> {
  const url = new URL(req.url)
  const path = url.pathname.replace(/\/+$/, '')
  const editable = !!env.passcode
  const authorized = () => editable && sameSecret(req.headers.get('x-crosswalk-passcode') ?? '', env.passcode)
  try {
    if (req.method === 'GET' && path === '/api/health') return json({ ok: true, editable })

    if (req.method === 'POST' && path === '/api/auth') return authorized() ? empty() : fail(401, 'Wrong passcode')

    if (req.method === 'GET' && path === '/api/state') {
      const schema = int(url.searchParams.get('schema'))
      if (!Number.isFinite(schema)) return fail(400, 'schema is required')
      const generation = await env.repo.latestGeneration(schema)
      if (!generation) return json({ generation: null, snapshot: null, changes: [], head: 0 })
      const snapshot = await env.repo.latestSnapshot(generation.id)
      const changes = await env.repo.changesAfter(generation.id, snapshot?.seq ?? 0, 5000)
      return json({ generation, snapshot, changes, head: changes.at(-1)?.seq ?? snapshot?.seq ?? 0 })
    }

    if (req.method === 'GET' && path === '/api/changes') {
      const gen = int(url.searchParams.get('generation')), after = int(url.searchParams.get('after')), schema = int(url.searchParams.get('schema'))
      if (!Number.isFinite(gen) || !Number.isFinite(after) || !Number.isFinite(schema)) return fail(400, 'generation, after and schema are required')
      const latest = await env.repo.latestGeneration(schema)
      if (!latest || latest.id !== gen) return json({ reset: true, generation: latest })
      const changes = await env.repo.changesAfter(gen, after)
      return json({ generation: latest, changes, head: changes.at(-1)?.seq ?? after })
    }

    if (req.method === 'POST' && (path === '/api/changes' || path === '/api/snapshot' || path === '/api/generation')) {
      if (!editable) return fail(403, 'Editing is turned off on this server')
      if (!authorized()) return fail(401, 'Wrong passcode')

      if (path === '/api/changes') {
        const b = await body<{ generation: number; seq: number; actor: string; summary: string; patches: unknown[] }>(req, MAX_CHANGE_BYTES)
        if (!b || !Number.isInteger(b.generation) || !Number.isInteger(b.seq) || !Array.isArray(b.patches)) return fail(400, 'Bad change')
        try {
          await env.repo.appendChange(b.generation, b.seq, str(b.actor), str(b.summary, 500), JSON.stringify(b.patches))
          return json({ seq: b.seq })
        } catch (e) {
          if (e instanceof Conflict) return json({ head: await env.repo.head(b.generation) }, 409)
          throw e
        }
      }
      if (path === '/api/snapshot') {
        const b = await body<{ generation: number; seq: number; stateGz: string }>(req, MAX_SNAPSHOT_BYTES)
        if (!b || !Number.isInteger(b.generation) || !Number.isInteger(b.seq) || typeof b.stateGz !== 'string') return fail(400, 'Bad snapshot')
        if (b.seq > (await env.repo.head(b.generation))) return fail(400, 'Snapshot is ahead of the change log')
        await env.repo.addSnapshot(b.generation, b.seq, b.stateGz)
        return empty()
      }
      const b = await body<{ schema: number; reason: string; actor: string; stateGz: string }>(req, MAX_SNAPSHOT_BYTES)
      if (!b || !Number.isInteger(b.schema) || typeof b.stateGz !== 'string' || !b.stateGz) return fail(400, 'Bad generation')
      const generation = await env.repo.createGeneration(b.schema, str(b.actor), str(b.reason, 500), b.stateGz)
      return json({ generation })
    }

    return fail(404, 'Not found')
  } catch (e) {
    console.error('crosswalk api', e)
    return fail(500, 'Server error')
  }
}

// Plain-SQL migration runner (AGENT_RULES §5 rule 19): applies migrations/NNN_*.sql in order, once each,
// and records them in schema_migrations. Used by scripts/migrate.mjs at deploy time and by the tests.
import type { Exec } from './repo'

export interface Migration { name: string; sql: string }

export async function migrate(exec: Exec, runScript: (sql: string) => Promise<unknown>, migrations: Migration[]): Promise<string[]> {
  await runScript('create table if not exists schema_migrations (name text primary key, applied_at timestamptz not null default now())')
  const done = new Set((await exec('select name from schema_migrations')).map((r) => String(r.name)))
  const applied: string[] = []
  for (const m of [...migrations].sort((a, b) => a.name.localeCompare(b.name))) {
    if (done.has(m.name)) continue
    await runScript(m.sql)
    await exec('insert into schema_migrations (name) values ($1)', [m.name])
    applied.push(m.name)
  }
  return applied
}

// parameterized query and returns rows: Neon's HTTP driver in production, PGlite in tests.

export type Exec = (sql: string, params?: unknown[]) => Promise<Record<string, unknown>[]>

export interface Generation { id: number; schemaVersion: number; startedAt: string; startedBy: string; reason: string }
export interface ChangeRow { seq: number; at: string; actor: string; summary: string; patches: string }
export interface SnapshotRow { seq: number; stateGz: string }

export class Conflict extends Error { constructor() { super('conflict') } }

const n = (v: unknown) => Number(v)
const iso = (v: unknown) => (v instanceof Date ? v.toISOString() : String(v))

export function sqlRepo(exec: Exec) {
  return {
    async latestGeneration(schemaVersion: number): Promise<Generation | null> {
      const r = await exec('select id, schema_version, started_at, started_by, reason from generations where schema_version = $1 order by id desc limit 1', [schemaVersion])
      if (!r.length) return null
      const g = r[0]
      return { id: n(g.id), schemaVersion: n(g.schema_version), startedAt: iso(g.started_at), startedBy: String(g.started_by), reason: String(g.reason) }
    },
    /** New generation with its starting snapshot (seq 0). */
    async createGeneration(schemaVersion: number, startedBy: string, reason: string, stateGz: string): Promise<Generation> {
      const r = await exec('insert into generations (schema_version, started_by, reason) values ($1, $2, $3) returning id, started_at', [schemaVersion, startedBy, reason])
      const id = n(r[0].id)
      await exec('insert into snapshots (generation_id, seq, state_gz) values ($1, 0, $2)', [id, stateGz])
      return { id, schemaVersion, startedAt: iso(r[0].started_at), startedBy, reason }
    },
    async head(generationId: number): Promise<number> {
      const r = await exec('select coalesce(max(seq), 0) as head from changes where generation_id = $1', [generationId])
      return n(r[0].head)
    },
    async latestSnapshot(generationId: number): Promise<SnapshotRow | null> {
      const r = await exec('select seq, state_gz from snapshots where generation_id = $1 order by seq desc limit 1', [generationId])
      return r.length ? { seq: n(r[0].seq), stateGz: String(r[0].state_gz) } : null
    },
    async changesAfter(generationId: number, after: number, limit = 500): Promise<ChangeRow[]> {
      const r = await exec('select seq, at, actor, summary, patches from changes where generation_id = $1 and seq > $2 order by seq limit $3', [generationId, after, limit])
      return r.map((c) => ({ seq: n(c.seq), at: iso(c.at), actor: String(c.actor), summary: String(c.summary), patches: String(c.patches) }))
    },
    /** Append change `seq`; throws Conflict when someone else already wrote that seq (or the client is behind). */
    async appendChange(generationId: number, seq: number, actor: string, summary: string, patches: string): Promise<void> {
      if (seq !== (await this.head(generationId)) + 1) throw new Conflict()
      try {
        await exec('insert into changes (generation_id, seq, actor, summary, patches) values ($1, $2, $3, $4, $5)', [generationId, seq, actor, summary, patches])
      } catch (e) {
        if (/duplicate key|unique/i.test(String((e as Error).message))) throw new Conflict()
        throw e
      }
    },
    async addSnapshot(generationId: number, seq: number, stateGz: string): Promise<void> {
      await exec('insert into snapshots (generation_id, seq, state_gz) values ($1, $2, $3) on conflict do nothing', [generationId, seq, stateGz])
    },
  }
}
export type Repo = ReturnType<typeof sqlRepo>
