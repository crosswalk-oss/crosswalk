type View = 'register' | 'quick' | 'historical' | 'library' | 'accuracy' | 'analogs' | 'admin'
const TAB_KEY = 'crosswalk:estimating:tab'
const ASSET_LABEL: Record<AssetType, string> = { 'transmission-pipeline': 'Transmission pipeline', 'distribution-main': 'Distribution main', 'compressor-station': 'Compressor station', 'valve-site': 'Valve site', 'storage-field': 'Storage field', 'regulator-station': 'Regulator station' }
const METHOD_LABEL: Record<WorkMethod, string> = { 'replace-open-trench': 'Replace, open trench', 'replace-hdd': 'Replace, HDD', hydrotest: 'Hydrotest', abandon: 'Abandon', 'valve-automation': 'Valve automation', 'station-rebuild': 'Station rebuild', 'compressor-replacement': 'Compressor replacement', 'inline-inspection': 'In-line inspection' }

export default function Estimating() {
  const s = useStore()
  const [view, setView] = useState<View>(() => { try { return (prefs.getRaw(TAB_KEY) as View) || 'register' } catch { return 'register' } })
  const pick = (v: View) => { setView(v); try { prefs.setRaw(TAB_KEY, v) } catch { /* storage blocked */ } }
  const latest = s.projects.map((p) => latestEstimate(s, p.id)).filter(Boolean)
  const inReview = s.estimates.filter((e) => e.status === 'in-review')
  const drafts = s.estimates.filter((e) => e.status === 'draft')
  const pts = useMemo(() => accuracyPoints(s), [s])
  const c3 = summarize(pts.filter((x) => x.cls === 3))
  const c5 = summarize(pts.filter((x) => x.cls === 5))
  const reviewValue = sum(inReview.map((e) => estimateTotals(e).total))
  const quickReview = inReview.filter((e) => e.method === 'quick').length
  return (
    <div>
      <PageHeader eyebrow="Controls" title="Estimating" subtitle="Historical quick estimates, Class 5 rate-table and Class 3 line-item estimates, the tables they draw from, and how accurate they turned out to be.">
        <div className="grid grid-cols-2 gap-3 md:grid-cols-4">
          <Stat label="Estimates in review" value={inReview.length} sub={`${money(reviewValue)} awaiting approval`} onClick={() => pick('register')} />
          <Stat label="Drafts in progress" value={drafts.length} sub={`${latest.length} projects estimated`} />
          <Stat label="Class 3 mean accuracy" value={c3.n ? `${c3.mean > 0 ? '+' : ''}${pct(c3.mean, 1)}` : '—'} sub={c3.n ? `${pct(c3.within)} within AACE −20%/+30% · n=${c3.n}` : 'No outcomes yet'} tone={c3.n ? (c3.within >= 0.8 ? 'good' : 'warn') : undefined} onClick={() => pick('accuracy')} />
          <Stat label="Class 5 mean accuracy" value={c5.n ? `${c5.mean > 0 ? '+' : ''}${pct(c5.mean, 1)}` : '—'} sub={c5.n ? `${pct(c5.within)} within AACE −50%/+100% · n=${c5.n}` : 'No outcomes yet'} onClick={() => pick('accuracy')} />
        </div>
      </PageHeader>
      <Tabs<View> className="mb-4" value={view} onChange={pick} tabs={[
        { id: 'register', label: 'Estimate register', count: inReview.length || undefined },
        { id: 'quick', label: 'Quick estimate', count: quickReview || undefined },
        { id: 'historical', label: 'Historical costs', count: s.historicalCosts.filter((h) => h.active).length },
        { id: 'library', label: 'Rate table', count: s.rates.filter((r) => !r.region).length },
        { id: 'accuracy', label: 'Accuracy' },
        { id: 'analogs', label: 'Analog finder' },
        { id: 'admin', label: 'Administration' },
      ]} />
      {view === 'register' && <Register s={s} />}
      {view === 'quick' && <QuickTab />}
      {view === 'historical' && <HistoricalCosts />}
      {view === 'library' && <Library s={s} onAdmin={() => pick('admin')} />}
      {view === 'accuracy' && <Accuracy s={s} pts={pts} />}
      {view === 'analogs' && <Analogs s={s} />}
      {view === 'admin' && <EstimatingAdmin s={s} />}
    </div>
  )
}

  const methods = Array.from(new Set(base.filter((r) => !asset || r.assetType === asset).map((r) => r.workMethod)))

  const m = (v: number) => money(v, v < 1e4 ? 'full' : 'compact')
  const usage = useMemo(() => { const u: Record<string, number> = {}; for (const e of s.estimates) for (const l of e.lines) if (l.rateId) u[l.rateId] = (u[l.rateId] || 0) + 1; return u }, [s.estimates])

  const cols: Column<UnitCostRate>[] = [
    { key: 'd', header: 'Rate', sort: (r) => r.description, render: (r) => <div className="min-w-[220px]"><div className="text-[12.5px] font-semibold">{r.description}</div><div className="text-[11px] text-faint">{ASSET_LABEL[r.assetType]} · {METHOD_LABEL[r.workMethod]}</div></div> },

    { key: 'u', header: 'Unit', render: (r) => r.unit },

    { key: 'p50', header: 'P50 base', num: true, sort: (r) => r.p50, render: (r) => <b>{m(r.p50 * f(r))}</b> },
    { key: 'p90', header: 'P90', num: true, sort: (r) => r.p90, render: (r) => <span className="text-muted">{m(r.p90 * f(r))}</span> },
    { key: 'reg', header: <span title="Effective P50 by region: regional row, else base × factor">By region ({regions.join(' / ')})</span>, render: (r) => { const e = effectiveByRegion(s, r); return <span className="whitespace-nowrap text-[11.5px] tabular-nums text-muted">{regions.map((g, i) => <span key={g} className={cx(e[g].regional && 'font-semibold text-ink')}>{i ? ' / ' : ''}{m(e[g].value * f(r))}</span>)}</span> } },

Estimates escalate the direct-plus-tax-plus-loaders base once, from the basis year to the project's spend midpoint, using the portfolio escalation table ({Object.entries(s.settings.escalationByYear).map(([y, v]) => `${y} ${v}%`).join(' · ')}). Toggle "Show in {year}$" to view rates escalated to today ({escalationTo(s, 2025, year)}% from 2025$). The estimating team maintains the table, its regional rows and every other estimating table under Administration; each publish creates a new rate table version.
      </Callout>
    </div>
  )
}

// ───────────────────────────── Accuracy (ES-13)
function Accuracy({ s, pts: all }: { s: AppState; pts: AccuracyPoint[] }) {
  const [basis, setBasis] = useState<'all' | 'final'>('all')
  const [zoom, setZoom] = useState<'data' | 'aace'>('data')
  const [prog, setProg] = useState('')
  const pts = all.filter((x) => (basis === 'all' || x.basis === 'final') && (!prog || x.programId === prog))
  const classes = ([5, 4, 3, 2, 1] as AaceClass[]).filter((c) => pts.some((x) => x.cls === c))
  const accs = pts.map((x) => x.acc)
  const domain: [number, number] = zoom === 'aace' ? [Math.min(-0.55, ...accs), Math.max(1.05, ...accs)] : [Math.min(-0.25, ...accs.map((a) => a - 0.04)), Math.max(0.35, ...accs.map((a) => a + 0.04))]
  const person = (id: string) => personName(s, id)
  const projects = Array.from(new Set(pts.map((x) => x.projectId))).map((id) => s.projects.find((p) => p.id === id)!).sort((a, b) => a.code.localeCompare(b.code))
  const groupTable = (key: (x: AccuracyPoint) => string, label: (k: string) => React.ReactNode, title: string) => {
    const g = groupBy(pts, key)
    return (
      <Panel pad={false} title={title}>
        <table className="tbl">
          <thead><tr><th>{title.replace('By ', '')}</th><th className="num">C5 mean</th><th className="num">C3 mean</th><th className="num">n</th><th className="num">Within range</th></tr></thead>
          <tbody>
            {Object.keys(g).sort().map((k) => {
              const a = summarize(g[k].filter((x) => x.cls === 5)), b = summarize(g[k].filter((x) => x.cls === 3)), t = summarize(g[k])
              return <tr key={k}><td>{label(k)}</td><td className="num">{a.n ? <AccPct v={a.mean} cls={5} /> : '—'}</td><td className="num">{b.n ? <AccPct v={b.mean} cls={3} /> : '—'}</td><td className="num">{t.n}</td><td className="num">{pct(t.within)}</td></tr>
            })}
          </tbody>
        </table>
      </Panel>
    )
  }
  return (
    <div className="flex flex-col gap-4">
      <div className="flex flex-wrap items-center gap-3">
        <SegButtons value={basis} onChange={setBasis} options={[{ id: 'all', label: 'Final cost + current EAC' }, { id: 'final', label: 'Closed projects only' }]} />
        <div className="w-56"><ProgramSelect id="acc-prog" value={prog} onChange={setProg} /></div>
        <span className="text-xs text-muted">Accuracy = outcome ÷ approved estimate − 1. In-flight projects use the current EAC; their accuracy is final at closeout.</span>
      </div>

    else q.push({ label: 'Valve sites automated', qty: i.sites, unit: 'ea' })
  }
  return { ...p, asset: { ...p.asset, lengthFt: i.lengthFt, diameterIn: i.diameterIn }, location: { ...p.location, terrain: i.terrain ?? p.location.terrain, district: i.region ?? p.location.district }, scope: { ...p.scope, quantities: q } }
}

export function describeInputs(p: Project, i: ParametricInputs): string {
  const reg = i.region ? `${i.region} region` : `${i.terrain} terrain`
  if (p.asset.type === 'valve-site') return `${i.sites} valve site${i.sites === 1 ? '' : 's'}, ${i.diameterIn}" line, ${reg}`
  if (p.asset.type === 'compressor-station' || p.asset.type === 'storage-field') return `One station (lump-sum rate-table rows)${i.hp ? `, ${i.hp.toLocaleString('en-US')} hp` : ''}, ${reg}`
  return `${i.lengthFt.toLocaleString('en-US')} ft of ${i.diameterIn}" pipe${i.tieIns ? `, ${i.tieIns} tie-ins` : ''}, ${reg}`
}

export function blankMaturity(items: MaturityItem[] = MATURITY_ITEMS): Record<string, boolean> {
  return Object.fromEntries(items.map((m) => [m.key, m.forClass === 5]))
}

/** Build a new Class 5 rate-table draft (ES-01) from the scope statement. Not yet pushed into state. */
export function buildParametric(s: AppState, p: Project, inputs: ParametricInputs, byId: string): Estimate {
  const proj = projectForInputs(p, inputs)
  const region = inputs.region ?? regionOf(p)
  const lines = parametricLines(proj, s.rates, { region, regionFactors: s.settings.estimating.regionFactors }, { lengthFt: inputs.lengthFt, sites: inputs.sites, hp: inputs.hp, tieIns: inputs.tieIns, services: inputs.services })
  const items = maturityItems(s)
  const maturity = blankMaturity(items)
  const cls = computeClass(maturity, items)
  const mid = midpointYear(s, p)
  const by = basisYearOf(s, lines)
  const esc = escalationPctFor(s, by, mid)
  const cont = contingencyForClass(s, cls)
  const tax = salesTaxFor(s, p)
  const rf = s.settings.estimating.regionFactors[region] ?? 1
  return {
    id: uid('est'), projectId: p.id, version: nextVersion(s, p.id), title: 'Class 5 rate-table estimate', cls, status: 'draft', method: 'parametric',
    createdAt: nowISO(), createdById: byId, lines, maturity,
    loaders: currentLoaders(s),
    escalationPct: esc,
    contingency: { method: 'deterministic', pct: cont },
    rateTableVersion: s.settings.estimating.rateTableVersion,
    region,
    salesTax: tax,
    basis: {
      scopeBasis: `${p.scope.summary} Rate-table basis: ${describeInputs(p, inputs)}.`,
      assumptions: [
        'Quantities read from the project scope statement',
        `Unit costs at the rate-table median (P50), rate table v${s.settings.estimating.rateTableVersion}, for the ${region} region (region factor ×${rf.toFixed(2)} where no regional rate is kept)`,
        `Escalated from ${by}$ to the ${mid} spend midpoint using the portfolio escalation table`,
        `Sales tax ${tax.pct}% on material (${tax.counties.map((c) => `${c.county} ${Math.round(c.share * 100)}%`).join(', ')})`,
      ],
      exclusions: ['Customer-side work', 'Unknown contamination beyond Phase I findings', 'Escalation beyond the planned in-service year'],
      contingencyRationale: `Deterministic ${cont}% for Class ${cls} per the estimating standard.`,
      classJustification: classJustification(maturity, items),
    },
  }
}

/** Default work packages for a Class 3 split, by asset type. */
export function defaultPackages(p: Project): string[] {
  if (p.asset.type === 'compressor-station') return ['WP1 Equipment', 'WP2 EPC construction', 'WP3 Electrical']
  if (p.asset.type === 'valve-site') return ['WP1 Sites']
  return ['WP1 Segment A', 'WP2 Segment B']
}

/** Split parametric lines into work packages: linear quantities spread evenly, lump sums and counts go to the first package. */
export function splitIntoPackages(lines: EstimateLine[], wps: string[]): EstimateLine[] {
  const pk = wps.length ? wps : ['WP1']
  const out: EstimateLine[] = []
  for (const l of lines) {
    if ((l.unit === 'ft' || l.unit === 'mi') && pk.length > 1) {
      let left = l.qty
      pk.forEach((wp, i) => {
        const q = i === pk.length - 1 ? Math.round(left * 100) / 100 : Math.round((l.qty / pk.length) * 100) / 100
        left -= q
        out.push({ ...l, id: uid('ln'), workPackage: wp, qty: q })
      })
    } else out.push({ ...l, id: uid('ln'), workPackage: pk[0] })
  }
  return out
}

/** New draft version copied from an existing estimate (ES-02). */
export function buildFromExisting(s: AppState, src: Estimate, byId: string, opts: { split?: string[]; toClass3?: boolean } = {}): Estimate {
  const lines = opts.split ? splitIntoPackages(src.lines, opts.split) : src.lines.map((l) => ({ ...l, id: uid('ln') }))
  const maturity = { ...src.maturity }
  const cls = computeClass(maturity, maturityItems(s))
  const p = s.projects.find((x) => x.id === src.projectId)
  // Loaders and escalation are portfolio tables, locked per project: a new version carries the ones in force.
  // Unit costs are copied as they were, so the version keeps the rate table they were priced on.
  const fromQuick = src.method === 'quick'
  const tables = p && !fromQuick ? { escalationPct: escalationPctFor(s, basisYearOf(s, lines), spendMidpoint(s, p)), salesTax: salesTaxFor(s, p), region: regionOf(p) } : {}
  return {
    ...JSON.parse(JSON.stringify(src)) as Estimate,
    id: uid('est'), version: nextVersion(s, src.projectId), status: 'draft', createdAt: nowISO(), createdById: byId,
    approvedAt: undefined, approvedById: undefined, workflowInstanceId: undefined, repricedFrom: undefined,
    title: opts.toClass3 ? 'Class 3 line-item estimate' : fromQuick ? 'Class 5 estimate (from quick estimate)' : src.title,
    method: opts.toClass3 || fromQuick ? 'line-item' : src.method,
    quick: undefined, // a new version is worked as line items; the quick estimate record stays on its own version
    lines, maturity, cls,
    loaders: currentLoaders(s).length ? currentLoaders(s) : src.loaders,
    rateTableVersion: fromQuick ? undefined : pricedOnVersion(s, src),
    ...tables,
  }
}

// ───────────────────────────── Basis of estimate
export function classJustification(m: Record<string, boolean>, items: MaturityItem[] = MATURITY_ITEMS): string {
  const cls = computeClass(m, items)
  const done = items.filter((i) => m[i.key]).map((i) => i.label.toLowerCase())
  const nextMissing = cls > 1 ? missingFor(m, (cls - 1) as AaceClass, items) : []
  const r = CLASS_RANGES[cls]
  return `Class ${cls} (AACE 18R-97; project definition ${r.definition}). Complete: ${done.length ? done.join('; ') : 'none'}.` +
    (nextMissing.length ? ` Not yet complete for Class ${cls - 1}: ${nextMissing.map((i) => i.label.toLowerCase()).join('; ')}.` : '') +
    ` Expected accuracy ${pct(r.low)} to +${pct(r.high)}.`
}

export function contingencyRationale(e: Pick<Estimate, 'contingency' | 'cls'>, mc?: Pick<MonteCarloResult, 'costP' | 'iterations'>, riskCount?: number, guide?: number): string {
  if (e.contingency.method === 'deterministic') {
    const g = guide ?? CONTINGENCY_GUIDE[e.cls]
    return `Deterministic ${e.contingency.pct}% of base` + (e.contingency.pct === g ? ` per the estimating standard for Class ${e.cls}.` : ` (standard for Class ${e.cls} is ${g}%; estimator adjustment documented in assumptions).`)
  }
  return `Risk-based: ${e.contingency.pct}% of base sets the budget at the ${e.contingency.confidence ?? 'P80'} of a Monte Carlo simulation` +
    (mc ? ` (${mc.iterations.toLocaleString('en-US')} iterations; P50 ${money(mc.costP.P50)}, P80 ${money(mc.costP.P80)})` : '') +
    (riskCount != null ? ` over ${riskCount} active quantified risks on the project register.` : ' over the active quantified risk register.')
}

export function sourceSummary(e: Pick<Estimate, 'lines'>): { source: LineSource; n: number; amount: number }[] {
  const g = groupBy(e.lines, (l) => l.source)
  return (['library', 'quote', 'analog', 'judgment'] as LineSource[]).map((src) => ({ source: src, n: g[src]?.length ?? 0, amount: sum((g[src] ?? []).map(lineTotal)) }))
}

export const workPackages = (e: Pick<Estimate, 'lines'>) => Array.from(new Set(e.lines.map((l) => l.workPackage)))

/** Lines more than 25% from the rate-table median for the estimate's region. */
export function benchmarkExceptions(e: Pick<Estimate, 'lines'> & { region?: string }, rates: AppState['rates'], regionFactors?: Record<string, number>) {
  const pr = e.region && regionFactors ? { region: e.region, regionFactors } : undefined
  return e.lines.map((l) => ({ line: l, b: benchmark(l, rates, pr) })).filter((x) => x.b?.flag) as { line: EstimateLine; b: { pct: number; flag: boolean; ref: number } }[]
}

export function activeRisks(s: Pick<AppState, 'risks'>, pid: string): Risk[] {
  return s.risks.filter((r) => r.projectId === pid && isActive(r))
}

// ───────────────────────────── Outcomes & accuracy (ES-13)
export interface Outcome { value: number; basis: 'final' | 'eac' }

/** What a project actually costs: final actuals once closed, else the current EAC once spending has started. */
export function outcomeFor(s: AppState, p: Project): Outcome | null {
  const pos = position(s, p.id)
  if (p.status === 'closed' && pos.actualToDate > 0) return { value: pos.actualToDate, basis: 'final' }
  if (pos.actualToDate > 0 && pos.eac > 0) return { value: pos.eac, basis: 'eac' }
  return null
}

export interface AccuracyPoint {
  projectId: string; code: string; programId: string; estimatorId: string; estimateId: string; version: number
  cls: AaceClass; estimate: number; low: number; high: number; outcome: number; basis: Outcome['basis']; acc: number; within: boolean; approvedAt?: string
}

export function accuracyForProject(s: AppState, p: Project): AccuracyPoint[] {
  const o = outcomeFor(s, p)
  if (!o) return []
  return everApproved(s, p.id).map((e) => {
    const t = estimateTotals(e)
    const acc = o.value / t.total - 1
    const r = CLASS_RANGES[e.cls]
    return { projectId: p.id, code: p.code, programId: p.programId, estimatorId: e.createdById, estimateId: e.id, version: e.version, cls: e.cls, estimate: t.total, low: t.low, high: t.high, outcome: o.value, basis: o.basis, acc, within: acc >= r.low && acc <= r.high, approvedAt: e.approvedAt }
  })
}

export function accuracyPoints(s: AppState, opts: { includeInFlight: boolean } = { includeInFlight: true }): AccuracyPoint[] {
  return s.projects.flatMap((p) => accuracyForProject(s, p)).filter((x) => opts.includeInFlight || x.basis === 'final')
}

export function summarize(points: AccuracyPoint[]) {
  if (!points.length) return { n: 0, mean: 0, meanAbs: 0, min: 0, max: 0, within: 0 }
  const accs = points.map((p) => p.acc)
  return { n: points.length, mean: sum(accs) / accs.length, meanAbs: sum(accs.map(Math.abs)) / accs.length, min: Math.min(...accs), max: Math.max(...accs), within: points.filter((p) => p.within).length / points.length }
}

// ───────────────────────────── Analogs (ES-14)
export interface UnitBasis { unit: 'ft' | 'site' | 'hp' | 'station'; qty: number }
export function unitBasis(p: Project): UnitBasis {
  if (p.asset.type === 'valve-site') return { unit: 'site', qty: p.scope.quantities.find((q) => q.unit === 'ea')?.qty ?? 1 }
  if (p.asset.type === 'compressor-station' || p.asset.type === 'storage-field') {
    const hp = p.scope.quantities.find((q) => q.unit === 'hp')?.qty
    return hp ? { unit: 'hp', qty: hp } : { unit: 'station', qty: 1 }
  }
  return { unit: 'ft', qty: Math.max(1, p.asset.lengthFt) }
}

export interface CostBasis { value: number; basis: 'final' | 'eac' | 'estimate'; label: string; cls?: AaceClass }
export function bestCost(s: AppState, p: Project): CostBasis | null {
  const o = outcomeFor(s, p)
  if (o) return { value: o.value, basis: o.basis, label: o.basis === 'final' ? 'Final cost' : 'Current EAC' }
  const e = latestApproved(s, p.id) ?? latestEstimate(s, p.id)
  if (!e) return null
  return { value: estimateTotals(e).total, basis: 'estimate', label: `Class ${e.cls} estimate`, cls: e.cls }
}

export interface Analog { project: Project; score: number; reasons: string[]; cost: CostBasis | null; perUnit: number | null; unit: UnitBasis }

export function analogs(s: AppState, target: Project): Analog[] {
  const tu = unitBasis(target)
  return s.projects.filter((p) => p.id !== target.id).map((p) => {
    const reasons: string[] = []
    let score = 0
    if (p.asset.type === target.asset.type) { score += 35; reasons.push('Same asset type') }
    if (p.workMethod === target.workMethod) { score += 30; reasons.push('Same work method') }
    const d1 = target.asset.diameterIn, d2 = p.asset.diameterIn
    if (d1 && d2) { const f = 1 - Math.abs(d1 - d2) / Math.max(d1, d2); score += 15 * f; if (d1 === d2) reasons.push(`Same ${d1}" diameter`); else reasons.push(`${d2}" vs ${d1}"`) }
    else if (!d1 && !d2) score += 15
    if (p.location.terrain === target.location.terrain) { score += 10; reasons.push(`${p.location.terrain[0].toUpperCase() + p.location.terrain.slice(1)} terrain`) }
    const u = unitBasis(p)
    if (u.unit === tu.unit) { const f = Math.min(u.qty, tu.qty) / Math.max(u.qty, tu.qty); score += 10 * f }
    if (p.location.district === target.location.district) { score += 3; reasons.push(`${p.location.district} district`) }
    const cost = bestCost(s, p)
    const perUnit = cost && u.unit === tu.unit ? cost.value / u.qty : null
    return { project: p, score: Math.round(Math.min(100, score)), reasons, cost, perUnit, unit: u }
  }).sort((a, b) => b.score - a.score || (b.cost?.basis === 'final' ? 1 : 0) - (a.cost?.basis === 'final' ? 1 : 0))
}

export function perUnitLabel(u: UnitBasis['unit']) {
  return u === 'ft' ? '/ft' : u === 'site' ? '/site' : u === 'hp' ? '/hp' : '/station'
}
export function fmtPerUnit(v: number | null, u: UnitBasis['unit']) {
  if (v == null) return '—'
  return (v >= 1e4 ? money(v) : '$' + Math.round(v).toLocaleString('en-US')) + perUnitLabel(u)
}

/**
 * Start the estimate-approval workflow for a saved draft (ES-09). If the current user holds the
 * first "prepare" step (or is an admin) it is submitted on their behalf. Returns a toast message.
 */
export function submitEstimate(d: AppState, estimateId: string, mc?: MonteCarloResult): string {
  const e = d.estimates.find((x) => x.id === estimateId)
  if (!e || e.status !== 'draft') return 'Only drafts can be submitted'
  const p = d.projects.find((x) => x.id === e.projectId)!
  e.cls = computeClass(e.maturity, maturityItems(d))
  const total = estimateTotals(e).total
  if (e.contingency.method === 'risk-based' && mc) d.monteCarlo.push({ ...mc, id: uid('mc') })
  const inst = startWorkflow(d, 'estimate-approval', { title: `${p.code} Class ${e.cls} estimate v${e.version}`, recordType: 'estimate', recordId: e.id, projectId: p.id, data: { amount: total, cls: String(e.cls) } })
  if (!inst) return 'No published estimate-approval workflow'
  const slot = inst.active[0]
  const actor = d.people.find((x) => x.id === d.currentUserId)
  const name = (id: string) => d.people.find((x) => x.id === id)?.name ?? id
  if (slot && (slot.assigneeIds.includes(d.currentUserId) || actor?.roles.includes('admin'))) {
    actOnWorkflow(d, inst.id, slot.nodeId, d.currentUserId, 'submit', 'Estimate and basis of estimate prepared')
    const next = inst.active[0]
    return next ? `Submitted for approval — now with ${next.assigneeIds.map(name).join(', ')}` : 'Submitted and approved'
  }
  return slot ? `Sent to ${slot.assigneeIds.map(name).join(', ')} to complete and submit` : 'Submitted for approval'
}

export function AccuracyStrip({ rows, domain }: { rows: StripRow[]; domain: [number, number] }) {

  for (let v = Math.ceil(d0 / step) * step; v <= d1 + 1e-9; v += step) ticks.push(Math.round(v * 100) / 100)

    <svg viewBox={`0 0 ${W} ${H}`} className="w-full" role="img" aria-label="Estimate accuracy by class">

function blankRow(s: AppState): HistoricalCost {
  return { id: '', assetType: 'transmission-pipeline', workMethod: 'replace-open-trench', diameterBand: '20-24', terrain: 'suburban', unit: 'ft', p10: 0, p50: 0, p90: 0, n: 1, baseYear: Number(s.settings.asOf.slice(0, 4)), mix: {}, sources: [], updatedAt: s.settings.asOf, updatedById: s.currentUserId, active: true }

function changesOf(a: HistoricalCost, b: HistoricalCost): string[] {
  const out: string[] = []
  for (const k of ['assetType', 'workMethod', 'diameterBand', 'terrain', 'unit', 'p10', 'p50', 'p90', 'n', 'baseYear', 'active', 'note'] as const) if (a[k] !== b[k]) out.push(`${k} ${a[k] ?? '—'} → ${b[k] ?? '—'}`)
  if (a.sources.join(',') !== b.sources.join(',')) out.push(`sources ${a.sources.length} → ${b.sources.length}`)
  if (JSON.stringify(a.mix) !== JSON.stringify(b.mix)) out.push('category mix')
  return out
}

/** Keep inputs consistent after a dropdown changes: method valid for the asset, unit from the catalog, band only when banded. */
export function normalizeInputs(i: QuickInputs, prevAsset?: AssetType): QuickInputs {

// ───────────────────────────── Result interpretation
export interface MatchNote { tone: 'good' | 'warn' | 'bad'; label: string; detail: string }

/** How the historical row was found, and whether the reviewer should look harder. */
export function matchNote(s: Pick<AppState, 'historicalCosts'>, r: Pick<QuickResult, 'row' | 'match'>, i: QuickInputs): MatchNote {
  if (!r.row) return { tone: 'bad', label: 'No historical data', detail: `The table has no active row for ${ASSET_LABEL[i.assetType].toLowerCase()} · ${METHOD_LABEL[i.workMethod].toLowerCase()}. Ask the estimating team to add one, or prepare a parametric estimate.` }
  const row = r.row
  const small = row.n < 5
  const sample = small ? ` Only ${row.n} closed project${row.n === 1 ? '' : 's'} behind it — small sample.` : ''
  const allDia = isBanded(i.assetType) && row.diameterBand === 'n/a' ? ' The row covers all diameters.' : ''
  if (r.match === 'nearest-band') return { tone: 'warn', label: 'Fallback: nearest diameter band', detail: `No row for ${bandLabel(i.diameterBand)}; priced from ${bandLabel(row.diameterBand)} (${TERRAIN_LABEL[row.terrain].toLowerCase()}).${sample}` }
  if (r.match === 'any-terrain') {
    const pool = s.historicalCosts.filter((h) => h.active && h.assetType === i.assetType && h.workMethod === i.workMethod)
    const split = pool.some((h) => h.terrain !== 'any')
    if (split) return { tone: 'warn', label: 'Fallback: all settings', detail: `No ${TERRAIN_LABEL[i.terrain].toLowerCase()} row; priced from the row that pools every setting.${allDia}${sample}` }
    return { tone: small ? 'warn' : 'good', label: small ? 'Small sample' : 'Matched', detail: `${ASSET_LABEL[i.assetType]} · ${METHOD_LABEL[i.workMethod].toLowerCase()} is kept as one row across settings.${allDia}${sample}` }
  }
  return { tone: small ? 'warn' : 'good', label: small ? 'Small sample' : 'Exact match', detail: `Same asset type, method${isBanded(i.assetType) && !allDia ? ', diameter band' : ''} and setting.${allDia}${sample}` }
}

/** Historical P10–P90 spread around the point, beside the AACE Class 5 expected range. */
export function aaceCompare(r: Pick<QuickResult, 'total' | 'low' | 'high'>) {
  const c5 = CLASS_RANGES[5]
  const lo = r.total ? r.low / r.total - 1 : 0
  const hi = r.total ? r.high / r.total - 1 : 0
  const inside = lo >= c5.low && hi <= c5.high
  return { lo, hi, aaceLo: c5.low, aaceHi: c5.high, aaceLow: Math.round(r.total * (1 + c5.low)), aaceHigh: Math.round(r.total * (1 + c5.high)), inside }
}

// ───────────────────────────── Eligibility, request, revise
export function quickEligibility(s: AppState, pid: string): { ok: boolean; reason?: string } {
  const p = s.projects.find((x) => x.id === pid)
  if (!p) return { ok: false, reason: 'Pick a project' }
  if (p.status === 'closed' || p.status === 'cancelled') return { ok: false, reason: `${p.code} is ${p.status}` }
  const ests = s.estimates.filter((e) => e.projectId === pid)
  const review = ests.find((e) => e.status === 'in-review')
  if (review) return { ok: false, reason: `v${review.version} is already in review` }
  const better = ests.find((e) => e.status === 'approved' && e.cls < 5)
  if (better) return { ok: false, reason: `${p.code} already has an approved Class ${better.cls} estimate (v${better.version})` }
  const draft = ests.find((e) => e.status === 'draft')
  if (draft) return { ok: false, reason: `Finish or submit draft v${draft.version} first` }
  return { ok: true }
}

export interface RequestArgs { inputs: QuickInputs | FactorInputs | ParametricInputs; note: string; needBy?: string; byId?: string }
const isParametric = (i: RequestArgs['inputs']): i is ParametricInputs => 'risks' in i && !!(i as ParametricInputs).risks
export interface RequestResult { estimateId?: string; instId?: string; message: string; ok: boolean }

const names = (d: AppState, ids: string[]) => ids.map((id) => d.people.find((p) => p.id === id)?.name ?? id).join(', ')

/** Route an existing quick estimate (status draft) to the estimating team: start class5-review and submit the prep step. */
function route(d: AppState, e: Estimate, note: string, byId: string): RequestResult {
  const p = d.projects.find((x) => x.id === e.projectId)!
  const total = estimateTotals(e).total
  const inst = startWorkflow(d, 'class5-review', { title: `${p.code} Class 5 quick estimate`, recordType: 'estimate', recordId: e.id, projectId: p.id, data: { amount: total, note }, startedById: byId })
  if (!inst) return { ok: false, estimateId: e.id, message: 'No published Class 5 quick estimate review process' }
  const slot = inst.active[0]
  if (slot && slot.assigneeIds.includes(byId)) actOnWorkflow(d, inst.id, slot.nodeId, byId, 'submit', note || 'Quick estimate requested')
  const next = d.workflows.find((w) => w.id === inst.id)?.active[0]
  return { ok: true, estimateId: e.id, instId: inst.id, message: next ? `Sent to the estimating team — ${names(d, next.assigneeIds)}` : 'Submitted' }
}

/** New quick estimate on a project, routed for review. Mutates the draft. */
export function submitQuickRequest(d: AppState, pid: string, a: RequestArgs): RequestResult {
  const byId = a.byId ?? d.currentUserId
  const p = d.projects.find((x) => x.id === pid)
  if (isParametric(a.inputs)) {
    const pe = createParametricEstimate(d, pid, a.inputs, a.note.trim(), byId, a.needBy || undefined)
    if (!pe) return { ok: false, message: 'The rate table has no rows for this pipe and work method' }
    return route(d, pe, a.note.trim(), byId)
  }
  if (!publishedModel(d)) return { ok: false, message: 'No published factor model — ask the estimating lead' }
  const e = createFactorEstimate(d, pid, toFactorInputs(d, p, a.inputs as QuickInputs), a.note.trim(), byId, a.needBy || undefined)
  if (!e) return { ok: false, message: 'The factor model has no base rate for this asset and method' }
  return route(d, e, a.note.trim(), byId)
}

/** Sent back by the estimating team: the workflow is waiting on the requester at the prep step again. */
export function returnedToRequester(s: Pick<AppState, 'workflows'>, e: Estimate) {
  const inst = s.workflows.find((w) => w.id === e.workflowInstanceId)
  if (!inst || inst.status !== 'running') return

}

/**
 * Revise a quick estimate that came back (returned to the requester, or recalled to draft): reprice it in place
 * from new inputs and send it to the estimating team again.
 */
export function reviseQuickRequest(d: AppState, estimateId: string, a: RequestArgs): RequestResult {
  const e = d.estimates.find((x) => x.id === estimateId)
  if (!e || !canRevise(d, e)) return { ok: false, message: 'Only a returned or recalled quick estimate can be revised' }
  const byId = a.byId ?? d.currentUserId

  if (!fresh) return { ok: false, message: par ? 'The rate table has no rows for this pipe and work method' : 'The factor model has no base rate for these inputs' }
  // Use the engine's pricing and basis text, but keep this record (same id and version).
  d.estimates.splice(d.estimates.indexOf(fresh), 1)
  d.audit.splice(nAudit)
  e.lines = fresh.lines; e.loaders = fresh.loaders; e.contingency = fresh.contingency; e.basis = fresh.basis; e.quick = fresh.quick; e.factor = fresh.factor; e.parametric = fresh.parametric; e.salesTax = fresh.salesTax; e.region = fresh.region; e.escalationPct = fresh.escalationPct; e.rateTableVersion = fresh.rateTableVersion; e.title = fresh.title; e.cls = 5
  const what = par ? `rate table + risk ratings (${(a.inputs as ParametricInputs).lengthFt.toLocaleString('en-US')} ft, ${(a.inputs as ParametricInputs).region})` : `factor model v${e.factor?.modelVersion ?? '—'} (${describeQuickInputs(a.inputs as QuickInputs)})`
  d.audit.push({ id: uid('aud'), at: nowISO(), actorId: byId, entity: 'estimate', entityId: e.id, projectId: e.projectId, action: 'revise', summary: `Revised Class 5 quick estimate v${e.version} on ${what}: $${(estimateTotals(e).total / 1e6).toFixed(2)}M` })
  const back = returnedToRequester(d, e)
  if (back) {
    back.inst.data.amount = estimateTotals(e).total
    back.inst.data.note = a.note.trim()
    actOnWorkflow(d, back.inst.id, 'prep', byId, 'submit', `Revised: ${a.note.trim()}`)

  }

  return route(d, e, a.note.trim(), byId)
}

/** How a saved quick estimate's row relates to its inputs (the same categories the request form showed). */
export function storedMatch(s: Pick<AppState, 'historicalCosts'>, e: Estimate): QuickResult['match'] {
  const q = e.quick
  if (q && e.factor) {
    const m = (s as AppState).factorModels?.find((x) => x.id === e.factor!.modelId)
    const br = m?.baseRates.find((b) => b.id === e.factor!.baseRateId)
    if (!br) return 'none'
    return isBanded(q.inputs.assetType) && br.diameterBand !== 'n/a' && br.diameterBand !== q.inputs.diameterBand ? 'nearest-band' : 'exact'
  }
  const row = q && s.historicalCosts.find((h) => h.id === q.historicalCostId)
  if (!q || !row) return 'none'
  const band = isBanded(q.inputs.assetType) ? q.inputs.diameterBand : 'n/a'
  if (row.diameterBand !== band && row.diameterBand !== 'n/a') return 'nearest-band'
  return row.terrain === 'any' ? 'any-terrain' : 'exact'
}

export const histKey = (h: Pick<HistoricalCost, 'assetType' | 'workMethod' | 'diameterBand' | 'terrain'>) => `${h.assetType}|${h.workMethod}|${h.diameterBand}|${h.terrain}`
export const mixTotal = (mix: Record<string, number>) => Math.round(sum(Object.values(mix)) * 10000) / 100 // percent

export function validateHistorical(h: HistoricalCost, all: HistoricalCost[]): string[] {
  const out: string[] = []
  const c = catalogOf(h.assetType)
  if (!c) out.push(`${ASSET_LABEL[h.assetType] ?? h.assetType} is not in the quick estimate catalog`)
  else {
    if (!c.methods.includes(h.workMethod)) out.push(`${METHOD_LABEL[h.workMethod] ?? h.workMethod} is not a work method for ${c.label.toLowerCase()}`)
    if (h.unit !== c.unit) out.push(`Unit must be ${c.unit} for ${c.label.toLowerCase()}`)
  }
  if (!(h.p50 > 0)) out.push('P50 must be greater than zero')
  if (h.p10 > h.p50 || h.p50 > h.p90) out.push('P10 ≤ P50 ≤ P90')
  if (!(h.n >= 1)) out.push('n must be at least 1')
  if (!(h.baseYear >= 2000 && h.baseYear <= 2100)) out.push('Base year looks wrong')
  if (Math.abs(mixTotal(h.mix) - 100) > 0.5) out.push(`Category mix must total 100% (now ${mixTotal(h.mix).toFixed(1)}%)`)
  if (h.active && all.some((x) => x.id !== h.id && x.active && histKey(x) === histKey(h))) out.push('Another active row already covers this asset, method, band and setting')
  return out
}

'yes' : 'no', note: h.note ?? '' }))

/** Minimal RFC 4180 CSV reader (quoted fields, doubled quotes, CRLF). */
export function readCsv(text: string): string[][] {
  const out: string[][] = []
  let row: string[] = [], f = '', q = false
  for (let i = 0; i < text.length; i++) {
    const c = text[i]
    if (q) {
      if (c === '"') { if (text[i + 1] === '"') { f += '"'; i++ } else q = false } else f += c
    } else if (c === '"') q = true
    else if (c === ',') { row.push(f); f = '' }
    else if (c === '\n' || c === '\r') { if (c === '\r' && text[i + 1] === '\n') i++; row.push(f); f = ''; if (row.some((x) => x.trim() !== '')) out.push(row); row = [] }
    else f += c
  }
  row.push(f)
  if (row.some((x) => x.trim() !== '')) out.push(row)
  return out
}

/** Parse an exported (or hand-edited) historical cost CSV. Rows with errors are reported, not imported. */
export function parseHistoricalCsv(text: string, s: Pick<AppState, 'settings' | 'currentUserId'>): { rows: HistoricalCost[]; errors: string[] } {
  const grid = readCsv(text.replace(/^﻿/, ''))
  if (!grid.length) return { rows: [], errors: ['The file is empty'] }
  const head = grid[0].map((h) => h.trim())
  const need = ['assetType', 'workMethod', 'diameterBand', 'terrain', 'p10', 'p50', 'p90', 'n', 'baseYear', 'mix']
  const missing = need.filter((k) => !head.includes(k))
  if (missing.length) return { rows: [], errors: [`Missing column${missing.length > 1 ? 's' : ''}: ${missing.join(', ')}`] }
  const rows: HistoricalCost[] = []
  const errors: string[] = []
  grid.slice(1).forEach((cells, k) => {
    const g = (key: string) => (cells[head.indexOf(key)] ?? '').trim()
    const line = k + 2
    const assetType = g('assetType') as AssetType
    const c = catalogOf(assetType)
    const mix: Record<string, number> = {}
    for (const part of g('mix').split(/[;|]/).map((x) => x.trim()).filter(Boolean)) {
      const [cat, v] = part.split(':')
      const n = Number(v)
      if (!cat || !Number.isFinite(n)) { errors.push(`Line ${line}: mix entry "${part}" should look like cat-contract:0.42`); return }
      mix[cat.trim()] = n > 1.0001 ? n / 100 : n
    }
    const num = (key: string) => Number(g(key).replace(/[$,\s]/g, ''))
    const h: HistoricalCost = {
      id: g('id') || '', assetType, workMethod: g('workMethod') as WorkMethod, diameterBand: (g('diameterBand') || 'n/a') as DiameterBand,
      terrain: (g('terrain') || 'any') as HistoricalCost['terrain'], unit: (g('unit') || c?.unit || 'ft') as HistUnit,
      p10: num('p10'), p50: num('p50'), p90: num('p90'), n: Math.round(num('n')), baseYear: Math.round(num('baseYear')), mix,
      sources: g('sources').split(/[;|]/).map((x) => x.trim()).filter(Boolean), updatedAt: s.settings.asOf, updatedById: s.currentUserId,
      active: !/^(no|false|0|inactive)$/i.test(g('active')), note: g('note') || undefined,
    }
    if (!c) { errors.push(`Line ${line}: unknown asset type "${assetType}"`); return }
    if (!DIAMETER_BANDS.some((b) => b.id === h.diameterBand)) { errors.push(`Line ${line}: unknown diameter band "${h.diameterBand}"`); return }
    if (!['urban', 'suburban', 'rural', 'any'].includes(h.terrain)) { errors.push(`Line ${line}: unknown setting "${h.terrain}"`); return }
    const v = validateHistorical(h, []).filter((x) => !x.startsWith('Another active row'))
    if (v.length) { errors.push(`Line ${line}: ${v.join('; ')}`); return }
    rows.push(h)
  })
  const keys = rows.filter((r) => r.active).map(histKey)
  const dup = keys.filter((k, i) => keys.indexOf(k) !== i)
  if (dup.length) errors.push(`Duplicate active rows in the file: ${Array.from(new Set(dup)).map((k) => k.replace(/\|/g, ' · ')).join('; ')}`)
  return { rows, errors }
}

export interface ImportPlan { add: HistoricalCost[]; update: { from: HistoricalCost; to: HistoricalCost }[]; deactivate: HistoricalCost[]; unchanged: number }

/** What an import would do. Rows match on id, else on asset/method/band/setting. Replace deactivates rows not in the file (never deletes). */
export function planImport(existing: HistoricalCost[], incoming: HistoricalCost[], mode: 'merge' | 'replace'): ImportPlan {
  const plan: ImportPlan = { add: [], update: [], deactivate: [], unchanged: 0 }
  const touched = new Set<string>()
  const canon = (h: HistoricalCost) => JSON.stringify([histKey(h), h.unit, h.p10, h.p50, h.p90, h.n, h.baseYear, Object.entries(h.mix).filter(([, v]) => v).sort(), h.sources, h.active, h.note ?? ''])
  const same = (a: HistoricalCost, b: HistoricalCost) => canon(a) === canon(b)
  for (const r of incoming) {
    const hit = (r.id && existing.find((x) => x.id === r.id)) || existing.find((x) => histKey(x) === histKey(r) && (x.active || !r.active))
    if (hit) {
      touched.add(hit.id)
      const to = { ...r, id: hit.id }
      if (same(hit, to)) plan.unchanged++
      else plan.update.push({ from: hit, to })
    } else plan.add.push({ ...r, id: r.id && !existing.some((x) => x.id === r.id) ? r.id : '' })
  }
  if (mode === 'replace') plan.deactivate = existing.filter((x) => x.active && !touched.has(x.id))
  return plan
}

export function applyImport(d: AppState, plan: ImportPlan, byId = d.currentUserId) {
  let k = d.historicalCosts.length
  for (const r of plan.add) {
    let id = r.id
    while (!id || d.historicalCosts.some((x) => x.id === id)) id = `hc-${++k}`
    d.historicalCosts.push({ ...r, id, updatedAt: d.settings.asOf, updatedById: byId })
  }
  for (const u of plan.update) { const x = d.historicalCosts.find((h) => h.id === u.from.id); if (x) Object.assign(x, u.to, { updatedAt: d.settings.asOf, updatedById: byId }) }
  for (const r of plan.deactivate) { const x = d.historicalCosts.find((h) => h.id === r.id); if (x) { x.active = false; x.updatedAt = d.settings.asOf; x.updatedById = byId } }
}

export interface DerivedPoint { projectId: string; code: string; perUnitBase: number; perUnitNominal: number; year: number; qty: number }
export interface Suggestion { key: string; row?: HistoricalCost; points: DerivedPoint[]; current?: Pick<HistoricalCost, 'p10' | 'p50' | 'p90' | 'n'>; suggested: HistoricalCost }

/**
 * Closed projects whose final actuals are not yet in the table, grouped by the row they price into.
 * Unit cost is normalized to the row's base year; the suggested P50 is the n-weighted blend, P10/P90 move with it
 * and widen to cover any new point outside the band.
 */
export function deriveSuggestions(s: AppState): Suggestion[] {
  const groups = new Map<string, { row?: HistoricalCost; inputs: QuickInputs; points: DerivedPoint[] }>()
  for (const p of s.projects.filter((x) => x.status === 'closed')) {
    const dv = deriveFromClosed(s, p)
    if (!dv) continue
    const inputs = inputsFromProject(s, p)
    const m = findHistorical(s, inputs)
    const row = m.match === 'exact' || m.match === 'any-terrain' ? m.row : undefined
    if (row?.sources.includes(p.code)) continue
    const baseYear = row?.baseYear ?? dv.year
    const f = dv.year >= baseYear ? 1 / escalationFactor(s, baseYear, dv.year) : escalationFactor(s, dv.year, baseYear)
    const key = row ? row.id : histKey({ ...inputs, terrain: inputs.terrain })
    const g = groups.get(key) ?? { row, inputs, points: [] }
    g.points.push({ projectId: p.id, code: p.code, perUnitBase: Math.round(dv.perUnit * f), perUnitNominal: Math.round(dv.perUnit), year: dv.year, qty: dv.qty })
    groups.set(key, g)
  }
  return Array.from(groups.entries()).map(([key, g]) => {
    const vals = g.points.map((x) => x.perUnitBase)
    const codes = g.points.map((x) => x.code)
    if (g.row) {
      const r = g.row
      const n = r.n + vals.length
      const p50 = Math.round((r.p50 * r.n + sum(vals)) / n)
      const ratio = p50 / r.p50
      const p10 = Math.min(Math.round(r.p10 * ratio), ...vals)
      const p90 = Math.max(Math.round(r.p90 * ratio), ...vals)
      return { key, row: r, points: g.points, current: { p10: r.p10, p50: r.p50, p90: r.p90, n: r.n }, suggested: { ...r, p10, p50, p90, n, sources: [...r.sources, ...codes], note: r.note } }
    }
    const near = findHistorical(s, g.inputs).row
    const v = Math.round(sum(vals) / vals.length)
    return {
      key, points: g.points,
      suggested: { id: '', assetType: g.inputs.assetType, workMethod: g.inputs.workMethod, diameterBand: g.inputs.diameterBand, terrain: g.inputs.terrain, unit: g.inputs.unit, p10: Math.min(...vals), p50: v, p90: Math.max(...vals), n: vals.length, baseYear: g.points[0].year, mix: near ? { ...near.mix } : {}, sources: codes, updatedAt: s.settings.asOf, updatedById: s.currentUserId, active: true, note: `New row from ${codes.join(', ')}${near ? `; category mix copied from ${bandLabel(near.diameterBand)} ${TERRAIN_LABEL[near.terrain].toLowerCase()} row` : ''} — set P10/P90 before relying on it` },
    }
  })
}

          {inbox ? <QuickHeadline total={t.total} low={q.range.low} high={q.range.high} perUnit={q.inputs.qty ? t.total / q.inputs.qty : 0} unit={q.inputs.unit} size="md" /> : requestPanel}

        </div>
      </div>}

      {adjustable && (e.parametric ? <ParametricAdjustPanel s={s} e={e} /> : e.factor ? <FactorAdjustPanel s={s} e={e} /> : <AdjustPanel s={s} e={e} />)}

      {!inbox && e.workflowInstanceId && <WorkflowStatus instanceId={e.workflowInstanceId} compact={e.status !== 'in-review'} />}

      {revise && (
        <Modal open wide onClose={() => setRevise(false)} title={`Revise quick estimate v${e.version}`}>
          <QuickRequest reviseId={e.id} onCancel={() => setRevise(false)} onDone={() => setRevise(false)} />
        </Modal>
      )}
    </div>
  )
}

/** Estimating-team adjustment of a factor estimate: %, optional percentile basis, reason from the override list. */

  const [reason, setReason] = useState('')
  const [note, setNote] = useState('')
  const before = estimateTotals(e).total
  const r = previewAdjust(s, e, basis, pctAdj)
  const after = r ? Math.round(r.total) : before

          <Field label="Note (optional)" hint="What about this project the drivers do not capture"><TextArea id="adj-note" className="min-h-[52px]" placeholder="e.g. Two HDD crossings under the flood-control channel." value={note} onChange={(ev) => setNote(ev.target.value)} /></Field>

            <Button variant="primary" disabled={!ok} onClick={apply}>Apply adjustment</Button>
            {!changed && <span className="text-[12px] text-muted">Change the % or the percentile to adjust.</span>}
            {changed && !reason && <span className="text-[12px] text-warn">Choose an override reason to apply.</span>}
            <span className="text-[12px] text-muted">Then approve or return below.</span>
          </div>
        </div>
        <div className="rounded-md border border-line bg-sunken/40 p-3">
          <div className="eyebrow">Before → after</div>
          <div className="mt-1 flex flex-wrap items-baseline gap-2 font-display tabular-nums">
            <span className="text-[17px] text-muted line-through decoration-faint">{money(before)}</span>
            <span aria-hidden className="text-muted">→</span>
            <span className="text-[22px] font-bold">{money(after)}</span>

          <div className={cx('text-[12.5px] font-semibold tabular-nums', after > before ? 'text-bad' : after < before ? 'text-good' : 'text-muted')}>{after === before ? 'No change' : `${moneyDelta(after - before)} (${after > before ? '+' : ''}${pct(after / before - 1, 1)})`}</div>

  const par = e.parametric!
  const reasons = s.settings.factorModel?.overrideReasons ?? []
  const [risks, setRisks] = useState<Record<string, RiskLevel>>({ ...par.inputs.risks })
  const [pctAdj, setPct] = useState<number>(par.adjustment?.pct ?? 0)
  const [reason, setReason] = useState('')
  const [note, setNote] = useState('')
  const before = estimateTotals(e).total
  const r = previewParametric(s, e, risks, pctAdj)
  const after = r ? r.total : before
  const changedRisks = Object.keys(risks).filter((k) => risks[k] !== par.inputs.risks[k])
  const changed = changedRisks.length > 0 || pctAdj !== (par.adjustment?.pct ?? 0)
  const ok = changed && !!reason
  const apply = () => {
    if (!ok) return
    mutate((d) => applyAdjustment(d, e.id, { pct: pctAdj, basis: 'p50', reason, note: note.trim() || undefined, risks }, me.id))
    toast(`Adjusted: ${money(before)} → ${money(after)}`)
    setReason(''); setNote('')
  }
  return (
    <Panel className="border-marker/70" title="Review and adjust (estimating team)" subtitle="Change any risk rating and/or apply a ± % to direct cost. The Monte Carlo re-runs; pick a reason. Kept with the estimate, in its BOE and in the audit trail.">
      <div className="flex flex-col gap-4">
        <div className="flex min-w-0 flex-col gap-3">
          <RiskRatings s={s} value={risks} onChange={setRisks} result={r} original={par.requestedRisks ?? par.inputs.risks} idPrefix="adj-risk" />
          <div className="flex flex-wrap items-end gap-3">
            <Field label="Adjustment % of direct" className="w-36"><NumInput id="adj-pct" value={pctAdj} onChange={(v) => setPct(Math.max(-50, Math.min(100, Math.round(v * 10) / 10)))} /></Field>
            <div className="flex flex-wrap gap-1 pb-0.5">{[-10, -5, 0, 5, 10, 15].map((v) => <button key={v} type="button" onClick={() => setPct(v)} className={cx('rounded border px-1.5 py-0.5 text-[11.5px] tabular-nums', pctAdj === v ? 'border-ink bg-ink text-surface' : 'border-line text-muted hover:text-ink')}>{v > 0 ? '+' : ''}{v}%</button>)}</div>
          </div>
          <Field label="Reason (required)">{reasons.length ? <Select id="adj-reason" value={reason} onChange={(ev) => setReason(ev.target.value)} options={[{ value: '', label: 'Choose a reason…' }, ...reasons.map((x) => ({ value: x, label: x }))]} /> : <TextArea id="adj-reason" value={reason} onChange={(ev) => setReason(ev.target.value)} />}</Field>
          <Field label="Note (optional)"><TextArea id="adj-note" className="min-h-[52px]" placeholder="e.g. Geotech borings found rock along the first 2,000 ft." value={note} onChange={(ev) => setNote(ev.target.value)} /></Field>
          <div className="flex flex-wrap items-center gap-2">
            <Button variant="primary" disabled={!ok} onClick={apply}>Apply</Button>
            {!changed && <span className="text-[12px] text-muted">Change a rating or the % to adjust.</span>}
            {changed && !reason && <span className="text-[12px] text-warn">Choose a reason to apply.</span>}
            <span className="text-[12px] text-muted">Then approve or return below.</span>
          </div>
        </div>
        <div className="flex flex-col gap-2 rounded-md border border-line bg-sunken/40 p-3 sm:max-w-sm">
          <div className="eyebrow">Before → after (P80)</div>
          <div className="flex flex-wrap items-baseline gap-2 font-display tabular-nums">
            <span className="text-[17px] text-muted line-through decoration-faint">{money(before)}</span><span aria-hidden className="text-muted">→</span><span className="text-[22px] font-bold">{money(after)}</span>
          </div>
          <div className={cx('text-[12.5px] font-semibold tabular-nums', after > before ? 'text-bad' : after < before ? 'text-good' : 'text-muted')}>{after === before ? 'No change' : `${moneyDelta(after - before)} (${after > before ? '+' : ''}${pct(after / before - 1, 1)})`}</div>
          {r && <div className="text-[11.5px] text-muted">New P10–P90 {money(r.mc.p.P10)} – {money(r.mc.p.P90)}</div>}
          {changedRisks.length > 0 && <div className="text-[11.5px] text-muted">{changedRisks.length} rating{changedRisks.length > 1 ? 's' : ''} changed</div>}
          <VerifyTag>BOE updated — verify</VerifyTag>
        </div>
      </div>
    </Panel>
  )
}
