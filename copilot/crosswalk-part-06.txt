const selKey = (pid: string) => `crosswalk:est-sel:${pid}`
export function openEstimate(pid: string, estimateId: string) {
  try { prefs.setRaw(selKey(pid), estimateId) } catch { /* storage blocked */ }
  go(projectHref(pid, 'estimates'))
}

function defaultProject(s: AppState): string {
  const act = s.projects.filter((p) => p.status === 'active')
  const pick = act.filter((p) => p.stageIndex <= 1).find((p) => quickEligibility(s, p.id).ok) ?? act.find((p) => quickEligibility(s, p.id).ok)
  return pick?.id ?? NEW_NEED
}

/** Build-up of a saved quick estimate (from what was stored when it was priced or adjusted). */
function savedBuildUp(s: AppState, e: Estimate): BuildUpData | undefined {
  const q = e.quick
  if (!q) return
  const t = estimateTotals(e)
  const row = s.historicalCosts.find((h) => h.id === q.historicalCostId)
  return { baseRate: q.baseRate, basis: q.basis.toUpperCase(), baseYear: row?.baseYear, qty: q.inputs.qty, unit: q.inputs.unit, district: q.inputs.district, regionFactor: q.regionFactor, escalationFactor: q.escalationFactor, midpointYear: q.inputs.midpointYear, adjustPct: q.adjustment?.pct, direct: t.lines, loaders: t.loaders, loaderPct: sum(e.loaders.map((l) => l.pct)), contingency: t.contingency, contingencyPct: e.contingency.pct, total: t.total }
}

/** The site drivers a factor estimate was priced with, as short chips. */
function DriverSummary({ inputs }: { inputs: FactorInputs }) {
  const dv = { ...inputs.site, terrain: inputs.terrain }
  const keys = ['soil', 'crossings', 'permitComplexity', 'classLocation', 'hca', 'traffic', 'workHours', 'remoteness'] as const
  return (
    <div><div className="eyebrow mb-1">Site drivers</div>
      <div className="flex flex-wrap gap-1">{keys.map((k) => { const lv = levelOf(dv, k); const off = lv !== DRIVER_INFO[k].baseline; return <span key={k} className={cx('rounded border px-1.5 py-0.5 text-[11.5px]', off ? 'border-ink/40 text-ink' : 'border-line text-muted')}>{DRIVER_INFO[k].label}: <b>{k === 'crossings' ? inputs.site.crossings : levelLabel(k, lv)}</b></span> })}</div>
    </div>
  )
}

export type Unit = 1 | 1000

export function fmtUnit(v: number, unit: Unit): string {
  if (!v) return '–'
  const x = v / unit
  const neg = x < 0
  const a = Math.abs(x)
  const s = a.toLocaleString('en-US', { maximumFractionDigits: unit === 1 ? 0 : a < 100 ? 1 : 0 })
  return neg ? `(${s})` : s
}
const rawUnit = (v: number, unit: Unit) => (v ? String(Math.round((v / unit) * 1000) / 1000) : '')

export interface DrillTarget { lineId?: string; fnId?: string; months: string[]; amount: Money }

export function LineGrid(props: LineGridProps) {
  const { s, groups, lines, held, editable, budget, budgetContingency, actual, actualMonths, actualGrid, futureMonths, closedThrough, collapsed, onToggleYear, pastCollapsed, onTogglePastYear, expanded, onToggleFn, profiles, unit, onChange, onHeld, onProfile, onLineTools, onDrill } = props
  const ref = useRef<HTMLDivElement>(null)
  const pastCols = useMemo(() => gridCols(actualMonths, pastCollapsed), [actualMonths, pastCollapsed])
  const cols = useMemo(() => gridCols(futureMonths, collapsed), [futureMonths, collapsed])
  const editRows = groups.filter((g) => expanded.has(g.fn.id)).flatMap((g) => g.lines.map((l) => l.id))
  const fnOfLine = (lid: string) => groups.find((g) => g.lines.some((l) => l.id === lid))?.fn.id
  const val = (ids: string[], c: GridCol) => sum(ids.flatMap((id) => c.months.map((m) => lines[id]?.[m] || 0)))
  const pastVal = (ids: string[], c: GridCol) => sum(ids.flatMap((id) => c.months.map((m) => actualGrid[id]?.[m] || 0)))
  const allIds = groups.flatMap((g) => g.lines.map((l) => l.id))

  const focusCell = (r: number, c: number) => {
    const el = ref.current?.querySelector<HTMLInputElement>(`input[data-r="${r}"][data-c="${c}"]`)
    if (el) { el.focus(); el.select() }
  }
  const commit = (lid: string, ci: number, v: number, label: string) => {
    const col = cols[ci]
    if (!col) return
    if (v === val([lid], col)) return
    onChange(col.kind === 'year' ? setYearTotal(lines, lid, col.months, v) : setCell(lines, lid, col.key, v), `${label} ${col.kind === 'year' ? col.key + ' total' : shortMonth(col.key)}`, [fnOfLine(lid) ?? ''])
  }
  const paste = (r: number, c: number, text: string) => {
    const block = parseClipboard(text)
    const res = applyPaste(lines, editRows, cols.map((x) => x.key), r, c, block, unit, futureMonths)
    const touched = editRows.slice(r, r + block.length).map(fnOfLine).filter(Boolean) as string[]
    onChange(res.spread, `pasted ${res.filled} cells`, [...new Set(touched)])
    toast(res.rejected.length ? `Pasted ${res.filled} cells · ${res.rejected.length} not numbers skipped` : `Pasted ${res.filled} cells`, res.rejected.length ? 'bad' : 'good')
  }

  const colTotals = cols.map((c) => val(allIds, c))
  const pastTotals = pastCols.map((c) => pastVal(allIds, c))
  const budTotal = sum(allIds.map((id) => budget[id] || 0)) + budgetContingency
  const actTotal = sum(allIds.map((id) => actual[id] || 0))
  const etcTotal = sum(allIds.map((id) => rowTotal(lines, id))) + held
  const eacTotal = actTotal + etcTotal

  const yearGroups = (cs: GridCol[]) => {
    const g: { year: string; span: number; collapsed: boolean }[] = []
    for (const c of cs) {
      const y = c.key.slice(0, 4)
      const last = g[g.length - 1]
      if (last && last.year === y) last.span++
      else g.push({ year: y, span: 1, collapsed: c.kind === 'year' })
    }
    return g
  }
  const pastGroups = yearGroups(pastCols)
  const futGroups = yearGroups(cols)
  const stickyTh = 'sticky left-0 z-[3] bg-sunken'

function CellInput({ value, unit, r, c, label, warn, onCommit, onPaste, onNav, maxR, maxC }: {
  value: number; unit: Unit; r: number; c: number; label: string; warn?: boolean
  onCommit: (v: number) => void; onPaste?: (text: string) => void; onNav: (r: number, c: number) => void; maxR: number; maxC: number
}) {
  const [txt, setTxt] = useState<string | null>(null)
  const done = () => {
    if (txt === null) return
    const v = parseMoney(txt, unit)
    setTxt(null)
    if (v === null) { toast(`"${txt}" is not an amount — try 125000, $1.2M or 450K`, 'bad'); return }
    if (v !== value) onCommit(v)
  }
  const key = (e: React.KeyboardEvent<HTMLInputElement>) => {
    const el = e.currentTarget
    const len = el.value.length
    const all = el.selectionStart === 0 && el.selectionEnd === len
    const go = (dr: number, dc: number) => {
      e.preventDefault()
      const nr = Math.max(0, Math.min(maxR, r + dr)), nc = Math.max(0, Math.min(maxC, c + dc))
      if (nr === r && nc === c) return
      el.blur()
      onNav(nr, nc)
    }
    switch (e.key) {
      case 'Enter': return e.shiftKey ? go(-1, 0) : go(1, 0)
      case 'ArrowDown': return go(1, 0)
      case 'ArrowUp': return go(-1, 0)
      case 'ArrowLeft': if (all || (el.selectionStart === 0 && el.selectionEnd === 0)) go(0, -1); return
      case 'ArrowRight': if (all || (el.selectionStart === len && el.selectionEnd === len)) go(0, 1); return
      case 'Escape': e.preventDefault(); setTxt(rawUnit(value, unit)); setTimeout(() => el.select()); return
    }
  }
  const editing = txt !== null
  return (
    <input
      data-r={r} data-c={c} aria-label={label} inputMode="decimal" spellCheck={false}
      value={editing ? txt! : value ? fmtUnit(value, unit) : ''}
      placeholder="–"
      onFocus={(e) => { setTxt(rawUnit(value, unit)); const t = e.currentTarget; setTimeout(() => t.select()) }}
      onChange={(e) => setTxt(e.target.value)}
      onBlur={done}
      onKeyDown={key}
      onPaste={(e) => { const t = e.clipboardData.getData('text'); if (onPaste && isBlock(t)) { e.preventDefault(); setTxt(null); onPaste(t) } }}
      className={cx('h-7 w-full min-w-[76px] rounded-none border border-transparent bg-transparent px-2 text-right text-[12px] tabular-nums outline-none placeholder:text-faint hover:border-line focus:border-link focus:bg-surface focus:ring-1 focus:ring-link/40',
        value < 0 && !editing && 'bg-bad/10 text-bad', warn && !editing && 'bg-warn/15')}
    />
  )
}

/**
 * Parse what a person types (or pastes) into a money cell: `1234`, `$1,234`, `1.2M`, `450k`, `(5,000)`, `-5k`.
 * Blank means zero. Returns null when the text is not a number. `scale` applies to plain numbers only
 * (1000 when the grid shows $K); a K/M/B suffix is always absolute.
 */
export function parseMoney(raw: string, scale = 1): number | null {
  let t = (raw ?? '').trim()
  if (!t || t === '-' || t === '—' || t === '–') return 0
  let neg = false
  if (/^\(.*\)$/.test(t)) { neg = true; t = t.slice(1, -1) }
  t = t.replace(/[$,\s]/g, '').replace(/^[−–]/, '-')
  if (t.startsWith('-')) { neg = !neg; t = t.slice(1) }
  if (t.startsWith('$')) t = t.slice(1)
  const m = /^(\d+\.?\d*|\.\d+)([kmb])?$/i.exec(t)
  if (!m) return null
  const mult = ({ k: 1e3, m: 1e6, b: 1e9 } as Record<string, number>)[(m[2] ?? '').toLowerCase()] ?? scale
  const v = Math.round(parseFloat(m[1]) * mult)
  return neg ? -v : v
}

/** A tab-separated block copied from Excel → rows of cells. Trailing blank line dropped. */

export const cloneSpread = (s?: Spread): Spread => Object.fromEntries(Object.entries(s ?? {}).map(([k, v]) => [k, { ...v }]))

export function setCell(spread: Spread, fnId: string, month: string, v: Money): Spread {

  const row = (out[fnId] ||= {})
  if (v) row[month] = Math.round(v)

export const rowTotal = (spread: Spread, fnId: string) => sum(Object.values(spread[fnId] ?? {}))

/**
 * Paste a block at (row r0, column c0). `rows` are function ids; `cols` are the grid's editable columns in
 * order — a month key ('YYYY-MM') or a collapsed year ('YYYY', which sets that year's total across `months`).
 */
export function applyPaste(spread: Spread, rows: string[], cols: string[], r0: number, c0: number, block: string[][], scale = 1, months: string[] = []): { spread: Spread; filled: number; rejected: string[] } {
  let out = cloneSpread(spread)
  let filled = 0
  const rejected: string[] = []
  block.forEach((line, i) => {
    const fn = rows[r0 + i]
    if (!fn) return
    line.forEach((cell, j) => {
      const col = cols[c0 + j]
      if (!col) return
      const v = parseMoney(cell, scale)
      if (v === null) { rejected.push(cell); return }
      out = col.length === 4 ? setYearTotal(out, fn, months.filter((m) => m.startsWith(col)), v) : setCell(out, fn, col, v)
      filled++
    })
  })
  return { spread: out, filled, rejected }
}

/** Set a row's total for one year: keeps the year's monthly shape when it has one, else spreads evenly. */
export function setYearTotal(spread: Spread, fnId: string, yearMonths: string[], v: Money): Spread {
  const row = spread[fnId] ?? {}
  const shape = yearMonths.map((m) => row[m] || 0)
  const vals = shape.some((x) => x > 0) ? distribute(Math.round(v), shape) : distribute(Math.round(v), yearMonths.map(() => 1))
  let out = cloneSpread(spread)
  yearMonths.forEach((m, i) => { out = setCell(out, fnId, m, vals[i]) })
  return out
}

/** Integer split of `total` by weights that sums exactly to total. */
export function distribute(total: Money, weights: number[]): Money[] {
  const W = sum(weights.map((w) => Math.max(0, w)))
  if (!weights.length) return []
  const ws = W > 0 ? weights.map((w) => Math.max(0, w)) : weights.map(() => 1)
  const T = W > 0 ? W : weights.length
  let acc = 0
  return ws.map((w, i) => {
    const v = i === ws.length - 1 ? total - acc : Math.round((total * w) / T)
    acc += v
    return v
  })
}

/** Spread `amount` from `from` to `to` (inclusive) by a curve. Replaces the whole row. */
export function spreadRow(spread: Spread, fnId: string, months: string[], amount: Money, method: SpreadMethod, from: string, to: string): Spread {
  const range = months.filter((m) => m >= from && m <= to)
  const out = cloneSpread(spread)
  const vals = range.length ? spreadAmount(Math.round(amount), range, method) : {}
  out[fnId] = Object.fromEntries(Object.entries(vals).filter(([, v]) => v))
  return out
}

export function clearRow(spread: Spread, fnId: string): Spread {
  const out = cloneSpread(spread)
  out[fnId] = {}
  return out
}

/** Set a row's ETC to `target`: keep the row's shape when it has one, else spread by `method` over `months`. */
export function scaleRow(spread: Spread, fnId: string, months: string[], target: Money, method: SpreadMethod): Spread {
  const row = spread[fnId] ?? {}
  const keys = Object.keys(row).filter((k) => row[k]).sort()
  const out = cloneSpread(spread)
  if (keys.length && rowTotal(spread, fnId) > 0) {
    const vals = distribute(Math.round(target), keys.map((k) => row[k]))
    out[fnId] = Object.fromEntries(keys.map((k, i) => [k, vals[i]]).filter(([, v]) => v))
  } else {
    out[fnId] = Object.fromEntries(Object.entries(spreadAmount(Math.round(target), months, method)).filter(([, v]) => v))
  }
  return out
}

}

export const statusOf = (f: Forecast): VersionStatus => f.status ?? (f.locked ? 'approved' : 'draft')
export const isEditable = (f: Forecast) => !f.locked && (statusOf(f) === 'draft' || statusOf(f) === 'returned')

/** Newest first; drafts after a same-day approved version (matches latestForecast). */
export function versionsOf(s: AppState, pid: string): Forecast[] {
  return s.forecasts.filter((f) => f.projectId === pid).sort((a, b) => b.date.localeCompare(a.date) || (statusOf(a) === 'draft' ? -1 : 1))
}

const monthBefore = (m: string) => monthKey(addMonths(m + '-01', -1))
const monthAfter = (m: string) => monthKey(addMonths(m + '-01', 1))

/** Last month treated as actuals for a version: its actualsThrough, or the month before its first ETC month. */
export function versionCutoff(s: AppState, f: Forecast): string {
  const months = Object.values(f.spread ?? {}).flatMap((r) => Object.keys(r)).sort()
  const a = f.actualsThrough ?? actualsThrough(s, f.projectId)
  const b = months.length ? monthBefore(months[0]) : a
  return a > b ? a : b
}

/**
 * Actuals by function through a cutoff month. Periods without a recorded split are allocated by budget share in
 * the engine, which can leave a few dollars of rounding; the residual goes to the largest function so the
 * functions always sum to the ERP actual.
 */
export function actualsByFunction2(s: AppState, pid: string, through?: string): Record<string, Money> {
  const m = actualsByFunctionMonth(s, pid)
  const out = Object.fromEntries(Object.entries(m).map(([fn, row]) => [fn, sum(Object.entries(row).filter(([k]) => !through || k <= through).map(([, v]) => v))]))
  const truth = sum(s.costPeriods.filter((c) => c.projectId === pid && (!through || c.period <= through)).map((c) => c.actual))
  const resid = truth - sum(Object.values(out))
  if (resid) {
    const big = Object.keys(out).sort((a, b) => out[b] - out[a])[0]
    if (big) out[big] += resid
  }
  return out
}

/** Calendar-year EAC: actuals by year + ETC by year (+ contingency held in the final year). */
export function byYearOf(s: AppState, pid: string, spread: Spread, held: Money): Record<string, Money> {
  const out: Record<string, Money> = {}
  for (const c of s.costPeriods.filter((x) => x.projectId === pid)) out[c.period.slice(0, 4)] = (out[c.period.slice(0, 4)] || 0) + c.actual
  for (const [y, v] of Object.entries(spreadByYear(spread))) out[y] = (out[y] || 0) + v
  const ys = Object.keys(out).sort()
  if (held) { if (ys.length) out[ys[ys.length - 1]] += held; else out[s.settings.asOf.slice(0, 4)] = held }
  return out
}

export type SlipMode = 'net' | 'all' | 'drop'

/**
 * Carry a spread forward into a new actuals period. ETC sitting in months that are now closed is
 * re-spread across the row's remaining months (in proportion to what is already there, else evenly).
 * `net`: only the part not already consumed by that function's actuals in those months moves forward.
 * `all`: the whole amount moves forward. `drop`: stale ETC is removed (the actuals replace it).
 */
export function carryForward(s: AppState, p: Project, src: Spread, srcCutoff: string, slip: SlipMode = 'net'): { spread: Spread; moved: Money; movedBy: Record<string, Money> } {
  return carryForwardGrid(s, p, src, srcCutoff, slip, actualsByFunctionMonth(s, p.id))
}
/** Same as carryForward at forecast line level (actuals by line from SAP / the mapping table). */
export function carryForwardLines(s: AppState, p: Project, src: Spread, srcCutoff: string, slip: SlipMode = 'net'): { spread: Spread; moved: Money; movedBy: Record<string, Money> } {
  return carryForwardGrid(s, p, src, srcCutoff, slip, actualsByLineMonth(s, p.id))
}
function carryForwardGrid(s: AppState, p: Project, src: Spread, srcCutoff: string, slip: SlipMode, act: Spread): { spread: Spread; moved: Money; movedBy: Record<string, Money> } {
  const through = actualsThrough(s, p.id)
  const months = forecastMonths(s, p)
  const out: Spread = {}
  const movedBy: Record<string, Money> = {}
  let moved = 0
  for (const [fn, row] of Object.entries(src)) {
    const future: Record<string, Money> = {}
    let stale = 0
    for (const [m, v] of Object.entries(row)) {
      if (m > through) future[m] = v
      else stale += v
    }
    if (slip === 'drop') stale = 0
    if (stale && slip === 'net') {
      const spent = sum(Object.entries(act[fn] ?? {}).filter(([m]) => m > srcCutoff && m <= through).map(([, v]) => v))
      stale = Math.max(0, stale - spent)
    }
    if (stale) {
      const keys = Object.keys(future).filter((k) => future[k] > 0).sort()
      if (keys.length) {
        const add = distribute(stale, keys.map((k) => future[k]))
        keys.forEach((k, i) => { future[k] += add[i] })
      } else {
        const add = spreadAmount(stale, months, 'even')
        for (const [k, v] of Object.entries(add)) future[k] = (future[k] || 0) + v
      }
      moved += stale
      movedBy[fn] = stale
    }
    out[fn] = Object.fromEntries(Object.entries(future).filter(([, v]) => v))
  }
  return { spread: out, moved, movedBy }
}

/**
 * A new draft version copied from `src` (as of today) at forecast-line level, with stale ETC carried forward
 * and each function's profile carried over (window clamped to the open months; marked adjusted when ETC moved).
 * With no source, the draft spreads each line's remaining budget by its function's typical timing.
 */
export function copyVersion(s: AppState, p: Project, src: Forecast | undefined, opts: { byId: string; note?: string; slip?: SlipMode }): { forecast: Forecast; moved: Money } {
  const held = src?.contingencyHeld ?? 0
  const months = forecastMonths(s, p)
  const hasSrc = !!src && (Object.keys(src.lineSpread ?? {}).length > 0 || Object.keys(src.spread ?? {}).length > 0)
  let lines: Spread, moved = 0, profiles: Record<string, FnProfile>
  if (hasSrc) {
    const r = carryForwardLines(s, p, stripZeros(lineGridOf(s, p, src)), versionCutoff(s, src!), opts.slip ?? 'net')
    lines = r.spread
    moved = r.moved
    const movedFns = new Set(Object.keys(r.movedBy).map((l) => functionOfLine(s, l)))
    profiles = carryProfiles(src!.profiles, months, movedFns)
  } else {
    const d = defaultLineSpread(s, p)
    lines = stripZeros(d.lineSpread)
    profiles = d.profiles
  }
  const forecast: Forecast = {
    id: uid('fc'), projectId: p.id, date: s.settings.asOf, eac: 0, byYear: {},
    note: opts.note ?? '', byId: opts.byId, locked: false, spread: {}, lineSpread: lines, profiles, contingencyHeld: held,
    actualsThrough: actualsThrough(s, p.id), status: 'draft',
  }
  syncForecast(s, forecast)
  return { forecast, moved }
}

/** Carry function profiles into a new period: windows start no earlier than the first open month. */
export function carryProfiles(src: Record<string, FnProfile> | undefined, months: string[], adjustedFns: Set<string | undefined> = new Set()): Record<string, FnProfile> {
  const out: Record<string, FnProfile> = {}
  const first = months[0]
  for (const [fn, pr] of Object.entries(src ?? {})) {
    const start = first && pr.start < first ? first : pr.start
    const end = pr.end < start ? start : pr.end
    out[fn] = { ...pr, start, end, adjusted: pr.adjusted || adjustedFns.has(fn) }
  }
  return out
}

/** Drop zero cells (profiles write explicit zeros outside their window). */
export function stripZeros(g?: Spread): Spread {
  return Object.fromEntries(Object.entries(g ?? {}).map(([k, row]) => [k, Object.fromEntries(Object.entries(row).filter(([, v]) => v))]))
}

/** Write a spread back to a forecast record (draft of the store) with recomputed EAC and calendar years. */
export function applySpread(d: AppState, f: Forecast, spread: Spread, held: Money, note: string) {
  f.spread = cloneSpread(spread)
  f.contingencyHeld = Math.round(held)
  f.note = note
  f.eac = eacOf(d, f.projectId, spread, held)
  f.byYear = byYearOf(d, f.projectId, spread, held)
}

/** Audit text for a save: which functions' ETC moved and by how much. */
export function diffSummary(fns: { id: string; code: string; name: string }[], before: Spread, after: Spread, heldBefore: Money, heldAfter: Money): string {
  const parts: string[] = []
  let cells = 0
  const ids = Array.from(new Set([...Object.keys(before), ...Object.keys(after)]))
  for (const id of ids) {
    const a = before[id] ?? {}, b = after[id] ?? {}
    for (const m of new Set([...Object.keys(a), ...Object.keys(b)])) if ((a[m] || 0) !== (b[m] || 0)) cells++
    const d = rowTotal(after, id) - rowTotal(before, id)
    const f = fns.find((x) => x.id === id)
    if (d) parts.push(`${f?.code ?? id} ${f?.name ?? ''} ${signed(d)}`.replace(/\s+/g, ' '))
  }
  if (heldAfter !== heldBefore) parts.push(`contingency held ${signed(heldAfter - heldBefore)}`)
  const etc = spreadTotal(after) - spreadTotal(before) + heldAfter - heldBefore
  if (!parts.length && !cells) return 'No changes to the spread'
  return `${cells} cell${cells === 1 ? '' : 's'} changed; ETC ${signed(etc)}${parts.length ? ' · ' + parts.join(', ') : ' (re-phased only)'}`
}
const signed = (n: number) => (n > 0 ? '+' : n < 0 ? '−' : '') + '$' + Math.abs(Math.round(n)).toLocaleString('en-US')

// ───────────────────────────── Function lists (including functions retired since a version was made)

export interface FnRef { id: string; code: string; name: string; retired: boolean; timing: WbsFunction['timing'] }
export function fnRefs(s: AppState, ...spreads: (Spread | undefined)[]): FnRef[] {
  const out: FnRef[] = s.settings.wbsFunctions.map((f) => ({ id: f.id, code: f.code, name: f.name, retired: false, timing: f.timing }))
  for (const sp of spreads) for (const id of Object.keys(sp ?? {})) {
    if (!out.some((x) => x.id === id) && rowTotal(sp!, id)) out.push({ id, code: id.replace(/^wbs-/, ''), name: 'Retired function', retired: true, timing: 'flat' })
  }
  return out
}

// ───────────────────────────── Compare two versions

export interface CompareRow { id: string; code: string; name: string; etcA: Money; etcB: Money; eacA: Money; eacB: Money; delta: Money }
export function compareVersions(s: AppState, a: Forecast, b: Forecast) {
  const fns = fnRefs(s, a.spread, b.spread)
  const actA = actualsByFunction2(s, a.projectId, versionCutoff(s, a))
  const actB = actualsByFunction2(s, b.projectId, versionCutoff(s, b))
  const ea = spreadByFunction(a.spread), eb = spreadByFunction(b.spread)
  const rows: CompareRow[] = fns.map((f) => {
    const eacA = (actA[f.id] || 0) + (ea[f.id] || 0), eacB = (actB[f.id] || 0) + (eb[f.id] || 0)
    return { id: f.id, code: f.code, name: f.name, etcA: ea[f.id] || 0, etcB: eb[f.id] || 0, eacA, eacB, delta: eacB - eacA }
  })
  const ma = spreadByMonth(a.spread), mb = spreadByMonth(b.spread)
  const months = Array.from(new Set([...Object.keys(ma), ...Object.keys(mb)])).sort().map((m) => ({ month: m, a: ma[m] || 0, b: mb[m] || 0, delta: (mb[m] || 0) - (ma[m] || 0) }))
  const movers = rows.filter((r) => r.delta).sort((x, y) => Math.abs(y.delta) - Math.abs(x.delta)).slice(0, 5)
  return { rows, months, movers, heldA: a.contingencyHeld ?? 0, heldB: b.contingencyHeld ?? 0, eacA: a.eac, eacB: b.eac }
}

// ───────────────────────────── Monthly cycle

/** The cycle opens on this day, after the prior month's actuals close in the ERP. */
export const CYCLE_OPEN_DAY = 21
/** Which monthly cycle a version dated `date` belongs to ('YYYY-MM'). */
export const cycleOf = (date: string) => (Number(date.slice(8, 10)) >= CYCLE_OPEN_DAY ? monthAfter(date.slice(0, 7)) : date.slice(0, 7))
export const currentCycle = (s: AppState) => cycleOf(s.settings.asOf)

export const inCycleScope = (s: AppState, p: Project) => p.status === 'active' && !!activeBudget(s, p.id)

export function cycleVersion(s: AppState, pid: string, cycle: string): Forecast | undefined {
  return versionsOf(s, pid).find((f) => cycleOf(f.date) === cycle)
}
export function cycleStatusOf(f?: Forecast): CycleStatus {
  if (!f) return 'not-started'
  const st = statusOf(f)
  return st === 'submitted' ? 'in-review' : st
}
/** Latest approved version other than `excludeId`. */
export function priorApproved(s: AppState, pid: string, excludeId?: string): Forecast | undefined {
  return versionsOf(s, pid).find((f) => f.id !== excludeId && statusOf(f) === 'approved')
}

export interface CycleRow { p: Project; status: CycleStatus; version?: Forecast; latest?: Forecast; prior?: Forecast; eac: Money; delta?: Money; budget: Money }
export function cycleRows(s: AppState, cycle = currentCycle(s)): CycleRow[] {
  return s.projects.filter((p) => inCycleScope(s, p)).map((p) => {
    const version = cycleVersion(s, p.id, cycle)
    const latest = versionsOf(s, p.id)[0]
    const prior = priorApproved(s, p.id, version?.id)
    const eac = (version ?? latest)?.eac ?? 0
    return { p, status: cycleStatusOf(version), version, latest, prior, eac, delta: prior && version ? version.eac - prior.eac : undefined, budget: budgetByFunction(s, p.id).total }
  })
}

/**
 * Open the cycle: a draft for every in-scope project that has no version this cycle, copied from its
 * latest version (stale ETC carried forward). Mutates `d`; returns the codes it created.
 */
export function startCycle(d: AppState, byId: string, slip: SlipMode = 'net'): { codes: string[]; moved: Money } {
  const cycle = currentCycle(d)
  const codes: string[] = []
  let moved = 0
  for (const p of d.projects.filter((x) => inCycleScope(d, x))) {
    if (cycleVersion(d, p.id, cycle)) continue
    const src = versionsOf(d, p.id)[0]
    const { forecast, moved: mv } = copyVersion(d, p, src, { byId: p.team.pm ?? byId, slip, note: `${monthLabel(cycle)} cycle — opened from ${src ? `the ${src.date} version` : 'remaining budget'}` })
    d.forecasts.push(forecast)
    codes.push(p.code)
    moved += mv
  }
  return { codes, moved }
}
const MONTHS = ['January', 'February', 'March', 'April', 'May', 'June', 'July', 'August', 'September', 'October', 'November', 'December']
export const monthLabel = (k: string) => `${MONTHS[Number(k.slice(5, 7)) - 1]} ${k.slice(0, 4)}`

// ───────────────────────────── Portfolio roll-up (function × calendar year)

export type RollupBasis = 'latest' | 'approved'
export interface Rollup {
  years: string[]
  fns: FnRef[]
  actual: Record<string, Record<string, Money>> // fn → year → actuals
  forecast: Record<string, Record<string, Money>> // fn → year → ETC
  contingency: Record<string, Money> // year → held contingency
  projects: { p: Project; version: Forecast }[]
}
export function versionFor(s: AppState, pid: string, basis: RollupBasis): Forecast | undefined {
  return basis === 'approved' ? priorApproved(s, pid) : versionsOf(s, pid)[0]
}
export function rollup(s: AppState, projects: Project[], basis: RollupBasis = 'latest'): Rollup {
  const picked = projects.map((p) => ({ p, version: versionFor(s, p.id, basis)! })).filter((x) => x.version)
  const fns = fnRefs(s, ...picked.map((x) => x.version.spread))
  const actual: Rollup['actual'] = {}, forecast: Rollup['forecast'] = {}, contingency: Rollup['contingency'] = {}
  const years = new Set<string>()
  const add = (t: Record<string, Record<string, Money>>, fn: string, y: string, v: Money) => { if (!v) return; (t[fn] ||= {})[y] = (t[fn][y] || 0) + v; years.add(y) }
  for (const { p, version } of picked) {
    const cut = versionCutoff(s, version)
    const pys = new Set<string>()
    for (const [fn, row] of Object.entries(actualsByFunctionMonth(s, p.id))) for (const [m, v] of Object.entries(row)) if (m <= cut) { add(actual, fn, m.slice(0, 4), v); pys.add(m.slice(0, 4)) }
    for (const [fn, row] of Object.entries(version.spread ?? {})) for (const [m, v] of Object.entries(row)) { add(forecast, fn, m.slice(0, 4), v); if (v) pys.add(m.slice(0, 4)) }
    const held = version.contingencyHeld ?? 0
    if (held) { const y = [...pys].sort().pop() ?? s.settings.asOf.slice(0, 4); contingency[y] = (contingency[y] || 0) + held; years.add(y) }
  }
  return { years: [...years].sort(), fns, actual, forecast, contingency, projects: picked }
}
export const rollupCell = (r: Rollup, fn: string, y: string) => (r.actual[fn]?.[y] || 0) + (r.forecast[fn]?.[y] || 0)
export const rollupFnTotal = (r: Rollup, fn: string) => sum(r.years.map((y) => rollupCell(r, fn, y)))
export const rollupYearTotal = (r: Rollup, y: string) => sum(r.fns.map((f) => rollupCell(r, f.id, y))) + (r.contingency[y] || 0)

/** Finance CSV rows: function × year with actual / forecast split and totals. */
export function rollupCsvRows(r: Rollup): Record<string, unknown>[] {
  const rows: Record<string, unknown>[] = r.fns.map((f) => ({
    'WBS code': f.code, 'WBS function': f.name + (f.retired ? ' (retired)' : ''),
    ...Object.fromEntries(r.years.map((y) => [y, rollupCell(r, f.id, y)])),
    Total: rollupFnTotal(r, f.id),
  }))
  rows.push({ 'WBS code': '990', 'WBS function': 'Contingency held', ...Object.fromEntries(r.years.map((y) => [y, r.contingency[y] || 0])), Total: sum(Object.values(r.contingency)) })
  rows.push({ 'WBS code': 'TOTAL', 'WBS function': 'Total', ...Object.fromEntries(r.years.map((y) => [y, rollupYearTotal(r, y)])), Total: sum(r.years.map((y) => rollupYearTotal(r, y))) })
  return rows
}
/** Long-format detail for finance pivots: project × function × year, actual vs forecast. */
export function rollupDetailRows(s: AppState, r: Rollup): Record<string, unknown>[] {
  const out: Record<string, unknown>[] = []
  for (const { p, version } of r.projects) {
    const cut = versionCutoff(s, version)
    const act = actualsByFunctionMonth(s, p.id)
    const prog = s.programs.find((x) => x.id === p.programId)?.code ?? ''
    for (const f of r.fns) {
      const ys = new Set<string>()
      const a: Record<string, Money> = {}, e: Record<string, Money> = {}
      for (const [m, v] of Object.entries(act[f.id] ?? {})) if (m <= cut) { a[m.slice(0, 4)] = (a[m.slice(0, 4)] || 0) + v; ys.add(m.slice(0, 4)) }
      for (const [m, v] of Object.entries(version.spread?.[f.id] ?? {})) { e[m.slice(0, 4)] = (e[m.slice(0, 4)] || 0) + v; ys.add(m.slice(0, 4)) }
      for (const y of [...ys].sort()) if (a[y] || e[y]) out.push({ Project: p.code, Program: prog, 'Forecast date': version.date, Status: statusOf(version), 'WBS code': f.code, 'WBS function': f.name, Year: y, Actual: a[y] || 0, Forecast: e[y] || 0, Total: (a[y] || 0) + (e[y] || 0) })
    }
    if (version.contingencyHeld) out.push({ Project: p.code, Program: prog, 'Forecast date': version.date, Status: statusOf(version), 'WBS code': '990', 'WBS function': 'Contingency held', Year: '', Actual: 0, Forecast: version.contingencyHeld, Total: version.contingencyHeld })
  }
  return out
}

// ───────────────────────────── Grid export

/** Functions × months CSV for one version (actuals columns for past months optional). */
export function gridCsvRows(s: AppState, p: Project, f: Forecast, months: string[]): Record<string, unknown>[] {
  const fns = fnRefs(s, f.spread)
  const bud = budgetByFunction(s, p.id)
  const act = actualsByFunction2(s, p.id, versionCutoff(s, f))
  const actTotal = sum(Object.values(act))
  const rows: Record<string, unknown>[] = fns.map((fn) => {
    const etc = rowTotal(f.spread ?? {}, fn.id)
    const eac = (act[fn.id] || 0) + etc
    return { 'WBS code': fn.code, 'WBS function': fn.name, Budget: bud.byFn[fn.id] || 0, 'Actual to date': act[fn.id] || 0, ...Object.fromEntries(months.map((m) => [m, f.spread?.[fn.id]?.[m] || 0])), ETC: etc, EAC: eac, Variance: (bud.byFn[fn.id] || 0) - eac }
  })
  const held = f.contingencyHeld ?? 0
  rows.push({ 'WBS code': '990', 'WBS function': 'Contingency held', Budget: bud.contingency, 'Actual to date': 0, ...Object.fromEntries(months.map((m) => [m, 0])), ETC: held, EAC: held, Variance: bud.contingency - held })
  const byM = spreadByMonth(f.spread)
  const etcT = spreadTotal(f.spread) + held
  rows.push({ 'WBS code': 'TOTAL', 'WBS function': 'Total', Budget: bud.total, 'Actual to date': actTotal, ...Object.fromEntries(months.map((m) => [m, byM[m] || 0])), ETC: etcT, EAC: actTotal + etcT, Variance: bud.total - actTotal - etcT })
  return rows
}

/** Same values cell by cell (zero and missing are equal). */
export function sameSpread(a?: Spread, b?: Spread): boolean {
  const A = a ?? {}, B = b ?? {}
  for (const fn of new Set([...Object.keys(A), ...Object.keys(B)])) {
    const x = A[fn] ?? {}, y = B[fn] ?? {}
    for (const m of new Set([...Object.keys(x), ...Object.keys(y)])) if ((x[m] || 0) !== (y[m] || 0)) return false
  }
  return true
}

/** 'Nov ’26' style short month label for charts. */
export const shortMonth = (k: string) => `${['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'][Number(k.slice(5, 7)) - 1]} '${k.slice(2, 4)}`

/** Grid columns: months, with any collapsed year shown as one year-total column. */
export interface GridCol { key: string; kind: 'month' | 'year'; months: string[] }
export function gridCols(months: string[], collapsed: Set<string> | string[]): GridCol[] {
  const c = new Set(collapsed)
  const out: GridCol[] = []
  for (const m of months) {
    const y = m.slice(0, 4)
    if (c.has(y)) {
      const last = out[out.length - 1]
      if (last?.kind === 'year' && last.key === y) last.months.push(m)
      else out.push({ key: y, kind: 'year', months: [m] })
    } else out.push({ key: m, kind: 'month', months: [m] })
  }
  return out
}

// ───────────────────────────── Forecast line items (line × month is what people edit; functions roll up)

export interface LineRef { id: string; code: string; name: string; fnId: string; inactive: boolean }

/** Active lines in WBS order, plus any inactive line that still carries ETC in one of the grids. */
export function lineRefs(s: AppState, ...grids: (Spread | undefined)[]): LineRef[] {
  const order = new Map(s.settings.wbsFunctions.map((f, i) => [f.id, i]))
  const ids = new Set(s.settings.forecastLines.filter((l) => l.active).map((l) => l.id))
  for (const g of grids) for (const [id, row] of Object.entries(g ?? {})) if (sum(Object.values(row))) ids.add(id)
  return [...ids].map((id): LineRef => {
    const l = lineById(s, id)
    return l ? { id, code: l.code, name: l.name, fnId: l.functionId, inactive: !l.active } : { id, code: id, name: 'Removed line', fnId: '', inactive: true }
  }).sort((a, b) => (order.get(a.fnId) ?? 99) - (order.get(b.fnId) ?? 99) || a.code.localeCompare(b.code))
}

export interface FnGroup { fn: FnRef; lines: LineRef[] }
/** Functions with their lines, in WBS order (functions without lines are kept so the table always shows nine). */
export function fnGroups(s: AppState, lines: LineRef[]): FnGroup[] {
  const groups: FnGroup[] = fnRefs(s).map((fn) => ({ fn, lines: lines.filter((l) => l.fnId === fn.id) }))
  const orphans = lines.filter((l) => !groups.some((g) => g.fn.id === l.fnId))
  if (orphans.length) groups.push({ fn: { id: 'fn-orphan', code: '—', name: 'Lines without a function', retired: true, timing: 'flat' }, lines: orphans })
  return groups
}

/** Actuals by line through a cutoff month (exact: line actuals sum to the ERP actual). */
export function lineActuals(s: AppState, pid: string, through?: string): Record<string, Money> {
  return Object.fromEntries(Object.entries(actualsByLineMonth(s, pid)).map(([k, row]) => [k, sum(Object.entries(row).filter(([m]) => !through || m <= through).map(([, v]) => v))]))
}

/** ETC/EAC by function (engine) with actuals cut off at a version's actuals month, so older versions read as they were made. */
export function etcEacAt(s: AppState, p: Project, f: Forecast, through?: string) {
  const st = through ? { ...s, costPeriods: s.costPeriods.filter((c) => c.projectId !== p.id || c.period <= through) } : s
  return etcEacByFunction(st, p, f)
}

/** Spread the edited line grid back into a stored forecast (draft of the store) and re-derive spread, EAC and years. */
export function applyLines(d: AppState, f: Forecast, lines: Spread, held: Money, note: string, profiles: Record<string, FnProfile>) {

/** Mark a function's profile as hand-adjusted (no-op when it has no profile or already is). */

/**
 * Apply a profile to one function: its ETC (`total`, default = current) goes to its lines by each line's current share
 * and is phased by the profile between start and end (engine applyProfile). Returns the new grid and profile entry.
 */
export function applyFnProfile(s: AppState, p: Project, lines: Spread, fnId: string, prof: Pick<FnProfile, 'profileId' | 'start' | 'end'>, total?: Money, at = new Date().toISOString()): { lines: Spread; profile: FnProfile } {

}

export interface PortfolioFnRow {
  fnId: string; code: string; name: string
  budget: Money; actual: Money; etc: Money; eac: Money; vac: Money
  lines: { id: string; code: string; name: string; budget: Money; actual: Money; etc: Money; eac: Money; vac: Money }[]
  over: { p: Project; vac: Money }[] // projects whose EAC exceeds budget in this function, worst first
}
export function portfolioEtcEac(s: AppState, projects: Project[], basis: RollupBasis = 'latest') {
  const picked = projects.map((p) => ({ p, version: versionFor(s, p.id, basis)! })).filter((x) => x.version)
  const map = new Map<string, PortfolioFnRow>()
  for (const fn of s.settings.wbsFunctions) map.set(fn.id, { fnId: fn.id, code: fn.code, name: fn.name, budget: 0, actual: 0, etc: 0, eac: 0, vac: 0, lines: [], over: [] })
  let held = 0, budCont = 0
  for (const { p, version } of picked) {
    const t = etcEacAt(s, p, version, versionCutoff(s, version))
    held += t.contingencyHeld
    budCont += t.totals.budget - sum(t.rows.map((r) => r.budget))
    for (const r of t.rows as FunctionRow[]) {
      const row = map.get(r.fnId)
      if (!row) continue
      row.budget += r.budget; row.actual += r.actual; row.etc += r.etc; row.eac += r.eac; row.vac += r.vac
      if (r.vac < 0 && r.eac > 0) row.over.push({ p, vac: r.vac })
      for (const l of r.lines) {
        let lr = row.lines.find((x) => x.id === l.line.id)
        if (!lr) { lr = { id: l.line.id, code: l.line.code, name: l.line.name, budget: 0, actual: 0, etc: 0, eac: 0, vac: 0 }; row.lines.push(lr) }
        lr.budget += l.budget; lr.actual += l.actual; lr.etc += l.etc; lr.eac += l.eac; lr.vac += l.vac
      }
    }
  }
  const rows = [...map.values()].map((r) => ({ ...r, over: r.over.sort((a, b) => a.vac - b.vac) }))
  const T = (k: 'budget' | 'actual' | 'etc' | 'eac') => sum(rows.map((r) => r[k]))
  const totals = { budget: T('budget') + budCont, actual: T('actual'), etc: T('etc') + held, eac: T('eac') + held, vac: T('budget') + budCont - T('eac') - held }
  return { rows, contingencyHeld: held, budgetContingency: budCont, totals, projects: picked }
}

  }

  out.push({ 'WBS code': 'TOTAL', 'WBS function': 'Total', 'Line code': '', 'Line item': '', Budget: r.totals.budget, 'Actual to date': r.totals.actual, ETC: r.totals.etc, EAC: r.totals.eac, VAC: r.totals.vac, 'Projects over budget': '' })

/** What one SAP transaction contributes to a line (or to all lines of a function): its amount, or its allocated share. */
export function txnShare(s: AppState, t: ActualTxn, o: { lineId?: string; fnId?: string }): Money {
  if (o.lineId) return t.lineId === o.lineId ? t.amount : t.allocation?.[o.lineId] ?? 0
  if (o.fnId) {
    if (t.lineId) return functionOfLine(s, t.lineId) === o.fnId ? t.amount : 0
    return sum(Object.entries(t.allocation ?? {}).filter(([l]) => functionOfLine(s, l) === o.fnId).map(([, v]) => v))
  }
  return t.amount
}

/** Small bar sketch of a weight shape. Zero bars (black-out months) show as a tick on the baseline. */
export function Shape({ weights, w = 64, h = 18, className, labels }: { weights: number[]; w?: number; h?: number; className?: string; labels?: string[] }) {
  const n = weights.length
  const mx = Math.max(1e-9, ...weights)
  const bw = n ? w / n : w
  return (
    <svg width={w} height={h} viewBox={`0 0 ${w} ${h}`} className={cx('shrink-0', className)} aria-hidden>
      <line x1={0} x2={w} y1={h - 0.5} y2={h - 0.5} className="stroke-line" strokeWidth={1} />
      {weights.map((x, i) => {
        const bh = x > 0 ? Math.max(1.5, ((h - 2) * x) / mx) : 0
        return x > 0
          ? <rect key={i} x={i * bw + (bw > 3 ? 0.5 : 0)} y={h - 1 - bh} width={Math.max(0.8, bw - (bw > 3 ? 1 : 0.2))} height={bh} rx={bw > 4 ? 1 : 0} className="fill-link/70">{labels && <title>{labels[i]}</title>}</rect>
          : <rect key={i} x={i * bw + bw / 2 - 0.75} y={h - 3} width={1.5} height={2} className="fill-bad/70">{labels && <title>{`${labels[i]} — no spend`}</title>}</rect>
      })}
    </svg>
  )
}

const esc = (t: string) => t.replace(/[&<>"]/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' })[c]!)
const $ = (n: number) => money(n, 'full')
const td = (n: number, bold = false) => `<td class="num">${bold ? '<b>' : ''}${n ? $(n) : '–'}${bold ? '</b>' : ''}</td>`

export function forecastReportHtml(s: AppState, p: Project, f: Forecast): string {
  const spread: Spread = f.spread ?? {}
  const held = f.contingencyHeld ?? 0
  const fns = fnRefs(s, spread)
  const bud = budgetByFunction(s, p.id)
  const cut = versionCutoff(s, f)
  const ee = etcEacAt(s, p, { ...f, lineSpread: f.lineSpread ?? lineGridOf(s, p, f) }, cut)
  const actT = ee.totals.actual
  const etcT = ee.totals.etc + held
  const eac = ee.totals.eac
  const months = Object.keys(spreadByMonth(spread)).sort()
  const years = Array.from(new Set(months.map((m) => m.slice(0, 4)))).sort()
  const yr = (fn: string, y: string) => sum(Object.entries(spread[fn] ?? {}).filter(([m]) => m.startsWith(y)).map(([, v]) => v))
  const prior = priorApproved(s, p.id, f.id)
  const cmp = prior ? compareVersions(s, prior, { ...f, spread, contingencyHeld: held, eac }) : undefined
  const inst = s.workflows.find((w) => w.id === f.workflowInstanceId)

${ee.rows.map((r) => `<tr><td><b>${r.code}</b></td><td><b>${esc(r.name)}</b></td><td>${prof(r.fnId)}</td>${td(r.budget, true)}${td(r.actual, true)}${years.map((y) => td(yr(r.fnId, y), true)).join('')}${td(r.etc, true)}${td(r.eac, true)}<td class="num"${r.vac < 0 ? ' style="color:#c02828"' : ''}><b>${$(r.vac)}</b></td><td class="num">${pctS(r.actual, r.eac)}</td></tr>

<tr><td>990</td><td>Contingency held</td><td></td>${td(bud.contingency)}<td class="num">–</td>${years.map(() => '<td></td>').join('')}${td(held)}${td(held)}${td(bud.contingency - held)}<td></td></tr>

function ReviewCalendar({ s, reviews }: { s: AppState; reviews: GateReview[] }) {
  const months = Array.from(new Set([s.settings.asOf.slice(0, 7), ...reviews.map((g) => g.scheduled.slice(0, 7))])).sort().slice(0, 4)
  const [mi, setMi] = useState(0)
  const m = months[Math.min(mi, months.length - 1)]
  const first = new Date(m + '-01T12:00:00Z')
  const startDow = first.getUTCDay()
  const daysIn = new Date(Date.UTC(first.getUTCFullYear(), first.getUTCMonth() + 1, 0)).getUTCDate()
  const cells = Array.from({ length: Math.ceil((startDow + daysIn) / 7) * 7 }, (_, i) => i - startDow + 1)
  const byDay = groupBy(reviews.filter((g) => g.scheduled.startsWith(m)), (g) => String(Number(g.scheduled.slice(8, 10))))
  const today = s.settings.asOf.startsWith(m) ? Number(s.settings.asOf.slice(8, 10)) : -1
  return (
    <div className="p-3">
      <div className="mb-2 flex items-center justify-between">
        <Button size="sm" variant="ghost" disabled={mi === 0} onClick={() => setMi(mi - 1)}>← Prev</Button>
        <span className="font-display text-[15px] font-semibold">{fmtMonth(m)}</span>
        <Button size="sm" variant="ghost" disabled={mi >= months.length - 1} onClick={() => setMi(mi + 1)}>Next →</Button>
      </div>
      <div className="grid grid-cols-7 gap-px overflow-hidden rounded border border-line bg-line text-[11px]">
        {['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'].map((d) => <div key={d} className="bg-sunken px-1.5 py-1 font-semibold text-muted">{d}</div>)}
        {cells.map((d, i) => (
          <div key={i} className={cx('min-h-[72px] bg-surface p-1', (d < 1 || d > daysIn) && 'bg-sunken/40')}>
            {d >= 1 && d <= daysIn && <div className={cx('mb-0.5 text-[11px] tabular-nums', d === today ? 'inline-block rounded-full bg-marker px-1.5 font-bold text-markerink' : 'text-faint')}>{d}</div>}
            {(byDay[String(d)] ?? []).map((g) => {
              const p = s.projects.find((x) => x.id === g.projectId)!
              const r = readiness(s, p)
              return <a key={g.id} href={projectHref(p.id, 'lifecycle')} title={`${p.name} · ${r.pct}% ready`} className={cx('mb-0.5 block truncate rounded-sm border-l-2 bg-sunken px-1 py-0.5 font-mono text-[10.5px] font-semibold hover:bg-line', r.pct >= 80 ? 'border-good' : r.pct >= 50 ? 'border-warn' : 'border-bad')}>{p.code.slice(4)} G{g.stageIndex + 1}</a>
            })}
          </div>
        ))}
      </div>
    </div>
  )
}

          {rows.map(({ p, r, defs, dls, review }) => (

// Schedule (or reschedule) a gate review board date for a project's current gate. Used by the gate
// calendar and the project's lifecycle tab.
import { useEffect, useState } from 'react'
import { mutate, useStore } from '@/core/store'
import { templateFor } from '@/core/engine/gates'
import { addDays, fmtDate, uid } from '@/core/util'
import { Button, Callout, Checkbox, Field, Input, Modal } from '@/ui/kit'
import { toast } from '@/ui/toast'
import { ProjectSelect } from '../shared/pickers'

export function ScheduleReviewModal({ open, onClose, projectId: fixed }: { open: boolean; onClose: () => void; projectId?: string }) {
  const s = useStore()
  const [pid, setPid] = useState(fixed ?? '')
  const [date, setDate] = useState(addDays(s.settings.asOf, 21))
  const [board, setBoard] = useState<string[]>([])
  const p = s.projects.find((x) => x.id === pid)
  const existing = p ? s.gateReviews.find((g) => g.projectId === p.id && g.stageIndex === p.stageIndex && g.status === 'scheduled') : undefined
  useEffect(() => {
    if (!open) return
    const id = fixed ?? pid
    setPid(id)
    const proj = s.projects.find((x) => x.id === id)
    const ex = proj ? s.gateReviews.find((g) => g.projectId === proj.id && g.stageIndex === proj.stageIndex && g.status === 'scheduled') : undefined
    setBoard(ex?.board ?? (proj ? [proj.team.sponsor, 'u-pmo2'].filter(Boolean) as string[] : []))
    if (ex) setDate(ex.scheduled)
  }, [open, fixed, pid]) // eslint-disable-line react-hooks/exhaustive-deps
  const eligible = s.people.filter((x) => !x.company && (x.roles.includes('gate-board') || x.roles.includes('sponsor') || x.roles.includes('exec')))
  const gate = p ? templateFor(s, p).stages[p.stageIndex]?.gate : undefined
  const active = s.projects.filter((x) => x.status === 'active')
  const save = () => {
    if (!p || !gate) return
    if (!board.length) { toast('Pick at least one board member', 'bad'); return }
    if (date < s.settings.asOf) { toast('Pick a date from today on', 'bad'); return }
    if (existing) {
      mutate((d) => { const g = d.gateReviews.find((x) => x.id === existing.id)!; g.scheduled = date; g.board = board }, { entity: 'gate-review', entityId: existing.id, projectId: p.id, action: 'reschedule', summary: `${gate.name} review moved ${existing.scheduled} → ${date}` })
      toast(`Gate review moved to ${fmtDate(date)}`)
    } else {
      const id = uid('gr')
      mutate((d) => { d.gateReviews.push({ id, projectId: p.id, stageIndex: p.stageIndex, scheduled: date, board, status: 'scheduled' }) }, { entity: 'gate-review', entityId: id, projectId: p.id, action: 'schedule', summary: `${gate.name} review scheduled for ${date}` })
      toast(`Gate review scheduled for ${fmtDate(date)}`)
    }
    onClose()
  }
  const cancelReview = () => {
    if (!existing || !p) return
    mutate((d) => { d.gateReviews.find((x) => x.id === existing.id)!.status = 'cancelled' }, { entity: 'gate-review', entityId: existing.id, projectId: p.id, action: 'cancel', summary: `${gate?.name} review on ${existing.scheduled} cancelled` })
    toast('Gate review cancelled')
    onClose()
  }
  return (
    <Modal open={open} onClose={onClose} title={existing ? 'Reschedule gate review' : 'Schedule gate review'}
      footer={<>{existing && <Button variant="ghost" className="mr-auto text-bad" onClick={cancelReview}>Cancel review</Button>}<Button variant="ghost" onClick={onClose}>Close</Button><Button variant="primary" disabled={!p} onClick={save}>{existing ? 'Save new date' : 'Schedule review'}</Button></>}>
      <div className="flex flex-col gap-3">
        {!fixed && <Field label="Project"><ProjectSelectActive value={pid} onChange={setPid} ids={active.map((x) => x.id)} /></Field>}
        {p && gate && <Callout tone="info"><b>{p.code}</b> · {gate.name}. Decision roles: {gate.decisionRoles.map((r) => s.roles.find((x) => x.id === r)?.name ?? r).join(', ')}.</Callout>}
        {existing && <Callout tone="warn">Already scheduled for {fmtDate(existing.scheduled)}. Saving moves it.</Callout>}
        <Field label="Board date"><Input id="gr-date" type="date" value={date} min={s.settings.asOf} onChange={(e) => setDate(e.target.value)} /></Field>
        <div>
          <div className="mb-1 text-[12px] font-semibold text-muted">Board members</div>
          <div className="grid grid-cols-1 gap-1.5 sm:grid-cols-2">
            {eligible.map((x) => <Checkbox key={x.id} checked={board.includes(x.id)} onChange={(v) => setBoard(v ? [...board, x.id] : board.filter((b) => b !== x.id))} label={<span>{x.name}<span className="block text-[11px] text-muted">{x.title}</span></span>} />)}
          </div>
        </div>
      </div>
    </Modal>
  )
}

        }} />
    </div>
  )
}

const code = (s: AppState, pid?: string) => s.projects.find((p) => p.id === pid)?.code ?? ''

function dueText(s: AppState, due?: string): { text: string; tone: 'bad' | 'warn' | 'muted' } {
  if (!due) return { text: 'No SLA', tone: 'muted' }
  const d = daysBetween(s.settings.asOf, due)
  if (d < 0) return { text: `Overdue ${-d} day${d === -1 ? '' : 's'}`, tone: 'bad' }
  if (d === 0) return { text: 'Due today', tone: 'warn' }
  if (d === 1) return { text: 'Due tomorrow', tone: 'warn' }
  return { text: `Due in ${d} days`, tone: d <= 3 ? 'warn' : 'muted' }
}

/** Reminder / escalation state of a task for the person it is assigned to. */
function noticeOf(s: AppState, t: Task, personId: string) {
  const slot = t.inst.active.find((a) => a.nodeId === t.node.id)
  if (!slot) return undefined
  const n = taskNoticeState(t.inst, slot, t.actingFor ?? personId)
  const esc = n.escalated ?? n.escalations[0]
  return { reminded: n.overdueReminded ?? n.reminded, overdue: !!n.overdueReminded, escalatedTo: esc ? personName(s, esc.to) : undefined, escalatedAt: esc?.at }
}

/** Act on a step and stamp the e-signature as password re-authenticated (WF-10). */
function signAndAct(actorId: string, t: Task, decision: Decision, comment: string | undefined, reauth: boolean) {
  mutate((d) => {
    const before = d.workflows.find((w) => w.id === t.inst.id)?.history.length ?? 0
    actOnWorkflow(d, t.inst.id, t.node.id, actorId, decision, comment)
    const inst = d.workflows.find((w) => w.id === t.inst.id)!
    const ev = inst.history.slice(before).find((h) => h.actorId === actorId && h.nodeId === t.node.id)
    if (ev && decision === 'approve' && reauth) {
      ev.signature = { method: 'password-reauth', hash: signatureHash([inst.id, t.node.id, actorId, ev.at, decision, comment ?? '']) }
      d.audit.push({ id: uid('aud'), at: nowISO(), actorId, entity: 'workflow', entityId: inst.id, projectId: inst.projectId, action: 'e-sign', summary: `E-signed “${t.node.label}” on ${inst.title} after password re-authentication (signature ${ev.signature.hash})` })
    }
  })
}

function TaskDetail({ s, me, task, onDone }: { s: AppState; me: Person; task: Task; onDone: () => void }) {
  const [comment, setComment] = useState('')
  const [reauth, setReauth] = useState(false)
  const inst = task.inst
  const href = recordHref(inst)
  const due = dueText(s, task.due)
  const slot = inst.active.find((a) => a.nodeId === task.node.id)
  const others = slot ? slot.assigneeIds.filter((x) => x !== me.id && x !== task.actingFor) : []
  const nt = noticeOf(s, task, me.id)
  const away = task.actingFor ? s.people.find((p) => p.id === task.actingFor) : undefined
  const act = (decision: Decision) => {
    if ((decision === 'reject' || decision === 'return') && !comment.trim()) { toast('Add a comment explaining why.', 'bad'); document.getElementById('inbox-comment')?.focus(); return }
    if (decision === 'approve') { setReauth(true); return }
    try {
      signAndAct(me.id, task, decision, comment.trim() || undefined, false)
      toast(decision === 'submit' ? 'Submitted' : decision === 'return' ? 'Returned for revision' : 'Rejected', decision === 'reject' ? 'bad' : 'good')
      onDone()
    } catch (e: any) { toast(e.message, 'bad') }
  }
  return (
    <div className="flex min-w-0 flex-col gap-3">
      <Panel title={<span className="flex flex-wrap items-center gap-2">{task.node.label}<Badge tone={task.node.type === 'approval' ? 'marker' : 'neutral'}>{nodeTypeLabel(task.node.type)}</Badge></span>}
        subtitle={<>{inst.title}</>}
        actions={href ? <a href={href} className="text-[12.5px] font-semibold text-link">Open record →</a> : undefined}>
        <div className="flex flex-col gap-3">
          <div className="flex flex-wrap gap-x-5 gap-y-1 text-[12px] text-muted">
            <span>{recordTypeLabel(inst.recordType)}</span>
            <span>Started by <b className="text-ink">{personName(s, inst.startedById)}</b> {relTime(inst.startedAt, nowMs(s))}{inst.startedById !== me.id && <> · <a href={dmHref(inst.startedById)} className="text-link">Message</a></>}</span>
            <span>Waiting since {fmtDateTime(task.since)}</span>
            <span className={due.tone === 'bad' ? 'font-semibold text-bad' : due.tone === 'warn' ? 'font-semibold text-warn' : ''}>{due.text}{task.due ? ` (${fmtDate(task.due)}, ${task.node.slaDays}-day SLA)` : ''}</span>
          </div>
          {(nt?.reminded || nt?.escalatedTo) && (
            <div className="flex flex-wrap items-center gap-2 text-[12px]">
              {nt.reminded && <Badge tone={nt.overdue ? 'warn' : 'info'}>{nt.overdue ? 'Overdue notice' : 'Reminder'} sent {fmtDate(nt.reminded)}</Badge>}
              {nt.escalatedTo && <Badge tone="bad">Escalated to {nt.escalatedTo} {fmtDate(nt.escalatedAt)}</Badge>}
              <a href="#admin-notify" className="text-[11.5px] text-faint hover:text-link">Reminder rules</a>
            </div>
          )}
          <ProjectLine s={s} projectId={inst.projectId} />
          <RecordSummary s={s} inst={inst} />
        </div>
      </Panel>
      <div className="flex flex-col gap-2 rounded-md border border-marker/70 bg-marker/5 p-3">
        <div className="flex flex-wrap items-center gap-2"><b className="text-[13px]">Your decision</b>{task.node.type !== 'form' && <span className="text-[11.5px] text-muted">Approvals are e-signed after you re-enter your password.</span>}</div>
        {away && <Callout tone="info" title={`Acting for ${away.name}`}>{away.name} is out of office{away.outOfOffice ? ` until ${fmtDate(away.outOfOffice.to)}` : ''} and named you as delegate. Your decision is signed by you and recorded on the step as on behalf of {away.name.split(' ')[0]}.</Callout>}
        {task.node.description && <p className="text-[12.5px] text-muted">{task.node.description}</p>}
        {others.length > 0 && <p className="text-[12px] text-muted">{task.node.rule === 'all' ? `All must act: ${slot!.assigneeIds.map((x) => `${personName(s, x)}${slot!.actedIds.includes(x) ? ' ✓' : ''}`).join(', ')}` : `Also assigned: ${others.map((x) => personName(s, x)).join(', ')} — the first to act decides.`}</p>}
        <TextArea id="inbox-comment" placeholder={task.node.type === 'form' ? 'Note for the next reviewer (optional)' : 'Comment (required to return or reject)'} value={comment} onChange={(e) => setComment(e.target.value)} className="min-h-[60px] bg-surface" />
        <div className="flex flex-wrap gap-2">
          {task.node.type === 'form'
            ? <Button variant="primary" onClick={() => act('submit')}>Submit</Button>
            : <Button variant="primary" onClick={() => act('approve')}>Approve & sign</Button>}
          {task.node.type !== 'form' && <Button onClick={() => act('return')}>Return for revision</Button>}
          {task.node.type !== 'form' && <Button variant="ghost" className="text-bad" onClick={() => act('reject')}>Reject</Button>}
          <a href={workflowHref(inst.defKey)} className="ml-auto self-center text-[12px] text-link">Process v{inst.defVersion}</a>
        </div>
      </div>
      <Panel title="History" pad={false}><Timeline s={s} inst={inst} /></Panel>
      <ReauthModal open={reauth} onClose={() => setReauth(false)} me={me} title={`Sign: ${task.node.label}`} initialComment={comment}

          } catch (e: any) { toast(e.message, 'bad') }

function ReauthModal({ open, onClose, me, title, summary, onSign, initialComment = '', commentLabel = 'Comment (optional)' }: { open: boolean; onClose: () => void; me: Person; title: string; summary?: React.ReactNode; onSign: (comment: string) => void; initialComment?: string; commentLabel?: string }) {
  const [pw, setPw] = useState('')
  const [comment, setComment] = useState(initialComment)
  const [err, setErr] = useState(false)
  useEffect(() => { if (open) { setPw(''); setErr(false); setComment(initialComment) } }, [open]) // eslint-disable-line react-hooks/exhaustive-deps
  const submit = () => { if (!pw) { setErr(true); return } onSign(comment.trim()) }
  return (
    <Modal open={open} onClose={onClose} title={title}
      footer={<><Button variant="ghost" onClick={onClose}>Cancel</Button><Button variant="primary" onClick={submit}>Sign and approve</Button></>}>
      <form className="flex flex-col gap-3" onSubmit={(e) => { e.preventDefault(); submit() }}>
        {summary}
        <Field label={commentLabel}><TextArea className="min-h-[52px]" value={comment} onChange={(e) => setComment(e.target.value)} /></Field>
        <div className="flex items-center gap-2 rounded border border-line px-3 py-2"><Avatar name={me.name} size={26} /><div className="min-w-0 text-[12.5px]"><b>{me.name}</b><div className="truncate text-muted">{me.email}</div></div></div>
        <Field label="Enter your password to sign" hint={err ? <span className="text-bad">Enter your password to apply your electronic signature.</span> : 'Your signature is bound to this record, step, time and comment. Demo: any password works.'}>
          <Input id="reauth-password" type="password" autoComplete="current-password" value={pw} onChange={(e) => { setPw(e.target.value); setErr(false) }} autoFocus />
        </Field>
        <button type="submit" className="hidden" />
      </form>
    </Modal>
  )
}

function StartedByMe({ s, me }: { s: AppState; me: Person }) {
  const mine = s.workflows.filter((w) => w.startedById === me.id).sort((a, b) => (a.status === 'running' ? 0 : 1) - (b.status === 'running' ? 0 : 1) || b.startedAt.localeCompare(a.startedAt))
  const [open, setOpen] = useState<string | undefined>()
  const running = mine.filter((w) => w.status === 'running')
  const recent = mine.filter((w) => w.status !== 'running').slice(0, 15)
  if (!mine.length) return <Empty title="You have not started any requests">Requests you submit — change requests, estimates, drawdowns — show here with where they sit and who holds them.</Empty>
  const row = (w: WorkflowInstance) => {
    const def = s.workflowDefs.find((d) => d.id === w.defId)
    const p = s.projects.find((x) => x.id === w.projectId)
    const late = w.active.some((a) => a.due && a.due < s.settings.asOf)
    return (
      <li key={w.id} className="border-b border-line/70 last:border-0">
        <button className="flex w-full flex-wrap items-start gap-x-3 gap-y-1 px-4 py-2.5 text-left hover:bg-sunken/60" onClick={() => setOpen(open === w.id ? undefined : w.id)}>
          <div className="min-w-0 flex-1">
            <div className="text-[13px] font-semibold">{w.title}</div>
            <div className="text-[11.5px] text-muted">{def?.name} v{w.defVersion}{p && <> · <span className="font-mono">{p.code}</span></>} · started {fmtDate(w.startedAt.slice(0, 10))}</div>
          </div>
          {w.status === 'running' ? (
            <div className="flex flex-col items-end gap-0.5 text-right text-[12px]">
              {w.active.map((a) => <span key={a.nodeId}><Badge tone={late ? 'bad' : 'marker'}>{def?.nodes.find((n) => n.id === a.nodeId)?.label}</Badge> <span className="text-muted">{a.assigneeIds.map((x) => personName(s, x)).join(', ')}</span></span>)}
              <span className="text-[11px] text-faint">{fmtDur((nowMs(s) - Date.parse(w.startedAt)) / 36e5)} in flight</span>
            </div>
          ) : <StatusBadge status={w.status} />}
        </button>
        {open === w.id && (
          <div className="border-t border-line bg-sunken/30">
            <Timeline s={s} inst={w} />
            <div className="flex flex-wrap gap-2 px-4 pb-3">
              {recordHref(w) && <a className="text-[12.5px] font-semibold text-link" href={recordHref(w)}>Open record</a>}
              {w.status === 'running' && <Button size="sm" variant="ghost" className="text-bad" onClick={() => { mutate((d) => recallWorkflow(d, w.id, me.id), { entity: 'workflow', entityId: w.id, action: 'recall', summary: `Recalled ${w.title}`, projectId: w.projectId }); toast('Request recalled') }}>Recall request</Button>}
            </div>
          </div>
        )}

    )
  }
  return (
    <div className="flex flex-col gap-4">
      <Panel title="In flight" subtitle="Where each of your requests sits now" pad={false}>{running.length ? <ul>{running.map(row)}</ul> : <div className="px-4 py-3 text-[13px] text-muted">None of your requests are in flight.</div>}</Panel>
      {recent.length > 0 && <Panel title="Finished" pad={false}><ul>{recent.map(row)}</ul></Panel>}
    </div>
  )
}

function Delegation({ s, me, tasks }: { s: AppState; me: Person; tasks: Task[] }) {
  const oo = me.outOfOffice
  const [from, setFrom] = useState(oo?.from ?? s.settings.asOf)
  const [to, setTo] = useState(oo?.to ?? addDays(s.settings.asOf, 7))
  const [delegate, setDelegate] = useState(oo?.delegateId ?? '')
  useEffect(() => { setFrom(me.outOfOffice?.from ?? s.settings.asOf); setTo(me.outOfOffice?.to ?? addDays(s.settings.asOf, 7)); setDelegate(me.outOfOffice?.delegateId ?? '') }, [me.id]) // eslint-disable-line react-hooks/exhaustive-deps
  const candidates = s.people.filter((p) => p.id !== me.id && !p.company)
  const suggested = candidates.filter((p) => p.roles.some((r) => me.roles.includes(r) && r !== 'admin'))
  const dp = s.people.find((p) => p.id === delegate)
  const authorityGap = dp && dp.dollarAuthority < me.dollarAuthority
  const chain = dp?.outOfOffice && dp.outOfOffice.delegateId === me.id
  const active = oo && isOutOfOffice(me, s.settings.asOf)
  const coveringFor = s.people.filter((p) => p.outOfOffice?.delegateId === me.id)
  const save = () => {
    if (!delegate || !from || !to || to < from) { toast('Pick a delegate and a valid date range', 'bad'); return }
    const starts = from <= s.settings.asOf && to >= s.settings.asOf
    const mine = tasks.filter((t) => !t.actingFor).length
    mutate((d) => {
      const p = d.people.find((x) => x.id === me.id)!
      p.outOfOffice = { from, to, delegateId: delegate }
      d.notifications.push({ id: uid('ntf'), at: nowISO(), toId: delegate, text: `${me.name} is out of office ${fmtDate(from)} – ${fmtDate(to)} and named you as delegate. Their workflow steps appear in your My work.`, link: '#inbox', read: false, kind: 'info' })
    }, { entity: 'person', entityId: me.id, action: 'delegate', summary: `Out of office ${from} to ${to}; delegate ${personName(s, delegate)}` })
    toast(starts && mine ? `Delegation saved — ${dp?.name} can act on your ${mine} open step${mine > 1 ? 's' : ''}` : 'Delegation saved')
  }
  const clear = () => {
    mutate((d) => { const p = d.people.find((x) => x.id === me.id)!; delete p.outOfOffice }, { entity: 'person', entityId: me.id, action: 'delegate-end', summary: 'Ended out-of-office delegation' })
    toast('Delegation ended — new steps route to you again')
  }
  return (
    <div className="grid grid-cols-1 gap-4 xl:grid-cols-[minmax(0,1fr)_380px]">
      <Panel title="Out of office" subtitle="While you are away, new steps that would come to you route to your delegate, and they can act on the steps already waiting on you. Also available from your avatar in the top bar.">
        <div className="flex flex-col gap-3">
          {oo ? <Callout tone={active ? 'marker' : 'info'} title={active ? 'Delegation in force' : oo.to < s.settings.asOf ? 'Past delegation' : 'Scheduled delegation'}>{fmtDate(oo.from)} – {fmtDate(oo.to)} → {personName(s, oo.delegateId)}</Callout> : <Callout>No delegation set. Steps route to you.</Callout>}
          <div className="grid grid-cols-1 gap-3 sm:grid-cols-3">
            <Field label="From"><Input type="date" value={from} onChange={(e) => setFrom(e.target.value)} /></Field>
            <Field label="To"><Input type="date" value={to} min={from} onChange={(e) => setTo(e.target.value)} /></Field>
            <Field label="Delegate">
              <Select id="delegate" value={delegate} onChange={(e) => setDelegate(e.target.value)} options={[{ value: '', label: '— Select —' }, ...suggested.map((p) => ({ value: p.id, label: `${p.name} · ${p.title} (same role)` })), ...candidates.filter((p) => !suggested.includes(p)).map((p) => ({ value: p.id, label: `${p.name} · ${p.title}` }))]} />
            </Field>
          </div>
          {authorityGap && <Callout tone="warn">{dp!.name}'s dollar authority ({money(dp!.dollarAuthority)}) is below yours ({money(me.dollarAuthority)}). Dollar-authority steps still route by the DOA matrix, so approvals above their limit go to the next tier.</Callout>}
          {chain && <Callout tone="bad">{dp!.name} has delegated to you for overlapping dates — pick someone else to avoid a loop.</Callout>}
          <p className="text-[12.5px] text-muted">Your <b className="text-ink">{tasks.filter((t) => !t.actingFor).length}</b> open step{tasks.filter((t) => !t.actingFor).length === 1 ? '' : 's'} stay assigned to you; while the delegation is in force they also appear in your delegate's My work as “acting for {me.name}”, and either of you can act. Steps nobody acted on come back to you when you return.</p>
          <div className="flex flex-wrap gap-2">
            <Button variant="primary" disabled={!delegate || !!chain} onClick={save}>Save delegation</Button>
            {oo && <Button variant="ghost" onClick={clear}>End delegation</Button>}
          </div>
        </div>
      </Panel>
      <div className="flex flex-col gap-4">
        <Panel title="You are covering for" pad={false}>
          {coveringFor.length ? <ul className="divide-y divide-line">{coveringFor.map((p) => <li key={p.id} className="flex items-center gap-2 px-4 py-2 text-[13px]"><Avatar name={p.name} /><span className="min-w-0 flex-1"><b>{p.name}</b><span className="block text-[11.5px] text-muted">{fmtDate(p.outOfOffice!.from)} – {fmtDate(p.outOfOffice!.to)}</span></span>{isOutOfOffice(p, s.settings.asOf) ? <Badge tone="marker">active</Badge> : <Badge>scheduled</Badge>}</li>)}</ul> : <div className="px-4 py-3 text-[13px] text-muted">Nobody has named you as their delegate.</div>}
        </Panel>
        <Panel title="Delegations across the organization" pad={false}>
          {s.people.filter((p) => p.outOfOffice).length ? <ul className="divide-y divide-line">{s.people.filter((p) => p.outOfOffice).map((p) => <li key={p.id} className="px-4 py-2 text-[12.5px]"><b>{p.name}</b> → {personName(s, p.outOfOffice!.delegateId)}<span className="block text-[11.5px] text-muted">{fmtDate(p.outOfOffice!.from)} – {fmtDate(p.outOfOffice!.to)}</span></li>)}</ul> : <div className="px-4 py-3 text-[13px] text-muted">No one is out of office.</div>}
        </Panel>
      </div>
    </div>
  )
}

function Notifications({ s, me, onOpenTask }: { s: AppState; me: Person; onOpenTask: (instId: string) => void }) {
  const list = s.notifications.filter((n) => n.toId === me.id).slice().reverse()
  const [unreadOnly, setUnreadOnly] = useState(false)
  const rows = unreadOnly ? list.filter((n) => !n.read) : list
  const markAll = () => { mutate((d) => d.notifications.forEach((n) => { if (n.toId === me.id) n.read = true })); toast('All notifications marked read') }
  const open = (id: string, text: string, link: string) => {
    mutate((d) => { const n = d.notifications.find((x) => x.id === id); if (n) n.read = true })
    const inst = s.workflows.find((w) => text.includes(w.title))
    if (link.startsWith('#') && link !== '#inbox') go(link)
    else if (inst && inst.status === 'running' && inst.active.some((a) => a.assigneeIds.includes(me.id))) onOpenTask(inst.id)
    else if (inst?.projectId) go(recordHref(inst) ?? '#inbox')
  }
  return (
    <Panel title="Notifications" pad={false} actions={<><Checkbox checked={unreadOnly} onChange={setUnreadOnly} label={<span className="text-[12px]">Unread only</span>} /><Button size="sm" disabled={!list.some((n) => !n.read)} onClick={markAll}>Mark all read</Button></>}>
      {rows.length ? (
        <ul className="divide-y divide-line">
          {rows.map((n) => (
            <li key={n.id}>
              <button onClick={() => open(n.id, n.text, n.link)} className="flex w-full items-start gap-3 px-4 py-2.5 text-left hover:bg-sunken/60">
                <span className={cx('mt-1.5 h-2 w-2 shrink-0 rounded-full', n.read ? 'bg-transparent' : 'bg-link')} />
                <span className="min-w-0 flex-1"><span className={cx('block text-[13px]', !n.read && 'font-semibold')}>{n.text}</span><span className="text-[11px] text-faint">{fmtDateTime(n.at)}</span></span>
                <Icon name="arrow" className="mt-1 h-4 w-4 shrink-0 text-faint" />
              </button>
            </li>
          ))}
        </ul>
      ) : <div className="px-4 py-6 text-center text-[13px] text-muted">{unreadOnly ? 'No unread notifications.' : 'No notifications yet.'}</div>}
    </Panel>
  )
}

function EscalatedToMe({ s, rows }: { s: AppState; rows: ReturnType<typeof openEscalations> }) {
  return (
    <Panel title="Escalated to me" subtitle="Steps assigned to people who report to you that are past the escalation threshold. Follow up with the assignee; the step stays theirs." pad={false}>
      <div className="scroll-x">
        <table className="tbl">
          <thead><tr><th>Step</th><th>Request</th><th>Waiting on</th><th className="num">Overdue</th><th>Escalated</th><th /></tr></thead>
          <tbody>
            {rows.map((e) => {
              const waiting = e.slot.assigneeIds.filter((x) => !e.slot.actedIds.includes(x))
              const late = daysBetween(e.slot.due ?? s.settings.asOf, s.settings.asOf)
              const href = recordHref(e.inst)
              return (
                <tr key={`${e.inst.id}:${e.slot.nodeId}`}>
                  <td className="whitespace-nowrap font-semibold">{e.stepLabel}</td>
                  <td className="min-w-[220px]">{href ? <a className="text-link" href={href}>{e.inst.title}</a> : e.inst.title}<div className="text-[11.5px] text-muted">{recordTypeLabel(e.inst.recordType)}{code(s, e.inst.projectId) && <> · <span className="font-mono">{code(s, e.inst.projectId)}</span></>}</div></td>
                  <td className="whitespace-nowrap text-[12.5px]">{waiting.map((x) => personName(s, x)).join(', ')}</td>
                  <td className="num font-semibold text-bad">{late} d</td>
                  <td className="whitespace-nowrap text-[12px]">{fmtDate(e.at)}</td>
                  <td className="whitespace-nowrap text-right">{waiting[0] && <a href={dmHref(e.assigneeId)} className="text-[12.5px] font-semibold text-link">Message {personName(s, e.assigneeId).split(' ')[0]}</a>}</td>
                </tr>
              )
            })}
          </tbody>
        </table>
      </div>
    </Panel>
  )
}
