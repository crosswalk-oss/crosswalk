// Pure helpers for the SAP actuals pages: import review roll-ups, exception reasons, rule labels and the
// coverage checks the mapping-table admin runs. No React; everything takes AppState so it can be unit-tested.

export const TXN_STATUS: Record<TxnStatus, { label: string; tone: 'good' | 'info' | 'neutral' | 'bad' | 'warn'; hint: string }> = {
  mapped: { label: 'Mapped', tone: 'good', hint: 'A mapping rule (or a person) sent it to one forecast line' },
  allocated: { label: 'Allocated', tone: 'info', hint: 'Loader / overhead settlement spread over the period’s direct lines pro-rata' },
  excluded: { label: 'Excluded', tone: 'neutral', hint: 'Kept out of project actuals (balance-sheet or non-project cost)' },
  unmapped: { label: 'Unmapped', tone: 'bad', hint: 'No active rule covers the cost element — blocks posting' },
  'no-project': { label: 'No project', tone: 'bad', hint: 'The SAP WBS element matches no project — blocks posting' },
}

export const importTxns = (s: AppState, importId: string) => s.actualTxns.filter((t) => t.importId === importId)

export interface ImportFigures { rows: number; total: Money; mapped: Money; allocated: Money; excluded: Money; unmapped: Money; noProject: Money; exceptions: number; resolved: number; credits: number; projects: number; periods: string[] }
export function importFigures(txns: ActualTxn[]): ImportFigures {
  const t = (st: TxnStatus) => Math.round(sum(txns.filter((x) => x.status === st).map((x) => x.amount)))
  return {
    rows: txns.length,
    total: Math.round(sum(txns.map((x) => x.amount))),
    mapped: t('mapped'), allocated: t('allocated'), excluded: t('excluded'), unmapped: t('unmapped'), noProject: t('no-project'),
    exceptions: txns.filter(isException).length,
    resolved: txns.filter((x) => x.manual).length,
    credits: txns.filter((x) => x.amount < 0 && x.status !== 'excluded').length,
    projects: new Set(txns.map((x) => x.projectId).filter(Boolean)).size,
    periods: [...new Set(txns.map((x) => x.period))].sort(),
  }
}

export const isException = (t: ActualTxn) => t.status === 'unmapped' || t.status === 'no-project'

/** Plain-language reason a row needs a person. */
export function exceptionReason(s: AppState, t: ActualTxn): string {
  if (t.status === 'no-project') return `WBS element ${t.sapWbs} does not belong to any project (no project has that ERP WBS or a parent of it)`
  if (t.status === 'unmapped') {
    const r = s.settings.sapMappings.find((x) => x.id === t.ruleId)
    if (r?.target === 'line') return `Rule P${r.priority} matches but targets a line that is missing or inactive`
    return `No active mapping rule covers cost element ${t.costElement}${t.costElementName ? ` (${t.costElementName})` : ''}`
  }
  return ''
}

/** "P30 · 5200000–5200999 → 06.10 Contract construction" */
export function ruleLabel(s: AppState, r?: SapMapping): string {
  if (!r) return ''
  return `P${r.priority} · ${r.costElementFrom}–${r.costElementTo}${r.wbsSuffix ? ` on *${r.wbsSuffix}` : ''}${r.textContains ? ` text “${r.textContains}”` : ''} → ${targetLabel(s, r)}`
}
export function targetLabel(s: AppState, r: Pick<SapMapping, 'target' | 'lineId'>): string {
  if (r.target === 'allocate') return 'Allocate pro-rata'
  if (r.target === 'exclude') return 'Exclude'
  const l = lineById(s, r.lineId)
  return l ? `${l.code} ${l.name}` : 'Missing line'
}
export const ruleOf = (s: AppState, t: ActualTxn) => s.settings.sapMappings.find((x) => x.id === t.ruleId)

/** How a transaction landed: rule, manual resolution or status. */
export function howMapped(s: AppState, t: ActualTxn): string {
  if (t.manual) return `Resolved by hand${t.manual.note ? ` — ${t.manual.note}` : ''}`
  const r = ruleOf(s, t)
  return r ? ruleLabel(s, r) : TXN_STATUS[t.status].hint
}

export interface ProjectRoll {
  p?: Project
  projectId: string
  byLine: Record<string, { direct: Money; allocated: Money }>
  direct: Money; allocated: Money; excluded: Money; exceptions: Money
  rows: number
}
/**
 * Roll an import up by project and forecast line. Allocated rows use their posted allocation; on a staged import
 * they are previewed the same way posting will do it (pro-rata to the project-period's mapped lines).
 */
export function projectLineRollup(s: AppState, txns: ActualTxn[]): ProjectRoll[] {
  const out = new Map<string, ProjectRoll>()
  const get = (pid: string) => {
    let r = out.get(pid)
    if (!r) { r = { p: s.projects.find((x) => x.id === pid), projectId: pid, byLine: {}, direct: 0, allocated: 0, excluded: 0, exceptions: 0, rows: 0 }; out.set(pid, r) }
    return r
  }
  const cell = (r: ProjectRoll, lid: string) => (r.byLine[lid] ??= { direct: 0, allocated: 0 })
  for (const t of txns) {
    const r = get(t.projectId ?? '—')
    r.rows++
    if (t.status === 'mapped' && t.lineId) { cell(r, t.lineId).direct += t.amount; r.direct += t.amount }
    else if (t.status === 'excluded') r.excluded += t.amount
    else if (isException(t)) r.exceptions += t.amount
  }
  for (const t of txns.filter((x) => x.status === 'allocated')) {
    const r = get(t.projectId ?? '—')
    const share = t.allocation ?? previewAllocation(txns, t)
    for (const [lid, v] of Object.entries(share)) cell(r, lid).allocated += v
    r.allocated += t.amount
  }
  return [...out.values()].sort((a, b) => (a.p?.code ?? 'ZZZ').localeCompare(b.p?.code ?? 'ZZZ'))
}

/** Pro-rata split of an allocated row over its project-period's mapped lines (mirrors postImport). */
export function previewAllocation(txns: ActualTxn[], t: ActualTxn): Record<string, Money> {
  const direct: Record<string, Money> = {}
  for (const x of txns) if (x.status === 'mapped' && x.lineId && x.projectId === t.projectId && x.period === t.period) direct[x.lineId] = (direct[x.lineId] || 0) + x.amount
  const ids = Object.keys(direct)
  if (!ids.length) return {}
  const base = sum(Object.values(direct))
  const out: Record<string, Money> = {}
  let acc = 0
  ids.forEach((k, i) => { const v = i === ids.length - 1 ? Math.round((t.amount - acc) * 100) / 100 : Math.round(t.amount * (base ? direct[k] / base : 1 / ids.length) * 100) / 100; out[k] = v; acc += v })
  return out
}

export interface Overlap { a: SapMapping; b: SapMapping; kind: 'refines' | 'shadowed' | 'conflict'; text: string }
const n = (x: string) => Number(x)
const covers = (a: SapMapping, b: SapMapping) => n(a.costElementFrom) <= n(b.costElementFrom) && n(a.costElementTo) >= n(b.costElementTo)
const intersects = (a: SapMapping, b: SapMapping) => n(a.costElementFrom) <= n(b.costElementTo) && n(b.costElementFrom) <= n(a.costElementTo)
const hasFilter = (r: SapMapping) => !!(r.wbsSuffix || r.textContains)

/** Overlapping cost-element ranges among active rules, classified: a narrower rule first (fine), a rule that can never match, or a real conflict. */
export function ruleOverlaps(rules: SapMapping[]): Overlap[] {
  const act = rules.filter((r) => r.active).sort((a, b) => a.priority - b.priority)
  const out: Overlap[] = []
  for (let i = 0; i < act.length; i++) for (let j = i + 1; j < act.length; j++) {
    const a = act[i], b = act[j] // a runs first
    if (!intersects(a, b)) continue
    const sameTarget = a.target === b.target && a.lineId === b.lineId
    if (covers(a, b) && !hasFilter(a)) out.push({ a, b, kind: 'shadowed', text: `P${b.priority} never matches: P${a.priority} runs first and covers its whole range with no WBS or text filter` })
    else if (hasFilter(a) || covers(b, a)) out.push({ a, b, kind: 'refines', text: `P${a.priority} is narrower and runs first; P${b.priority} catches the rest` })
    else out.push({ a, b, kind: sameTarget ? 'refines' : 'conflict', text: sameTarget ? `Ranges overlap but both send cost to the same place` : `Ranges partly overlap with different targets — rows in ${a.costElementFrom > b.costElementFrom ? a.costElementFrom : b.costElementFrom}–${a.costElementTo < b.costElementTo ? a.costElementTo : b.costElementTo} follow P${a.priority}` })
  }
  return out
}

/** Active forecast lines no active rule sends cost to. */
export const linesWithoutRule = (s: AppState) => forecastLines(s).filter((l) => !s.settings.sapMappings.some((r) => r.active && r.target === 'line' && r.lineId === l.id))

/** Rules pointing at a missing or inactive line. */
export const brokenRules = (s: AppState) => s.settings.sapMappings.filter((r) => r.active && r.target === 'line' && !forecastLines(s).some((l) => l.id === r.lineId))

/** Cost elements in staged imports that no rule covers (what to add next), with row count and amount. */
export function unmappedElements(s: AppState): { costElement: string; name: string; rows: number; amount: Money }[] {
  const staged = new Set(s.actualsImports.filter((x) => x.status === 'staged').map((x) => x.id))
  const m = new Map<string, { costElement: string; name: string; rows: number; amount: Money }>()
  for (const t of s.actualTxns) if (staged.has(t.importId) && t.status === 'unmapped' && !t.manual) {
    const e = m.get(t.costElement) ?? { costElement: t.costElement, name: t.costElementName, rows: 0, amount: 0 }
    e.rows++; e.amount += t.amount
    m.set(t.costElement, e)
  }
  return [...m.values()]
}

/** Test one row against the mapping table: the winning rule, every rule that would match, and the project. */
export function testRow(s: AppState, row: { costElement: string; sapWbs: string; text: string }, rules = s.settings.sapMappings) {
  const r = { ...row, vendor: '' }
  const winner = matchRule(rules, r)
  const all = [...rules].filter((x) => x.active).sort((a, b) => a.priority - b.priority).filter((x) => matchRule([x], r))
  const p = row.sapWbs.trim() ? projectForWbs(s, row.sapWbs) : undefined
  const line = winner?.target === 'line' ? lineById(s, winner.lineId) : undefined
  return { winner, all, project: p, line, fnId: line ? functionOfLine(s, line.id) : undefined }
}

function forecastCell(p: ReconProject) {
  const inAct = p.periods.every((x) => x.forecast.inActuals)
  const none = p.periods.every((x) => !x.forecast.inActuals && !x.forecast.etc)
  if (none) return <span className="text-faint">no forecast</span>
  return inAct ? <span title="The latest forecast treats these months as actuals">{$(p.forecastActual)}<div className="text-[10.5px] text-faint">as actuals</div></span>
    : <span className="text-warn" title="The latest forecast still holds estimate-to-complete in these months — roll the forecast forward">{$(p.forecastEtc)}<div className="text-[10.5px]">still ETC</div></span>
}

function Fig({ label, value, sub, tone }: { label: string; value: string; sub?: string; tone?: 'good' | 'bad' }) {
  return (
    <div className="min-w-0 bg-surface px-3 py-2.5 sm:px-4">
      <div className="eyebrow truncate">{label}</div>
      <div className={cx('mt-0.5 truncate font-display text-[17px] font-bold leading-tight tabular-nums', tone === 'bad' ? 'text-bad' : tone === 'good' ? 'text-good' : 'text-ink')}>{value}</div>
      {sub && <div className="truncate text-[11px] text-muted" title={sub}>{sub}</div>}
    </div>

function StatusNote({ pp }: { pp: ReconPeriod }) {
  const un = pp.diff - pp.explained
  if (pp.status === 'ties') return <Callout tone="good">Crosswalk’s {fmtMonth(pp.period)} actuals tie to SAP in total and on every forecast line.</Callout>
  if (pp.status === 'not-posted') return <Callout tone="warn" title="Not posted">Crosswalk has no actuals for {fmtMonth(pp.period)} on this project. Post the extract to bring them in.</Callout>
  if (pp.status === 'lines-differ') return <Callout tone="warn" title="Total ties, lines differ">The total agrees with SAP, but some cost sits on different forecast lines than the mapping table sends it — usually a hand resolution. See Reconciling rows.</Callout>
  return (
    <Callout tone="bad" title={`Crosswalk ${pp.diff > 0 ? 'is higher' : 'is lower'} than SAP by ${$(Math.abs(pp.diff))}`}>
      {pp.source === 'summary' ? 'Crosswalk holds monthly summary actuals for this month, not SAP line items; posting the extract replaces them.' : Math.abs(un) <= RECON_TOLERANCE ? 'Every dollar of the gap is explained by the reconciling rows below.' : `${$(Math.abs(un))} is not explained by row-level matching — check for edits to the cost period outside an SAP import.`}
    </Callout>
  )
}

/** Local editable copy of a value; `dirty` when it differs from the source. Resets when the source changes. */
export function useDraft<T>(source: T): [T, React.Dispatch<React.SetStateAction<T>>, boolean, () => void] {
  const key = JSON.stringify(source) ?? 'null'
  const [d, setD] = useState<T>(() => JSON.parse(key))
  useEffect(() => { setD(JSON.parse(key)) }, [key])
  return [d, setD, JSON.stringify(d) !== key, () => setD(JSON.parse(key))]
}

export function validateImport(raw: unknown): { ok: true; state: AppState } | { ok: false; error: string } {
  if (!raw || typeof raw !== 'object') return { ok: false, error: 'The file is not a JSON object.' }
  const s = raw as Partial<AppState>
  if (s.schemaVersion !== SCHEMA_VERSION) return { ok: false, error: `Schema version ${s.schemaVersion ?? 'missing'} does not match this release (v${SCHEMA_VERSION}). Export again from the same release.` }
  const missing = REQUIRED.filter((k) => !(k in s))
  if (missing.length) return { ok: false, error: `Missing sections: ${missing.join(', ')}.` }
  if (!Array.isArray(s.projects) || !Array.isArray(s.audit)) return { ok: false, error: 'Projects and audit must be lists.' }
  return { ok: true, state: s as AppState }
}

import { mutate, useStore } from '@/core/store'
import { CONT_CAT, ESC_CAT } from '@/core/engine/estimates'

const SYSTEM = [ESC_CAT, CONT_CAT] // overhead (cat-loaders) is assignable, e.g. to 900 Project Overhead

/** Validation of the line table against cost categories and the WBS function mapping. */
export function lineIssues(lines: ForecastLine[], cats: CostCategory[], fns: WbsFunction[]) {
  const act = lines.filter((l) => l.active)
  const direct = cats.filter((c) => !SYSTEM.includes(c.id))
  const owners = (id: string) => act.filter((l) => l.categoryIds.includes(id))
  const fnOfCat = (id: string) => fns.find((f) => f.categoryIds.includes(id))
  return {
    unassigned: direct.filter((c) => owners(c.id).length === 0),
    multi: direct.filter((c) => owners(c.id).length > 1),
    mismatch: act.flatMap((l) => l.categoryIds.map((c) => ({ l, c, fn: fnOfCat(c) })).filter((x) => x.fn && x.fn.id !== l.functionId)),
    dupCodes: act.filter((l, i) => act.findIndex((x) => x.code.trim() === l.code.trim()) !== i).map((l) => l.code),
    emptyFns: fns.filter((f) => !act.some((l) => l.functionId === f.id)),
  }
}

    const out: Record<string, { forecasts: number; periods: number; txns: number; rules: number }> = {}
    const u = (id: string) => (out[id] ||= { forecasts: 0, periods: 0, txns: 0, rules: 0 })
    for (const f of s.forecasts) for (const [id, row] of Object.entries(f.lineSpread ?? {})) if (Object.values(row).some(Boolean)) u(id).forecasts++
    for (const c of s.costPeriods) for (const [id, v] of Object.entries(c.byLine ?? {})) if (v) u(id).periods++
    for (const t of s.actualTxns) { if (t.lineId) u(t.lineId).txns++; for (const id of Object.keys(t.allocation ?? {})) u(id).txns++ }
    for (const r of s.settings.sapMappings) if (r.lineId) u(r.lineId).rules++
    return out
  }, [s.forecasts, s.costPeriods, s.actualTxns, s.settings.sapMappings])
  const inUse = (id: string) => { const x = usage[id]; return !!x && x.forecasts + x.periods + x.txns + x.rules > 0 }
  const catName = (id: string) => s.categories.find((c) => c.id === id)

    const same = d.fields.filter((x) => x.section === f.section)
    const i = same.indexOf(f), j = i + dir
    if (j < 0 || j >= same.length) return

    const fields = d.fields.slice(); fields[a] = same[j]; fields[b] = f

  }

  const problems: string[] = []
  const secNames = d.sections.map((x) => x.trim())
  if (secNames.some((x) => !x)) problems.push('A section has no name')

function AddFieldModal({ section, onClose, taken, onAdd, stageOptions, bindOptions }: {
  section: string | null; onClose: () => void; taken: string[]; onAdd: (f: InfoFieldDef) => void
  stageOptions: { value: string; label: string }[]; bindOptions: { value: string; label: string }[]
}) {
  const [label, setLabel] = useState('')
  const [key, setKey] = useState('')
  const [keyTouched, setKeyTouched] = useState(false)
  const [type, setType] = useState<InfoFieldDef['type']>('text')
  const [options, setOptions] = useState('')
  const [help, setHelp] = useState('')
  const [req, setReq] = useState('')
  const [bind, setBind] = useState('')
  const [readOnly, setReadOnly] = useState(false)
  const k = keyTouched ? key : keyFromLabel(label, taken)
  const err = !label.trim() ? 'Enter a label' : !validKey(k) ? 'Key must start with a letter and use letters and digits only' : taken.includes(k) ? 'That key is already used' : type === 'select' && !options.trim() ? 'Enter at least one option' : ''
  const reset = () => { setLabel(''); setKey(''); setKeyTouched(false); setType('text'); setOptions(''); setHelp(''); setReq(''); setBind(''); setReadOnly(false) }
  const close = () => { reset(); onClose() }
  const add = () => {
    if (err || !section) return
    onAdd({ section, key: k, label: label.trim(), type, options: type === 'select' ? options.split('\n').map((o) => o.trim()).filter(Boolean) : undefined, help: help.trim() || undefined, requiredAtStage: req === '' ? undefined : Number(req), bind: bind || undefined, readOnly: readOnly || undefined })
    reset()
  }
  const pickBind = (path: string) => {
    setBind(path)
    const b = BIND_PATHS.find((x) => x.path === path)
    if (b) { setType(b.type); if (b.options) setOptions(b.options.join('\n')); if (!label.trim()) setLabel(b.label) }
  }
  return (
    <Modal open={section !== null} onClose={close} title={`Add field to ${section ?? ''}`}
      footer={<><Button variant="ghost" onClick={close}>Cancel</Button><Button variant="primary" disabled={!!err} onClick={add}>Add field</Button></>}>
      <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
        <Field label="Label"><Input value={label} onChange={(e) => setLabel(e.target.value)} placeholder="e.g. Environmental lead" /></Field>
        <Field label="Key" hint="Permanent once saved"><Input className="font-mono" value={k} onChange={(e) => { setKey(e.target.value); setKeyTouched(true) }} /></Field>
        <Field label="Stored on"><Select value={bind} onChange={(e) => pickBind(e.target.value)} options={bindOptions} /></Field>
        <Field label="Type"><Select value={type} onChange={(e) => setType(e.target.value as InfoFieldDef['type'])} options={FIELD_TYPES.map((t) => ({ value: t.id, label: t.label }))} /></Field>
        {type === 'select' && <Field label="Options" hint="One per line" className="sm:col-span-2"><textarea className="inp" rows={4} value={options} onChange={(e) => setOptions(e.target.value)} /></Field>}
        <Field label="Required at" className="sm:col-span-2"><Select value={req} onChange={(e) => setReq(e.target.value)} options={stageOptions} /></Field>
        <Field label="Help text" className="sm:col-span-2"><Input value={help} onChange={(e) => setHelp(e.target.value)} /></Field>
        <label className="flex items-center gap-2 text-[13px] sm:col-span-2"><input type="checkbox" className="h-4 w-4" checked={readOnly} onChange={(e) => setReadOnly(e.target.checked)} />Read-only on the record</label>
      </div>
      {req !== '' && <div className="mt-3"><Callout tone="warn">Projects already at or past this stage with the field empty will show it as a gate blocker as soon as you save.</Callout></div>}
      {err && label && <div className="mt-2 text-[12px] text-bad">{err}</div>}
    </Modal>
  )

// Configuration: the SAP → forecast line mapping table. Rules run in priority order (lowest first) and the first
// match wins: cost-element range, optional WBS-element suffix, optional text/vendor match → a forecast line,
// allocate pro-rata (loader settlements) or exclude (non-project cost). Includes a row tester and coverage checks.
import { useMemo, useState } from 'react'
import type { SapMapping } from '@/core/types'
import { mutate, useStore } from '@/core/store'
import { forecastLines, lineById } from '@/core/engine/lines'
import { remap } from '@/core/engine/sap'
import { cx, money, uid } from '@/core/util'
import { Badge, Button, Callout, Checkbox, ConfirmButton, Field, Input, Modal, Panel, SegButtons, Select, TextArea } from '@/ui/kit'
import { toast } from '@/ui/toast'
import { Icon } from '@/ui/icons'
import { brokenRules, linesWithoutRule, ruleOverlaps, targetLabel, testRow, unmappedElements } from '../actuals/lib'

export default function SapMappings(_props: { projectId?: string }) {
  const s = useStore()
  const rules = useMemo(() => [...s.settings.sapMappings].sort((a, b) => a.priority - b.priority), [s.settings.sapMappings])
  const [edit, setEdit] = useState<SapMapping | null>(null)
  const overlaps = ruleOverlaps(rules)
  const noRule = linesWithoutRule(s)
  const broken = brokenRules(s)
  const missing = unmappedElements(s)
  const staged = s.actualsImports.filter((x) => x.status === 'staged')
  const hits = useMemo(() => { const out: Record<string, number> = {}; for (const t of s.actualTxns) if (t.ruleId && !t.manual) out[t.ruleId] = (out[t.ruleId] || 0) + 1; return out }, [s.actualTxns])

  const save = (r: SapMapping, isNew: boolean) => {
    mutate((d) => {
      const i = d.settings.sapMappings.findIndex((x) => x.id === r.id)
      if (i >= 0) d.settings.sapMappings[i] = r
      else d.settings.sapMappings.push(r)
    }, { entity: 'sap-mapping', entityId: r.id, action: isNew ? 'create' : 'update', summary: `${isNew ? 'Added' : 'Updated'} SAP mapping rule P${r.priority}: ${r.costElementFrom}–${r.costElementTo}${r.wbsSuffix ? ` WBS *${r.wbsSuffix}` : ''}${r.textContains ? ` text "${r.textContains}"` : ''} → ${targetLabel(s, r)}` })
    toast(isNew ? `Rule P${r.priority} added` : `Rule P${r.priority} saved`)
    setEdit(null)
  }
  const swap = (i: number, dir: -1 | 1) => {
    const a = rules[i], b = rules[i + dir]
    if (!a || !b) return
    mutate((d) => {
      const x = d.settings.sapMappings.find((y) => y.id === a.id)!, y = d.settings.sapMappings.find((z) => z.id === b.id)!
      const pa = x.priority
      if (pa === y.priority) x.priority = pa + dir
      else { x.priority = y.priority; y.priority = pa }
    }, { entity: 'sap-mapping', entityId: a.id, action: 'reorder', summary: `Moved SAP mapping rule ${a.costElementFrom}–${a.costElementTo} ${dir < 0 ? 'ahead of' : 'after'} ${b.costElementFrom}–${b.costElementTo}` })
  }
  const toggle = (r: SapMapping) => {
    mutate((d) => { const x = d.settings.sapMappings.find((y) => y.id === r.id); if (x) x.active = !x.active }, { entity: 'sap-mapping', entityId: r.id, action: r.active ? 'deactivate' : 'activate', summary: `${r.active ? 'Deactivated' : 'Activated'} SAP mapping rule P${r.priority} ${r.costElementFrom}–${r.costElementTo}` })
    toast(r.active ? `P${r.priority} deactivated` : `P${r.priority} activated`)
  }
  const remove = (r: SapMapping) => {
    mutate((d) => { d.settings.sapMappings = d.settings.sapMappings.filter((x) => x.id !== r.id) }, { entity: 'sap-mapping', entityId: r.id, action: 'delete', summary: `Deleted SAP mapping rule P${r.priority} ${r.costElementFrom}–${r.costElementTo} → ${targetLabel(s, r)}` })
    toast(`P${r.priority} deleted`)
  }
  const newRule = (patch: Partial<SapMapping> = {}) => {
    const maxBelowAlloc = Math.max(0, ...rules.filter((r) => r.target === 'line').map((r) => r.priority))
    setEdit({ id: uid('map'), priority: maxBelowAlloc + 1, costElementFrom: '', costElementTo: '', target: 'line', lineId: forecastLines(s)[0]?.id, note: '', active: true, ...patch })
  }
  const rerunAll = () => {
    mutate((d) => { for (const imp of d.actualsImports.filter((x) => x.status === 'staged')) remap(d, imp.id) }, { entity: 'sap-mapping', entityId: 'remap', action: 'remap', summary: `Re-ran the mapping table over ${staged.length} staged SAP extract(s)` })
    toast(`Mapping re-run on ${staged.length} staged extract${staged.length === 1 ? '' : 's'}`)
  }

  return (
    <div className="flex flex-col gap-4">
      <div className="grid grid-cols-1 gap-4 xl:grid-cols-[minmax(0,1.4fr)_minmax(0,1fr)]">
        <RowTester />
        <Panel title="Coverage checks" subtitle="What the table misses, and where rules overlap">
          <div className="flex flex-col gap-2 text-[12.5px]">
            {missing.length > 0 && (
              <Callout tone="bad" title={`${missing.length} cost element${missing.length === 1 ? '' : 's'} in staged extracts match no rule`}>
                <ul className="flex flex-col gap-1">{missing.map((m) => <li key={m.costElement} className="flex flex-wrap items-center gap-2"><span className="font-mono">{m.costElement}</span><span className="text-muted">{m.name} · {m.rows} row{m.rows === 1 ? '' : 's'} · {money(m.amount)}</span><Button size="sm" onClick={() => newRule({ costElementFrom: m.costElement, costElementTo: m.costElement, note: m.name })}>Add rule</Button></li>)}</ul>
              </Callout>
            )}
            {broken.map((r) => <Callout key={r.id} tone="bad">P{r.priority} targets a missing or inactive line — rows it matches stay unmapped.</Callout>)}
            {noRule.length > 0 ? <Callout tone="warn" title="Lines no rule sends cost to">{noRule.map((l) => `${l.code} ${l.name}`).join(', ')}. SAP actuals can still reach them by hand, but not automatically.</Callout> : <div className="flex items-center gap-1.5 text-good"><Icon name="check" className="h-3.5 w-3.5" />Every active forecast line has at least one rule.</div>}
            {overlaps.filter((o) => o.kind !== 'refines').map((o, i) => <Callout key={i} tone={o.kind === 'shadowed' ? 'bad' : 'warn'}>{o.text}.</Callout>)}
            {overlaps.filter((o) => o.kind === 'refines').length > 0 && (
              <div><div className="eyebrow mb-1">Intended overlaps</div><ul className="flex flex-col gap-0.5 text-muted">{overlaps.filter((o) => o.kind === 'refines').map((o, i) => <li key={i}><span className="font-mono text-[11.5px] text-ink">P{o.a.priority} / P{o.b.priority}</span> — {o.text}</li>)}</ul></div>
            )}
            {staged.length > 0 && <div className="flex flex-wrap items-center gap-2 border-t border-line pt-2"><span className="text-muted">{staged.length} staged extract{staged.length === 1 ? '' : 's'} waiting.</span><Button size="sm" onClick={rerunAll}>Re-run mapping on staged</Button><a href="#actuals" className="text-link hover:underline">Open SAP actuals</a></div>}
          </div>
        </Panel>
      </div>

      <Panel title="Mapping rules" subtitle={`${rules.filter((r) => r.active).length} active of ${rules.length} · lowest priority number runs first; the first matching rule wins`} pad={false}
        actions={<Button size="sm" variant="primary" onClick={() => newRule()} icon={<Icon name="plus" className="h-3.5 w-3.5" />}>Add rule</Button>}>
        <div className="scroll-x">
          <table className="tbl">
            <thead><tr><th>Order</th><th>Cost element</th><th>WBS suffix</th><th>Text contains</th><th>Target</th><th>Note</th><th className="num">Rows matched</th><th>Active</th><th /></tr></thead>
            <tbody>
              {rules.map((r, i) => (
                <tr key={r.id} className={cx(!r.active && 'text-faint')}>
                  <td className="whitespace-nowrap">
                    <span className="mr-1 inline-block w-9 font-mono text-[12px] font-semibold">P{r.priority}</span>
                    <button className="rounded px-1 text-muted hover:bg-sunken hover:text-ink disabled:opacity-30" disabled={i === 0} onClick={() => swap(i, -1)} aria-label="Run earlier">▲</button>
                    <button className="rounded px-1 text-muted hover:bg-sunken hover:text-ink disabled:opacity-30" disabled={i === rules.length - 1} onClick={() => swap(i, 1)} aria-label="Run later">▼</button>
                  </td>
                  <td className="whitespace-nowrap font-mono text-[12px]">{r.costElementFrom}{r.costElementTo !== r.costElementFrom && <>–{r.costElementTo}</>}</td>
                  <td className="font-mono text-[12px]">{r.wbsSuffix ? `*${r.wbsSuffix}` : <span className="text-faint">any</span>}</td>
                  <td className="text-[12px]">{r.textContains ? `“${r.textContains}”` : <span className="text-faint">any</span>}</td>
                  <td className="whitespace-nowrap"><Badge tone={r.target === 'line' ? (lineById(s, r.lineId)?.active ? 'neutral' : 'bad') : r.target === 'allocate' ? 'info' : 'warn'}>{targetLabel(s, r)}</Badge></td>
                  <td className="min-w-[200px] text-[12px] text-muted">{r.note}</td>
                  <td className="num text-muted">{hits[r.id] ?? 0}</td>
                  <td><input type="checkbox" className="h-4 w-4" checked={r.active} onChange={() => toggle(r)} aria-label={`Rule P${r.priority} active`} /></td>
                  <td className="whitespace-nowrap text-right"><Button size="sm" variant="ghost" onClick={() => setEdit(r)}>Edit</Button><ConfirmButton size="sm" variant="ghost" confirmLabel="Delete rule" onConfirm={() => remove(r)}>Delete</ConfirmButton></td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
        <div className="border-t border-line px-4 py-2 text-[11.5px] text-muted">Changing rules does not touch posted actuals. Staged extracts pick up changes when mapping is re-run; rows resolved by hand are kept.</div>
      </Panel>
      {edit && <RuleModal rule={edit} isNew={!s.settings.sapMappings.some((x) => x.id === edit.id)} onClose={() => setEdit(null)} onSave={save} />}
    </div>

function RuleModal({ rule, isNew, onClose, onSave }: { rule: SapMapping; isNew: boolean; onClose: () => void; onSave: (r: SapMapping, isNew: boolean) => void }) {
  const s = useStore()
  const [r, setR] = useState<SapMapping>(rule)
  const lines = forecastLines(s)
  const clash = s.settings.sapMappings.find((x) => x.id !== r.id && x.priority === r.priority)
  const submit = () => {
    const from = r.costElementFrom.trim(), to = (r.costElementTo.trim() || from)
    if (!/^\d{4,10}$/.test(from) || !/^\d{4,10}$/.test(to)) { toast('Cost elements are numbers, e.g. 5200000', 'bad'); return }
    if (Number(to) < Number(from)) { toast('The range ends before it starts', 'bad'); return }
    if (r.target === 'line' && !r.lineId) { toast('Choose the forecast line', 'bad'); return }
    if (!r.note.trim()) { toast('Add a note — what the rule is for', 'bad'); return }
    onSave({ ...r, costElementFrom: from, costElementTo: to, wbsSuffix: r.wbsSuffix?.trim() || undefined, textContains: r.textContains?.trim() || undefined, lineId: r.target === 'line' ? r.lineId : undefined, note: r.note.trim() }, isNew)
  }
  const affected = useMemo(() => {
    const t = s.actualTxns.filter((x) => s.actualsImports.find((i) => i.id === x.importId)?.status === 'staged' && !x.manual)
    const from = Number(r.costElementFrom), to = Number(r.costElementTo || r.costElementFrom)
    return t.filter((x) => Number(x.costElement) >= from && Number(x.costElement) <= to).length
  }, [s, r.costElementFrom, r.costElementTo])
  return (
    <Modal open onClose={onClose} title={isNew ? 'Add mapping rule' : `Edit rule P${rule.priority}`}
      footer={<><Button variant="ghost" onClick={onClose}>Cancel</Button><Button variant="primary" onClick={submit}>{isNew ? 'Add rule' : 'Save rule'}</Button></>}>
      <div className="flex flex-col gap-3">
        <div className="grid grid-cols-1 gap-3 sm:grid-cols-3">
          <Field label="Priority" hint={clash ? `Same as P${clash.priority} (${clash.costElementFrom}–${clash.costElementTo})` : 'Lower runs first'}><Input type="number" value={r.priority} onChange={(e) => setR({ ...r, priority: Number(e.target.value) })} /></Field>

        </div>
        <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
          <Field label="WBS element ends with" hint="e.g. .02 for the engineering sub-WBS; blank = any"><Input className="font-mono" value={r.wbsSuffix ?? ''} onChange={(e) => setR({ ...r, wbsSuffix: e.target.value })} /></Field>
          <Field label="Text or vendor contains" hint="Case-insensitive; blank = any"><Input value={r.textContains ?? ''} onChange={(e) => setR({ ...r, textContains: e.target.value })} /></Field>
        </div>
        <Field label="Target">
          <div className="flex flex-col gap-1.5">
            <SegButtons value={r.target} onChange={(v) => setR({ ...r, target: v })} options={[{ id: 'line', label: 'Forecast line' }, { id: 'allocate', label: 'Allocate pro-rata' }, { id: 'exclude', label: 'Exclude' }]} />

''} onChange={(e) => setR({ ...r, lineId: e.target.value })} options={lines.map((l) => ({ value: l.id, label: `${l.code} ${l.name} (WBS ${s.settings.wbsFunctions.find((f) => f.id === l.functionId)?.code})` }))} />}
            {r.target === 'allocate' && <span className="text-[12px] text-muted">For overhead, benefits and AFUDC settlements: spread over the project-period’s mapped lines in proportion to their direct cost.</span>}

        </Field>
        <Field label="Note (why this rule exists)"><TextArea rows={2} value={r.note} onChange={(e) => setR({ ...r, note: e.target.value })} /></Field>
        <Checkbox checked={r.active} onChange={(v) => setR({ ...r, active: v })} label="Active" />
        {affected > 0 && <Callout tone="info">{affected} row{affected === 1 ? '' : 's'} in staged extracts fall in this range. Re-run mapping after saving to apply the rule to them.</Callout>}
      </div>
    </Modal>
  )
}

// with optional black-out calendar months (e.g. a holiday paving moratorium).

/** Parse "3, 8, 12" into weights (non-negative numbers). Returns null when any entry is not a number. */
export function parseWeights(text: string): number[] | null {
  const parts = text.split(/[\s,;]+/).map((x) => x.trim()).filter(Boolean)
  if (!parts.length) return null
  const out = parts.map(Number)
  return out.every((x) => Number.isFinite(x) && x >= 0) ? out : null
}

/** Weights a profile produces over `months`, as shares (sum 1), using a draft profile that may not be saved yet. */
export function previewShares(s: AppState, prof: SpreadProfile | string, months: string[]): number[] {
  const st = typeof prof === 'string' ? s : { ...s, settings: { ...s.settings, spreadProfiles: [...s.settings.spreadProfiles.filter((x) => x.id !== prof.id), prof] } }
  const w = profileWeights(st, typeof prof === 'string' ? prof : prof.id, months)
  const W = sum(w) || 1
  return w.map((x) => x / W)
}

/** Drag-to-set bar editor for a weight shape. */
function BarEditor({ weights, onChange }: { weights: number[]; onChange: (w: number[]) => void }) {
  const ref = useRef<HTMLDivElement>(null)
  const [drag, setDrag] = useState(false)
  const top = Math.max(1, ...weights) * 1.25
  const set = (e: React.PointerEvent) => {
    const el = ref.current
    if (!el) return
    const r = el.getBoundingClientRect()
    const i = Math.max(0, Math.min(weights.length - 1, Math.floor(((e.clientX - r.left) / r.width) * weights.length)))
    const v = Math.max(0, Math.round((1 - (e.clientY - r.top) / r.height) * top * 10) / 10)
    if (weights[i] === v) return
    onChange(weights.map((x, j) => (j === i ? v : x)))
  }
  return (
    <div ref={ref} role="group" aria-label="Weight bars" className="flex h-28 cursor-crosshair touch-none select-none items-end gap-px rounded border border-line bg-sunken/50 p-1"
      onPointerDown={(e) => { setDrag(true); (e.target as HTMLElement).setPointerCapture?.(e.pointerId); set(e) }} onPointerMove={(e) => drag && set(e)} onPointerUp={() => setDrag(false)} onPointerLeave={() => setDrag(false)}>
      {weights.map((w, i) => <div key={i} title={`Point ${i + 1}: ${w}`} className={cx('flex-1 rounded-sm', w > 0 ? 'bg-link/70' : 'h-[2px] bg-bad/60')} style={w > 0 ? { height: `${(w / top) * 100}%` } : undefined} />)}
    </div>
  )
}
