// Export / import / reset of the full dataset, plus the demo note.
import { useState } from 'react'
import type { AppState } from '@/core/types'
import { mutate, strip, useStore, useSync } from '@/core/store'
import { SCHEMA_VERSION } from '@/core/seed'
import { saveFile } from '@/core/download'
import { fmtDate, num } from '@/core/util'
import { go } from '@/core/router'
import { Button, Callout, ConfirmButton, KV, Panel } from '@/ui/kit'
import { toast } from '@/ui/toast'
import { FileButton } from './common'

const REQUIRED: (keyof AppState)[] = ['settings', 'people', 'programs', 'templates', 'projects', 'estimates', 'budgets', 'snapshots', 'workflows', 'workflowDefs', 'audit']

export function validateImport(raw: unknown): { ok: true; state: AppState } | { ok: false; error: string } {
  if (!raw || typeof raw !== 'object') return { ok: false, error: 'The file is not a JSON object.' }
  const s = raw as Partial<AppState>
  if (s.schemaVersion !== SCHEMA_VERSION) return { ok: false, error: `Schema version ${s.schemaVersion ?? 'missing'} does not match this release (v${SCHEMA_VERSION}). Export again from the same release.` }
  const missing = REQUIRED.filter((k) => !(k in s))
  if (missing.length) return { ok: false, error: `Missing sections: ${missing.join(', ')}.` }
  if (!Array.isArray(s.projects) || !Array.isArray(s.audit)) return { ok: false, error: 'Projects and audit must be lists.' }
  return { ok: true, state: s as AppState }
}

// Configuration: forecast line items. Each line rolls up to exactly one WBS function and owns the CBS categories
// whose budget belongs to it (a category belongs to one line). SAP actuals land on lines through the mapping table.
import { useMemo, useState } from 'react'
import type { CostCategory, ForecastLine, WbsFunction } from '@/core/types'
import { mutate, useStore } from '@/core/store'
import { CONT_CAT, ESC_CAT } from '@/core/engine/estimates'
import { cx, uid } from '@/core/util'
import { Badge, Button, Callout, ConfirmButton, Field, Input, Modal, Panel, Select, TextArea } from '@/ui/kit'
import { toast } from '@/ui/toast'
import { Icon } from '@/ui/icons'

const SYSTEM = [ESC_CAT, CONT_CAT] // overhead (cat-loaders) is assignable, e.g. to 900 Project Overhead

/** Validation of the line table against cost categories and the WBS function mapping. */
export function lineIssues(lines: ForecastLine[], cats: CostCategory[], fns: WbsFunction[]) {
  const act = lines.filter((l) => l.active)
  const direct = cats.filter((c) => !SYSTEM.includes(c.id))
  const owners = (id: string) => act.filter((l) => l.categoryIds.includes(id))
  const fnOfCat = (id: string) => fns.find((f) => f.categoryIds.includes(id))
  return {
    unassigned: direct.filter((c) => owners(c.id).length === 0),
    multi: direct.filter((c) => owners(c.id).length > 1),
    mismatch: act.flatMap((l) => l.categoryIds.map((c) => ({ l, c, fn: fnOfCat(c) })).filter((x) => x.fn && x.fn.id !== l.functionId)),
    dupCodes: act.filter((l, i) => act.findIndex((x) => x.code.trim() === l.code.trim()) !== i).map((l) => l.code),
    emptyFns: fns.filter((f) => !act.some((l) => l.functionId === f.id)),
  }
}

export default function ForecastLines(_props: { projectId?: string }) {
  const s = useStore()
  const lines = s.settings.forecastLines
  const fns = s.settings.wbsFunctions
  const [edit, setEdit] = useState<ForecastLine | null>(null)
  const [showInactive, setShowInactive] = useState(true)
  const issues = lineIssues(lines, s.categories, fns)
  const usage = useMemo(() => {
    const out: Record<string, { forecasts: number; periods: number; txns: number; rules: number }> = {}
    const u = (id: string) => (out[id] ||= { forecasts: 0, periods: 0, txns: 0, rules: 0 })
    for (const f of s.forecasts) for (const [id, row] of Object.entries(f.lineSpread ?? {})) if (Object.values(row).some(Boolean)) u(id).forecasts++
    for (const c of s.costPeriods) for (const [id, v] of Object.entries(c.byLine ?? {})) if (v) u(id).periods++
    for (const t of s.actualTxns) { if (t.lineId) u(t.lineId).txns++; for (const id of Object.keys(t.allocation ?? {})) u(id).txns++ }
    for (const r of s.settings.sapMappings) if (r.lineId) u(r.lineId).rules++
    return out
  }, [s.forecasts, s.costPeriods, s.actualTxns, s.settings.sapMappings])
  const inUse = (id: string) => { const x = usage[id]; return !!x && x.forecasts + x.periods + x.txns + x.rules > 0 }
  const catName = (id: string) => s.categories.find((c) => c.id === id)

  const save = (l: ForecastLine, isNew: boolean) => {
    const moved: string[] = []
    mutate((d) => {
      for (const o of d.settings.forecastLines) if (o.id !== l.id && o.active) {
        const keep = o.categoryIds.filter((c) => !l.categoryIds.includes(c))
        if (keep.length !== o.categoryIds.length) { moved.push(o.code); o.categoryIds = keep }
      }
      const i = d.settings.forecastLines.findIndex((x) => x.id === l.id)
      if (i >= 0) d.settings.forecastLines[i] = l
      else d.settings.forecastLines.push(l)
      d.settings.forecastLines.sort((a, b) => a.code.localeCompare(b.code, undefined, { numeric: true }))
    }, { entity: 'forecast-line', entityId: l.id, action: isNew ? 'create' : 'update', summary: `${isNew ? 'Added' : 'Updated'} forecast line ${l.code} ${l.name} (WBS ${fns.find((f) => f.id === l.functionId)?.code}; categories ${l.categoryIds.map((c) => catName(c)?.code ?? c).join(', ') || 'none'})${moved.length ? `; categories moved from ${moved.join(', ')}` : ''}` })
    toast(isNew ? `Line ${l.code} added` : `Line ${l.code} saved`)
    setEdit(null)
  }
  const setActive = (l: ForecastLine, active: boolean) => {
    mutate((d) => { const x = d.settings.forecastLines.find((y) => y.id === l.id); if (x) x.active = active }, { entity: 'forecast-line', entityId: l.id, action: active ? 'activate' : 'deactivate', summary: `${active ? 'Reactivated' : 'Deactivated'} forecast line ${l.code} ${l.name}` })
    toast(active ? `${l.code} reactivated` : `${l.code} deactivated — history kept`)
  }
  const remove = (l: ForecastLine) => {
    mutate((d) => { d.settings.forecastLines = d.settings.forecastLines.filter((x) => x.id !== l.id) }, { entity: 'forecast-line', entityId: l.id, action: 'delete', summary: `Deleted unused forecast line ${l.code} ${l.name}` })
    toast(`${l.code} deleted`)
  }
  const add = () => {
    const fn = fns[0]
    const n = lines.filter((l) => l.functionId === fn.id).length
    setEdit({ id: uid('fl'), code: `${fn.code}.${String((n + 1) * 10).padStart(2, '0')}`, name: '', functionId: fn.id, categoryIds: [], active: true })
  }

  const shown = lines.filter((l) => showInactive || l.active)
  return (
    <div className="flex flex-col gap-4">
      <Callout tone="info" title="How lines are used">
        Forecasts are entered by line item and roll up to the WBS functions (CBS Level 3 codes 100–999). A line’s budget is the CBS categories it owns (overhead goes to 900 Project Overhead; escalation is allocated pro-rata); SAP actuals reach it through the rules on the <b>SAP mapping</b> tab. A line that has forecasts, actuals or rules can be deactivated but not deleted, so history stays traceable.
      </Callout>
      {(issues.unassigned.length > 0 || issues.multi.length > 0 || issues.mismatch.length > 0 || issues.dupCodes.length > 0 || issues.emptyFns.length > 0) ? (
        <Callout tone="warn" title="Coverage checks">
          <ul className="list-disc pl-4">
            {issues.unassigned.length > 0 && <li>{issues.unassigned.length} cost categor{issues.unassigned.length === 1 ? 'y has' : 'ies have'} no line, so their budget is left out of line totals: {issues.unassigned.map((c) => c.name).join(', ')}</li>}
            {issues.multi.map((c) => <li key={c.id}>{c.name} belongs to more than one active line</li>)}
            {issues.mismatch.map((x, i) => <li key={i}>{x.l.code} owns {catName(x.c)?.name}, which the WBS configuration maps to {x.fn?.code} {x.fn?.name} — budget and actuals will land in different functions</li>)}
            {issues.dupCodes.map((c) => <li key={c}>Duplicate line code {c}</li>)}
            {issues.emptyFns.length > 0 && <li>No active line under {issues.emptyFns.map((f) => `${f.code} ${f.name}`).join(', ')}</li>}
          </ul>
        </Callout>
      ) : <Callout tone="good">Every cost category belongs to exactly one active line, and every line sits under the WBS function its categories map to.</Callout>}

      <Panel title="Forecast line items" subtitle={`${lines.filter((l) => l.active).length} active · ${lines.filter((l) => !l.active).length} inactive · under ${fns.length} WBS functions`} pad={false}
        actions={<>
          <label className="flex items-center gap-1.5 text-[12px] text-muted"><input type="checkbox" className="h-3.5 w-3.5" checked={showInactive} onChange={(e) => setShowInactive(e.target.checked)} />Show inactive</label>
          <Button size="sm" variant="primary" onClick={add} icon={<Icon name="plus" className="h-3.5 w-3.5" />}>Add line</Button>
        </>}>
        <div className="scroll-x">
          <table className="tbl">
            <thead><tr><th>Code</th><th>Line item</th><th>Cost categories (budget)</th><th className="num">SAP rules</th><th>In use</th><th>Status</th><th /></tr></thead>
            <tbody>
              {fns.map((fn) => {
                const ls = shown.filter((l) => l.functionId === fn.id)
                return [
                  <tr key={fn.id} className="bg-sunken/50"><td colSpan={7} className="py-1.5 font-semibold"><span className="font-mono text-[11px] text-faint">{fn.code}</span> {fn.name}{!ls.length && <span className="ml-2 text-[12px] font-normal text-warn">no lines</span>}</td></tr>,
                  ...ls.map((l) => {
                    const u = usage[l.id]
                    return (
                      <tr key={l.id} className={cx(!l.active && 'text-muted')}>
                        <td className="whitespace-nowrap font-mono text-[12px] font-semibold">{l.code}</td>
                        <td className="min-w-[180px]"><div className={cx('font-semibold', !l.active && 'line-through')}>{l.name}</div>{l.note && <div className="text-[11.5px] text-muted">{l.note}</div>}</td>
                        <td className="min-w-[160px]"><div className="flex flex-wrap gap-1">{l.categoryIds.length ? l.categoryIds.map((c) => <Badge key={c} title={catName(c)?.name}>{catName(c)?.code ?? c} {catName(c)?.name}</Badge>) : <span className="text-[12px] text-faint">none — no budget</span>}</div></td>
                        <td className="num">{u?.rules ? u.rules : <span className="text-warn" title="No SAP rule sends cost here">0</span>}</td>
                        <td className="whitespace-nowrap text-[12px] text-muted">{u ? [u.forecasts && `${u.forecasts} forecasts`, u.periods && `${u.periods} actual periods`, u.txns && `${u.txns} SAP rows`].filter(Boolean).join(' · ') || '—' : '—'}</td>

function LineModal({ line, isNew, onClose, onSave }: { line: ForecastLine; isNew: boolean; onClose: () => void; onSave: (l: ForecastLine, isNew: boolean) => void }) {
  const s = useStore()
  const [l, setL] = useState<ForecastLine>(line)
  const others = s.settings.forecastLines.filter((x) => x.id !== l.id && x.active)
  const ownerOf = (c: string) => others.find((x) => x.categoryIds.includes(c))
  const direct = s.categories.filter((c) => !SYSTEM.includes(c.id))
  const fn = s.settings.wbsFunctions.find((f) => f.id === l.functionId)
  const moves = l.categoryIds.map((c) => ({ c, o: ownerOf(c) })).filter((x) => x.o)
  const dup = others.some((x) => x.code.trim() === l.code.trim())
  const submit = () => {
    if (!l.code.trim() || !l.name.trim()) { toast('A line needs a code and a name', 'bad'); return }
    if (dup) { toast(`Code ${l.code} is already used`, 'bad'); return }
    onSave({ ...l, code: l.code.trim(), name: l.name.trim(), note: l.note?.trim() || undefined }, isNew)
  }
  const toggle = (c: string) => setL({ ...l, categoryIds: l.categoryIds.includes(c) ? l.categoryIds.filter((x) => x !== c) : [...l.categoryIds, c] })
  return (
    <Modal open onClose={onClose} title={isNew ? 'Add forecast line' : `Edit ${line.code} ${line.name}`}
      footer={<><Button variant="ghost" onClick={onClose}>Cancel</Button><Button variant="primary" onClick={submit}>{isNew ? 'Add line' : 'Save line'}</Button></>}>
      <div className="flex flex-col gap-3">
        <div className="grid grid-cols-1 gap-3 sm:grid-cols-[120px_minmax(0,1fr)]">
          <Field label="Code" hint={dup ? 'Already used' : 'e.g. 06.30'}><Input className="font-mono" value={l.code} onChange={(e) => setL({ ...l, code: e.target.value })} /></Field>
          <Field label="Name"><Input value={l.name} onChange={(e) => setL({ ...l, name: e.target.value })} placeholder="e.g. Directional drilling" /></Field>
        </div>
        <Field label="Rolls up to WBS function"><Select value={l.functionId} onChange={(e) => setL({ ...l, functionId: e.target.value })} options={s.settings.wbsFunctions.map((f) => ({ value: f.id, label: `${f.code} ${f.name}` }))} /></Field>
        <Field label="Cost categories whose budget belongs to this line" hint="A category belongs to one line; choosing one owned by another line moves it here when you save.">
          <div className="flex flex-wrap gap-1">
            {direct.map((c) => {
              const on = l.categoryIds.includes(c.id)
              const o = ownerOf(c.id)
              const fnOf = s.settings.wbsFunctions.find((f) => f.categoryIds.includes(c.id))
              return (
                <button key={c.id} type="button" aria-pressed={on} onClick={() => toggle(c.id)} title={`${c.name}${o ? ` — now on ${o.code}` : ''}${fnOf ? ` · WBS ${fnOf.code}` : ''}`}
                  className={cx('rounded border px-2 py-0.5 text-[12px] font-semibold', on ? 'border-ink bg-ink text-surface' : o ? 'border-dashed border-line bg-surface text-faint hover:text-ink' : 'border-line bg-surface text-muted hover:border-ink/40 hover:text-ink')}>
                  {c.name}{o && !on ? <span className="ml-1 font-mono text-[10.5px]">{o.code}</span> : null}
                </button>
              )
            })}
          </div>
        </Field>
        {moves.length > 0 && <Callout tone="warn">{moves.map((x) => `${s.categories.find((c) => c.id === x.c)?.name} moves from ${x.o!.code}`).join('; ')}.</Callout>}
        {l.categoryIds.some((c) => { const f = s.settings.wbsFunctions.find((x) => x.categoryIds.includes(c)); return f && f.id !== l.functionId }) && <Callout tone="warn">Some of these categories map to a different WBS function than {fn?.code} {fn?.name} in the WBS configuration.</Callout>}
        <Field label="Note"><TextArea rows={2} value={l.note ?? ''} onChange={(e) => setL({ ...l, note: e.target.value })} placeholder="What belongs on this line" /></Field>

  )
}

const KINDS: { value: MapLayer['kind']; label: string }[] = [
  { value: 'arcgis-feature', label: 'Esri feature layer (…/FeatureServer/0 or …/MapServer/0)' },

export default function GisSettings() {

  const [editing, setEditing] = useState<string | null>(null)

  const dirty = JSON.stringify(draft) !== JSON.stringify(gis)

function FieldValues() {

  const [vals, setVals, dirty] = useDraft(Object.fromEntries(projects.map((p) => [p.id, p.customFields?.[key] ?? ''])))

      { entity: 'project', entityId: key, action: 'update', summary: `Set ${def.label} on ${changed.length} project${changed.length === 1 ? '' : 's'}: ${changed.slice(0, 6).map((p) => `${p.code}=${vals[p.id] || '(blank)'}`).join(', ')}` })
    toast(`${def.label} saved on ${changed.length} project${changed.length === 1 ? '' : 's'}`)
  }

      <div className="mb-2 flex flex-wrap items-end gap-2">

            </label>
          )
        })}
      </div>

  )
}

function FundingGrid() {

  const [draft, setDraft, dirty, discard] = useDraft(funding)
  const planned = useMemo(() => {
    const out: Record<string, number> = {}
    for (const p of projects) {
      if (p.status === 'cancelled') continue

      for (const [y, v] of Object.entries(b?.byYear ?? {})) out[`${p.programId}|${y}`] = (out[`${p.programId}|${y}`] || 0) + v
    }
    return out

    const i = d.findIndex((f) => f.programId === pid && f.year === y)

// Configuration: the SAP → forecast line mapping table. Rules run in priority order (lowest first) and the first
// match wins: cost-element range, optional WBS-element suffix, optional text/vendor match → a forecast line,
// allocate pro-rata (loader settlements) or exclude (non-project cost). Includes a row tester and coverage checks.
import { useMemo, useState } from 'react'
import type { SapMapping } from '@/core/types'
import { mutate, useStore } from '@/core/store'
import { forecastLines, lineById } from '@/core/engine/lines'
import { remap } from '@/core/engine/sap'
import { cx, money, uid } from '@/core/util'
import { Badge, Button, Callout, Checkbox, ConfirmButton, Field, Input, Modal, Panel, SegButtons, Select, TextArea } from '@/ui/kit'
import { toast } from '@/ui/toast'
import { Icon } from '@/ui/icons'
import { brokenRules, linesWithoutRule, ruleOverlaps, targetLabel, testRow, unmappedElements } from '../actuals/lib'

export default function SapMappings(_props: { projectId?: string }) {
  const s = useStore()
  const rules = useMemo(() => [...s.settings.sapMappings].sort((a, b) => a.priority - b.priority), [s.settings.sapMappings])
  const [edit, setEdit] = useState<SapMapping | null>(null)
  const overlaps = ruleOverlaps(rules)
  const noRule = linesWithoutRule(s)
  const broken = brokenRules(s)
  const missing = unmappedElements(s)
  const staged = s.actualsImports.filter((x) => x.status === 'staged')
  const hits = useMemo(() => { const out: Record<string, number> = {}; for (const t of s.actualTxns) if (t.ruleId && !t.manual) out[t.ruleId] = (out[t.ruleId] || 0) + 1; return out }, [s.actualTxns])

  const save = (r: SapMapping, isNew: boolean) => {
    mutate((d) => {
      const i = d.settings.sapMappings.findIndex((x) => x.id === r.id)
      if (i >= 0) d.settings.sapMappings[i] = r
      else d.settings.sapMappings.push(r)
    }, { entity: 'sap-mapping', entityId: r.id, action: isNew ? 'create' : 'update', summary: `${isNew ? 'Added' : 'Updated'} SAP mapping rule P${r.priority}: ${r.costElementFrom}–${r.costElementTo}${r.wbsSuffix ? ` WBS *${r.wbsSuffix}` : ''}${r.textContains ? ` text "${r.textContains}"` : ''} → ${targetLabel(s, r)}` })
    toast(isNew ? `Rule P${r.priority} added` : `Rule P${r.priority} saved`)
    setEdit(null)
  }
  const swap = (i: number, dir: -1 | 1) => {
    const a = rules[i], b = rules[i + dir]
    if (!a || !b) return
    mutate((d) => {
      const x = d.settings.sapMappings.find((y) => y.id === a.id)!, y = d.settings.sapMappings.find((z) => z.id === b.id)!
      const pa = x.priority
      if (pa === y.priority) x.priority = pa + dir
      else { x.priority = y.priority; y.priority = pa }
    }, { entity: 'sap-mapping', entityId: a.id, action: 'reorder', summary: `Moved SAP mapping rule ${a.costElementFrom}–${a.costElementTo} ${dir < 0 ? 'ahead of' : 'after'} ${b.costElementFrom}–${b.costElementTo}` })
  }
  const toggle = (r: SapMapping) => {
    mutate((d) => { const x = d.settings.sapMappings.find((y) => y.id === r.id); if (x) x.active = !x.active }, { entity: 'sap-mapping', entityId: r.id, action: r.active ? 'deactivate' : 'activate', summary: `${r.active ? 'Deactivated' : 'Activated'} SAP mapping rule P${r.priority} ${r.costElementFrom}–${r.costElementTo}` })
    toast(r.active ? `P${r.priority} deactivated` : `P${r.priority} activated`)
  }
  const remove = (r: SapMapping) => {
    mutate((d) => { d.settings.sapMappings = d.settings.sapMappings.filter((x) => x.id !== r.id) }, { entity: 'sap-mapping', entityId: r.id, action: 'delete', summary: `Deleted SAP mapping rule P${r.priority} ${r.costElementFrom}–${r.costElementTo} → ${targetLabel(s, r)}` })
    toast(`P${r.priority} deleted`)
  }
  const newRule = (patch: Partial<SapMapping> = {}) => {
    const maxBelowAlloc = Math.max(0, ...rules.filter((r) => r.target === 'line').map((r) => r.priority))
    setEdit({ id: uid('map'), priority: maxBelowAlloc + 1, costElementFrom: '', costElementTo: '', target: 'line', lineId: forecastLines(s)[0]?.id, note: '', active: true, ...patch })
  }
  const rerunAll = () => {
    mutate((d) => { for (const imp of d.actualsImports.filter((x) => x.status === 'staged')) remap(d, imp.id) }, { entity: 'sap-mapping', entityId: 'remap', action: 'remap', summary: `Re-ran the mapping table over ${staged.length} staged SAP extract(s)` })
    toast(`Mapping re-run on ${staged.length} staged extract${staged.length === 1 ? '' : 's'}`)
  }

  return (
    <div className="flex flex-col gap-4">
      <div className="grid grid-cols-1 gap-4 xl:grid-cols-[minmax(0,1.4fr)_minmax(0,1fr)]">
        <RowTester />
        <Panel title="Coverage checks" subtitle="What the table misses, and where rules overlap">
          <div className="flex flex-col gap-2 text-[12.5px]">
            {missing.length > 0 && (
              <Callout tone="bad" title={`${missing.length} cost element${missing.length === 1 ? '' : 's'} in staged extracts match no rule`}>
                <ul className="flex flex-col gap-1">{missing.map((m) => <li key={m.costElement} className="flex flex-wrap items-center gap-2"><span className="font-mono">{m.costElement}</span><span className="text-muted">{m.name} · {m.rows} row{m.rows === 1 ? '' : 's'} · {money(m.amount)}</span><Button size="sm" onClick={() => newRule({ costElementFrom: m.costElement, costElementTo: m.costElement, note: m.name })}>Add rule</Button></li>)}</ul>
              </Callout>
            )}
            {broken.map((r) => <Callout key={r.id} tone="bad">P{r.priority} targets a missing or inactive line — rows it matches stay unmapped.</Callout>)}
            {noRule.length > 0 ? <Callout tone="warn" title="Lines no rule sends cost to">{noRule.map((l) => `${l.code} ${l.name}`).join(', ')}. SAP actuals can still reach them by hand, but not automatically.</Callout> : <div className="flex items-center gap-1.5 text-good"><Icon name="check" className="h-3.5 w-3.5" />Every active forecast line has at least one rule.</div>}
            {overlaps.filter((o) => o.kind !== 'refines').map((o, i) => <Callout key={i} tone={o.kind === 'shadowed' ? 'bad' : 'warn'}>{o.text}.</Callout>)}
            {overlaps.filter((o) => o.kind === 'refines').length > 0 && (
              <div><div className="eyebrow mb-1">Intended overlaps</div><ul className="flex flex-col gap-0.5 text-muted">{overlaps.filter((o) => o.kind === 'refines').map((o, i) => <li key={i}><span className="font-mono text-[11.5px] text-ink">P{o.a.priority} / P{o.b.priority}</span> — {o.text}</li>)}</ul></div>
            )}
            {staged.length > 0 && <div className="flex flex-wrap items-center gap-2 border-t border-line pt-2"><span className="text-muted">{staged.length} staged extract{staged.length === 1 ? '' : 's'} waiting.</span><Button size="sm" onClick={rerunAll}>Re-run mapping on staged</Button><a href="#actuals" className="text-link hover:underline">Open SAP actuals</a></div>}
          </div>
        </Panel>
      </div>

      <Panel title="Mapping rules" subtitle={`${rules.filter((r) => r.active).length} active of ${rules.length} · lowest priority number runs first; the first matching rule wins`} pad={false}
        actions={<Button size="sm" variant="primary" onClick={() => newRule()} icon={<Icon name="plus" className="h-3.5 w-3.5" />}>Add rule</Button>}>
        <div className="scroll-x">
          <table className="tbl">
            <thead><tr><th>Order</th><th>Cost element</th><th>WBS suffix</th><th>Text contains</th><th>Target</th><th>Note</th><th className="num">Rows matched</th><th>Active</th><th /></tr></thead>
            <tbody>
              {rules.map((r, i) => (
                <tr key={r.id} className={cx(!r.active && 'text-faint')}>
                  <td className="whitespace-nowrap">
                    <span className="mr-1 inline-block w-9 font-mono text-[12px] font-semibold">P{r.priority}</span>
                    <button className="rounded px-1 text-muted hover:bg-sunken hover:text-ink disabled:opacity-30" disabled={i === 0} onClick={() => swap(i, -1)} aria-label="Run earlier">▲</button>
                    <button className="rounded px-1 text-muted hover:bg-sunken hover:text-ink disabled:opacity-30" disabled={i === rules.length - 1} onClick={() => swap(i, 1)} aria-label="Run later">▼</button>
                  </td>
                  <td className="whitespace-nowrap font-mono text-[12px]">{r.costElementFrom}{r.costElementTo !== r.costElementFrom && <>–{r.costElementTo}</>}</td>
                  <td className="font-mono text-[12px]">{r.wbsSuffix ? `*${r.wbsSuffix}` : <span className="text-faint">any</span>}</td>
                  <td className="text-[12px]">{r.textContains ? `“${r.textContains}”` : <span className="text-faint">any</span>}</td>
                  <td className="whitespace-nowrap"><Badge tone={r.target === 'line' ? (lineById(s, r.lineId)?.active ? 'neutral' : 'bad') : r.target === 'allocate' ? 'info' : 'warn'}>{targetLabel(s, r)}</Badge></td>

                </tr>

      </Panel>
      {edit && <RuleModal rule={edit} isNew={!s.settings.sapMappings.some((x) => x.id === edit.id)} onClose={() => setEdit(null)} onSave={save} />}

/** Parse "3, 8, 12" into weights (non-negative numbers). Returns null when any entry is not a number. */
export function parseWeights(text: string): number[] | null {
  const parts = text.split(/[\s,;]+/).map((x) => x.trim()).filter(Boolean)
  if (!parts.length) return null
  const out = parts.map(Number)
  return out.every((x) => Number.isFinite(x) && x >= 0) ? out : null
}

/** Weights a profile produces over `months`, as shares (sum 1), using a draft profile that may not be saved yet. */
export function previewShares(s: AppState, prof: SpreadProfile | string, months: string[]): number[] {
  const st = typeof prof === 'string' ? s : { ...s, settings: { ...s.settings, spreadProfiles: [...s.settings.spreadProfiles.filter((x) => x.id !== prof.id), prof] } }
  const w = profileWeights(st, typeof prof === 'string' ? prof : prof.id, months)
  const W = sum(w) || 1
  return w.map((x) => x / W)
}

/** Drag-to-set bar editor for a weight shape. */
function BarEditor({ weights, onChange }: { weights: number[]; onChange: (w: number[]) => void }) {
  const ref = useRef<HTMLDivElement>(null)
  const [drag, setDrag] = useState(false)
  const top = Math.max(1, ...weights) * 1.25
  const set = (e: React.PointerEvent) => {
    const el = ref.current
    if (!el) return
    const r = el.getBoundingClientRect()
    const i = Math.max(0, Math.min(weights.length - 1, Math.floor(((e.clientX - r.left) / r.width) * weights.length)))
    const v = Math.max(0, Math.round((1 - (e.clientY - r.top) / r.height) * top * 10) / 10)
    if (weights[i] === v) return
    onChange(weights.map((x, j) => (j === i ? v : x)))
  }
  return (
    <div ref={ref} role="group" aria-label="Weight bars" className="flex h-28 cursor-crosshair touch-none select-none items-end gap-px rounded border border-line bg-sunken/50 p-1"
      onPointerDown={(e) => { setDrag(true); (e.target as HTMLElement).setPointerCapture?.(e.pointerId); set(e) }} onPointerMove={(e) => drag && set(e)} onPointerUp={() => setDrag(false)} onPointerLeave={() => setDrag(false)}>
      {weights.map((w, i) => <div key={i} title={`Point ${i + 1}: ${w}`} className={cx('flex-1 rounded-sm', w > 0 ? 'bg-link/70' : 'h-[2px] bg-bad/60')} style={w > 0 ? { height: `${(w / top) * 100}%` } : undefined} />)}
    </div>
  )
}

function diffTemplate(a: LifecycleTemplate, b: LifecycleTemplate): string[] {
  const out: string[] = []
  if (a.name !== b.name) out.push(`renamed to "${b.name}"`)
  if (a.description !== b.description) out.push('description edited')
  b.stages.forEach((sb, i) => {
    const sa = a.stages[i]
    if (!sa) return
    const parts: string[] = []
    if (sa.name !== sb.name) parts.push('name')
    if (sa.purpose !== sb.purpose) parts.push('purpose')
    if (sa.gate.name !== sb.gate.name) parts.push('gate name')
    if (JSON.stringify(sa.gate.decisionRoles) !== JSON.stringify(sb.gate.decisionRoles)) parts.push(`decision roles → ${sb.gate.decisionRoles.join(', ')}`)
    if (sa.gate.requiredEstimateClass !== sb.gate.requiredEstimateClass) parts.push(`estimate class → ${sb.gate.requiredEstimateClass ? 'Class ' + sb.gate.requiredEstimateClass : 'none'}`)
    if (sa.gate.budgetEvent !== sb.gate.budgetEvent) parts.push(`budget event → ${sb.gate.budgetEvent ?? 'none'}`)
    if (JSON.stringify(sa.gate.unlocks) !== JSON.stringify(sb.gate.unlocks)) parts.push(`unlocks → ${sb.gate.unlocks.join(', ') || 'none'}`)
    const ida = sa.gate.deliverables.map((x) => x.id), idb = sb.gate.deliverables.map((x) => x.id)
    const added = sb.gate.deliverables.filter((x) => !ida.includes(x.id)).map((x) => x.name)
    const removed = sa.gate.deliverables.filter((x) => !idb.includes(x.id)).map((x) => x.name)
    const edited = sb.gate.deliverables.filter((x) => { const o = sa.gate.deliverables.find((y) => y.id === x.id); return o && JSON.stringify(o) !== JSON.stringify(x) }).map((x) => x.name)
    if (added.length) parts.push(`added ${added.join(', ')}`)
    if (removed.length) parts.push(`removed ${removed.join(', ')}`)
    if (edited.length) parts.push(`edited ${edited.join(', ')}`)
    if (!added.length && !removed.length && ida.filter((x) => idb.includes(x)).join() !== idb.filter((x) => ida.includes(x)).join()) parts.push('deliverables reordered')
    if (parts.length) out.push(`Stage ${i + 1} ${sb.name}: ${parts.join(', ')}`)
  })
  return out
}

// ───────────────────────────── Backtest
function Metric({ label, value, ok, limit }: { label: string; value: string; ok?: boolean; limit?: string }) {
  return (
    <div className={cx('rounded-md border px-3 py-2', ok === false ? 'border-bad/50 bg-bad/5' : 'border-line')}>
      <div className="text-[10.5px] font-semibold uppercase tracking-wide text-faint">{label}</div>
      <div className="font-display text-[20px] font-bold tabular-nums">{value}</div>
      {limit && <div className={cx('text-[11px]', ok ? 'text-good' : 'text-bad')}>{ok ? 'Meets' : 'Fails'} {limit}</div>}
    </div>
  )
}

function BacktestView({ s, m, run, onRun }: { s: AppState; m: FactorModel; run: { result: BacktestResult; dirty: boolean } | null; onRun: () => void }) {
  if (!run) return (
    <Panel title="Backtest" subtitle={`${s.backtestProjects.length} fictional closed projects with known drivers and final cost`}>

'+' : ''}{m.backtest.bias}%, P10–P90 coverage {m.backtest.coverage}% (n = {m.backtest.n}) — {m.backtest.passed ? 'passed the gate' : 'did not pass'}.</p>}

            <Metric label="MAPE" value={`${bt.mape.toFixed(1)}%`} ok={g.checks[0].ok} limit={g.checks[0].limit} />
            <Metric label="Median abs % error" value={`${bt.medianApe.toFixed(1)}%`} />

          <div><div className="eyebrow mb-1">Predicted vs actual (log scale)</div><Scatter points={bt.points} /></div>
        </div>
      </Panel>
      <div className="grid grid-cols-1 gap-3 xl:grid-cols-2">
        <Panel pad={false} title="By asset type">

'+' : ''}{r.bias.toFixed(1)}%</td><td className="num">{r.coverage.toFixed(0)}%</td></tr>)}</tbody>
          </table></div>
          {bt.byAsset.some((r) => r.n < 5) && <p className="px-4 pb-3 text-[11.5px] text-muted">Asset types with fewer than 5 projects are reported but carry little weight.</p>}
        </Panel>

  )

/** Predicted vs actual on log axes, with the 1:1 line and ±25% guides; points outside P10–P90 in red. */
function Scatter({ points }: { points: BacktestResult['points'] }) {
  const W = 720, H = 300, L = 64, R = 16, T = 12, B = 34
  const vals = points.flatMap((p) => [p.predicted, p.actual]).filter((v) => v > 0)
  const lo = Math.log10(Math.min(...vals) * 0.8), hi = Math.log10(Math.max(...vals) * 1.25)
  const X = (v: number) => L + ((Math.log10(v) - lo) / (hi - lo)) * (W - L - R)
  const Y = (v: number) => T + (1 - (Math.log10(v) - lo) / (hi - lo)) * (H - T - B)
  const ticks = Array.from({ length: Math.floor(hi) - Math.ceil(lo) + 1 }, (_, k) => Math.pow(10, Math.ceil(lo) + k))
  const a = Math.pow(10, lo), b = Math.pow(10, hi)
  return (
    <div>
      <svg viewBox={`0 0 ${W} ${H}`} className="w-full" role="img" aria-label="Predicted versus actual cost">
        {ticks.map((t) => <g key={t}><line x1={L} x2={W - R} y1={Y(t)} y2={Y(t)} stroke="rgb(var(--line))" /><text x={L - 6} y={Y(t) + 4} textAnchor="end" fontSize={11} fill="rgb(var(--muted))">{money(t)}</text><text x={X(t)} y={H - 12} textAnchor="middle" fontSize={11} fill="rgb(var(--muted))">{money(t)}</text></g>)}
        <line x1={X(a)} y1={Y(a)} x2={X(b)} y2={Y(b)} stroke="rgb(var(--ink))" strokeWidth={1.2} />
        <line x1={X(a)} y1={Y(a * 1.25)} x2={X(b / 1.25)} y2={Y(b)} stroke="rgb(var(--muted))" strokeDasharray="4 4" />
        <line x1={X(a * 1.25)} y1={Y(a)} x2={X(b)} y2={Y(b / 1.25)} stroke="rgb(var(--muted))" strokeDasharray="4 4" />
        {points.map((p) => <circle key={p.id} cx={X(p.actual)} cy={Y(p.predicted)} r={4} fill={p.inside ? 'rgb(var(--link))' : 'rgb(var(--bad))'} opacity={0.75}><title>{`${p.code} ${p.name}: predicted ${money(p.predicted)}, actual ${money(p.actual)} (${p.err > 0 ? '+' : ''}${(p.err * 100).toFixed(0)}%)`}</title></circle>)}
      </svg>
      <div className="mt-1 flex flex-wrap gap-x-4 gap-y-1 text-[11px] text-muted">
        <span>x: actual · y: predicted</span>
        <span className="inline-flex items-center gap-1"><span className="inline-block h-2.5 w-2.5 rounded-full bg-link" />Actual inside P10–P90</span>
        <span className="inline-flex items-center gap-1"><span className="inline-block h-2.5 w-2.5 rounded-full bg-bad" />Outside</span>
        <span>Solid line 1:1 · dashed ±25%</span>
      </div>
    </div>
  )
}

// ───────────────────────────── Compare
function Compare({ s, m, pub }: { s: AppState; m: FactorModel; pub?: FactorModel }) {
  const data = useMemo(() => {
    if (!pub || pub.id === m.id) return
    const a = backtest(s, pub), b = backtest(s, m)
    const projects = s.projects.filter((p) => p.status === 'active' && p.stageIndex <= 1).slice(0, 10).map((p) => {
      const i = factorInputsFromProject(s, p)
      const x = priceFactor(s, pub, i), y = priceFactor(s, m, i)
      return { p, a: x.total, b: y.total }
    }).filter((r) => r.a > 0)
    return { a, b, diff: diffModels(pub, m), projects }
  }, [s, m, pub])
  if (!data || !pub) return <Panel title="Draft vs published"><p className="text-[13px] text-muted">{pub && pub.id === m.id ? 'You are viewing the published model. Open the draft (or start one) to compare it with the published version.' : 'No published model to compare with.'}</p></Panel>
  const row = (label: string, x: number, y: number, unit = '%', better: 'lower' | 'abs' | 'none' = 'lower') => {
    const d = y - x
    const good = better === 'lower' ? d < 0 : better === 'abs' ? Math.abs(y) < Math.abs(x) : false
    return <tr key={label}><td className="text-[12.5px]">{label}</td><td className="num">{x.toFixed(1)}{unit}</td><td className="num font-semibold">{y.toFixed(1)}{unit}</td><td className={cx('num', d && better !== 'none' ? (good ? 'text-good' : 'text-bad') : 'text-muted')}>{d > 0 ? '+' : ''}{d.toFixed(1)}</td></tr>
  }
  return (
    <div className="grid grid-cols-1 gap-3 xl:grid-cols-2">
      <Panel pad={false} title="Backtest: published vs this version">
        <div className="scroll-x"><table className="tbl">
          <thead><tr><th>Metric</th><th className="num">v{pub.version}</th><th className="num">v{m.version}</th><th className="num">Δ</th></tr></thead>
          <tbody>
            {row('MAPE', data.a.mape, data.b.mape)}
            {row('Median abs % error', data.a.medianApe, data.b.medianApe)}
            {row('Bias', data.a.bias, data.b.bias, '%', 'abs')}
            {row('Inside P10–P90', data.a.coverage, data.b.coverage, '%', 'none')}
          </tbody>
        </table></div>
      </Panel>
      <Panel title={`What changed (v${pub.version} → v${m.version})`} subtitle="Weights, multipliers, uncertainty and base rates, including unsaved edits">
        {data.diff.length ? <ul className="flex flex-col gap-1 text-[12.5px]">{data.diff.map((x) => <li key={x} className="rounded border border-line px-2 py-1 font-mono text-[11.5px]">{x}</li>)}</ul> : <p className="text-[13px] text-muted">No differences.</p>}
      </Panel>
      <Panel pad={false} title="Effect on early-stage projects" subtitle="Class 5 total under each version, same inputs" className="xl:col-span-2">
        <div className="scroll-x"><table className="tbl">
          <thead><tr><th>Project</th><th className="num">v{pub.version}</th><th className="num">v{m.version}</th><th className="num">Δ</th></tr></thead>
          <tbody>{data.projects.map((r) => <tr key={r.p.id}><td className="min-w-[200px] text-[12.5px]"><span className="font-mono text-[11.5px] text-muted">{r.p.code}</span> {r.p.name}</td><td className="num">{money(r.a)}</td><td className="num font-semibold">{money(r.b)}</td><td className={cx('num', r.b > r.a ? 'text-bad' : r.b < r.a ? 'text-good' : 'text-muted')}>{r.b === r.a ? '—' : `${r.b > r.a ? '+' : ''}${pct(r.b / r.a - 1, 1)}`}</td></tr>)}</tbody>
        </table></div>
      </Panel>
    </div>
  )
}

function History({ s }: { s: AppState }) {
  const versions = modelsByVersion(s)
  const log = s.audit.filter((a) => a.entity === 'factor-model').slice().sort((a, b) => b.at.localeCompare(a.at)).slice(0, 25)
  const [open, setOpen] = useState<string | null>(null)
  return (
    <div className="flex flex-col gap-3">
      <Panel pad={false} title="Versions" subtitle="Exactly one version is published at a time; approval retires the one it replaces. Records are never deleted.">
        <div className="scroll-x"><table className="tbl">
          <thead><tr><th>Version</th><th>Status</th><th>Created</th><th>Approved</th><th className="num">Backtest</th><th>Changes from parent</th></tr></thead>
          <tbody>
            {versions.map((m) => {
              const parent = s.factorModels.find((x) => x.version === m.parentVersion)
              const diff = parent ? diffModels(parent, m) : []

              return (
                <tr key={m.id}>
                  <td className="font-mono font-semibold">v{m.version}</td>

                  <td className="whitespace-nowrap text-[12px]">{m.approvedAt ? <>{fmtDate(m.approvedAt.slice(0, 10))}<div className="text-muted">{personName(s, m.approvedById)}</div></> : '—'}{m.retiredAt && <div className="text-faint">retired {fmtDate(m.retiredAt.slice(0, 10))}</div>}</td>
                  <td className="num whitespace-nowrap text-[12px]">{m.backtest ? <>MAPE {m.backtest.mape}%<div className="text-muted">bias {m.backtest.bias > 0 ? '+' : ''}{m.backtest.bias}% · cov {m.backtest.coverage}%</div></> : <span className="text-faint">not run at submission</span>}</td>
                  <td className="min-w-[260px] text-[11.5px]">
                    {m.notes && <div className="mb-1 text-[12px]">{m.notes}</div>}
                    {shown.map((x) => <div key={x} className="font-mono text-muted">{x}</div>)}
                    {diff.length > 3 && <button className="text-[11.5px] font-semibold text-link" onClick={() => setOpen(open === m.id ? null : m.id)}>{open === m.id ? 'Show less' : `+${diff.length - 3} more`}</button>}
                    {!parent && <span className="text-faint">First version</span>}
                  </td>
                </tr>
              )
            })}
          </tbody>
        </table></div>
      </Panel>
      <Panel pad={false} title="Audit trail" subtitle="Drafts, edits (with weight diffs), submissions, approvals and retirements">

          <thead><tr><th>When</th><th>Who</th><th>Action</th><th>Summary</th></tr></thead>

        </table></div>
      </Panel>
    </div>
  )
}

// ───────────────────────────── Override reasons and gate
function SettingsEditor({ s, canEdit }: { s: AppState; canEdit: boolean }) {
  const me = useMe()
  const wc = useDraft<FactorModelSettings>(s.settings.factorModel)
  const c = wc.draft
  const [add, setAdd] = useState('')
  const editable = canEdit && (me.roles.includes('estimating-lead') || me.roles.includes('admin') || me.roles.includes('estimator'))
  const problems = [!c.overrideReasons.length && 'Keep at least one override reason', !(c.gate.minCoverage < c.gate.maxCoverage) && 'Coverage minimum must be below the maximum', !(c.gate.maxMape > 0) && 'MAPE limit must be above 0', !(c.iterations >= 500 && c.iterations <= 50000) && 'Monte Carlo draws: 500–50,000'].filter(Boolean) as string[]
  const save = () => {
    if (problems.length) return
    const before = s.settings.factorModel
    const changes: string[] = []
    for (const r of c.overrideReasons) if (!before.overrideReasons.includes(r)) changes.push(`added reason "${r}"`)
    for (const r of before.overrideReasons) if (!c.overrideReasons.includes(r)) changes.push(`removed reason "${r}"`)
    for (const k of Object.keys(c.gate) as (keyof typeof c.gate)[]) if (c.gate[k] !== before.gate[k]) changes.push(`gate ${k} ${before.gate[k]} → ${c.gate[k]}`)
    if (c.iterations !== before.iterations) changes.push(`Monte Carlo draws ${before.iterations} → ${c.iterations}`)
    mutate((d) => { d.settings.factorModel = JSON.parse(JSON.stringify(c)) }, { entity: 'factor-model', entityId: 'factor-model-settings', action: 'settings', summary: `Factor model settings: ${changes.join('; ') || 'no changes'}` })
    wc.commit()
    toast('Saved')
  }
  const g = (k: keyof FactorModelSettings['gate'], label: string, hint?: string) => <Field label={label} hint={hint}>{editable ? <NumInput id={`gate-${k}`} value={c.gate[k]} min={0} onChange={(v) => wc.set((x) => ({ ...x, gate: { ...x.gate, [k]: Math.round(v * 10) / 10 } }))} /> : <b className="font-mono">{c.gate[k]}</b>}</Field>
  return (
    <div className="grid grid-cols-1 gap-3 xl:grid-cols-2">
      <Panel title="Override reasons" subtitle="An estimator adjusting a quick estimate must pick one of these.">
        <ul className="flex flex-col gap-1.5">
          {c.overrideReasons.map((r, k) => (
            <li key={k} className="flex items-center gap-2">
              {editable ? <Input aria-label={`Reason ${k + 1}`} className="h-8 py-0.5 text-[13px]" value={r} onChange={(e) => wc.set((x) => ({ ...x, overrideReasons: x.overrideReasons.map((y, j) => (j === k ? e.target.value : y)) }))} /> : <span className="text-[13px]">{r}</span>}
              {editable && <Button size="sm" variant="ghost" onClick={() => wc.set((x) => ({ ...x, overrideReasons: x.overrideReasons.filter((_, j) => j !== k) }))}>Remove</Button>}
            </li>
          ))}
        </ul>
        {editable && <div className="mt-2 flex gap-2"><Input aria-label="New reason" className="h-8 py-0.5 text-[13px]" placeholder="Add a reason" value={add} onChange={(e) => setAdd(e.target.value)} /><Button size="sm" disabled={!add.trim() || c.overrideReasons.includes(add.trim())} onClick={() => { wc.set((x) => ({ ...x, overrideReasons: [...x.overrideReasons, add.trim()] })); setAdd('') }}>Add</Button></div>}
      </Panel>
      <Panel title="Publish gate and simulation" subtitle="A draft can be submitted only when its backtest meets every threshold.">
        <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
          {g('maxMape', 'Max MAPE (%)')}
          {g('maxAbsBias', 'Max |bias| (%)')}
          {g('minCoverage', 'Min P10–P90 coverage (%)', 'Below this the range is too narrow')}
          {g('maxCoverage', 'Max P10–P90 coverage (%)', 'Above this the range is too wide')}
          <Field label="Monte Carlo draws">{editable ? <NumInput id="mc-iter" value={c.iterations} min={500} onChange={(v) => wc.set((x) => ({ ...x, iterations: Math.round(v) }))} /> : <b className="font-mono">{c.iterations.toLocaleString('en-US')}</b>}</Field>
        </div>
      </Panel>
      {editable && wc.dirty && (
        <div className="flex flex-wrap items-center gap-2 rounded-md border border-line bg-marker/10 px-4 py-2 xl:col-span-2">
          <Badge tone="marker">Unsaved changes</Badge>
          {problems.map((p) => <span key={p} className="text-[12px] text-bad">{p}</span>)}
          <span className="ml-auto flex gap-2"><Button size="sm" variant="ghost" onClick={() => wc.reset()}>Discard</Button><Button size="sm" variant="primary" disabled={problems.length > 0} onClick={save}>Save changes</Button></span>
        </div>
      )}
    </div>
  )
}

/** Inbox summary for a factor-model approval task: backtest at submission and what changed. */
export function FactorModelReviewSummary({ s, id }: { s: AppState; id: string }) {
  const m = s.factorModels.find((x) => x.id === id)
  if (!m) return <p className="text-[13px] text-muted">Model not found.</p>
  const parent = s.factorModels.find((x) => x.version === m.parentVersion)
  const diff = parent ? diffModels(parent, m) : []
  return (
    <div className="flex flex-col gap-2.5">
      <div className="flex flex-wrap items-center gap-2"><b className="font-display text-[16px]">Factor model v{m.version}</b><StatusBadge status={m.status} />{parent && <span className="text-[12px] text-muted">replaces v{parent.version}</span>}</div>
      {m.backtest && <div className="grid grid-cols-2 gap-2 sm:grid-cols-4"><Metric label="MAPE" value={`${m.backtest.mape}%`} /><Metric label="Bias" value={`${m.backtest.bias > 0 ? '+' : ''}${m.backtest.bias}%`} /><Metric label="Coverage" value={`${m.backtest.coverage}%`} /><Metric label="Projects" value={String(m.backtest.n)} /></div>}
      {m.notes && <p className="text-[13px]">{m.notes}</p>}
      {diff.length > 0 && <ul className="text-[11.5px]">{diff.map((x) => <li key={x} className="font-mono text-muted">· {x}</li>)}</ul>}
      <a href="#estimating" className="text-[12.5px] font-semibold text-link">Open Estimating → Administration → Factor model</a>
    </div>
  )
}
