const KINDS: DeliverableKind[] = ['document', 'estimate', 'risk-register', 'safety-plan', 'schedule', 'checklist', 'permit', 'records']

export default function Templates() {
  const templates = useStore((s) => s.templates)
  const programs = useStore((s) => s.programs)
  const [sel, setSel] = useState(templates[0]?.id)
  const tpl = templates.find((t) => t.id === sel) ?? templates[0]
  const duplicate = () => {
    if (!tpl) return
    const copy: LifecycleTemplate = clone(tpl)
    const suffix = uid('x').slice(-4)
    copy.id = uid('tpl')
    copy.name = `${tpl.name} (copy)`
    copy.stages.forEach((st) => { st.id = `${st.id}-${suffix}`; st.gate.id = `${st.gate.id}-${suffix}`; st.gate.deliverables.forEach((d) => { d.id = `${d.id}-${suffix}` }) })
    mutate((d) => { d.templates.push(copy) }, { entity: 'template', entityId: copy.id, action: 'create', summary: `Duplicated lifecycle template "${tpl.name}" as "${copy.name}"` })
    setSel(copy.id)
    toast('Template duplicated')
  }
  return (
    <div className="grid grid-cols-1 gap-4 lg:grid-cols-[260px_minmax(0,1fr)]">
      <div className="flex flex-col gap-2">
        <div className="eyebrow px-1">Templates</div>
        {templates.map((t) => {
          const using = programs.filter((p) => p.lifecycleTemplateId === t.id)
          return (
            <button key={t.id} onClick={() => setSel(t.id)} className={cx('rounded-md border px-3 py-2.5 text-left', t.id === tpl?.id ? 'border-ink bg-surface shadow-sm' : 'border-line bg-surface hover:border-ink/40')}>
              <div className="text-[13px] font-semibold leading-snug">{t.name}</div>
              <div className="mt-1 flex flex-wrap items-center gap-1 text-[11px] text-muted">
                <span>{t.stages.length} stages · {t.stages.reduce((a, st) => a + st.gate.deliverables.length, 0)} deliverables</span>
              </div>
              <div className="mt-1.5 flex flex-wrap gap-1">{using.length ? using.map((p) => <ProgramTag key={p.id} programId={p.id} />) : <span className="text-[11px] text-faint">Not assigned to a program</span>}</div>
            </button>
          )
        })}
        <Button size="sm" onClick={duplicate}>Duplicate selected template</Button>
        <p className="px-1 text-[11px] text-faint">Duplicate a template to tailor it for a new program, then assign it on the Programs tab.</p>
      </div>
      {tpl ? <TemplateEditor key={tpl.id} tpl={tpl} onDeleted={() => setSel(templates[0]?.id)} /> : <Empty title="No templates" />}

function TemplateEditor({ tpl, onDeleted }: { tpl: LifecycleTemplate; onDeleted: () => void }) {
  const programs = useStore((s) => s.programs)
  const allProjects = useStore((s) => s.projects)
  const roles = useStore((s) => s.roles)
  const [draft, setDraft, dirty, discard] = useDraft(tpl)
  const [stageIdx, setStageIdx] = useState(0)
  const [editing, setEditing] = useState<{ def: DeliverableDef; isNew: boolean } | null>(null)
  const using = programs.filter((p) => p.lifecycleTemplateId === tpl.id)
  const projects = useMemo(() => allProjects.filter((p) => using.some((u) => u.id === p.programId)), [allProjects, using])
  const st = draft.stages[stageIdx] ?? draft.stages[0]
  const atStage = projects.filter((p) => p.stageIndex === stageIdx && p.status === 'active')

  const setStage = (fn: (s: typeof st) => void) => setDraft((d) => { const n = clone(d); fn(n.stages[stageIdx]); return n })
  const moveDel = (i: number, dir: -1 | 1) => setStage((s) => { const a = s.gate.deliverables; const j = i + dir; if (j < 0 || j >= a.length) return; [a[i], a[j]] = [a[j], a[i]] })
  const removeDel = (i: number) => setStage((s) => { s.gate.deliverables.splice(i, 1) })

  const save = () => {
    const changes = diffTemplate(tpl, draft)
    let created = 0
    mutate((d) => {
      const i = d.templates.findIndex((t) => t.id === tpl.id)
      d.templates[i] = clone(draft)
      const before = d.deliverables.length
      for (const p of d.projects) if (p.status === 'active' && using.some((u) => u.id === p.programId)) ensureDeliverables(d, p)
      created = d.deliverables.length - before
    }, { entity: 'template', entityId: tpl.id, action: 'update', summary: `Updated lifecycle template "${draft.name}": ${changes.join('; ') || 'no field changes'}` })
    toast(created ? `Template saved · ${created} new deliverable${created > 1 ? 's' : ''} assigned to active projects` : 'Template saved')
  }
  const remove = () => {
    mutate((d) => { d.templates = d.templates.filter((t) => t.id !== tpl.id) }, { entity: 'template', entityId: tpl.id, action: 'delete', summary: `Deleted unused lifecycle template "${tpl.name}"` })
    toast('Template deleted')
    onDeleted()
  }

  return (
    <div className="flex min-w-0 flex-col gap-4">
      {dirty && (
        <div className="sticky top-0 z-10 flex flex-wrap items-center justify-between gap-2 rounded-md border border-marker bg-marker/10 px-3 py-2">
          <span className="text-[13px] font-semibold">Unsaved changes to this template</span>
          <span className="flex gap-2"><Button size="sm" variant="ghost" onClick={discard}>Discard</Button><Button size="sm" variant="primary" onClick={save}>Save changes</Button></span>
        </div>
      )}
      <Panel title="Template" subtitle={`${draft.stages.length} stages · used by ${using.length ? using.map((u) => u.code).join(', ') : 'no programs'} · ${projects.length} projects`}
        actions={using.length === 0 ? <ConfirmButton size="sm" confirmLabel="Delete template" onConfirm={remove}>Delete</ConfirmButton> : undefined}>
        <div className="grid grid-cols-1 gap-3 md:grid-cols-2">
          <Field label="Template name"><Input value={draft.name} onChange={(e) => setDraft({ ...draft, name: e.target.value })} /></Field>
          <Field label="Description"><Input value={draft.description} onChange={(e) => setDraft({ ...draft, description: e.target.value })} /></Field>
        </div>
      </Panel>

      <div className="scroll-x">
        <ol className="flex min-w-max gap-1">
          {draft.stages.map((x, i) => (
            <li key={x.id}>
              <button onClick={() => setStageIdx(i)} className={cx('flex w-[190px] flex-col rounded-md border px-3 py-2 text-left', i === stageIdx ? 'border-marker bg-marker/10' : 'border-line bg-surface hover:border-ink/40')}>
                <span className="eyebrow">Stage {i + 1}</span>
                <span className="truncate text-[13px] font-semibold">{x.name}</span>
                <span className="truncate text-[11px] text-muted">{x.gate.name} · {x.gate.deliverables.length} deliverables</span>
              </button>
            </li>
          ))}
        </ol>
      </div>

      <div className="grid grid-cols-1 gap-4 xl:grid-cols-2">
        <Panel title={`Stage ${stageIdx + 1}`} subtitle={`${atStage.length} active project${atStage.length === 1 ? '' : 's'} in this stage`}>
          <div className="flex flex-col gap-3">
            <Field label="Stage name"><Input value={st.name} onChange={(e) => setStage((s) => { s.name = e.target.value })} /></Field>
            <Field label="Purpose"><TextArea rows={2} value={st.purpose} onChange={(e) => setStage((s) => { s.purpose = e.target.value })} /></Field>

            <Field label="Decision roles"><MultiCheck options={roles.filter((r) => ['sponsor', 'exec', 'gate-board', 'program-manager', 'pmo', 'finance'].includes(r.id) || st.gate.decisionRoles.includes(r.id)).map((r) => ({ id: r.id, label: r.name }))} value={st.gate.decisionRoles} onChange={(v) => setStage((s) => { s.gate.decisionRoles = v as RoleId[] })} /></Field>

function DeliverableDrawer({ editing, projects, onClose, onSave }: { editing: { def: DeliverableDef; isNew: boolean } | null; projects: ReturnType<typeof useStore.getState>['projects']; onClose: () => void; onSave: (d: DeliverableDef) => void }) {
  const [d, setD] = useDraft<DeliverableDef | null>(editing?.def ?? null)
  if (!editing || !d) return null
  return (
    <Drawer open onClose={onClose} title={editing.isNew ? 'Add deliverable' : 'Edit deliverable'} width={600}
      footer={<><Button variant="ghost" onClick={onClose}>Cancel</Button><Button variant="primary" disabled={!d.name.trim()} onClick={() => onSave(d)}>{editing.isNew ? 'Add to gate' : 'Apply'}</Button></>}>
      <div className="flex flex-col gap-3">
        <Field label="Name"><Input value={d.name} onChange={(e) => setD({ ...d, name: e.target.value })} placeholder="e.g. Traffic management plan" /></Field>
        <Field label="Description"><TextArea rows={2} value={d.description} onChange={(e) => setD({ ...d, description: e.target.value })} /></Field>
        <div className="grid grid-cols-1 gap-3 sm:grid-cols-3">
          <Field label="Kind"><Select value={d.kind} onChange={(e) => setD({ ...d, kind: e.target.value as DeliverableKind })} options={KINDS} /></Field>
          <Field label="Owner role"><RoleSelect value={d.ownerRole} onChange={(r) => setD({ ...d, ownerRole: r })} /></Field>
          <Field label="Reviewer role"><RoleSelect value={d.reviewerRole} onChange={(r) => setD({ ...d, reviewerRole: r })} /></Field>
        </div>
        <Field label="Acceptance criteria" hint="Shown to the reviewer when the deliverable is submitted"><TextArea rows={2} value={d.acceptance} onChange={(e) => setD({ ...d, acceptance: e.target.value })} /></Field>
        <ConditionEditor value={d.condition} onChange={(c) => setD({ ...d, condition: c })} projects={projects} />
        {d.ownerRole === d.reviewerRole && <Callout tone="warn">Owner and reviewer are the same role. Separation of duties is stronger when a different role reviews.</Callout>}
      </div>
    </Drawer>
  )
}

function diffTemplate(a: LifecycleTemplate, b: LifecycleTemplate): string[] {
  const out: string[] = []
  if (a.name !== b.name) out.push(`renamed to "${b.name}"`)
  if (a.description !== b.description) out.push('description edited')
  b.stages.forEach((sb, i) => {
    const sa = a.stages[i]
    if (!sa) return
    const parts: string[] = []
    if (sa.name !== sb.name) parts.push('name')
    if (sa.purpose !== sb.purpose) parts.push('purpose')
    if (sa.gate.name !== sb.gate.name) parts.push('gate name')
    if (JSON.stringify(sa.gate.decisionRoles) !== JSON.stringify(sb.gate.decisionRoles)) parts.push(`decision roles → ${sb.gate.decisionRoles.join(', ')}`)
    if (sa.gate.requiredEstimateClass !== sb.gate.requiredEstimateClass) parts.push(`estimate class → ${sb.gate.requiredEstimateClass ? 'Class ' + sb.gate.requiredEstimateClass : 'none'}`)
    if (sa.gate.budgetEvent !== sb.gate.budgetEvent) parts.push(`budget event → ${sb.gate.budgetEvent ?? 'none'}`)
    if (JSON.stringify(sa.gate.unlocks) !== JSON.stringify(sb.gate.unlocks)) parts.push(`unlocks → ${sb.gate.unlocks.join(', ') || 'none'}`)
    const ida = sa.gate.deliverables.map((x) => x.id), idb = sb.gate.deliverables.map((x) => x.id)
    const added = sb.gate.deliverables.filter((x) => !ida.includes(x.id)).map((x) => x.name)
    const removed = sa.gate.deliverables.filter((x) => !idb.includes(x.id)).map((x) => x.name)
    const edited = sb.gate.deliverables.filter((x) => { const o = sa.gate.deliverables.find((y) => y.id === x.id); return o && JSON.stringify(o) !== JSON.stringify(x) }).map((x) => x.name)
    if (added.length) parts.push(`added ${added.join(', ')}`)
    if (removed.length) parts.push(`removed ${removed.join(', ')}`)
    if (edited.length) parts.push(`edited ${edited.join(', ')}`)
    if (!added.length && !removed.length && ida.filter((x) => idb.includes(x)).join() !== idb.filter((x) => ida.includes(x)).join()) parts.push('deliverables reordered')
    if (parts.length) out.push(`Stage ${i + 1} ${sb.name}: ${parts.join(', ')}`)
  })
  return out
}

// Configuration: the WBS functions the forecast is spread against, and which cost categories roll up to each.
// Loaders, escalation and contingency are system categories allocated pro-rata (contingency held separately).

const SYSTEM = [ESC_CAT, CONT_CAT] // overhead (cat-loaders) is assignable, e.g. to 900 Project Overhead
const TIMING: { value: WbsFunction['timing']; label: string }[] = [
  { value: 'early', label: 'Early (front-loaded)' },
  { value: 'mid', label: 'Mid (S-curve)' },
  { value: 'late', label: 'Late (back-loaded)' },
  { value: 'flat', label: 'Flat (even)' },
]

/** Validation of a WBS configuration against the cost categories. */
export function wbsIssues(fns: WbsFunction[], cats: { id: string; code: string; name: string }[]) {
  const direct = cats.filter((c) => !SYSTEM.includes(c.id))
  const owners = (id: string) => fns.filter((f) => f.categoryIds.includes(id))
  return {
    unmapped: direct.filter((c) => owners(c.id).length === 0),
    multi: direct.filter((c) => owners(c.id).length > 1),
    dupCodes: fns.filter((f, i) => fns.findIndex((x) => x.code.trim() === f.code.trim()) !== i).map((f) => f.code),
    blank: fns.filter((f) => !f.code.trim() || !f.name.trim()),
  }
}

  const save = () => {

    toast(issues.unmapped.length ? `WBS saved — ${issues.unmapped.length} categories still unmapped` : 'WBS functions saved', issues.unmapped.length ? 'bad' : 'good')

/** Predicted vs actual on log axes, with the 1:1 line and ±25% guides; points outside P10–P90 in red. */
function Scatter({ points }: { points: BacktestResult['points'] }) {
  const W = 720, H = 300, L = 64, R = 16, T = 12, B = 34
  const vals = points.flatMap((p) => [p.predicted, p.actual]).filter((v) => v > 0)
  const lo = Math.log10(Math.min(...vals) * 0.8), hi = Math.log10(Math.max(...vals) * 1.25)
  const X = (v: number) => L + ((Math.log10(v) - lo) / (hi - lo)) * (W - L - R)
  const Y = (v: number) => T + (1 - (Math.log10(v) - lo) / (hi - lo)) * (H - T - B)
  const ticks = Array.from({ length: Math.floor(hi) - Math.ceil(lo) + 1 }, (_, k) => Math.pow(10, Math.ceil(lo) + k))
  const a = Math.pow(10, lo), b = Math.pow(10, hi)
  return (
    <div>
      <svg viewBox={`0 0 ${W} ${H}`} className="w-full" role="img" aria-label="Predicted versus actual cost">
        {ticks.map((t) => <g key={t}><line x1={L} x2={W - R} y1={Y(t)} y2={Y(t)} stroke="rgb(var(--line))" /><text x={L - 6} y={Y(t) + 4} textAnchor="end" fontSize={11} fill="rgb(var(--muted))">{money(t)}</text><text x={X(t)} y={H - 12} textAnchor="middle" fontSize={11} fill="rgb(var(--muted))">{money(t)}</text></g>)}
        <line x1={X(a)} y1={Y(a)} x2={X(b)} y2={Y(b)} stroke="rgb(var(--ink))" strokeWidth={1.2} />
        <line x1={X(a)} y1={Y(a * 1.25)} x2={X(b / 1.25)} y2={Y(b)} stroke="rgb(var(--muted))" strokeDasharray="4 4" />
        <line x1={X(a * 1.25)} y1={Y(a)} x2={X(b)} y2={Y(b / 1.25)} stroke="rgb(var(--muted))" strokeDasharray="4 4" />
        {points.map((p) => <circle key={p.id} cx={X(p.actual)} cy={Y(p.predicted)} r={4} fill={p.inside ? 'rgb(var(--link))' : 'rgb(var(--bad))'} opacity={0.75}><title>{`${p.code} ${p.name}: predicted ${money(p.predicted)}, actual ${money(p.actual)} (${p.err > 0 ? '+' : ''}${(p.err * 100).toFixed(0)}%)`}</title></circle>)}
      </svg>
      <div className="mt-1 flex flex-wrap gap-x-4 gap-y-1 text-[11px] text-muted">
        <span>x: actual · y: predicted</span>
        <span className="inline-flex items-center gap-1"><span className="inline-block h-2.5 w-2.5 rounded-full bg-link" />Actual inside P10–P90</span>
        <span className="inline-flex items-center gap-1"><span className="inline-block h-2.5 w-2.5 rounded-full bg-bad" />Outside</span>
        <span>Solid line 1:1 · dashed ±25%</span>
      </div>
    </div>
  )
}

// ───────────────────────────── Compare
function Compare({ s, m, pub }: { s: AppState; m: FactorModel; pub?: FactorModel }) {
  const data = useMemo(() => {
    if (!pub || pub.id === m.id) return
    const a = backtest(s, pub), b = backtest(s, m)
    const projects = s.projects.filter((p) => p.status === 'active' && p.stageIndex <= 1).slice(0, 10).map((p) => {
      const i = factorInputsFromProject(s, p)
      const x = priceFactor(s, pub, i), y = priceFactor(s, m, i)
      return { p, a: x.total, b: y.total }
    }).filter((r) => r.a > 0)
    return { a, b, diff: diffModels(pub, m), projects }
  }, [s, m, pub])
  if (!data || !pub) return <Panel title="Draft vs published"><p className="text-[13px] text-muted">{pub && pub.id === m.id ? 'You are viewing the published model. Open the draft (or start one) to compare it with the published version.' : 'No published model to compare with.'}</p></Panel>
  const row = (label: string, x: number, y: number, unit = '%', better: 'lower' | 'abs' | 'none' = 'lower') => {
    const d = y - x
    const good = better === 'lower' ? d < 0 : better === 'abs' ? Math.abs(y) < Math.abs(x) : false
    return <tr key={label}><td className="text-[12.5px]">{label}</td><td className="num">{x.toFixed(1)}{unit}</td><td className="num font-semibold">{y.toFixed(1)}{unit}</td><td className={cx('num', d && better !== 'none' ? (good ? 'text-good' : 'text-bad') : 'text-muted')}>{d > 0 ? '+' : ''}{d.toFixed(1)}</td></tr>
  }
  return (
    <div className="grid grid-cols-1 gap-3 xl:grid-cols-2">
      <Panel pad={false} title="Backtest: published vs this version">
        <div className="scroll-x"><table className="tbl">
          <thead><tr><th>Metric</th><th className="num">v{pub.version}</th><th className="num">v{m.version}</th><th className="num">Δ</th></tr></thead>
          <tbody>
            {row('MAPE', data.a.mape, data.b.mape)}
            {row('Median abs % error', data.a.medianApe, data.b.medianApe)}
            {row('Bias', data.a.bias, data.b.bias, '%', 'abs')}
            {row('Inside P10–P90', data.a.coverage, data.b.coverage, '%', 'none')}
          </tbody>
        </table></div>
      </Panel>
      <Panel title={`What changed (v${pub.version} → v${m.version})`} subtitle="Weights, multipliers, uncertainty and base rates, including unsaved edits">
        {data.diff.length ? <ul className="flex flex-col gap-1 text-[12.5px]">{data.diff.map((x) => <li key={x} className="rounded border border-line px-2 py-1 font-mono text-[11.5px]">{x}</li>)}</ul> : <p className="text-[13px] text-muted">No differences.</p>}
      </Panel>
      <Panel pad={false} title="Effect on early-stage projects" subtitle="Class 5 total under each version, same inputs" className="xl:col-span-2">
        <div className="scroll-x"><table className="tbl">
          <thead><tr><th>Project</th><th className="num">v{pub.version}</th><th className="num">v{m.version}</th><th className="num">Δ</th></tr></thead>
          <tbody>{data.projects.map((r) => <tr key={r.p.id}><td className="min-w-[200px] text-[12.5px]"><span className="font-mono text-[11.5px] text-muted">{r.p.code}</span> {r.p.name}</td><td className="num">{money(r.a)}</td><td className="num font-semibold">{money(r.b)}</td><td className={cx('num', r.b > r.a ? 'text-bad' : r.b < r.a ? 'text-good' : 'text-muted')}>{r.b === r.a ? '—' : `${r.b > r.a ? '+' : ''}${pct(r.b / r.a - 1, 1)}`}</td></tr>)}</tbody>
        </table></div>
      </Panel>
    </div>
  )
}

// ───────────────────────────── Override reasons and gate
function SettingsEditor({ s, canEdit }: { s: AppState; canEdit: boolean }) {
  const me = useMe()
  const wc = useDraft<FactorModelSettings>(s.settings.factorModel)
  const c = wc.draft
  const [add, setAdd] = useState('')
  const editable = canEdit && (me.roles.includes('estimating-lead') || me.roles.includes('admin') || me.roles.includes('estimator'))
  const problems = [!c.overrideReasons.length && 'Keep at least one override reason', !(c.gate.minCoverage < c.gate.maxCoverage) && 'Coverage minimum must be below the maximum', !(c.gate.maxMape > 0) && 'MAPE limit must be above 0', !(c.iterations >= 500 && c.iterations <= 50000) && 'Monte Carlo draws: 500–50,000'].filter(Boolean) as string[]
  const save = () => {
    if (problems.length) return
    const before = s.settings.factorModel
    const changes: string[] = []
    for (const r of c.overrideReasons) if (!before.overrideReasons.includes(r)) changes.push(`added reason "${r}"`)
    for (const r of before.overrideReasons) if (!c.overrideReasons.includes(r)) changes.push(`removed reason "${r}"`)
    for (const k of Object.keys(c.gate) as (keyof typeof c.gate)[]) if (c.gate[k] !== before.gate[k]) changes.push(`gate ${k} ${before.gate[k]} → ${c.gate[k]}`)
    if (c.iterations !== before.iterations) changes.push(`Monte Carlo draws ${before.iterations} → ${c.iterations}`)
    mutate((d) => { d.settings.factorModel = JSON.parse(JSON.stringify(c)) }, { entity: 'factor-model', entityId: 'factor-model-settings', action: 'settings', summary: `Factor model settings: ${changes.join('; ') || 'no changes'}` })
    wc.commit()
    toast('Saved')
  }
  const g = (k: keyof FactorModelSettings['gate'], label: string, hint?: string) => <Field label={label} hint={hint}>{editable ? <NumInput id={`gate-${k}`} value={c.gate[k]} min={0} onChange={(v) => wc.set((x) => ({ ...x, gate: { ...x.gate, [k]: Math.round(v * 10) / 10 } }))} /> : <b className="font-mono">{c.gate[k]}</b>}</Field>
  return (
    <div className="grid grid-cols-1 gap-3 xl:grid-cols-2">
      <Panel title="Override reasons" subtitle="An estimator adjusting a quick estimate must pick one of these.">
        <ul className="flex flex-col gap-1.5">
          {c.overrideReasons.map((r, k) => (
            <li key={k} className="flex items-center gap-2">
              {editable ? <Input aria-label={`Reason ${k + 1}`} className="h-8 py-0.5 text-[13px]" value={r} onChange={(e) => wc.set((x) => ({ ...x, overrideReasons: x.overrideReasons.map((y, j) => (j === k ? e.target.value : y)) }))} /> : <span className="text-[13px]">{r}</span>}
              {editable && <Button size="sm" variant="ghost" onClick={() => wc.set((x) => ({ ...x, overrideReasons: x.overrideReasons.filter((_, j) => j !== k) }))}>Remove</Button>}
            </li>
          ))}
        </ul>
        {editable && <div className="mt-2 flex gap-2"><Input aria-label="New reason" className="h-8 py-0.5 text-[13px]" placeholder="Add a reason" value={add} onChange={(e) => setAdd(e.target.value)} /><Button size="sm" disabled={!add.trim() || c.overrideReasons.includes(add.trim())} onClick={() => { wc.set((x) => ({ ...x, overrideReasons: [...x.overrideReasons, add.trim()] })); setAdd('') }}>Add</Button></div>}
      </Panel>
      <Panel title="Publish gate and simulation" subtitle="A draft can be submitted only when its backtest meets every threshold.">
        <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
          {g('maxMape', 'Max MAPE (%)')}
          {g('maxAbsBias', 'Max |bias| (%)')}
          {g('minCoverage', 'Min P10–P90 coverage (%)', 'Below this the range is too narrow')}
          {g('maxCoverage', 'Max P10–P90 coverage (%)', 'Above this the range is too wide')}
          <Field label="Monte Carlo draws">{editable ? <NumInput id="mc-iter" value={c.iterations} min={500} onChange={(v) => wc.set((x) => ({ ...x, iterations: Math.round(v) }))} /> : <b className="font-mono">{c.iterations.toLocaleString('en-US')}</b>}</Field>
        </div>
      </Panel>
      {editable && wc.dirty && (
        <div className="flex flex-wrap items-center gap-2 rounded-md border border-line bg-marker/10 px-4 py-2 xl:col-span-2">
          <Badge tone="marker">Unsaved changes</Badge>
          {problems.map((p) => <span key={p} className="text-[12px] text-bad">{p}</span>)}
          <span className="ml-auto flex gap-2"><Button size="sm" variant="ghost" onClick={() => wc.reset()}>Discard</Button><Button size="sm" variant="primary" disabled={problems.length > 0} onClick={save}>Save changes</Button></span>
        </div>
      )}
    </div>
  )
}

/** Inbox summary for a factor-model approval task: backtest at submission and what changed. */
export function FactorModelReviewSummary({ s, id }: { s: AppState; id: string }) {
  const m = s.factorModels.find((x) => x.id === id)
  if (!m) return <p className="text-[13px] text-muted">Model not found.</p>
  const parent = s.factorModels.find((x) => x.version === m.parentVersion)

function Lines({ id, value, onChange, disabled }: { id: string; value: string[]; onChange: (v: string[]) => void; disabled: boolean }) {
  const [txt, setTxt] = useState(value.join('\n'))
  const joined = value.join('\n')

  useEffect(() => { if (joined !== last.current) { setTxt(joined); last.current = joined } }, [joined])
  return <TextArea id={id} disabled={disabled} className="min-h-[260px] text-[12.5px] leading-relaxed" value={txt} onChange={(e) => { setTxt(e.target.value); const v = e.target.value.split('\n').map((x) => x.trim()).filter(Boolean); last.current = v.join('\n'); onChange(v) }} />

function RateDrawer({ s, rates, rate, canEdit, usage, onClose, onApply }: { s: AppState; rates: UnitCostRate[]; rate?: UnitCostRate; canEdit: boolean; usage: Record<string, number>; onClose: () => void; onApply: (rows: UnitCostRate[], removeIds: string[]) => void }) {
  const cfg = s.settings.estimating
  const year = Number(s.settings.asOf.slice(0, 4))
  const [r, setR] = useState<UnitCostRate>(() => rate ? { ...rate } : { id: `rate-${Date.now().toString(36)}`, assetType: 'transmission-pipeline', workMethod: 'replace-open-trench', categoryId: 'cat-contract', description: '', unit: 'ft', p10: 0, p50: 0, p90: 0, sampleSize: 1, basisYear: year })

  const unitOk = cfg.units.some((u) => u.id === r.unit)
  const order = (x: UnitCostRate) => x.p10 <= x.p50 && x.p50 <= x.p90 && x.p50 > 0

  const used = rate ? (usage[rate.id] ?? 0) : 0

// Bulk estimate update: after the rate table changes, list projects whose current estimate was priced on an
// older version, let the estimating team pick which to reprice (or keep on historical rates), and create the

export function BulkReprice({ s, canEdit }: { s: AppState; canEdit: boolean }) {

  const v = s.settings.estimating.rateTableVersion
  const [sel, setSel] = useState<Record<string, boolean>>({})
  const key = plans.map((p) => p.estimate.id + (p.kept ? 'k' : '')).join()
  useEffect(() => { setSel(Object.fromEntries(plans.map((p) => [p.estimate.id, p.eligible && !p.kept]))) }, [key]) // eslint-disable-line react-hooks/exhaustive-deps
  const [detail, setDetail] = useState<RepricePlan | null>(null)
  const [confirm, setConfirm] = useState<'reprice' | 'keep' | null>(null)
  const chosen = plans.filter((p) => p.eligible && sel[p.estimate.id])
  const eligible = plans.filter((p) => p.eligible)
  const before = sum(chosen.map((p) => p.before.total)), after = sum(chosen.map((p) => p.after.total))
  const allOn = eligible.length > 0 && eligible.every((p) => sel[p.estimate.id])
  const toggleAll = () => setSel(Object.fromEntries(plans.map((p) => [p.estimate.id, p.eligible && !allOn])))

  const doReprice = (note: string) => {
    let r = { created: 0, updated: 0 }
    const ids = chosen.map((p) => p.estimate.id)
    mutate((d) => { const x = applyReprice(d, ids, note); r = { created: x.created.length, updated: x.updated.length } }, { entity: 'estimating-config', entityId: 'bulk-reprice', action: 'reprice', summary: `Bulk update to rate table v${v}: ${ids.length} estimate${ids.length === 1 ? '' : 's'} repriced (${money(before)} → ${money(after)})${note ? ` — ${note}` : ''}` })
    setConfirm(null)
    toast(`${r.created} new draft version${r.created === 1 ? '' : 's'} created${r.updated ? `, ${r.updated} draft${r.updated === 1 ? '' : 's'} updated` : ''} on rate table v${v}`)
  }
  const doKeep = (note: string) => {
    const ids = chosen.map((p) => p.estimate.id)
    let n = 0
    mutate((d) => { n = keepHistorical(d, ids, note) }, { entity: 'estimating-config', entityId: 'bulk-reprice', action: 'keep-rates', summary: `Kept ${ids.length} estimate${ids.length === 1 ? '' : 's'} on historical rates for rate table v${v} — ${note}` })
    setConfirm(null)
    toast(`${n} estimate${n === 1 ? '' : 's'} kept on historical rates`)
  }

  if (!plans.length) return (
    <Panel title="Bulk estimate update"><Empty title={`Every current estimate is on rate table v${v}`}>When the estimating team publishes a new rate table, projects whose current estimate was priced on an older version appear here to reprice or keep on historical rates. Historical quick estimates price from the historical cost table and are not listed.</Empty></Panel>
  )
  return (
    <div className="flex flex-col gap-3">
      <div className="grid grid-cols-2 gap-3 md:grid-cols-4">
        <Stat label="Rate table in force" value={`v${v}`} sub={s.settings.estimating.rateTableHistory.slice(-1)[0]?.note} />

        <Stat label="Selected" value={chosen.length} sub={chosen.length ? `${money(before)} → ${money(after)}` : 'Pick projects below'} />
        <Stat label="Net change" value={chosen.length ? moneyDelta(after - before) : '—'} sub={chosen.length && before ? `${after >= before ? '+' : ''}${pct(after / before - 1, 1)} on the selected estimates` : undefined} tone={chosen.length ? (after > before ? 'bad' : 'good') : undefined} />
      </div>

Vendor quotes, analogs, estimator judgment and overrides keep their prices. Approved estimates stay as signed: a new draft version is created."
        actions={canEdit && <>
          <Button size="sm" disabled={!chosen.length} onClick={() => setConfirm('keep')}>Keep historical rates ({chosen.length})</Button>

                const n = p.lines.filter((x) => x.status === 'repriced').length
                const kept = p.lines.filter((x) => x.status === 'quote' || x.status === 'override' || x.status === 'manual').length
                return (

const describe = (a: RiskItemDef[], b: RiskItemDef[]) => {
  const out: string[] = []
  for (const x of b) {
    const y = a.find((i) => i.id === x.id)
    if (!y) { out.push(`added ${x.name}`); continue }
    if (y.name !== x.name) out.push(`${y.name} renamed ${x.name}`)
    for (const k of RISK_LEVELS) {
      const p = y.levels[k], q = x.levels[k]
      if (p.min !== q.min || p.mode !== q.mode || p.max !== q.max) out.push(`${x.name} ${k} ${p.min}/${p.mode}/${p.max}% → ${q.min}/${q.mode}/${q.max}%`)
      if (p.label !== q.label) out.push(`${x.name} ${k} description changed`)
    }
  }
  for (const y of a) if (!b.some((x) => x.id === y.id)) out.push(`removed ${y.name}`)
  return out
}

// Shared pieces for Estimating administration: a working copy with save/discard, the read-only notice,
// and the audited save into settings.estimating.
import { useEffect, useRef, useState } from 'react'
import type { AppState } from '@/core/types'
import { mutate } from '@/core/store'
import { Badge, Button } from '@/ui/kit'
import { toast } from '@/ui/toast'

const clone = <T,>(x: T): T => JSON.parse(JSON.stringify(x))

/** Working copy of a slice of the store. Follows the store while nothing is pending. */
export function useDraft<T>(source: T): { draft: T; set: (fn: (d: T) => T) => void; dirty: boolean; reset: () => void; commit: () => void } {
  const [draft, setDraft] = useState<T>(() => clone(source))
  const [dirty, setDirty] = useState(false)
  const key = JSON.stringify(source)
  const last = useRef(key)
  useEffect(() => { if (key !== last.current) { last.current = key; if (!dirty) setDraft(clone(source)) } }, [key]) // eslint-disable-line react-hooks/exhaustive-deps
  return {
    draft,
    set: (fn) => { setDraft((d) => fn(clone(d))); setDirty(true) },
    dirty,
    reset: () => { setDraft(clone(source)); setDirty(false) },
    commit: () => setDirty(false),
  }
}

/** Save a section of the estimating tables with one audit entry. */
export function saveSection(section: string, summary: string, fn: (d: AppState) => void, msg = 'Saved') {
  mutate(fn, { entity: 'estimating-config', entityId: section, action: 'update', summary })
  toast(msg)
}

/** Footer bar for a section with pending changes. */
export function SaveBar({ dirty, onSave, onDiscard, disabled, note }: { dirty: boolean; onSave: () => void; onDiscard: () => void; disabled?: boolean; note?: React.ReactNode }) {
  if (!dirty) return null
  return (
    <div className="flex flex-wrap items-center gap-2 border-t border-line bg-marker/10 px-4 py-2">
      <Badge tone="marker">Unsaved changes</Badge>
      {note && <span className="text-xs text-muted">{note}</span>}
      <span className="ml-auto flex gap-2"><Button size="sm" variant="ghost" onClick={onDiscard}>Discard</Button><Button size="sm" variant="primary" disabled={disabled} onClick={onSave}>Save changes</Button></span>
    </div>
  )
}

/** Compact list of what changed between two flat records, for audit summaries. */
export function diffRecord(a: Record<string, unknown>, b: Record<string, unknown>, fmt: (v: unknown) => string = String): string[] {
  const keys = Array.from(new Set([...Object.keys(a), ...Object.keys(b)]))
  return keys.filter((k) => a[k] !== b[k]).map((k) => a[k] == null ? `added ${k} ${fmt(b[k])}` : b[k] == null ? `removed ${k}` : `${k} ${fmt(a[k])} → ${fmt(b[k])}`)
}

/** Everything a BOE shows, as plain values. */
export function boeFacts(s: AppState, e: Estimate, opts: { mc?: MonteCarloResult } = {}) {
  const p = s.projects.find((x) => x.id === e.projectId)!
  const t = estimateTotals(e)
  const person = (id?: string) => s.people.find((x) => x.id === id)?.name ?? '-'
  const prog = s.programs.find((x) => x.id === p.programId)
  const r = CLASS_RANGES[e.cls]
  const wf = s.workflows.find((w) => w.id === e.workflowInstanceId)
  const def = wf ? s.workflowDefs.find((d) => d.id === wf.defId) : undefined
  const cats = Object.entries(t.direct).sort((a, b) => catCode(s, a[0]).localeCompare(catCode(s, b[0])))
  return {
    p, t, r, prog, wf, def, cats, person,
    title: `${p.code} Basis of Estimate - v${e.version} Class ${e.cls}`,
    rateVersion: e.method === 'quick' ? undefined : pricedOnVersion(s, e),
    srcs: sourceSummary(e),
    ex: benchmarkExceptions(e, s.rates, s.settings.estimating.regionFactors),
    overrides: e.lines.filter((l) => l.override),
    wps: workPackages(e),
    maturity: maturityItems(s),
    mc: opts.mc && e.contingency.method === 'risk-based' ? opts.mc : undefined,
    history: wf ? wf.history.filter((h) => h.action !== 'system').map((h) => ({ at: h.at.slice(0, 16).replace('T', ' '), who: h.actorId === 'system' ? 'System' : person(h.actorId), action: h.action + (h.signature ? ' (e-signed)' : ''), step: def?.nodes.find((n) => n.id === h.nodeId)?.label ?? '', comment: h.comment ?? '' })) : [],
  }
}

/** Characters the standard PDF fonts cannot draw are replaced with plain equivalents. */
const clean = (x: unknown) => String(x ?? '')
  .replace(/[→⇒]/g, '->').replace(/[←]/g, '<-').replace(/≥/g, '>=').replace(/≤/g, '<=').replace(/×/g, 'x')
  .replace(/[–—]/g, '-').replace(/[‘’]/g, "'").replace(/[“”]/g, '"').replace(/…/g, '...').replace(/[•·]/g, '-')
  .replace(/[^\x09\x0A\x0D\x20-\x7E\xA0-\xFF]/g, '')

type Align = 'left' | 'right'
interface Col { label: string; w: number; align?: Align }

/** A small page-flow writer over jsPDF: headings, wrapped paragraphs and tables with repeating headers. */
class Writer {
  doc = new jsPDF({ unit: 'pt', format: 'letter' })
  x = 48
  y = 56
  W = 612 - 96
  bottom = 792 - 54
  constructor(private footer: string) {}
  ensure(h: number) { if (this.y + h > this.bottom) { this.doc.addPage(); this.y = 56 } }
  h1(t: string) { this.doc.setFont('helvetica', 'bold'); this.doc.setFontSize(20); this.doc.setTextColor(20, 27, 36); this.doc.text(clean(t), this.x, this.y); this.y += 18 }
  h2(t: string) {
    this.ensure(40); this.y += 10
    this.doc.setFont('helvetica', 'bold'); this.doc.setFontSize(12.5); this.doc.setTextColor(20, 27, 36); this.doc.text(clean(t), this.x, this.y)
    this.y += 5; this.doc.setDrawColor(20, 27, 36); this.doc.setLineWidth(1.2); this.doc.line(this.x, this.y, this.x + this.W, this.y); this.y += 13
  }
  h3(t: string) { this.ensure(26); this.doc.setFont('helvetica', 'bold'); this.doc.setFontSize(10.5); this.doc.setTextColor(20, 27, 36); this.doc.text(clean(t), this.x, this.y); this.y += 13 }
  p(t: string, o: { size?: number; muted?: boolean; bold?: boolean } = {}) {
    const size = o.size ?? 9.5
    this.doc.setFont('helvetica', o.bold ? 'bold' : 'normal'); this.doc.setFontSize(size)
    if (o.muted) this.doc.setTextColor(79, 90, 105); else this.doc.setTextColor(20, 27, 36)
    const lines: string[] = this.doc.splitTextToSize(clean(t), this.W)
    for (const l of lines) { this.ensure(size * 1.35); this.doc.text(l, this.x, this.y); this.y += size * 1.35 }
    this.y += 3
  }
  bullets(items: string[]) {
    if (!items.length) { this.p('None recorded', { muted: true }); return }
    this.doc.setFont('helvetica', 'normal'); this.doc.setFontSize(9.5); this.doc.setTextColor(20, 27, 36)
    for (const it of items) {
      const lines: string[] = this.doc.splitTextToSize(clean(it), this.W - 14)
      lines.forEach((l, i) => { this.ensure(13); if (i === 0) this.doc.text('-', this.x + 3, this.y); this.doc.text(l, this.x + 14, this.y); this.y += 12.8 })
    }
    this.y += 3
  }
  table(cols: Col[], rows: string[][], o: { boldLast?: boolean; boldRows?: number[] } = {}) {
    const total = sum(cols.map((c) => c.w))
    const ws = cols.map((c) => (c.w / total) * this.W)
    const pad = 4, size = 8.5, lh = size * 1.25
    const header = () => {
      this.doc.setFillColor(233, 236, 240); this.doc.setDrawColor(214, 220, 227); this.doc.setLineWidth(0.5)
      this.doc.rect(this.x, this.y, this.W, lh + pad * 2, 'FD')
      this.doc.setFont('helvetica', 'bold'); this.doc.setFontSize(7.5); this.doc.setTextColor(79, 90, 105)
      let cx = this.x
      cols.forEach((c, i) => { const tx = clean(c.label).toUpperCase(); this.doc.text(tx, c.align === 'right' ? cx + ws[i] - pad : cx + pad, this.y + pad + 7.5, { align: c.align === 'right' ? 'right' : 'left' }); cx += ws[i] })
      this.y += lh + pad * 2
    }
    this.ensure(lh * 3 + pad * 4); header()
    rows.forEach((r, ri) => {
      const bold = (o.boldLast && ri === rows.length - 1) || o.boldRows?.includes(ri)
      this.doc.setFont('helvetica', bold ? 'bold' : 'normal'); this.doc.setFontSize(size)
      const wrapped = r.map((v, i) => this.doc.splitTextToSize(clean(v), ws[i] - pad * 2) as string[])
      const h = Math.max(...wrapped.map((w) => w.length)) * lh + pad * 2
      if (this.y + h > this.bottom) { this.doc.addPage(); this.y = 56; header(); this.doc.setFont('helvetica', bold ? 'bold' : 'normal'); this.doc.setFontSize(size) }
      this.doc.setDrawColor(214, 220, 227); this.doc.setLineWidth(0.5); this.doc.line(this.x, this.y + h, this.x + this.W, this.y + h)
      this.doc.setTextColor(20, 27, 36)
      let cx = this.x
      wrapped.forEach((w, i) => { w.forEach((l, k) => this.doc.text(l, cols[i].align === 'right' ? cx + ws[i] - pad : cx + pad, this.y + pad + 7 + k * lh, { align: cols[i].align === 'right' ? 'right' : 'left' })); cx += ws[i] })
      this.y += h
    })
    this.y += 8
  }
  finish(): Blob {
    const n = this.doc.getNumberOfPages()
    for (let i = 1; i <= n; i++) {
      this.doc.setPage(i); this.doc.setFont('helvetica', 'normal'); this.doc.setFontSize(7.5); this.doc.setTextColor(120, 128, 140)
      this.doc.text(clean(this.footer), this.x, 792 - 28)
      this.doc.text(`Page ${i} of ${n}`, this.x + this.W, 792 - 28, { align: 'right' })
    }
    return this.doc.output('blob')
  }
}

// Download the basis of estimate: PDF by default, CSV (opens in Excel) or HTML from the menu.
import { useState } from 'react'
import type { Estimate, MonteCarloResult, Project } from '@/core/types'
import { getState, mutate } from '@/core/store'
import { htmlDoc, saveFile } from '@/core/download'
import { cx } from '@/core/util'
import { Button } from '@/ui/kit'
import { toast } from '@/ui/toast'
import { boeCsv, boePdf } from './boe'
import { boeHtml } from './lib'

export type BoeFormat = 'pdf' | 'csv' | 'html'
const FORMATS: { id: BoeFormat; label: string; hint: string }[] = [
  { id: 'pdf', label: 'PDF', hint: 'Paginated document for the record' },
  { id: 'csv', label: 'CSV for Excel', hint: 'Every line, markup and category in one sheet' },
  { id: 'html', label: 'Web page (HTML)', hint: 'Opens in a browser' },
]

/** Generate and offer the BOE file. `est` is the version to export (a draft's unsaved working copy is allowed). */
export async function downloadBoe(fmt: BoeFormat, est: Estimate, project: Project, mc?: MonteCarloResult) {
  const s = getState()
  const base = `${project.code}-BOE-v${est.version}-Class${est.cls}`
  let r: Awaited<ReturnType<typeof saveFile>>
  try {
    if (fmt === 'pdf') r = await saveFile(`${base}.pdf`, boePdf(s, est, { mc }), 'application/pdf')
    else if (fmt === 'csv') r = await saveFile(`${base}.csv`, boeCsv(s, est, { mc }), 'text/csv')
    else { const doc = boeHtml(s, est, { mc }); r = await saveFile(`${base}.html`, htmlDoc(doc.title, doc.body), 'text/html') }
  } catch {
    toast('Could not build the file', 'bad')
    return
  }
  if (r === 'saved') {
    mutate(() => {}, { entity: 'estimate', entityId: est.id, projectId: project.id, action: 'export', summary: `Exported basis of estimate for v${est.version} as ${fmt.toUpperCase()}` })
    toast(`Basis of estimate downloaded (${fmt.toUpperCase()})`)
  } else toast(r === 'declined' ? 'Download cancelled' : 'Downloads are not available here', 'info')
}

/** Split button: the main action downloads the PDF; the arrow offers CSV and HTML. */
export function BoeDownload({ get, project, mc, size = 'md', variant = 'secondary' }: { get: () => Estimate; project: Project; mc?: MonteCarloResult; size?: 'sm' | 'md'; variant?: 'secondary' | 'primary' }) {
  const [open, setOpen] = useState(false)
  const [busy, setBusy] = useState(false)
  const run = async (f: BoeFormat) => { setOpen(false); setBusy(true); try { await downloadBoe(f, get(), project, mc) } finally { setBusy(false) } }
  return (
    <div className="relative inline-flex">
      <Button size={size} variant={variant} className="rounded-r-none" disabled={busy} onClick={() => run('pdf')}>{busy ? 'Preparing…' : 'Download BOE (PDF)'}</Button>
      <Button size={size} variant={variant} className="-ml-px rounded-l-none px-2" aria-label="More download formats" aria-haspopup="menu" aria-expanded={open} onClick={() => setOpen((x) => !x)}>
        <span aria-hidden className="text-[10px]">▾</span>
      </Button>
      {open && <>
        <button aria-label="Close menu" className="fixed inset-0 z-30 cursor-default" onClick={() => setOpen(false)} />
        <div role="menu" className="absolute right-0 top-full z-40 mt-1 w-64 overflow-hidden rounded-md border border-line bg-surface shadow-lg">
          {FORMATS.map((f) => (
            <button key={f.id} role="menuitem" className={cx('flex w-full flex-col items-start px-3 py-2 text-left hover:bg-sunken', f.id === 'pdf' && 'border-b border-line')} onClick={() => run(f.id)}>
              <span className="text-[13px] font-semibold">{f.label}{f.id === 'pdf' && <span className="ml-1.5 text-[11px] font-normal text-faint">default</span>}</span>
              <span className="text-[11.5px] text-muted">{f.hint}</span>
            </button>
          ))}
        </div>
      </>}
    </div>
  )
}

      <Modal open={confirmSubmit} onClose={() => setConfirmSubmit(false)} title={`Submit v${w.version} for approval`}
        footer={<><Button variant="ghost" onClick={() => setConfirmSubmit(false)}>Cancel</Button><Button variant="primary" disabled={issues.some((i) => i.block)} onClick={submit}>Submit for approval</Button></>}>
        <div className="flex flex-col gap-3 text-[13px]">
          <div className="grid grid-cols-2 gap-3 sm:grid-cols-3">
            <Mini label="Total" value={money(t.total)} />
            <Mini label="AACE class" value={<ClassBadge cls={cls} size="lg" />} />
            <Mini label="Expected range" value={`${money(t.low)}–${money(t.high)}`} />
          </div>
          <p className="text-muted">Routes through <b className="text-ink">Estimate approval</b>: estimating lead review, PM concurrence, project controls acceptance, then the version locks. The amount ({money(t.total)}) and class travel with the request.</p>
          {gate?.requiredEstimateClass && cls > gate.requiredEstimateClass && <Callout tone="warn" title={`${gate.name} needs Class ${gate.requiredEstimateClass}`}>This draft computes to Class {cls}. It can be approved, but it will not satisfy the gate until the maturity items for Class {gate.requiredEstimateClass} are complete.</Callout>}
          {issues.length > 0 ? (
            <ul className="flex flex-col gap-1">
              {issues.map((i) => <li key={i.text} className={cx('flex gap-2', i.block ? 'text-bad' : 'text-warn')}><span aria-hidden>{i.block ? '✕' : '!'}</span>{i.text}</li>)}
            </ul>
          ) : <Callout tone="good">All checks pass: lines priced, basis of estimate complete, no unexplained benchmark exceptions.</Callout>}
        </div>
      </Modal>
    </div>
  )
}

function Mini({ label, value }: { label: string; value: React.ReactNode }) {
  return <div className="min-w-0 rounded border border-line px-3 py-2"><div className="text-[11px] font-semibold uppercase tracking-wide text-faint">{label}</div><div className="mt-0.5 font-display text-[17px] font-bold tabular-nums">{value}</div></div>
}

function submitIssues(e: Estimate, t: ReturnType<typeof estimateTotals>) {
  const out: { text: string; block: boolean }[] = []
  if (!e.lines.length) out.push({ text: 'Add at least one line item.', block: true })
  const unpriced = e.lines.filter((l) => !(l.qty > 0) || !(l.unitCost > 0))
  if (unpriced.length) out.push({ text: `${unpriced.length} line(s) have zero quantity or unit cost.`, block: true })
  const nodesc = e.lines.filter((l) => !l.description.trim())
  if (nodesc.length) out.push({ text: `${nodesc.length} line(s) have no description.`, block: true })
  if (!e.basis.scopeBasis.trim()) out.push({ text: 'Basis of estimate needs a scope basis.', block: true })
  if (!e.basis.classJustification.trim()) out.push({ text: 'Basis of estimate needs a class justification.', block: false })
  if (!e.basis.contingencyRationale.trim()) out.push({ text: 'Basis of estimate needs a contingency rationale.', block: false })
  if (!e.basis.assumptions.length) out.push({ text: 'No assumptions recorded.', block: false })
  const st = getState()
  const ex = benchmarkExceptions(e, st.rates, st.settings.estimating.regionFactors).filter((x) => !x.line.override)
  if (ex.length && !e.basis.assumptions.some((a) => /benchmark|library|rate table|above|below|quote/i.test(a))) out.push({ text: `${ex.length} line(s) are more than 25% from the rate-table median; explain them in the assumptions.`, block: false })
  const units = new Set(st.settings.estimating.units.map((u) => u.id))
  const badUnit = e.lines.filter((l) => !units.has(l.unit))
  if (badUnit.length) out.push({ text: `${badUnit.length} line(s) use a unit that is not on the allowed-units list (${Array.from(new Set(badUnit.map((l) => l.unit))).join(', ')}).`, block: true })
  const p = st.projects.find((x) => x.id === e.projectId)
  const drift = p ? tablesDrift(st, p, e) : []
  if (drift.length) out.push({ text: `Markups differ from the portfolio tables (${drift.join('; ')}). Apply the current tables first.`, block: true })
  if (t.total <= 0) out.push({ text: 'Total is zero.', block: true })
  return out
}

      </div>
    </Modal>
  )
}

// ───────────────────────────── Maturity checklist (ES-04)
function evidenceFor(s: AppState, p: Project, e: Estimate, key: string): { text: string; ok?: boolean } | undefined {
  const dl = (ids: string[]) => s.deliverables.filter((x) => x.projectId === p.id && ids.includes(x.defId)).sort((a, b) => b.stageIndex - a.stageIndex)[0]
  const dtxt = (ids: string[], name: string) => { const d = dl(ids); return d ? { text: `${name}: ${d.status.replace(/-/g, ' ')}`, ok: d.status === 'accepted' || d.status === 'waived' } : undefined }
  const permits = s.permits.filter((x) => x.projectId === p.id)
  const land = s.landRights.filter((x) => x.projectId === p.id)
  const risks = activeRisks(s, p.id)
  switch (key) {
    case 'need-defined': return dtxt(['d-charter', 'ds-charter'], 'Charter')
    case 'route-selected': return dtxt(['d-options'], 'Option analysis') ?? (p.stageIndex >= 1 ? { text: 'Option gate passed', ok: true } : undefined)
    case 'scope-quantities': return { text: `${p.scope.quantities.length} quantities in the scope statement`, ok: p.scope.quantities.length > 0 }
    case 'design-30': return dtxt(['d-design', 'ds-design'], 'Design package')
    case 'survey-complete': return dtxt(['d-site'], 'Site assessment & survey')
    case 'permits-identified': return { text: `${permits.length} permit${permits.length === 1 ? '' : 's'} in the permit matrix`, ok: permits.length > 0 }
    case 'land-identified': return { text: `${land.length} parcel${land.length === 1 ? '' : 's'} identified`, ok: land.length > 0 }
    case 'material-quotes': { const n = e.lines.filter((l) => l.source === 'quote').length; return { text: `${n} line${n === 1 ? '' : 's'} priced from vendor quotes`, ok: n > 0 } }
    case 'risk-register': { const q = risks.filter((r) => r.cost.max > 0); return { text: `${q.length} active risk${q.length === 1 ? '' : 's'} with three-point cost`, ok: q.length >= 3 } }
    case 'schedule-baseline': return dtxt(['d-sched', 'ds-sched'], 'Schedule')
    case 'contractor-bids': { const c = s.contracts.filter((x) => x.projectId === p.id && x.type === 'Construction'); return c.length ? { text: `Construction contract: ${c[0].status}`, ok: c[0].status !== 'rfp' } : undefined }
    case 'ifc': return dtxt(['d-design', 'ds-design'], 'IFC design')
    case 'land-acquired': return land.length ? { text: `${land.filter((x) => x.status === 'acquired').length} of ${land.length} parcels acquired`, ok: land.every((x) => x.status === 'acquired') } : undefined
  }
}

They cannot be changed on a project estimate.</p>

                </div>
              </>
            ) : <Callout tone="info" title="No quantified risks">Add risks with three-point cost impacts on the project's risk register to size contingency from risk. Until then use the deterministic class percentage.</Callout>}
          </div>
        </div>
      </Panel>
    </div>
  )
}

interface BoeDraft { scopeBasis: string; assumptions: string[]; exclusions: string[]; contingencyRationale: string; classJustification: string }

function BoePanel({ s, project, e, editable, edit, mc, riskCount, download }: { s: AppState; project: Project; e: Estimate; editable: boolean; edit: EditFn; mc?: MonteCarloResult; riskCount: number; download: React.ReactNode }) {
  const ai = useAiAvailable()
  const [busy, setBusy] = useState(false)
  const [stream, setStream] = useState('')
  const [draft, setDraft] = useState<BoeDraft | null>(null)
  const [err, setErr] = useState('')
  const t = estimateTotals(e)
  const srcs = sourceSummary(e)
  const ex = benchmarkExceptions(e, s.rates, s.settings.estimating.regionFactors)
  const guide = contingencyForClass(s, e.cls)
  const runAi = async () => {
    setBusy(true); setErr(''); setDraft(null); setStream('Thinking…')
    const facts = estimateFacts(s, e)
    const prompt = `You are a senior cost estimator at a US gas utility writing the Basis of Estimate (BOE) for a capital project estimate, following AACE 34R-05 practice. Data is a fictional demo.
Write from the facts below only. Do not invent quantities, prices or permits that are not implied by the facts. Be specific and concise, utility-professional tone.

FACTS (JSON):
${JSON.stringify(facts)}

Return ONLY a JSON object, no prose and no code fences, with exactly these keys:
{"scopeBasis": string (2-4 sentences: what is estimated and from what definition), "assumptions": string[] (5-8 items), "exclusions": string[] (3-6 items), "contingencyRationale": string (1-3 sentences consistent with the contingency method and percent given), "classJustification": string (1-3 sentences tying the AACE class to the completed and missing maturity items)}`
    try {
      const txt = await ask(prompt, { onText: (x) => setStream(x) })
      const m = txt.match(/\{[\s\S]*\}/)

    if (!draft) return
    edit((d) => { d.basis = { ...draft } })

  }

  return (
    <div className="grid grid-cols-1 gap-3 lg:grid-cols-[minmax(0,1fr)_300px]">
      <Panel title="Basis of estimate" subtitle="What the number rests on. Exported with every line, markup, source and approval."
        actions={<>
          {editable && ai !== false && <Button size="sm" onClick={runAi} disabled={busy} icon={<span aria-hidden>✦</span>}>{busy ? 'Drafting…' : 'AI draft BOE'}</Button>}
          {download}
        </>}>
        {err && <div className="mb-3"><Callout tone="bad">{err}</Callout></div>}
        {busy && <div className="mb-3 whitespace-pre-wrap rounded border border-info/40 bg-info/5 px-3 py-2 font-mono text-[11.5px] text-muted">{stream.slice(-600)}</div>}
        {draft && (
          <div className="mb-4 rounded-md border border-info/50 bg-info/5 p-3">
            <div className="mb-2 flex flex-wrap items-center gap-2"><AiTag /><span className="text-[13px] font-semibold">Suggested basis of estimate</span><span className="text-xs text-muted">Nothing changes until you accept.</span>
              <span className="ml-auto flex gap-2"><Button size="sm" variant="ghost" onClick={() => setDraft(null)}>Discard</Button><Button size="sm" variant="primary" onClick={accept}>Accept into draft</Button></span></div>
            <div className="flex flex-col gap-2 text-[13px]">
              <p><b>Scope basis.</b> {draft.scopeBasis}</p>
              <div><b>Assumptions</b><ul className="ml-5 list-disc">{draft.assumptions.map((a) => <li key={a}>{a}</li>)}</ul></div>
              <div><b>Exclusions</b><ul className="ml-5 list-disc">{draft.exclusions.map((a) => <li key={a}>{a}</li>)}</ul></div>
              <p><b>Contingency.</b> {draft.contingencyRationale}</p>
              <p><b>Class.</b> {draft.classJustification}</p>
            </div>
          </div>
        )}
        <div className="flex flex-col gap-3">
          <Field label="Scope basis">{editable ? <TextArea id="boe-scope" value={e.basis.scopeBasis} onChange={(x) => edit((d) => { d.basis.scopeBasis = x.target.value })} /> : <p className="text-[13px]">{e.basis.scopeBasis || '—'}</p>}</Field>
          <div className="grid grid-cols-1 gap-3 md:grid-cols-2">
            <Field label="Assumptions" hint={editable ? 'One per line' : undefined}>{editable ? <ListArea id="boe-assume" value={e.basis.assumptions} onChange={(v) => edit((d) => { d.basis.assumptions = list(v) })} /> : <Bullets items={e.basis.assumptions} />}</Field>
            <Field label="Exclusions" hint={editable ? 'One per line' : undefined}>{editable ? <ListArea id="boe-excl" value={e.basis.exclusions} onChange={(v) => edit((d) => { d.basis.exclusions = list(v) })} /> : <Bullets items={e.basis.exclusions} />}</Field>
          </div>
          <Field label={<span className="flex items-center justify-between gap-2">Contingency rationale {editable && <button type="button" className="text-[11px] font-semibold text-link" onClick={(ev) => { ev.preventDefault(); edit((d) => { d.basis.contingencyRationale = contingencyRationale(e, mc, riskCount, guide) }) }}>Generate from contingency settings</button>}</span>}>
            {editable ? <TextArea id="boe-cont" className="min-h-[56px]" value={e.basis.contingencyRationale} onChange={(x) => edit((d) => { d.basis.contingencyRationale = x.target.value })} /> : <p className="text-[13px]">{e.basis.contingencyRationale || '—'}</p>}
          </Field>

            {editable ? <TextArea id="boe-class" className="min-h-[56px]" value={e.basis.classJustification} onChange={(x) => edit((d) => { d.basis.classJustification = x.target.value })} /> : <p className="text-[13px]">{e.basis.classJustification || '—'}</p>}
          </Field>
        </div>
      </Panel>
      <div className="flex flex-col gap-3">
        <Panel title="Sources of pricing" pad={false}>
          <table className="tbl">
            <thead><tr><th>Source</th><th className="num">Lines</th><th className="num">Amount</th><th className="num">Share</th></tr></thead>

          </table>
        </Panel>
        <Panel title="Benchmark exceptions" subtitle="Lines more than 25% from the rate-table P50 for the region">
          {ex.length === 0 ? <p className="text-[13px] text-good">None. Every rate-table line is within 25% of its median.</p> : (
            <ul className="flex flex-col gap-1.5 text-[12.5px]">
              {ex.map((x) => <li key={x.line.id} className="flex items-start justify-between gap-2"><span className="min-w-0"><span className="text-faint">{x.line.workPackage}</span> · {x.line.description}</span><Badge tone={x.b.pct > 0 ? 'bad' : 'warn'}>{x.b.pct > 0 ? '+' : ''}{pct(x.b.pct)}</Badge></li>)}
            </ul>
          )}
        </Panel>
      </div>
    </div>
  )
}

function ListArea({ id, value, onChange }: { id: string; value: string[]; onChange: (v: string) => void }) {
  const [txt, setTxt] = useState(value.join('\n'))
  const joined = value.join('\n')
  const last = useRef(joined)
  useEffect(() => { if (joined !== last.current) { setTxt(joined); last.current = joined } }, [joined])
  return <TextArea id={id} className="min-h-[120px]" value={txt} onChange={(e) => { setTxt(e.target.value); onChange(e.target.value); last.current = e.target.value.split('\n').map((x) => x.replace(/^\s*[-•*]\s*/, '').trim()).filter(Boolean).join('\n') }} />
}
