function RecordsBadge({ v }: { v: Project['asset']['recordsStatus'] }) {
  return v === 'complete' ? <Badge tone="good">Traceable, verifiable, complete</Badge> : v === 'gaps' ? <Badge tone="warn">Gaps identified</Badge> : <Badge tone="bad">Unknown</Badge>
}

// ───────────────────────────── Scope
function ScopePanel({ p, canEdit }: { p: Project; canEdit: boolean }) {
  const [edit, setEdit] = useState<ScopeStatement | null>(null)
  const lastEdit = useStore((s) => s.audit.filter((a) => a.projectId === p.id && a.entity === 'scope').pop())
  const people = useStore((s) => s.people)
  const qTypes = useStore((s) => s.settings.estimating.quantityTypes)
  const lib = useStore((s) => s.settings.estimating.scopeLibrary)
  const save = () => {
    if (!edit) return
    const clean: ScopeStatement = {
      summary: edit.summary.trim(),
      inScope: edit.inScope.map((x) => x.trim()).filter(Boolean),
      outOfScope: edit.outOfScope.map((x) => x.trim()).filter(Boolean),
      assumptions: edit.assumptions.map((x) => x.trim()).filter(Boolean),
      quantities: edit.quantities.map((q) => { const t = qTypes.find((x) => x.id === q.typeId); return t ? { ...q, label: t.label, unit: t.unit } : q }),
      capitalPct: edit.capitalPct,
    }
    const problems = scopeProblems(clean, qTypes.map((t) => t.id))
    if (problems.length) { toast(problems[0], 'bad'); return }
    const diffs: string[] = []
    if (clean.summary !== p.scope.summary) diffs.push('summary')
    const added = clean.inScope.filter((x) => !p.scope.inScope.includes(x)), removed = p.scope.inScope.filter((x) => !clean.inScope.includes(x))
    if (added.length) diffs.push(`added in-scope "${added.join('", "')}"`)
    if (removed.length) diffs.push(`removed in-scope "${removed.join('", "')}"`)
    for (const q of clean.quantities) { const o = p.scope.quantities.find((x) => x.label === q.label); if (!o) diffs.push(`added quantity ${q.label}`); else if (o.qty !== q.qty) diffs.push(`${q.label} ${o.qty.toLocaleString()} → ${q.qty.toLocaleString()} ${q.unit}`) }
    if (JSON.stringify(clean.outOfScope) !== JSON.stringify(p.scope.outOfScope)) diffs.push('out-of-scope list')
    if (JSON.stringify(clean.assumptions) !== JSON.stringify(p.scope.assumptions)) diffs.push('assumptions')
    for (const q of p.scope.quantities) if (!clean.quantities.some((x) => x.label === q.label)) diffs.push(`removed quantity ${q.label}`)
    if (clean.capitalPct !== p.scope.capitalPct) diffs.push(`capital ${p.scope.capitalPct ?? '—'}% → ${clean.capitalPct ?? '—'}% (O&M ${clean.capitalPct === undefined ? '—' : Math.round((100 - clean.capitalPct) * 10) / 10}%)`)
    if (!diffs.length) { setEdit(null); return }
    mutate((d) => { d.projects.find((x) => x.id === p.id)!.scope = clean }, { entity: 'scope', entityId: p.id, projectId: p.id, action: 'edit', summary: `Scope statement edited: ${diffs.join('; ')}` })
    toast('Scope statement saved')
    setEdit(null)
  }
  const hasControl = useStore((s) => s.budgets.some((b) => b.projectId === p.id && b.kind === 'control'))
  if (edit) {
    return (
      <Panel title="Edit scope statement" actions={<><Button variant="ghost" onClick={() => setEdit(null)}>Cancel</Button><Button variant="primary" onClick={save}>Save scope</Button></>}>
        <div className="flex flex-col gap-3">
          {hasControl && <Callout tone="warn" title="This project has a control budget">Scope changes that affect cost or schedule must go through a change request so the Current baseline moves with an approved decision. Use this edit for clarifications.</Callout>}
          <Field label="Summary"><TextArea id="scope-summary" className="min-h-[70px]" value={edit.summary} onChange={(e) => setEdit({ ...edit, summary: e.target.value })} /></Field>
          <p className="text-[12px] text-muted">Pick items from the estimating team's scope library so estimates read them consistently; your own wording is allowed and is outlined for the estimator.</p>
          <div className="grid grid-cols-1 gap-4 md:grid-cols-3">
            <ScopeListEditor k="inScope" label="In scope" items={edit.inScope} onChange={(v) => setEdit({ ...edit, inScope: v })} />
            <ScopeListEditor k="outOfScope" label="Out of scope" items={edit.outOfScope} onChange={(v) => setEdit({ ...edit, outOfScope: v })} />
            <ScopeListEditor k="assumptions" label="Assumptions" items={edit.assumptions} onChange={(v) => setEdit({ ...edit, assumptions: v })} />
          </div>
          <div className="grid grid-cols-1 gap-4 lg:grid-cols-[minmax(0,1.6fr)_minmax(0,1fr)]">
            <QuantityEditor quantities={edit.quantities} onChange={(q) => setEdit({ ...edit, quantities: q })} />
            <div>
              <div className="mb-1 text-[12px] font-semibold text-muted">Capital / O&M split</div>
              <CapitalSplitEditor value={edit.capitalPct} onChange={(v) => setEdit({ ...edit, capitalPct: v })} />
              <p className="mt-1 text-[11px] text-faint">Placeholder until internal orders exist; the Finance tab compares it with what the IOs carry.</p>
            </div>
          </div>
        </div>
      </Panel>
    )
  }
  return (
    <Panel title="Scope statement" subtitle={lastEdit ? `Last edited ${fmtDateTime(lastEdit.at)} by ${personName({ people }, lastEdit.actorId)}` : 'As chartered'}
      actions={canEdit ? <Button size="sm" onClick={() => setEdit(JSON.parse(JSON.stringify(p.scope)))}>Edit scope</Button> : <span className="text-xs text-faint">Editable by the PM</span>}>
      <p className="text-[14px] leading-relaxed">{p.scope.summary}</p>
      <div className="mt-3 grid grid-cols-1 gap-4 md:grid-cols-3">
        <ScopeList title="In scope" items={p.scope.inScope} lib={lib.inScope} mark="✓" tone="text-good" />
        <ScopeList title="Out of scope" items={p.scope.outOfScope} lib={lib.outOfScope} mark="✕" tone="text-faint" />
        <ScopeList title="Assumptions" items={p.scope.assumptions} lib={lib.assumptions} mark="•" tone="text-muted" />
      </div>
      <div className="mt-3 flex flex-wrap gap-2">
        {p.scope.quantities.map((q) => (
          <div key={q.label} className="rounded border border-line bg-sunken/50 px-3 py-1.5">
            <div className="text-[10.5px] font-semibold uppercase tracking-wide text-faint">{q.label}</div>
            <div className="font-display text-[17px] font-bold tabular-nums">{q.qty.toLocaleString()} <span className="text-[12px] font-semibold text-muted">{q.unit}</span></div>
          </div>
        ))}
        {p.scope.capitalPct !== undefined && (
          <div className="min-w-[150px] rounded border border-line bg-sunken/50 px-3 py-1.5">
            <div className="text-[10.5px] font-semibold uppercase tracking-wide text-faint">Capital / O&M</div>
            <div className="font-display text-[17px] font-bold tabular-nums">{p.scope.capitalPct}% <span className="text-[12px] font-semibold text-muted">/ {Math.round((100 - p.scope.capitalPct) * 10) / 10}%</span></div>
            <div className="mt-1 flex h-1 overflow-hidden rounded-full bg-line"><span className="bg-ink" style={{ width: `${p.scope.capitalPct}%` }} /></div>
          </div>
        )}
      </div>
    </Panel>
  )
}
function ScopeList({ title, items, lib, mark, tone }: { title: string; items: string[]; lib: string[]; mark: string; tone: string }) {
  return (
    <div className="min-w-0">
      <div className="eyebrow mb-1">{title}</div>
      {items.length ? <ul className="flex flex-col gap-1 text-[13px]">{items.map((x) => <li key={x} className="flex gap-1.5"><span className={cx('shrink-0 font-bold', tone)}>{mark}</span><span>{x}{!lib.includes(x) && <span className="ml-1.5 rounded-sm border border-info/40 px-1 align-middle text-[9.5px] font-semibold uppercase tracking-wide text-info" title="Free text, not from the scope library">custom</span>}</span></li>)}</ul> : <span className="text-[13px] text-faint">None stated</span>}
    </div>
  )
}

// ───────────────────────────── Milestones
function MilestonePanel({ p }: { p: Project }) {
  const asOf = useStore((s) => s.settings.asOf)
  const span = p.milestones.flatMap((m) => [m.baseline, m.forecast, m.actual].filter(Boolean) as string[]).sort()
  const lo = span[0], hi = span[span.length - 1]
  const total = Math.max(1, daysBetween(lo, hi))
  const X = (d: string) => `${(daysBetween(lo, d) / total) * 100}%`
  return (
    <Panel title="Milestones" subtitle="Baseline vs forecast vs actual. Forecast moves only through approved changes or the monthly forecast.">
      <div className="relative mb-4 h-9 rounded bg-sunken/60">
        {asOf >= lo && asOf <= hi && <div className="absolute inset-y-0 w-px bg-marker" style={{ left: X(asOf) }}><span className="absolute -top-0.5 left-1 text-[10px] font-semibold text-ink">Today</span></div>}
        {p.milestones.map((m) => {
          const slip = daysBetween(m.baseline, m.actual ?? m.forecast)
          return (
            <div key={m.id} title={`${m.name}: baseline ${m.baseline}, ${m.actual ? 'actual ' + m.actual : 'forecast ' + m.forecast}`}>
              <span className="absolute top-2 h-2.5 w-2.5 -translate-x-1/2 rotate-45 border border-faint bg-surface" style={{ left: X(m.baseline) }} />
              {slip !== 0 && <span className={cx('absolute top-[13px] h-0.5', slip > 0 ? 'bg-bad/50' : 'bg-good/50')} style={{ left: slip > 0 ? X(m.baseline) : X(m.actual ?? m.forecast), width: `${(Math.abs(slip) / total) * 100}%` }} />}
              <span className={cx('absolute top-2 h-2.5 w-2.5 -translate-x-1/2 rotate-45', m.actual ? 'bg-good' : slip > 30 ? 'bg-bad' : slip > 0 ? 'bg-warn' : 'bg-ink')} style={{ left: X(m.actual ?? m.forecast) }} />
            </div>
          )
        })}
        <span className="absolute bottom-0.5 left-1 text-[10px] text-faint">{fmtDate(lo)}</span>
        <span className="absolute bottom-0.5 right-1 text-[10px] text-faint">{fmtDate(hi)}</span>
      </div>
      <Table rows={p.milestones} rowKey={(m) => m.id} columns={[
        { key: 'n', header: 'Milestone', render: (m) => <span className="font-semibold">{m.name}</span> },
        { key: 'b', header: 'Baseline', render: (m) => <span className="tabular-nums">{fmtDate(m.baseline)}</span> },
        { key: 'f', header: 'Forecast', render: (m) => <span className="tabular-nums">{m.actual ? <span className="text-faint">—</span> : fmtDate(m.forecast)}</span> },
        { key: 'a', header: 'Actual', render: (m) => <span className="tabular-nums">{m.actual ? fmtDate(m.actual) : '—'}</span> },
        { key: 's', header: 'Slip', num: true, render: (m) => { const d = daysBetween(m.baseline, m.actual ?? m.forecast); return <span className={cx('font-semibold', d > 30 ? 'text-bad' : d > 0 ? 'text-warn' : d < 0 ? 'text-good' : 'text-faint')}>{d > 0 ? `+${d}` : d}d</span> } },
        { key: 'st', header: 'Status', render: (m) => {
          if (m.actual) return <Badge tone="good">Complete</Badge>
          const d = daysBetween(m.baseline, m.forecast)
          if (m.forecast < asOf) return <Badge tone="bad">Overdue</Badge>
          if (d > 30) return <Badge tone="bad">Late</Badge>
          if (d > 0) return <Badge tone="warn">At risk</Badge>
          return <Badge tone="neutral">On plan</Badge>
        } },
      ]} />
    </Panel>
  )
}

// ───────────────────────────── Team
function TeamPanel({ p, canEdit }: { p: Project; canEdit: boolean }) {
  const s = useStore()
  const [editing, setEditing] = useState<TeamPosition | null>(null)
  const [val, setVal] = useState('')
  const roleFor: Record<TeamPosition, Parameters<typeof PersonSelect>[0]['role']> = { sponsor: 'sponsor', pm: 'pm', engineer: 'engineer', estimator: 'estimator', projectControls: 'project-controls', scheduler: 'scheduler', constructionManager: 'construction-manager', safety: 'safety-advisor', environmental: 'environmental', land: 'land' }
  const save = (pos: TeamPosition) => {
    const from = personName(s, p.team[pos])
    mutate((d) => { d.projects.find((x) => x.id === p.id)!.team[pos] = val || undefined }, { entity: 'project', entityId: p.id, projectId: p.id, action: 'team', summary: `${POS_LABEL[pos]}: ${from} → ${personName(s, val)}` })
    toast(`${POS_LABEL[pos]} updated`)
    setEditing(null)
  }
  return (
    <Panel title="Team" subtitle="Positions route workflow steps and own deliverables" pad={false}>
      <ul className="divide-y divide-line">
        {(Object.keys(POS_LABEL) as TeamPosition[]).map((pos) => {
          const person = s.people.find((x) => x.id === p.team[pos])
          return (
            <li key={pos} className="flex items-center gap-2.5 px-4 py-1.5">
              {editing === pos ? (
                <div className="flex w-full flex-wrap items-center gap-2">
                  <span className="w-28 shrink-0 text-[12px] text-muted">{POS_LABEL[pos]}</span>
                  <div className="min-w-[160px] flex-1"><PersonSelect id={`team-${pos}`} role={roleFor[pos]} includeEmpty value={val} onChange={setVal} /></div>
                  <Button size="sm" variant="primary" onClick={() => save(pos)}>Save</Button><Button size="sm" variant="ghost" onClick={() => setEditing(null)}>Cancel</Button>
                </div>
              ) : (
                <>
                  {person ? <Avatar name={person.name} size={26} /> : <span className="h-[26px] w-[26px] shrink-0 rounded-full border border-dashed border-line" />}
                  <div className="min-w-0 flex-1">
                    <div className="truncate text-[13px] font-semibold">{person?.name ?? <span className="font-normal text-faint">Unassigned</span>}</div>
                    <div className="truncate text-[11px] text-muted">{POS_LABEL[pos]}{person ? ` · ${person.title}` : ''}{person?.outOfOffice && s.settings.asOf >= person.outOfOffice.from && s.settings.asOf <= person.outOfOffice.to ? ' · out of office' : ''}</div>
                  </div>
                  {person && person.dollarAuthority > 0 && <span className="hidden text-[11px] tabular-nums text-faint sm:inline" title="Dollar authority under the DOA matrix">DOA {money(person.dollarAuthority)}</span>}
                  {canEdit && <Button size="sm" variant="ghost" onClick={() => { setEditing(pos); setVal(p.team[pos] ?? '') }}>Change</Button>}
                </>
              )}
            </li>
          )
        })}
      </ul>
    </Panel>
  )
}

// ───────────────────────────── Custom fields
function CustomFieldsPanel({ p, canEdit }: { p: Project; canEdit: boolean }) {
  const defs = useStore((s) => s.settings.fieldDefs.filter((d) => d.programIds.includes(p.programId)))
  const tpl = useStore((s) => templateFor(s, p))
  const [vals, setVals] = useState<Record<string, string> | null>(null)
  if (!defs.length) return null
  const save = () => {
    const changed = defs.filter((d) => (vals![d.key] ?? '') !== (p.customFields[d.key] ?? ''))
    if (changed.length) {
      mutate((d) => { const x = d.projects.find((y) => y.id === p.id)!; for (const c of changed) x.customFields[c.key] = vals![c.key] ?? '' }, { entity: 'project', entityId: p.id, projectId: p.id, action: 'fields', summary: `Updated ${changed.map((c) => `${c.label} → "${vals![c.key]}"`).join(', ')}` })
      toast('Fields saved')
    }
    setVals(null)
  }
  return (
    <Panel title="Program fields" subtitle="Configured in settings for this program" actions={canEdit && !vals ? <Button size="sm" variant="ghost" onClick={() => setVals({ ...p.customFields })}>Edit</Button> : vals ? <><Button size="sm" variant="ghost" onClick={() => setVals(null)}>Cancel</Button><Button size="sm" variant="primary" onClick={save}>Save</Button></> : undefined}>
      <div className="flex flex-col gap-2.5">
        {defs.map((d) => {
          const v = (vals ?? p.customFields)[d.key]
          const required = d.requiredAtStage !== undefined
          const due = required && p.stageIndex >= d.requiredAtStage!
          const missing = due && !v?.trim()
          return (
            <div key={d.key} className="min-w-0">
              <div className="flex flex-wrap items-center gap-1.5 text-[11px] font-semibold uppercase tracking-wide text-faint">
                {d.label}
                {required && <Badge tone={missing ? 'bad' : due ? 'good' : 'neutral'}>{missing ? 'Required now' : `Required at ${tpl.stages[d.requiredAtStage!]?.name ?? 'stage ' + (d.requiredAtStage! + 1)}`}</Badge>}
              </div>

function Documents({ p, rows }: { p: Project; rows: ReturnType<typeof useStore.getState>['documents'] }) {

'Hide' : 'Conditions'}</Button></td>

                            const late = c.status === 'open' && c.due && c.due < asOf

                          })}

                            <AddCondition pm={pm} projectId={projectId} />

function AddCondition({ pm, projectId }: { pm: Permit; projectId: string }) {
  const [text, setText] = useState('')
  const [due, setDue] = useState('')
  const add = () => {
    if (!text.trim()) return
    mutate((d) => { d.permits.find((x) => x.id === pm.id)!.conditions.push({ id: uid('pc'), text: text.trim(), due: due || undefined, status: 'open' }) }, { entity: 'permit', entityId: pm.id, projectId, action: 'condition-added', summary: `${pm.type} ${pm.number}: added condition "${text.trim()}"${due ? ` due ${due}` : ''}` })
    setText(''); setDue(''); toast('Condition added')
  }
  return (
    <form className="flex min-w-0 flex-1 flex-wrap gap-2" onSubmit={(e) => { e.preventDefault(); add() }}>
      <Input className="h-7 min-w-[220px] flex-1 py-0.5" placeholder="New condition" value={text} onChange={(e) => setText(e.target.value)} />
      <Input className="h-7 w-[150px] py-0.5" type="date" value={due} onChange={(e) => setDue(e.target.value)} aria-label="Due date" />
      <Button size="sm" type="submit" disabled={!text.trim()}>Add condition</Button>
    </form>
  )
}

interface Extracted { text: string; due?: string; dueBasis?: string; accept: boolean }

function ExtractDrawer({ permit, onClose, projectName, asOf }: { permit: Permit | null; onClose: () => void; projectName: string; asOf: string }) {
  const ai = useAiAvailable()
  const [text, setText] = useState('')
  const [busy, setBusy] = useState(false)
  const [items, setItems] = useState<Extracted[]>([])
  const [err, setErr] = useState('')
  if (!permit) return null
  const sample = `CONDITIONS OF APPROVAL — ${permit.type.toUpperCase()} ${permit.number}\n1. Permittee shall notify the City Inspector at least 48 hours prior to commencing work.\n2. No lane closures on arterial streets between 7:00–9:00 AM or 4:00–6:00 PM.\n3. Submit a traffic control plan stamped by a registered engineer no later than 10 working days before start.\n4. All trenches in the roadway shall be plated or backfilled at the end of each work day.\n5. Final paving shall be completed within 30 days of trench backfill and shall extend curb to lane line.\n6. Provide written notice to all fronting residents and businesses 72 hours before work.`
  const run = async () => {
    setBusy(true); setErr(''); setItems([])
    try {
      const prompt = `You extract compliance conditions from a permit issued to a gas utility for a capital project.
Project: ${projectName}. Permit: ${permit.type} ${permit.number} from ${permit.agency}. Today is ${asOf}.
Return JSON exactly: {"conditions":[{"text":"<one obligation, imperative, under 140 chars>","due":"YYYY-MM-DD or null","dueBasis":"<short phrase if the deadline is relative, e.g. '48 h before start', else null>"}]}
Rules: one entry per distinct obligation the utility or its contractor must meet; skip boilerplate definitions and agency obligations; use a calendar date only if the text states one; do not invent conditions.
Already tracked (do not repeat): ${JSON.stringify(permit.conditions.map((c) => c.text))}
PERMIT TEXT:
"""${text.slice(0, 12000)}"""`
      const r = await askJSON<{ conditions: { text: string; due?: string | null; dueBasis?: string | null }[] }>(prompt)
      const list = (r?.conditions ?? []).filter((c) => c && typeof c.text === 'string' && c.text.trim())
      setItems(list.map((c) => ({ text: c.text.trim(), due: c.due && /^\d{4}-\d{2}-\d{2}$/.test(c.due) ? c.due : undefined, dueBasis: c.dueBasis ?? undefined, accept: true })))
      if (!list.length) setErr('No conditions found in that text.')
    } catch (e) { setErr(aiMessage(e)) } finally { setBusy(false) }
  }
  const accept = () => {
    const chosen = items.filter((x) => x.accept && x.text.trim())
    mutate((d) => { const pm = d.permits.find((x) => x.id === permit.id)!; for (const c of chosen) pm.conditions.push({ id: uid('pc'), text: c.text, due: c.due, status: 'open' }) },
      { entity: 'permit', entityId: permit.id, projectId: permit.projectId, action: 'conditions-accepted', summary: `${permit.type} ${permit.number}: accepted ${chosen.length} AI-extracted condition${chosen.length === 1 ? '' : 's'} after review` })
    toast(`${chosen.length} condition${chosen.length === 1 ? '' : 's'} added`); setItems([]); setText(''); onClose()
  }
  return (
    <Drawer open onClose={onClose} width={640} title={<span className="flex items-center gap-2">Extract permit conditions <AiTag>AI assist</AiTag></span>}
      footer={items.length ? <><Button variant="ghost" onClick={() => setItems([])}>Discard</Button><Button variant="primary" disabled={!items.some((x) => x.accept)} onClick={accept}>Accept {items.filter((x) => x.accept).length} condition{items.filter((x) => x.accept).length === 1 ? '' : 's'}</Button></> : undefined}>
      <div className="flex flex-col gap-3">
        <p className="text-[13px] text-muted">Paste the conditions-of-approval section of <b className="text-ink">{permit.type} {permit.number}</b>. AI assist drafts a list of obligations; you review each one before it is added to the tracker.</p>
        <TextArea rows={9} value={text} onChange={(e) => setText(e.target.value)} placeholder="Paste permit text…" />
        <div className="flex flex-wrap gap-2">
          <Button variant="primary" disabled={!text.trim() || busy || ai === false} onClick={run}>{busy ? 'Thinking…' : 'Extract conditions'}</Button>
          {!text && <Button variant="ghost" onClick={() => setText(sample)}>Use sample permit text</Button>}
        </div>
        {err && <Callout tone="bad">{err}</Callout>}
        {items.length > 0 && (
          <div className="flex flex-col gap-2 rounded-md border border-info/40 bg-info/5 p-3">
            <div className="flex items-center gap-2"><AiTag /> <span className="text-[13px] font-semibold">{items.length} conditions found — review before accepting</span></div>
            {items.map((c, i) => (
              <div key={i} className="grid grid-cols-[auto_1fr_150px] items-start gap-2">
                <input type="checkbox" className="mt-2 h-4 w-4" checked={c.accept} onChange={(e) => setItems(items.map((x, j) => (j === i ? { ...x, accept: e.target.checked } : x)))} aria-label="Accept" />
                <div><Input className="h-8" value={c.text} onChange={(e) => setItems(items.map((x, j) => (j === i ? { ...x, text: e.target.value } : x)))} />{c.dueBasis && <div className="mt-0.5 text-[11px] text-muted">Deadline: {c.dueBasis}</div>}</div>
                <Input className="h-8" type="date" value={c.due ?? ''} onChange={(e) => setItems(items.map((x, j) => (j === i ? { ...x, due: e.target.value || undefined } : x)))} aria-label="Due" />
              </div>
            ))}
          </div>
        )}
      </div>
    </Drawer>
  )
}

function ContractDrawer({ edit, onClose, code }: { edit: { c: Contract; isNew: boolean } | null; onClose: () => void; code: string }) {
  const [c, setC] = useDraft<Contract | null>(edit?.c ?? null)
  if (!edit || !c) return null
  const locked = !edit.isNew && edit.c.status !== 'rfp'
  const save = () => {
    if (!c.vendor.trim() && c.status !== 'rfp') { toast('Vendor is required once awarded', 'bad'); return }
    const prev = edit.c
    const parts: string[] = []
    if (prev.changeOrders !== c.changeOrders) parts.push(`change orders ${money(prev.changeOrders)} → ${money(c.changeOrders)}`)
    if (prev.invoiced !== c.invoiced) parts.push(`invoiced ${money(prev.invoiced)} → ${money(c.invoiced)}`)
    if (prev.status !== c.status) parts.push(`status ${prev.status} → ${c.status}`)
    if (prev.value !== c.value) parts.push(`value ${money(prev.value)} → ${money(c.value)}`)
    mutate((d) => { const i = d.contracts.findIndex((x) => x.id === c.id); if (i >= 0) d.contracts[i] = c; else d.contracts.push(c) },
      { entity: 'contract', entityId: c.id, projectId: c.projectId, action: edit.isNew ? 'create' : 'update', summary: edit.isNew ? `${code}: added ${c.type.toLowerCase()} contract ${c.number || '(RFP)'} ${c.vendor}` : `${code}: contract ${c.number} ${parts.join(', ') || 'updated'}` })
    toast(edit.isNew ? 'Contract added' : 'Contract saved'); onClose()

function AwardModal({ c, onClose, onStarted, code }: { c: Contract | null; onClose: () => void; onStarted: (id: string) => void; code: string }) {

    toast(instId ? 'Award submitted for approval' : 'No published contract-award workflow', instId ? 'good' : 'bad')

  return (

// Project Information record (PI-*): the controlled project data sheet. Every edit is a numbered
// revision with a reason, field-level from → to, and an audit entry; the record can be viewed as it
// stood at any revision and exported as a formal data sheet. Overview stays the summary page.
import { useMemo, useState } from 'react'
import { getState, mutate, useMe, useStore } from '@/core/store'
import type { AppState, InfoFieldDef, Project } from '@/core/types'
import { infoCompleteness, missingInfo, reviseInfo } from '@/core/engine/info'
import { templateFor } from '@/core/engine/gates'
import { deliverableCompleteness } from '@/core/engine/finance'
import { projectHref } from '@/core/router'
import { htmlDoc, saveFile } from '@/core/download'
import { askJSON, aiMessage, useAiAvailable } from '@/core/ai'

const fieldAnchor = (key: string) => `pi-field-${key}`
function jumpTo(key: string) {
  const el = document.getElementById(fieldAnchor(key))
  if (!el) return
  el.scrollIntoView({ behavior: 'smooth', block: 'center' })
  el.classList.add('ring-2', 'ring-marker')
  setTimeout(() => el.classList.remove('ring-2', 'ring-marker'), 1600)
}

function ProjectInfoInner({ projectId }: { projectId: string }) {
  const s = useStore()
  const me = useMe()
  const p = s.projects.find((x) => x.id === projectId)!
  const tpl = templateFor(s, p)
  const revs = useMemo(() => revisionsFor(s, p.id), [s.infoRevisions, p.id])
  const current = useMemo(() => currentValues(s, p), [s, p])
  const sections = sectionsOf(s)
  const comp = infoCompleteness(s, p)
  const dc = deliverableCompleteness(s, p)
  const missing = missingInfo(s, p)
  const gate = currentGateName(s, p)
  const latest = revs[0]
  const frozen = p.status === 'closed' || p.status === 'cancelled'
  const canEdit = !frozen && (me.roles.includes('admin') || me.roles.includes('pmo') || Object.values(p.team).includes(me.id))

  const [asOfRev, setAsOfRev] = useState<number | null>(null)
  const [editing, setEditing] = useState<string[]>([]) // section names in edit mode
  const [draft, setDraft] = useState<Record<string, string>>({})
  const [note, setNote] = useState('')
  const [aiKeys, setAiKeys] = useState<string[]>([])
  const [textOpen, setTextOpen] = useState(false)

  const viewing = asOfRev !== null
  const values = viewing ? valuesAsOf(current, revs, asOfRev!) : current
  const highlight = viewing ? changedKeys(revs.find((r) => r.rev === asOfRev)) : new Set<string>()
  const editable = s.settings.infoFields.filter((f) => !f.readOnly)
  const edits = Object.fromEntries(editable.filter((f) => f.key in draft && (draft[f.key] ?? '') !== (current[f.key] ?? '')).map((f) => [f.key, draft[f.key]]))
  const changedCount = Object.keys(edits).length
  const isEditing = editing.length > 0

  const startEdit = (secs: string[]) => {
    setAsOfRev(null)
    if (!isEditing) { setDraft({ ...current }); setNote(''); setAiKeys([]) }
    setEditing(Array.from(new Set([...editing, ...secs])))
  }
  const cancel = () => { setEditing([]); setDraft({}); setNote(''); setAiKeys([]) }
  const save = () => {
    if (!note.trim()) { toast('Enter a reason for the change', 'bad'); return }
    if (!changedCount) { toast('No changes to save', 'info'); return }
    let ok = false
    mutate((d) => { ok = !!reviseInfo(d, p.id, edits, note.trim()) })
    if (!ok) { toast('No changes to save', 'info'); return }
    const n = revisionsFor(getState(), p.id)[0]?.rev ?? 0
    cancel()
    toast(`Project information revised (rev ${n})`)
  }

  const exportSheet = async () => {
    const rev = asOfRev ?? latest?.rev ?? 0
    const html = htmlDoc(`${p.code} Project Information Sheet Rev ${rev}`, infoSheetHtml(s, p, { values, asOfRev: rev }))
    const r = await saveFile(`${p.code}-project-information-rev${rev}.html`, html, 'text/html')
    toast(r === 'saved' ? `Project information sheet exported (rev ${rev})` : r === 'declined' ? 'Export cancelled' : 'Downloads are not available here', r === 'saved' ? 'good' : 'info')
  }
  const exportCsv = async () => {
    const r = await saveFile(`${p.code}-project-information.csv`, infoCsv(s, p, values), 'text/csv')
    toast(r === 'saved' ? 'Project information CSV exported' : r === 'declined' ? 'Export cancelled' : 'Downloads are not available here', r === 'saved' ? 'good' : 'info')
  }
  const copyText = async () => {
    try { await navigator.clipboard.writeText(infoText(s, p, values)); toast('Project information copied as text') }
    catch { setTextOpen(true) }
  }

  const acceptProposal = (pr: DraftProposal) => {
    const def = s.settings.infoFields.find((f) => f.key === pr.key)
    if (!def) return
    if (!isEditing) setDraft({ ...current, [pr.key]: pr.value })
    else setDraft((d) => ({ ...d, [pr.key]: pr.value }))
    setEditing((e) => Array.from(new Set([...e, def.section])))
    setAsOfRev(null)
    setAiKeys((k) => Array.from(new Set([...k, pr.key])))
  }

  return (
    <div className="flex flex-col gap-4">
      <Header s={s} p={p} comp={comp} dc={dc} missing={missing} gate={gate} latest={latest} revCount={revs.length}
        canEdit={canEdit && !viewing} frozen={frozen} isEditing={isEditing}
        onEdit={() => startEdit(sections.map((x) => x.name))} onSheet={exportSheet} onCsv={exportCsv} onCopy={copyText}
        asOfRev={asOfRev} setAsOfRev={(r) => { if (!isEditing) setAsOfRev(r) }} revs={revs.map((r) => ({ rev: r.rev, at: r.at }))} />

      <LinkedRecords s={s} p={p} />

      {viewing && (
        <Callout tone="marker" title={asOfRev === 0 ? 'Viewing the record before revision 1' : `Viewing revision ${asOfRev} of ${latest?.rev ?? 0}`}>
          <div className="flex flex-wrap items-center gap-2">
            <span>{asOfRev === 0 ? 'Values as first recorded, before any tracked revision.' : `The record as it stood after rev ${asOfRev} (${fmtDateTime(revs.find((r) => r.rev === asOfRev)?.at)}). Fields changed in that revision are outlined.`} System fields show current values. Editing is paused while viewing history.</span>
            <Button size="sm" onClick={() => setAsOfRev(null)}>Back to current</Button>
          </div>
        </Callout>
      )}

      {canEdit && !viewing && <AiDraft s={s} p={p} current={current} onAccept={acceptProposal} draft={isEditing ? draft : current} />}

      <div className="grid grid-cols-1 gap-4 2xl:grid-cols-[minmax(0,1fr)_minmax(0,420px)]">
        <div className="flex min-w-0 flex-col gap-4">
          {sections.filter((sec) => sec.fields.length).map((sec) => {
            const secEditing = editing.includes(sec.name)
            const secMissing = sec.fields.filter((f) => requirement(f, values[f.key] ?? '', p.stageIndex) === 'blocking').length
            const filled = sec.fields.filter((f) => !f.readOnly && (values[f.key] ?? '').trim()).length
            const total = sec.fields.filter((f) => !f.readOnly).length
            return (
              <Panel key={sec.name} title={sec.name}
                subtitle={<span className="tabular-nums">{filled}/{total} completed{secMissing ? <span className="ml-2 font-semibold text-bad">{secMissing} required empty</span> : null}</span>}
                actions={canEdit && !viewing && !secEditing && <Button size="sm" variant="ghost" onClick={() => startEdit([sec.name])}>Edit section</Button>}>
                <div className="grid grid-cols-1 gap-x-6 gap-y-3.5 md:grid-cols-2">
                  {sec.fields.map((f) => (
                    <FieldRow key={f.key} s={s} p={p} def={f} value={values[f.key] ?? ''} tplStages={tpl.stages.length}
                      editing={secEditing && !f.readOnly} draftValue={draft[f.key] ?? ''} changed={secEditing && (draft[f.key] ?? '') !== (current[f.key] ?? '')}
                      ai={aiKeys.includes(f.key)} highlight={highlight.has(f.key)}
                      onChange={(v) => { setDraft((d) => ({ ...d, [f.key]: v })) }} />
                  ))}
                </div>
              </Panel>
            )
          })}
        </div>
        <div className="flex min-w-0 flex-col gap-4">
          <CostDrivers p={p} canEdit={canEdit && !viewing} />
          <RevisionHistory s={s} revs={revs} asOfRev={asOfRev} onViewRev={(r) => { if (!isEditing) { setAsOfRev(r === latest?.rev ? null : r); document.getElementById('main')?.scrollTo?.({ top: 0, behavior: 'smooth' }) } }} />
        </div>
      </div>

      {isEditing && (
        <div className="sticky bottom-0 z-20 -mx-1 rounded-md border border-line bg-surface p-3 shadow-lg">
          <div className="flex flex-col gap-2 md:flex-row md:items-end">
            <label className="flex min-w-0 flex-1 flex-col gap-1">

function AiDraft({ s, p, current, draft, onAccept }: { s: AppState; p: Project; current: Record<string, string>; draft: Record<string, string>; onAccept: (pr: DraftProposal) => void }) {
  const ai = useAiAvailable()

  const [err, setErr] = useState('')
  const [props, setProps] = useState<DraftProposal[] | null>(null)
  const [done, setDone] = useState<Record<string, 'accepted' | 'rejected'>>({})
  const empty = s.settings.infoFields.filter((f) => !f.readOnly && !(current[f.key] ?? '').trim() && !(draft[f.key] ?? '').trim())
  if (ai === false) return null
  const run = async () => {
    setBusy(true); setErr(''); setProps(null); setDone({})
    try {
      const raw = await askJSON<unknown>(draftPrompt(s, p, empty))
      setProps(cleanProposals(raw, empty))
    } catch (e) { setErr(aiMessage(e)) } finally { setBusy(false) }
  }
  const label = (k: string) => s.settings.infoFields.find((f) => f.key === k)
  const open = (props ?? []).filter((x) => !done[x.key])
  if (!empty.length && !props) return null
  return (

      actions={<>

        <Button size="sm" disabled={busy || !empty.length} onClick={run}>{busy ? 'Thinking…' : props ? 'Draft again' : 'Draft missing fields'}</Button>
      </>}

      {props && props.length > 0 && (
        <ul className="flex flex-col divide-y divide-line">
          {props.map((pr) => {
            const def = label(pr.key)
            const st = done[pr.key]
            return (
              <li key={pr.key} className={cx('flex flex-col gap-1 py-2 sm:flex-row sm:items-start sm:gap-3', st && 'opacity-60')}>
                <div className="min-w-0 flex-1">
                  <div className="flex items-center gap-2 text-[12px] font-semibold text-muted">{def?.label}<span className="font-normal text-faint">· {def?.section}</span></div>
                  <div className="whitespace-pre-wrap text-[13px]">{def ? displayValue(def, pr.value) : pr.value}</div>
                  {pr.basis && <div className="text-[11.5px] text-faint">Basis: {pr.basis}</div>}
                </div>
                <div className="flex shrink-0 gap-1">
                  {st ? <Badge tone={st === 'accepted' ? 'good' : 'neutral'}>{st === 'accepted' ? 'Added to edit form' : 'Rejected'}</Badge> : <>
                    <Button size="sm" onClick={() => { onAccept(pr); setDone((d) => ({ ...d, [pr.key]: 'accepted' })) }}>Accept</Button>
                    <Button size="sm" variant="ghost" onClick={() => setDone((d) => ({ ...d, [pr.key]: 'rejected' }))}>Reject</Button>
                  </>}
                </div>
              </li>
            )

  )
}

function RecordsMatrix({ p }: { p: Project }) {

    mutate((d) => { for (const seg of m.segs) for (const t of m.types) d.records.push({ id: uid('rec'), projectId: p.id, type: t, segment: seg, status: 'missing' }) },
      { entity: 'records', entityId: p.id, projectId: p.id, action: 'create', summary: `${p.code}: set up records matrix (${m.segs.length} segments × ${m.types.length} record types)` })

    if (!seg || segs.includes(seg)) return
    const ts = types.length ? types : defaultMatrix(p).types
    mutate((d) => { for (const t of ts) d.records.push({ id: uid('rec'), projectId: p.id, type: t, segment: seg, status: 'missing' }) }, { entity: 'records', entityId: p.id, projectId: p.id, action: 'add-segment', summary: `${p.code}: added segment "${seg}" to records matrix` })
    setNewSeg(''); toast('Segment added')
  }
  const addType = () => {
    if (!newType) return

  const verifyAllAttached = () => {

'' : 's'} as traceable, verifiable and complete` })

    return (

  return (

    mutate((d) => { const x = d.records.find((y) => y.id === r.id)!; x.status = 'missing'; x.verifiedById = undefined }, { entity: 'record', entityId: r.id, projectId: p.id, action: 'reject', summary: `${p.code}: rejected ${label} (${r.fileName}) — ${reason.trim()}` })

  }
  const hist = s.audit.filter((a) => a.entityId === r.id).slice().reverse()
  return (
    <Drawer open onClose={onClose} title={label} width={520}>
      <div className="flex flex-col gap-4">
        <KV items={[['Status', <StatusBadge status={r.status} />], ['File', r.fileName ? <Mono>{r.fileName}</Mono> : '—'], ['Verified by', r.verifiedById ? personName(s, r.verifiedById) : '—'], ['Segment', r.segment]]} />
        {r.status !== 'verified' && (
          <div className="flex flex-col gap-2 rounded-md border border-line p-3">
            <div className="text-[13px] font-semibold">{r.status === 'missing' ? 'Attach the record' : 'Replace the attached file'}</div>
            <p className="text-[12px] text-muted">PDF scans of signed field records, test charts or mill certificates. The file name is kept as evidence; verification is a separate step by Records.</p>
            <FileButton size="md" variant={r.status === 'missing' ? 'primary' : 'secondary'} label={r.status === 'missing' ? 'Choose file…' : 'Replace file…'} onFile={attach} accept=".pdf,.tif,.tiff,.jpg,.png,.xlsx,.csv,.dwg" />
          </div>
        )}
        {r.status === 'attached' && (
          canVerify ? (
            <div className="flex flex-col gap-2 rounded-md border border-marker/60 bg-marker/5 p-3">
              <div className="text-[13px] font-semibold">Records verification</div>
              <p className="text-[12px] text-muted">Confirm the record is traceable (tied to this segment), verifiable (signed, legible, complete) and matches the as-built.</p>

          ) : <Callout tone="info">Awaiting verification by the Records role.</Callout>
        )}

function Closeout({ p, items, rc, wfId, wfStatus }: { p: Project; items: CloseoutItem[]; rc: ReturnType<typeof recordsCompleteness>; wfId?: string; wfStatus?: string }) {
  const s = useStore()

  const pos = position(s, p.id)
  const openItems = items.filter((x) => x.status === 'open')
  const blocks: string[] = []
  if (!rc.total) blocks.push('No records matrix — records completeness cannot be shown')
  else if (rc.pct < 1) blocks.push(`${rc.total - rc.verified} of ${rc.total} records not verified (${rc.missing} missing, ${rc.attached} awaiting verification)`)
  if (!items.length) blocks.push('Closeout checklist not started')
  else if (openItems.length) blocks.push(`${openItems.length} checklist item${openItems.length === 1 ? '' : 's'} open`)
  if (pos.commitments > 0 && p.status !== 'closed') blocks.push(`${money(pos.commitments)} in open commitments`)
  const running = wfStatus === 'running'
  const approved = wfStatus === 'approved'
  const recordsBlock = blocks.filter((b) => !b.startsWith('$') && !b.includes('commitments'))

  const setStatus = (it: CloseoutItem, st: CloseoutItem['status']) => {

'complete' : st === 'n/a' ? 'not-applicable' : 'reopen', summary: `${p.code}: closeout item "${it.item}" ${st === 'done' ? 'completed' : st === 'n/a' ? 'marked not applicable' : 'reopened'}` })

    mutate((d) => { for (const [item, role] of STD_CLOSEOUT) d.closeout.push({ id: uid('co'), projectId: p.id, item, ownerRole: role, status: 'open' }) }, { entity: 'closeout', entityId: p.id, projectId: p.id, action: 'create', summary: `${p.code}: started closeout checklist (${STD_CLOSEOUT.length} standard items)` })

function Unitization({ p }: { p: Project }) {
  const s = useStore()
  const b = activeBudget(s, p.id)
  const pos = position(s, p.id)
  const rows = useMemo(() => {
    if (!b) return []
    const tot = sum(Object.values(b.byCategory))
    return Object.entries(b.byCategory).filter(([, v]) => v).map(([cid, v]) => {
      const c = s.categories.find((x) => x.id === cid)
      return { cid, code: c?.code ?? '', name: c?.name ?? cid, erp: c?.erpCostElement ?? '', budget: v, share: tot ? v / tot : 0, final: Math.round((tot ? v / tot : 0) * (pos.actualToDate || pos.eac)) }
    }).sort((a, z) => a.code.localeCompare(z.code))
  }, [b, s.categories, pos.actualToDate, pos.eac])
  const primary = p.scope.quantities[0]
  const basis = pos.actualToDate || pos.eac
  const exportCsv = async () => {
    const data = [
      ...p.scope.quantities.map((q) => ({ section: 'Asset unit', project: p.code, wbs: p.erpWbs, item: q.label, qty: q.qty, unit: q.unit, cost_element: '', amount: '' })),
      ...rows.map((r) => ({ section: 'Cost', project: p.code, wbs: p.erpWbs, item: `${r.code} ${r.name}`, qty: '', unit: '', cost_element: r.erp, amount: r.final })),
    ]
    const res = await saveFile(`${p.code}-unitization.csv`, toCSV(data), 'text/csv')
    if (res === 'saved') { mutate(() => {}, { entity: 'project', entityId: p.id, projectId: p.id, action: 'export-unitization', summary: `${p.code}: exported unitization package for plant accounting (${money(basis)} across ${rows.length} cost elements)` }); toast('Unitization package exported') }
    else toast('Download not available here', 'info')
  }
  if (!b) return <Panel title="Unitization summary"><Empty title="No budget yet">Unitization uses the control budget's cost breakdown once one exists.</Empty></Panel>
  return (
    <Panel title="Unitization summary" subtitle={`Asset units and cost by category for plant accounting · WBS ${p.erpWbs} · allocated on ${pos.actualToDate ? 'actual to date' : 'forecast'} ${money(basis)}`} pad={false}
      actions={<Button size="sm" onClick={exportCsv}>Export for plant accounting</Button>}>
      <div className="flex flex-wrap gap-x-6 gap-y-2 border-b border-line px-4 py-3">
        {p.scope.quantities.map((q) => <div key={q.label}><div className="text-[11px] font-semibold uppercase tracking-wide text-faint">{q.label}</div><div className="font-display text-[17px] font-bold tabular-nums">{num(q.qty)} <span className="text-[12px] font-normal text-muted">{q.unit}</span></div></div>)}
        {primary && primary.qty > 0 && <div><div className="text-[11px] font-semibold uppercase tracking-wide text-faint">Unit cost</div><div className="font-display text-[17px] font-bold tabular-nums">{money(basis / primary.qty)} <span className="text-[12px] font-normal text-muted">/ {primary.unit}</span></div></div>}
      </div>
      <div className="scroll-x" style={{ maxHeight: 340, overflowY: 'auto' }}>
        <table className="tbl">
          <thead><tr><th>Category</th><th>ERP element</th><th className="num">Budget</th><th className="num">Share</th><th className="num">Allocated</th></tr></thead>
          <tbody>
            {rows.map((r) => <tr key={r.cid}><td className="whitespace-nowrap"><Mono className="text-faint">{r.code}</Mono> {r.name}</td><td><Mono>{r.erp}</Mono></td><td className="num">{money(r.budget)}</td><td className="num text-muted">{pct(r.share, 1)}</td><td className="num font-semibold">{money(r.final)}</td></tr>)}
            <tr><td colSpan={2} className="font-semibold">Total (excl. contingency)</td><td className="num font-semibold">{money(sum(rows.map((r) => r.budget)))}</td><td className="num">100%</td><td className="num font-semibold">{money(sum(rows.map((r) => r.final)))}</td></tr>
          </tbody>
        </table>
      </div>
    </Panel>
  )
}

function ExposureByCategory({ risks }: { risks: RiskT[] }) {
  const g = groupBy(risks.filter((r) => isActive(r) && r.kind === 'threat'), (r) => r.category)
  const rows = Object.entries(g).map(([k, rs]) => ({ label: k, value: Math.round(sum(rs.map(expectedCost))) })).sort((a, b) => b.value - a.value).slice(0, 6)
  if (!rows.length) return null
  return <div className="flex flex-col gap-1.5"><span className="eyebrow">Threat exposure by category</span><HBarChart rows={rows} /></div>
}

// ───────────────────────────── Contingency drawdown & usage over time (RK-08)

// ───────────────────────────── Library (RK-03)
function LibraryModal({ open, onClose, projectId }: { open: boolean; onClose: () => void; projectId: string }) {
  const s = useStore()
  const p = s.projects.find((x) => x.id === projectId)!
  const lib = s.riskLibrary.filter((x) => x.appliesTo.includes(p.workMethod))
  const onReg = new Set(s.risks.filter((r) => r.projectId === projectId && isActive(r)).map((r) => r.libraryId))
  const [pick, setPick] = useState<Record<string, boolean>>({})
  const base = budgetBase(s, projectId)
  const add = () => {
    const items = lib.filter((x) => pick[x.id])
    if (!items.length) return
    mutate((d) => { items.forEach((it, i) => d.risks.push(riskFromLibrary(d, p, it, base, i))) }, { entity: 'risk', entityId: projectId, projectId, action: 'create', summary: `Added ${items.length} risk(s) from library: ${items.map((x) => x.title).join(', ')}` })
    toast(`${items.length} risk${items.length > 1 ? 's' : ''} added from the library`)
    setPick({}); onClose()
  }
  return (
    <Modal open={open} onClose={onClose} wide title="Add risks from the library"
      footer={<><span className="mr-auto self-center text-xs text-muted">Typical cost is the library % of {money(base)} budget; adjust on the risk after adding.</span><Button variant="ghost" onClick={onClose}>Cancel</Button><Button variant="primary" disabled={!Object.values(pick).some(Boolean)} onClick={add}>Add selected</Button></>}>
      <p className="mb-3 text-[13px] text-muted">Library risks for <b className="text-ink">{p.workMethod.replace(/-/g, ' ')}</b> work, drawn from closed projects.</p>
      <div className="flex flex-col divide-y divide-line">
        {lib.map((it) => {
          const ml = Math.max(5000, Math.round((Math.abs(it.typicalCostPct) / 100) * base / 1000) * 1000)
          const has = onReg.has(it.id)
          return (
            <div key={it.id} className="flex items-start gap-3 py-2.5">
              <Checkbox checked={!!pick[it.id]} disabled={has} onChange={(v) => setPick({ ...pick, [it.id]: v })} label={<span className="sr-only">{it.title}</span>} />
              <div className="min-w-0 flex-1">
                <div className="flex flex-wrap items-center gap-2"><b className="text-[13px]">{it.title}</b><Badge>{it.category}</Badge>{it.typicalCostPct < 0 && <Badge tone="good">Opportunity</Badge>}{has && <Badge tone="info">On register</Badge>}</div>
                <p className="text-xs text-muted">{it.cause} → {it.event} → {it.effect}</p>
              </div>
              <div className="shrink-0 text-right text-xs tabular-nums"><div className="font-semibold">{money(ml)}</div><div className="text-faint">{it.typicalCostPct}% · {it.typicalDays} d</div></div>
            </div>
          )
        })}
      </div>
    </Modal>
  )
}

      setItems((r.risks ?? []).map((x) => ({ ...x, key: uid('ai') })))
      if (!r.risks?.length) setErr('No new risks suggested.')
    } catch (e) { setErr(aiMessage(e)) } finally { setBusy(false) }
  }
  const accept = (x: AiRisk & { key: string }) => {
    const L = Math.min(5, Math.max(1, Math.round(x.likelihood))) as RiskT['likelihood']
    const C = Math.min(5, Math.max(1, Math.round(x.consequence))) as RiskT['consequence']
    const ml = Math.max(1000, Math.round(Number(x.costMostLikely) || 0))
    const days = Math.max(0, Math.round(Number(x.scheduleDays) || 0))
    mutate((d) => {
      d.risks.push({
        id: uid('rsk'), projectId, code: nextRiskCode(d, p), title: x.title, cause: x.cause, event: x.event, effect: x.effect, category: x.category, kind: x.kind === 'opportunity' ? 'opportunity' : 'threat',
        ownerId: p.team.pm ?? d.currentUserId, status: 'open', likelihood: L, consequence: C, consequenceDim: x.category === 'Environmental' ? 'Environmental' : x.category === 'Safety' ? 'Safety' : days > 40 ? 'Schedule' : 'Financial',
        probability: PROB_BY_L[L - 1], cost: { min: Math.round(ml * 0.4), ml, max: Math.round(ml * 2.2) }, scheduleDays: { min: Math.round(days * 0.4), ml: days, max: days * 2 },
        responses: x.response ? [{ id: uid('rr'), action: x.response, ownerId: p.team.pm ?? d.currentUserId, due: addDays(d.settings.asOf, 30), status: 'open', cost: 0 }] : [],
        residual: { likelihood: Math.max(1, L - 1) as RiskT['likelihood'], consequence: C }, createdAt: d.settings.asOf,
      })
    }, { entity: 'risk', entityId: projectId, projectId, action: 'create', summary: `Accepted AI-suggested risk: ${x.title}` })
    setItems(items.filter((i) => i.key !== x.key))
    toast('Risk added to register')
  }
  if (!ok) return null
  return (

      actions={<Button size="sm" onClick={go} disabled={busy}>{busy ? 'Thinking…' : items.length ? 'Suggest again' : 'Suggest risks'}</Button>}>
      {err && <Callout tone="warn">{err}</Callout>}
      {busy && <p className="text-[13px] text-muted">Thinking… reviewing {s.risks.filter((r) => r.projectId === projectId).length} existing risks against the scope.</p>}
      {!busy && items.length === 0 && !err && <p className="text-[13px] text-muted">No suggestions yet.</p>}
      <div className="flex flex-col divide-y divide-line">
        {items.map((x) => (
          <div key={x.key} className="flex flex-col gap-1.5 py-3 md:flex-row md:items-start">
            <div className="min-w-0 flex-1">
              <div className="flex flex-wrap items-center gap-2"><AiTag /><b className="text-[13px]">{x.title}</b><Badge>{x.category}</Badge>{x.kind === 'opportunity' && <Badge tone="good">Opportunity</Badge>}<ScoreChip l={x.likelihood} c={x.consequence} /><span className="text-xs tabular-nums text-muted">ML {money(x.costMostLikely)} · {x.scheduleDays} d</span></div>

          </div>
        ))}

import { addDays, cx, money, nowISO, uid } from '@/core/util'
import { Badge, Button, Callout, Checkbox, ClassBadge, Field, Input, KV, Mono, PageHeader, Panel, Select, TextArea } from '@/ui/kit'
import { RangeBar } from '@/ui/charts'
import { toast } from '@/ui/toast'
import { PersonSelect, ProgramTag } from '../shared/pickers'

const DRIVERS: Driver[] = ['Safety / compliance', 'Reliability', 'Capacity', 'Customer / new business', 'Relocation', 'Environmental']
const ASSETS: { id: AssetType; label: string; methods: WorkMethod[] }[] = [
  { id: 'transmission-pipeline', label: 'Transmission pipeline', methods: ['replace-open-trench', 'replace-hdd', 'hydrotest'] },
  { id: 'distribution-main', label: 'Distribution main', methods: ['replace-open-trench'] },
  { id: 'valve-site', label: 'Valve site', methods: ['valve-automation'] },
  { id: 'compressor-station', label: 'Compressor station', methods: ['compressor-replacement', 'station-rebuild'] },
]
const METHOD_LABEL: Record<WorkMethod, string> = {
  'replace-open-trench': 'Replace — open trench', 'replace-hdd': 'Replace — HDD', hydrotest: 'Hydrostatic test', abandon: 'Abandon', 'valve-automation': 'Valve automation',
  'station-rebuild': 'Station rebuild', 'compressor-replacement': 'Compressor replacement', 'inline-inspection': 'In-line inspection',
}
const DISTRICTS: Record<string, { box: [number, number, number, number]; county: string; cities: string[] }> = {
  West: { box: [90, 300, 330, 540], county: 'Coastline County', cities: ['Mesa Verde', 'Old Mission', 'Bellport', 'Cresthaven'] },
  East: { box: [360, 260, 620, 500], county: 'Arroyo County', cities: ['San Lucero', 'Rancho Alto', 'Las Piedras', 'Sycamore Flats', 'Westgate', 'Las Lomas'] },
  South: { box: [650, 220, 930, 520], county: 'Sierra Mesa County', cities: ['Palo Duro', 'Ironwood', 'Laguna Ciega', 'Del Arroyo'] },
  North: { box: [260, 50, 720, 210], county: 'Arroyo County', cities: ['Vista Linda', 'Cielo Grande', 'Juniper Ridge', 'Fairhollow'] },
}
const POSITIONS: { pos: TeamPosition; label: string; role: Parameters<typeof PersonSelect>[0]['role'] }[] = [
  { pos: 'sponsor', label: 'Sponsor (director)', role: 'sponsor' },
  { pos: 'pm', label: 'Project manager', role: 'pm' },
  { pos: 'engineer', label: 'Project engineer', role: 'engineer' },
  { pos: 'estimator', label: 'Estimator', role: 'estimator' },
  { pos: 'projectControls', label: 'Project controls', role: 'project-controls' },
  { pos: 'scheduler', label: 'Scheduler', role: 'scheduler' },
  { pos: 'constructionManager', label: 'Construction manager', role: 'construction-manager' },
  { pos: 'safety', label: 'Safety advisor', role: 'safety-advisor' },
  { pos: 'environmental', label: 'Environmental', role: 'environmental' },
  { pos: 'land', label: 'Land agent', role: 'land' },
]
const STEPS = ['Need & driver', 'Asset & location', 'Team', 'Review & create'] as const

interface Form {
  name: string; programId: string; driver: Driver; need: string; citation: string; proceeding: string; custom: Record<string, string>
  asset: AssetType; method: WorkMethod; lineId: string; diameterIn: number; lengthFt: number; sites: number; maopPsig: number; material: string; vintage: number; recordsStatus: Project['asset']['recordsStatus']
  district: string; city: string; terrain: Terrain; classLocation: 1 | 2 | 3 | 4; hca: boolean; dac: boolean; x: number; y: number
  team: Partial<Record<TeamPosition, string>>
  buildEstimate: boolean; startWorkflow: boolean
}

export default function NewProject(_props: { projectId?: string; id?: string }) {
  const s = useStore()
  const me = useMe()
  const [step, setStep] = useState(0)
  const [f, setF] = useState<Form>(() => {
    const prog = s.programs[0]
    return {
      name: '', programId: prog.id, driver: 'Safety / compliance', need: '', citation: prog.regulatoryDriver, proceeding: 'PSE balancing account (AV-PUC-2022-05 (fictional) review)', custom: {},
      asset: 'transmission-pipeline', method: 'replace-open-trench', lineId: '', diameterIn: 16, lengthFt: 8000, sites: 1, maopPsig: 720, material: 'Steel, Grade B', vintage: 1958, recordsStatus: 'gaps',
      district: 'East', city: 'San Lucero', terrain: 'suburban', classLocation: 2, hca: true, dac: false, x: 490, y: 380,
      team: { sponsor: 'u-dir1', pm: 'u-pm1', engineer: 'u-eng1', estimator: 'u-est1', projectControls: 'u-pc1', scheduler: 'u-sch', constructionManager: 'u-cm1', safety: 'u-saf1', environmental: 'u-env', land: 'u-land' },
      buildEstimate: true, startWorkflow: true,
    }
  })
  const set = <K extends keyof Form>(k: K, v: Form[K]) => setF((x) => ({ ...x, [k]: v }))
  const program = s.programs.find((p) => p.id === f.programId)!
  const tpl = s.templates.find((t) => t.id === program.lifecycleTemplateId)!
  const fieldDefs = s.settings.fieldDefs.filter((d) => d.programIds.includes(f.programId))
  const nextNum = Math.max(...s.projects.map((p) => Number(p.code.replace(/\D/g, '')))) + 1
  const code = `PRJ-${nextNum}`

  const setProgram = (id: string) => {
    const pr = s.programs.find((p) => p.id === id)!
    setF((x) => {
      const asset: AssetType = pr.code === 'VAP' ? 'valve-site' : pr.code === 'CSM' ? 'compressor-station' : pr.code === 'DRP' ? 'distribution-main' : 'transmission-pipeline'
      const methods = ASSETS.find((a) => a.id === asset)!.methods
      const sponsor = pr.code === 'CSM' ? 'u-dir2' : 'u-dir1'
      return {
        ...x, programId: id, citation: pr.regulatoryDriver, asset, method: methods.includes(x.method) ? x.method : methods[0],
        driver: pr.code === 'CSM' ? 'Environmental' : pr.code === 'DRP' ? 'Reliability' : 'Safety / compliance',
        proceeding: pr.code === 'PSE' ? 'PSE balancing account (AV-PUC-2022-05 (fictional) review)' : pr.code === 'CSM' ? 'Tier 2 advice letter on in-service' : 'TY2028 GRC (AV-PUC-2026-06 (fictional))',
        diameterIn: asset === 'distribution-main' ? 6 : asset === 'compressor-station' ? 0 : x.diameterIn || 16,
        maopPsig: asset === 'distribution-main' ? 60 : asset === 'compressor-station' ? 1000 : x.maopPsig,
        material: asset === 'distribution-main' ? 'Pre-1940 cast iron / bare steel' : asset === 'compressor-station' ? 'Gas turbine-driven reciprocating units' : 'Steel, Grade B',
        team: { ...x.team, sponsor, engineer: asset === 'compressor-station' ? 'u-eng2' : 'u-eng1', constructionManager: asset === 'compressor-station' ? 'u-cm2' : 'u-cm1' },
      }
    })
  }

  // Draft project object (used for preview estimate and creation)
  const draft = useMemo<Project>(() => buildProject(f, code, s.settings.asOf, tpl.stages.length === 3), [f, code, s.settings.asOf, tpl.stages.length])
  const lines = useMemo(() => parametricLines(draft, s.rates, pricingFor(s, draft), scopeQuantities(s, draft)), [draft, s.rates, s.settings])
  const estPreview = useMemo(() => {
    const maturity = Object.fromEntries((s.settings.estimating.maturityItems ?? MATURITY_ITEMS).map((m) => [m.key, m.key === 'need-defined']))
    const year = Number(s.settings.asOf.slice(0, 4))
    void year
    const cls = computeClass(maturity, s.settings.estimating.maturityItems)
    const e = { lines, loaders: currentLoaders(s), escalationPct: escalationPctFor(s, basisYearOf(s, lines), spendMidpoint(s, draft)), contingency: { method: 'deterministic' as const, pct: contingencyForClass(s, cls) }, cls, salesTax: salesTaxFor(s, draft), region: draft.location.district, rateTableVersion: s.settings.estimating.rateTableVersion }
    return { ...e, maturity, totals: estimateTotals(e) }
  }, [lines, s.settings])
  const defs = requiredDefs(s, draft, 0)

  const problems: string[][] = [
    [!f.name.trim() && 'Give the project a name', f.need.trim().length < 20 && 'Describe the need in at least a sentence', !f.citation.trim() && 'Add the regulatory citation or driver reference', ...fieldDefs.filter((d) => d.requiredAtStage === 0 && !f.custom[d.key]?.trim()).map((d) => `${d.label} is required at initiation`)].filter(Boolean) as string[],
    [!f.lineId.trim() && f.asset !== 'compressor-station' && 'Enter the line or facility ID', f.asset !== 'compressor-station' && f.asset !== 'valve-site' && f.lengthFt <= 0 && 'Length must be greater than zero', !f.city.trim() && 'Enter the city'].filter(Boolean) as string[],
    [!f.team.sponsor && 'Name a sponsor', !f.team.pm && 'Name a project manager'].filter(Boolean) as string[],
    [],
  ]
  const canNext = problems[step].length === 0

  const create = () => {
    const pid = `prj-${nextNum}`
    let instTitle = ''
    mutate((d) => {
      const p: Project = JSON.parse(JSON.stringify(draft))
      const rom = f.buildEstimate ? estPreview.totals.total : 0
      p.sizeTier = rom > 50_000_000 ? 'L' : rom > 5_000_000 || !rom ? 'M' : 'S'
      d.projects.push(p)
      ensureDeliverables(d, p)
      // charter deliverable starts in progress with the wizard output as its first draft
      const charter = d.deliverables.find((x) => x.projectId === pid && /charter/.test(x.defId))
      if (charter) { charter.status = 'in-progress'; charter.version = 1; charter.evidence = [{ id: uid('evd'), name: `${code}_Charter_draft.pdf`, sizeKb: 184, uploadedAt: nowISO(), byId: d.currentUserId }]; charter.updatedAt = nowISO() }
      if (f.buildEstimate && estPreview.lines.length) {
        const e: Estimate = {
          id: uid('est'), projectId: pid, version: 1, title: 'Class 5 rate-table estimate', cls: estPreview.cls, status: 'draft', method: 'parametric', createdAt: nowISO(), createdById: d.currentUserId,
          lines: estPreview.lines.map((l) => ({ ...l, id: uid('ln') })), maturity: estPreview.maturity, loaders: estPreview.loaders, escalationPct: estPreview.escalationPct, contingency: estPreview.contingency, salesTax: estPreview.salesTax, region: estPreview.region, rateTableVersion: estPreview.rateTableVersion,
          basis: {
            scopeBasis: p.scope.summary,
            assumptions: ['Quantities from GIS centerline length', 'Library P50 rates for matching asset, method, diameter and terrain', 'No land acquisition beyond temporary rights'],
            exclusions: ['Customer-side work', 'Unknown contamination', 'Escalation beyond the planned in-service year'],
            contingencyRationale: 'Deterministic 30% for Class 5 per estimating standard.',
            classJustification: 'Definition 0–2%: need, asset and length known; no design.',
          },
        }
        d.estimates.push(e)
        d.audit.push({ id: uid('aud'), at: nowISO(), actorId: d.currentUserId, entity: 'estimate', entityId: e.id, projectId: pid, action: 'create', summary: `Draft Class ${e.cls} rate-table estimate ${money(estPreview.totals.total)} created from initiation` })
      }
      if (f.startWorkflow) {
        const inst = startWorkflow(d, 'project-initiation', { title: `${code} ${p.name}`, recordType: 'initiation', recordId: pid, projectId: pid, data: { need: f.need, amount: estPreview.totals.total, program: program.code, driver: f.driver } })
        if (inst) {
          instTitle = inst.title
          const slot = inst.active[0]
          if (slot && slot.assigneeIds.includes(d.currentUserId)) actOnWorkflow(d, inst.id, slot.nodeId, d.currentUserId, 'submit', 'Submitted from the initiation wizard')
        }
      }
    }, { entity: 'project', entityId: pid, projectId: pid, action: 'create', summary: `Initiated ${code} ${draft.name} (${program.code}, ${tpl.stages[0].name})` })
    toast(instTitle ? `${code} created and sent for program screening` : `${code} created`)
    go(projectHref(pid))
  }

  return (
    <div className="flex flex-col gap-4">
      <PageHeader eyebrow={<a href="#projects" className="hover:underline">Projects</a>} title="Initiate a project"
        subtitle="Capture the need, the asset and the team. Crosswalk creates the project at Stage 1 on the program's lifecycle template, opens its gate deliverables, drafts a Class 5 rate-table estimate from the unit-cost library, and routes the request for program screening." />
      <ol className="grid grid-cols-2 gap-2 sm:grid-cols-4">
        {STEPS.map((label, i) => (
          <li key={label}>
            <button disabled={i > step && problems.slice(0, i).some((p) => p.length)} onClick={() => setStep(i)} className={cx('flex w-full items-center gap-2 rounded-md border px-3 py-2 text-left text-[13px] font-semibold disabled:cursor-not-allowed disabled:opacity-50', i === step ? 'border-ink bg-surface' : 'border-line bg-surface/60 text-muted hover:text-ink')}>
              <span className={cx('flex h-5 w-5 shrink-0 items-center justify-center rounded-full text-[11px]', i < step ? 'bg-good text-white' : i === step ? 'bg-marker text-markerink' : 'bg-sunken text-muted')}>{i < step ? '✓' : i + 1}</span>
              <span className="truncate">{label}</span>
            </button>
          </li>
        ))}
      </ol>

      <div className="grid grid-cols-1 gap-4 lg:grid-cols-[minmax(0,1fr)_320px]">
        <Panel title={STEPS[step]}>
          {step === 0 && (
            <div className="grid grid-cols-1 gap-3 md:grid-cols-2">
              <Field label="Project name" className="md:col-span-2" hint="Line or facility, work type and place, e.g. “Line AV-15 Replacement — Old Mission Blvd”"><Input id="np-name" value={f.name} onChange={(e) => set('name', e.target.value)} placeholder="Line AV-27 Replacement — Arroyo Seco Crossing" /></Field>
              <Field label="Program"><Select id="np-program" value={f.programId} onChange={(e) => setProgram(e.target.value)} options={s.programs.map((p) => ({ value: p.id, label: `${p.code} · ${p.name}` }))} /></Field>
              <Field label="Driver"><Select id="np-driver" value={f.driver} onChange={(e) => set('driver', e.target.value as Driver)} options={DRIVERS} /></Field>
              <Field label="Need statement" className="md:col-span-2" hint="What condition or requirement makes this work necessary? This becomes the charter need and the initiation request."><TextArea id="np-need" className="min-h-[90px]" value={f.need} onChange={(e) => set('need', e.target.value)} placeholder="Records review found no pressure test record for 1.6 miles of Line AV-27 in a Class 3 location; PU Code §958 requires test or replacement." /></Field>
              <Field label="Regulatory citation"><Input id="np-cite" value={f.citation} onChange={(e) => set('citation', e.target.value)} /></Field>
              <Field label="Funding proceeding"><Input id="np-proc" value={f.proceeding} onChange={(e) => set('proceeding', e.target.value)} /></Field>
              <div className="md:col-span-2"><KV cols={3} items={[['Recovery mechanism', program.recovery], ['Lifecycle template', tpl.name], ['DOA matrix', s.doaMatrices.find((m) => m.id === program.doaMatrixId)?.name ?? '—']]} /></div>
              {fieldDefs.map((d) => (
                <Field key={d.key} label={<span className="flex items-center gap-1.5">{d.label}{d.requiredAtStage !== undefined && <Badge tone={d.requiredAtStage === 0 ? 'bad' : 'neutral'}>Required by stage {d.requiredAtStage + 1}</Badge>}</span>}>
                  <Input id={`np-cf-${d.key}`} value={f.custom[d.key] ?? ''} onChange={(e) => set('custom', { ...f.custom, [d.key]: e.target.value })} />
                </Field>
              ))}
            </div>
          )}
          {step === 1 && (
            <div className="flex flex-col gap-4">
              <div className="grid grid-cols-1 gap-3 md:grid-cols-3">
                <Field label="Asset type"><Select id="np-asset" value={f.asset} onChange={(e) => { const a = e.target.value as AssetType; set('asset', a); set('method', ASSETS.find((x) => x.id === a)!.methods[0]) }} options={ASSETS.map((a) => ({ value: a.id, label: a.label }))} /></Field>
                <Field label="Work method"><Select id="np-method" value={f.method} onChange={(e) => set('method', e.target.value as WorkMethod)} options={ASSETS.find((a) => a.id === f.asset)!.methods.map((m) => ({ value: m, label: METHOD_LABEL[m] }))} /></Field>
                <Field label={f.asset === 'compressor-station' ? 'Facility ID' : 'Line ID'}><Input id="np-line" value={f.lineId} onChange={(e) => set('lineId', e.target.value)} placeholder={f.asset === 'compressor-station' ? 'STA-LAS' : '7100'} /></Field>
                {f.asset !== 'compressor-station' && <Field label="Diameter (in)"><Input id="np-dia" type="number" value={f.diameterIn} onChange={(e) => set('diameterIn', Number(e.target.value))} /></Field>}
                {f.asset !== 'compressor-station' && f.asset !== 'valve-site' && <Field label="Length (ft)" hint={`${(f.lengthFt / 5280).toFixed(2)} miles`}><Input id="np-len" type="number" value={f.lengthFt} onChange={(e) => set('lengthFt', Number(e.target.value))} /></Field>}
                {f.asset === 'valve-site' && <Field label="Valve sites"><Input id="np-sites" type="number" min={1} value={f.sites} onChange={(e) => set('sites', Math.max(1, Number(e.target.value)))} /></Field>}
                <Field label="MAOP (psig)"><Input id="np-maop" type="number" value={f.maopPsig} onChange={(e) => set('maopPsig', Number(e.target.value))} /></Field>
                <Field label="Material"><Input id="np-mat" value={f.material} onChange={(e) => set('material', e.target.value)} /></Field>
                <Field label="Vintage"><Input id="np-vin" type="number" value={f.vintage} onChange={(e) => set('vintage', Number(e.target.value))} /></Field>
                <Field label="Records status"><Select id="np-rec" value={f.recordsStatus} onChange={(e) => set('recordsStatus', e.target.value as Form['recordsStatus'])} options={[{ value: 'complete', label: 'Complete (TVC)' }, { value: 'gaps', label: 'Gaps identified' }, { value: 'unknown', label: 'Unknown' }]} /></Field>
              </div>
              <div className="grid grid-cols-1 gap-4 md:grid-cols-[minmax(0,1fr)_minmax(0,1.1fr)]">
                <div className="grid grid-cols-2 gap-3">
                  <Field label="District"><Select id="np-district" value={f.district} onChange={(e) => { const dn = e.target.value; const b = DISTRICTS[dn].box; setF((x) => ({ ...x, district: dn, city: DISTRICTS[dn].cities[0], x: Math.round((b[0] + b[2]) / 2), y: Math.round((b[1] + b[3]) / 2) })) }} options={Object.keys(DISTRICTS)} /></Field>
                  <Field label="City"><Input id="np-city" list="np-cities" value={f.city} onChange={(e) => set('city', e.target.value)} /><datalist id="np-cities">{DISTRICTS[f.district].cities.map((c) => <option key={c} value={c} />)}</datalist></Field>
                  <Field label="Terrain"><Select id="np-terrain" value={f.terrain} onChange={(e) => { const t = e.target.value as Terrain; setF((x) => ({ ...x, terrain: t, classLocation: t === 'urban' ? 3 : t === 'suburban' ? 2 : 1 })) }} options={['urban', 'suburban', 'rural']} /></Field>
                  <Field label="Class location"><Select id="np-class" value={String(f.classLocation)} onChange={(e) => set('classLocation', Number(e.target.value) as Form['classLocation'])} options={['1', '2', '3', '4'].map((c) => ({ value: c, label: `Class ${c}` }))} /></Field>
                  <div className="col-span-2 flex flex-col gap-2">
                    <Checkbox checked={f.hca} onChange={(v) => set('hca', v)} label="In a high consequence area (HCA)" />
                    <Checkbox checked={f.dac} onChange={(v) => set('dac', v)} label="Serves a disadvantaged community (DAC)" />
                  </div>
                </div>
                <div>
                  <div className="mb-1 text-[12px] font-semibold text-muted">Map location <span className="font-normal text-faint">— click to place the pin</span></div>
                  <PlacePin x={f.x} y={f.y} onPlace={(x, y) => { const dn = Object.entries(DISTRICTS).find(([, v]) => x >= v.box[0] - 20 && x <= v.box[2] + 20 && y >= v.box[1] - 20 && y <= v.box[3] + 20)?.[0]; setF((s0) => ({ ...s0, x, y, district: dn ?? s0.district })) }} />
                </div>
              </div>
            </div>
          )}
          {step === 2 && (
            <div className="flex flex-col gap-3">
              <p className="text-[13px] text-muted">Team positions drive workflow routing (for example, the controls review on a change request goes to the project's controls analyst) and deliverable ownership.</p>
              <div className="grid grid-cols-1 gap-3 md:grid-cols-2">
                {POSITIONS.map((x) => <Field key={x.pos} label={x.label}><PersonSelect id={`np-team-${x.pos}`} role={x.role} includeEmpty value={f.team[x.pos]} onChange={(v) => set('team', { ...f.team, [x.pos]: v || undefined })} /></Field>)}
              </div>
              <div>
                <div className="eyebrow mb-1.5">Stage 1 deliverables and owners</div>
                <div className="scroll-x"><table className="tbl"><thead><tr><th>Deliverable</th><th>Owner</th><th>Reviewer</th></tr></thead><tbody>
                  {defs.map((d) => <tr key={d.id}><td>{d.name}</td><td>{s.people.find((p) => p.id === ownerForRole(s, { ...draft }, d.ownerRole))?.name ?? '—'}</td><td className="text-muted">{s.people.find((p) => p.id === ownerForRole(s, { ...draft }, d.reviewerRole))?.name ?? '—'}</td></tr>)}
                </tbody></table></div>
              </div>
            </div>
          )}
          {step === 3 && (
            <div className="flex flex-col gap-4">
              <div>
                <div className="flex flex-wrap items-center gap-2 text-xs text-muted"><ProgramTag programId={f.programId} /><Mono className="font-semibold text-ink">{code}</Mono><span>{f.city}, {f.district} district</span></div>
                <h3 className="mt-1 font-display text-[19px] font-bold">{f.name || 'Untitled project'}</h3>
                <p className="mt-1 text-[13px]">{draft.scope.summary}</p>
              </div>
              <KV cols={3} items={[
                ['Driver', f.driver], ['Citation', f.citation], ['Proceeding', f.proceeding],
                ['Asset', `${ASSETS.find((a) => a.id === f.asset)!.label} · ${METHOD_LABEL[f.method]}`], ['Line / facility', f.lineId || '—'], ['MAOP', `${f.maopPsig} psig`],
                ['Location', `Class ${f.classLocation} · ${f.terrain}${f.hca ? ' · HCA' : ''}${f.dac ? ' · DAC' : ''}`], ['Sponsor', s.people.find((p) => p.id === f.team.sponsor)?.name], ['Project manager', s.people.find((p) => p.id === f.team.pm)?.name],
              ]} />
              <div className="rounded-md border border-line">
                <div className="flex flex-wrap items-center justify-between gap-2 border-b border-line px-3 py-2">
                  <Checkbox checked={f.buildEstimate} onChange={(v) => set('buildEstimate', v)} label={<span className="font-semibold">Build a draft Class 5 rate-table estimate</span>} />
                  <span className="flex items-center gap-2"><ClassBadge cls={estPreview.cls} /><span className="font-display text-[17px] font-bold tabular-nums">{money(estPreview.totals.total)}</span></span>
                </div>
                {f.buildEstimate && (
                  <div className="px-3 py-2">
                    {lines.length === 0 ? <Callout tone="warn">No library rates match this asset and work method. The estimator will build the estimate by hand.</Callout> : (
                      <>
                        <RangeBar low={estPreview.totals.low} point={estPreview.totals.total} high={estPreview.totals.high} />
                        <div className="mt-1 flex justify-between text-[11px] tabular-nums text-muted"><span>Low {money(estPreview.totals.low)} ({Math.round(CLASS_RANGES[estPreview.cls].low * 100)}%)</span><span>High {money(estPreview.totals.high)} (+{Math.round(CLASS_RANGES[estPreview.cls].high * 100)}%)</span></div>
                        <div className="scroll-x mt-2"><table className="tbl"><thead><tr><th>Library line</th><th className="num">Qty</th><th className="num">Unit P50</th><th className="num">Total</th></tr></thead><tbody>
                          {lines.map((l) => <tr key={l.id}><td>{l.description}</td><td className="num">{l.qty.toLocaleString()} {l.unit}</td><td className="num">{money(l.unitCost, 'full')}</td><td className="num">{money(Math.round(l.qty * l.unitCost))}</td></tr>)}
                          <tr><td className="text-muted">Loaders {estPreview.loaders.map((x) => x.pct).reduce((a, b) => a + b, 0).toFixed(1)}% · escalation {estPreview.escalationPct}% · contingency 30%</td><td /><td /><td className="num">{money(estPreview.totals.loaders + estPreview.totals.escalation + estPreview.totals.contingency)}</td></tr>
                        </tbody></table></div>
                      </>
                    )}
                  </div>
                )}
              </div>
              <Checkbox checked={f.startWorkflow} onChange={(v) => set('startWorkflow', v)} label={<span><b>Submit for program screening</b> — starts the Project initiation request workflow (program screening → portfolio planning → sponsor approval).</span>} />
              <Callout tone="info">The project opens at <b>{tpl.stages[0].name}</b> with {defs.length} gate deliverables. Its charter is marked in progress with this record as the first draft. Created by {me.name}; every field is recorded in the audit log.</Callout>
            </div>
          )}
          {problems[step].length > 0 && <div className="mt-4"><Callout tone="warn" title="Before you continue">{problems[step].map((p) => <div key={p}>• {p}</div>)}</Callout></div>}
          <div className="mt-4 flex flex-wrap justify-between gap-2 border-t border-line pt-3">
            <Button variant="ghost" onClick={() => (step === 0 ? go('#projects') : setStep(step - 1))}>{step === 0 ? 'Cancel' : 'Back'}</Button>
            {step < 3 ? <Button variant="primary" disabled={!canNext} onClick={() => setStep(step + 1)}>Continue</Button> : <Button variant="primary" disabled={problems.some((p) => p.length)} onClick={create}>Create {code}</Button>}
          </div>
        </Panel>

        <aside className="flex flex-col gap-3">
          <Panel title="What happens next">
            <ol className="flex flex-col gap-2.5 text-[13px]">
              {tpl.stages.map((st, i) => (
                <li key={st.id} className="flex gap-2">
                  <span className={cx('mt-0.5 flex h-5 w-5 shrink-0 items-center justify-center rounded-full text-[11px] font-bold', i === 0 ? 'bg-marker text-markerink' : 'bg-sunken text-muted')}>{i + 1}</span>
                  <span className="min-w-0"><b>{st.name}</b><span className="block text-xs text-muted">{st.gate.name}{st.gate.requiredEstimateClass ? ` · needs Class ${st.gate.requiredEstimateClass}` : ''}{st.gate.budgetEvent ? ` · sets ${st.gate.budgetEvent} budget` : ''}</span></span>
                </li>
              ))}
            </ol>
          </Panel>
          <Panel title="Planned milestones">
            <ul className="flex flex-col gap-1 text-[13px]">
              {draft.milestones.map((m) => <li key={m.id} className="flex justify-between gap-2"><span className="truncate">{m.name}</span><span className="shrink-0 tabular-nums text-muted">{m.baseline}</span></li>)}
            </ul>
            <p className="mt-2 text-[11px] text-faint">Indicative dates from typical stage durations; the scheduler sets the baseline at option selection.</p>
          </Panel>
        </aside>
      </div>
    </div>
  )
}

function PlacePin({ x, y, onPlace }: { x: number; y: number; onPlace: (x: number, y: number) => void }) {
  return (
    <svg viewBox="0 0 1000 600" className="w-full cursor-crosshair rounded-md border border-line bg-sunken/40" onClick={(e) => {
      const r = (e.currentTarget as SVGSVGElement).getBoundingClientRect()
      onPlace(Math.round(((e.clientX - r.left) / r.width) * 1000), Math.round(((e.clientY - r.top) / r.height) * 600))
    }}>
      {Object.entries(DISTRICTS).map(([n, v]) => (
        <g key={n}>
          <rect x={v.box[0] - 20} y={v.box[1] - 20} width={v.box[2] - v.box[0] + 40} height={v.box[3] - v.box[1] + 40} rx={40} fill="rgb(var(--surface))" stroke="rgb(var(--line))" strokeWidth={2} />
          <text x={(v.box[0] + v.box[2]) / 2} y={v.box[1] + 4} textAnchor="middle" fontSize={22} fontWeight={700} fill="rgb(var(--faint))" letterSpacing="0.12em">{n.toUpperCase()}</text>
        </g>
      ))}
      <path d="M30 420 C 140 410, 220 395, 300 375 S 470 352, 560 320 S 760 268, 980 250" fill="none" stroke="rgb(var(--steel))" strokeOpacity={0.3} strokeWidth={5} />
      <g transform={`translate(${x} ${y})`}><circle r={18} fill="rgb(var(--marker))" fillOpacity={0.35} stroke="rgb(var(--marker))" strokeWidth={3} /><circle r={5} fill="rgb(var(--ink))" /></g>
    </svg>
  )
}

function buildProject(f: Form, code: string, asOf: string, small: boolean): Project {
  const isStation = f.asset === 'compressor-station', isValve = f.asset === 'valve-site'
  const miles = +(f.lengthFt / 5280).toFixed(1)
  const lengthFt = isStation || isValve ? 0 : f.lengthFt
  const quantities = isValve
    ? [{ label: 'Valve sites automated', qty: f.sites, unit: 'ea' }]
    : isStation
      ? [{ label: 'Compressor units', qty: f.method === 'compressor-replacement' ? 4 : 2, unit: 'ea' }, { label: 'Station capacity', qty: f.method === 'compressor-replacement' ? 22000 : 9000, unit: 'hp' }]
      : f.method === 'hydrotest'
        ? [{ label: 'Pipe tested', qty: lengthFt, unit: 'ft' }, { label: 'Test sections', qty: Math.max(2, Math.round(lengthFt / 4200)), unit: 'ea' }]
        : [{ label: 'Pipe installed', qty: lengthFt, unit: 'ft' }, { label: 'Tie-ins', qty: 2, unit: 'ea' }]
  const summary = isValve
    ? `Install remote-control valves with SCADA at ${f.sites} mainline valve site(s) on Line ${f.lineId || '—'} to meet automatic shutoff requirements.`
    : isStation
      ? `${f.method === 'station-rebuild' ? 'Rebuild' : 'Replace compression at'} ${f.city} station with electric-drive units, new substation and controls; decommission legacy units.`
      : f.method === 'hydrotest'
        ? `Pressure test ${miles} miles of ${f.diameterIn}-inch Line ${f.lineId || '—'} to establish MAOP records; replace short sections that cannot be tested.`
        : `Replace ${miles} miles of ${f.diameterIn}-inch ${f.asset === 'distribution-main' ? 'distribution main' : `Line ${f.lineId || '—'}`} through ${f.city} by ${f.method === 'replace-hdd' ? 'horizontal directional drill' : 'open trench'}; abandon existing pipe in place.`
  const durs = small ? [150, 240, 300] : [160, 240, 320, isStation ? 720 : 300, 150]
  const names = small ? ['Plan approved', 'Construction authorized', 'In service & closed'] : ['Business case approved', 'Option selected', 'Construction authorized', 'In service', 'Closeout complete']
  let acc = asOf
  const milestones: Milestone[] = names.map((name, i) => { acc = addDays(acc, durs[i]); return { id: uid('ms'), name, baseline: acc, forecast: acc } })
  if (!small) { const auth = milestones[2].baseline; milestones.splice(3, 0, { id: uid('ms'), name: 'Construction start', baseline: addDays(auth, 30), forecast: addDays(auth, 30) }) }
  const n = code.slice(4)
  const progCode = f.programId.slice(4).toUpperCase()
  return {
    id: `prj-${n}`, code, info: { problemStatement: f.need ?? '' }, name: f.name.trim() || 'Untitled project', programId: f.programId, sizeTier: 'M', driver: f.driver, regulatoryCitation: f.citation, fundingProceeding: f.proceeding,
    location: { district: f.district, city: f.city, county: DISTRICTS[f.district].county, x: f.x, y: f.y, classLocation: f.classLocation, hca: f.hca, dac: f.dac, terrain: f.terrain },
    asset: { type: f.asset, lineId: f.lineId || (isStation ? 'STA-' + f.city.slice(0, 3).toUpperCase() : 'DIST'), diameterIn: isStation ? 0 : f.diameterIn, lengthFt, maopPsig: f.maopPsig, material: f.material, vintage: f.vintage, recordsStatus: f.recordsStatus },
    workMethod: f.method, team: { ...f.team }, stageIndex: 0, status: 'active',
    health: { scope: 'green', schedule: 'green', cost: 'green', risk: 'green', safety: 'green' },
    scope: {
      summary,
      inScope: isStation ? ['Compressor units and motors', 'Substation and interconnect', 'Station piping and controls', 'Demolition of legacy units', 'Air permit compliance']
        : isValve ? ['Valve and actuator installation', 'SCADA RTU and communications', 'Site civil work', 'Commissioning with gas control']
          : ['Engineering, survey and design', 'Land rights and permits', 'Construction and inspection', 'Tie-ins and gas handling', 'Restoration'],
      outOfScope: ['Customer meter work', 'Unrelated leak repairs'],
      quantities,
      assumptions: ['Outage windows available October–April', 'Work hours 7 AM–6 PM weekdays in residential areas'],
    },
    milestones, dates: { initiated: asOf }, erpWbs: `C.${progCode}.${n}`, scheduleId: `P6-${n}`, gisIds: isStation ? [`FAC-${n}`] : [`SEG-${f.lineId || 'NEW'}-001`],
    permitsNeeded: { ceqa: isStation || f.terrain === 'rural', encroachment: !isStation, hydrotestDischarge: f.method === 'hydrotest' },
    customFields: Object.fromEntries(Object.entries(f.custom).filter(([, v]) => v.trim())),
    lastStatusNote: 'Initiated; need submitted for program screening.',
  }
}
