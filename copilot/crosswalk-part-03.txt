// Default process catalog (Module 7). Each is a published, versioned WorkflowDef a client edits in the designer.
import type { Assignee, FormField, RecordType, RoleId, WfEdge, WfNode, WorkflowDef } from '../types'

type Step = { id: string; type: WfNode['type']; label: string; a?: Assignee; sla?: number; rule?: 'any' | 'all'; desc?: string; sys?: WfNode['systemAction'] }

const role = (roleId: RoleId): Assignee => ({ kind: 'role', roleId })
const team = (position: Extract<Assignee, { kind: 'team-position' }>['position']): Assignee => ({ kind: 'team-position', position })
const requester: Assignee = { kind: 'requester' }
const doa: Assignee = { kind: 'doa', field: 'amount' }

let eid = 0
const E = (from: string, to: string, extra: Partial<WfEdge> = {}): WfEdge => ({ id: `e${++eid}`, from, to, ...extra })

/** A straight-line process: start → steps → end. Approvals/reviews get a 'return' edge to the first form step. */
function linear(
  id: string, key: string, name: string, recordType: RecordType, ownerRole: RoleId, description: string, purpose: string, fields: FormField[], steps: Step[],
): WorkflowDef {
  const all: Step[] = [{ id: 'start', type: 'start', label: 'Start' }, ...steps, { id: 'end', type: 'end', label: 'Complete' }]
  const nodes: WfNode[] = all.map((s, i) => ({ id: s.id, type: s.type, label: s.label, x: 60 + i * 170, y: 160, assignee: s.a, slaDays: s.sla, rule: s.rule, description: s.desc, systemAction: s.sys }))
  const edges: WfEdge[] = []
  for (let i = 0; i < all.length - 1; i++) edges.push(E(all[i].id, all[i + 1].id, { outcome: all[i].type === 'approval' || all[i].type === 'review' ? 'approve' : undefined }))
  const firstForm = steps.find((s) => s.type === 'form')
  if (firstForm) for (const s of steps) if ((s.type === 'approval' || s.type === 'review') && s.id !== firstForm.id) edges.push(E(s.id, firstForm.id, { outcome: 'return', label: 'Return for revision' }))
  return { id, key, name, recordType, version: 1, status: 'published', effective: '2026-01-01', description, purpose, nodes, edges, fields, ownerRole }
}

    switch (route.view) {
      case 'project': return <ProjectView id={route.id!} tab={route.tab} />
      case 'workflow': return <Designer id={route.id} />
      case 'projects': return <Projects />
      case 'new-project': return <NewProject />
      case 'gates': return <Gates />
      case 'estimating': return <Estimating />
      case 'regulatory': return <Regulatory />
      case 'risk': return <RiskPortfolio />

      case 'field': return features?.fieldTailboard ? <FieldMode /> : <PhaseTwo name="Field tailboard" />
      case 'inbox': return <Inbox />
      case 'workflows': return <Catalog />
      case 'admin': return <Admin />

      case 'audit': return <Audit />
      case 'forecast': return <ForecastCycle />
      case 'map': return <MapPage />
      case 'actuals': return <Actuals />

        if (route.view.startsWith('admin-')) return <Admin id={route.view.slice(6)} />

  if (sync.mode === 'connecting') return <Connecting />
  const activeNav = route.view === 'project' ? 'projects' : route.view === 'workflow' ? 'workflows' : route.view === 'new-project' ? 'projects' : isMessagesView(route.view) ? 'messages' : route.view.startsWith('admin-') ? 'admin' : route.view
  return (
    <div className="flex h-full">
      <aside className={cx('fixed inset-y-0 left-0 z-40 flex w-[236px] flex-col bg-steel text-steelink transition-transform lg:static lg:translate-x-0', navOpen ? 'translate-x-0' : '-translate-x-full')} style={{ paddingTop: 'env(safe-area-inset-top)', paddingLeft: 'env(safe-area-inset-left)' }}>

const isAway = (p: Person | undefined, asOf: string) => !!p?.outOfOffice && asOf >= p.outOfOffice.from && asOf <= p.outOfOffice.to

/** Avatar menu: who you are, your out-of-office status and who you are covering for. */
function UserMenu({ me }: { me: Person }) {
  const [open, setOpen] = useState(false)
  const [ooo, setOoo] = useState(false)
  const asOf = useStore((s) => s.settings.asOf)
  const people = useStore((s) => s.people)
  const away = isAway(me, asOf)
  const oo = me.outOfOffice
  const scheduled = !!oo && oo.to >= asOf && !away
  const covering = people.filter((p) => p.outOfOffice?.delegateId === me.id && isAway(p, asOf))
  useEffect(() => setOpen(false), [me.id])
  const end = () => {
    mutate((d) => { const p = d.people.find((x) => x.id === me.id); if (p) delete p.outOfOffice }, { entity: 'person', entityId: me.id, action: 'delegate-end', summary: 'Ended out-of-office delegation' })
    toast('Out of office ended — steps route to you again')
    setOpen(false)
  }
  return (
    <div className="relative">
      <button id="user-menu" onClick={() => setOpen(!open)} className="relative flex items-center gap-1.5 rounded p-1 hover:bg-sunken" aria-haspopup="menu" aria-expanded={open} aria-label={`Account menu for ${me.name}${away ? ' (out of office)' : ''}`}>
        <Avatar name={me.name} size={24} />
        {away && <span className="hidden rounded-sm bg-warn/15 px-1.5 py-px text-[10.5px] font-bold uppercase tracking-wide text-warn sm:inline">Out of office</span>}
        {(away || covering.length > 0) && <span className={cx('absolute left-5 top-0.5 h-2 w-2 rounded-full border border-surface', away ? 'bg-warn' : 'bg-info')} />}
      </button>
      {open && (
        <>
          <div className="fixed inset-0 z-20" onClick={() => setOpen(false)} />
          <div role="menu" className="absolute right-0 top-10 z-30 w-[300px] max-w-[calc(100vw-24px)] rounded-md border border-line bg-surface shadow-xl">
            <div className="flex items-center gap-2.5 border-b border-line px-3 py-2.5">
              <Avatar name={me.name} size={32} />
              <div className="min-w-0"><div className="truncate text-[13px] font-semibold">{me.name}</div><div className="truncate text-[11.5px] text-muted">{me.title}{me.company ? ` · ${me.company}` : ''}</div><div className="truncate text-[11px] text-faint">{me.email}</div></div>
            </div>
            <div className="border-b border-line px-3 py-2.5 text-[12.5px]">
              <div className="eyebrow mb-1">Status</div>
              {away ? <div><Badge tone="warn">Out of office</Badge> <span className="text-muted">until {fmtDate(oo!.to)}</span><div className="mt-1 text-muted">{personName(getState(), oo!.delegateId)} is acting for you on workflow steps.</div></div>
                : scheduled ? <div><Badge tone="info">Scheduled</Badge> <span className="text-muted">away {fmtDate(oo!.from)} – {fmtDate(oo!.to)} · {personName(getState(), oo!.delegateId)} covering</span></div>
                  : <div className="text-muted"><span className="mr-1.5 inline-block h-2 w-2 rounded-full bg-good" />Available — steps route to you</div>}
              <div className="mt-2 flex flex-wrap gap-2">
                <Button size="sm" variant={away || scheduled ? 'secondary' : 'primary'} onClick={() => { setOoo(true); setOpen(false) }}>{away || scheduled ? 'Edit out of office' : 'Set out of office…'}</Button>
                {(away || scheduled) && <Button size="sm" variant="ghost" onClick={end}>End</Button>}
              </div>
            </div>
            {covering.length > 0 && (
              <div className="border-b border-line px-3 py-2.5 text-[12.5px]">
                <div className="eyebrow mb-1">Acting for</div>
                {covering.map((p) => <div key={p.id} className="flex items-center gap-2 py-0.5"><Avatar name={p.name} size={18} /><span className="min-w-0 flex-1 truncate">{p.name}</span><span className="text-[11px] text-faint">to {fmtDate(p.outOfOffice!.to)}</span></div>)}
              </div>
            )}
            <div className="flex flex-col py-1 text-[13px]">
              <a href="#inbox" onClick={() => setOpen(false)} className="px-3 py-1.5 hover:bg-sunken">My work</a>
              <a href="#messages" onClick={() => setOpen(false)} className="px-3 py-1.5 hover:bg-sunken">Messages</a>
            </div>
          </div>
        </>
      )}
      <OutOfOfficeModal open={ooo} onClose={() => setOoo(false)} me={me} />
    </div>
  )
}

function OutOfOfficeModal({ open, onClose, me }: { open: boolean; onClose: () => void; me: Person }) {
  const asOf = useStore((s) => s.settings.asOf)
  const people = useStore((s) => s.people)
  const openSteps = useStore((s) => tasksFor(s, me.id).filter((t) => !t.actingFor).length)
  const [from, setFrom] = useState(asOf)
  const [to, setTo] = useState(addDays(asOf, 7))
  const [delegate, setDelegate] = useState('')
  useEffect(() => {
    if (!open) return
    setFrom(me.outOfOffice?.from ?? asOf); setTo(me.outOfOffice?.to ?? addDays(asOf, 7)); setDelegate(me.outOfOffice?.delegateId ?? '')
  }, [open, me.id]) // eslint-disable-line react-hooks/exhaustive-deps
  const candidates = people.filter((p) => p.id !== me.id && (!!me.company === !!p.company))
  const sameRole = candidates.filter((p) => p.roles.some((r) => me.roles.includes(r) && r !== 'admin'))
  const dp = people.find((p) => p.id === delegate)
  const loop = dp?.outOfOffice && dp.outOfOffice.delegateId === me.id && dp.outOfOffice.from <= to && dp.outOfOffice.to >= from
  const dpAway = dp?.outOfOffice && dp.outOfOffice.from <= to && dp.outOfOffice.to >= from
  const bad = !delegate || !from || !to || to < from || !!loop
  const save = () => {
    if (bad) return
    mutate((d) => {
      const p = d.people.find((x) => x.id === me.id)!
      p.outOfOffice = { from, to, delegateId: delegate }
      d.notifications.push({ id: uid('ntf'), at: nowISO(), toId: delegate, text: `${me.name} is out of office ${fmtDate(from)} – ${fmtDate(to)} and named you as delegate. Their workflow steps appear in your My work.`, link: '#inbox', read: false, kind: 'info' })
    }, { entity: 'person', entityId: me.id, action: 'delegate', summary: `Out of office ${from} to ${to}; delegate ${dp?.name}` })
    toast(from <= asOf && to >= asOf ? `Out of office — ${dp?.name} is now acting for you` : `Out of office scheduled from ${fmtDate(from)}`)
    onClose()
  }
  return (
    <Modal open={open} onClose={onClose} title="Out of office"
      footer={<><Button variant="ghost" onClick={onClose}>Cancel</Button><Button variant="primary" disabled={bad} onClick={save}>Save</Button></>}>
      <div className="flex flex-col gap-3">
        <p className="text-[13px] text-muted">While you are away, new workflow steps route to your delegate, and the <b className="text-ink">{openSteps}</b> step{openSteps === 1 ? '' : 's'} already waiting on you show in their My work as “acting for {me.name}”. Every action they take is signed by them and recorded as on your behalf.</p>
        <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
          <Field label="From"><Input id="ooo-from" type="date" value={from} onChange={(e) => setFrom(e.target.value)} /></Field>
          <Field label="To"><Input id="ooo-to" type="date" value={to} min={from} onChange={(e) => setTo(e.target.value)} /></Field>
        </div>
        <Field label="Delegate" hint={sameRole.length ? 'People who share one of your roles are listed first.' : undefined}>
          <Select id="ooo-delegate" value={delegate} onChange={(e) => setDelegate(e.target.value)} options={[{ value: '', label: '— Select —' }, ...sameRole.map((p) => ({ value: p.id, label: `${p.name} · ${p.title} (same role)` })), ...candidates.filter((p) => !sameRole.includes(p)).map((p) => ({ value: p.id, label: `${p.name} · ${p.title}` }))]} />
        </Field>
        {to < from && <Callout tone="bad">The end date is before the start date.</Callout>}
        {loop && <Callout tone="bad">{dp!.name} has named you as their delegate for overlapping dates. Pick someone else.</Callout>}
        {!loop && dpAway && <Callout tone="warn">{dp!.name} is also out of office for part of this time ({fmtDate(dp!.outOfOffice!.from)} – {fmtDate(dp!.outOfOffice!.to)}).</Callout>}
        {dp && dp.dollarAuthority < me.dollarAuthority && <Callout tone="warn">{dp.name}'s dollar authority is lower than yours. Steps routed by the DOA matrix still go to someone with enough authority.</Callout>}
        <p className="text-[11.5px] text-faint">Email auto-reply in Outlook is set separately until the mail integration is connected.</p>
      </div>
    </Modal>
  )
}

const TEXT = 'rgb(var(--muted))'

function niceTicks(min: number, max: number, n = 4): number[] {
  if (max === min) max = min + 1
  const span = max - min
  const step0 = span / n
  const mag = Math.pow(10, Math.floor(Math.log10(step0)))
  const step = [1, 2, 2.5, 5, 10].map((m) => m * mag).find((s) => span / s <= n) ?? mag * 10
  const lo = Math.floor(min / step) * step
  const out: number[] = []
  for (let v = lo; v <= max + step * 0.5; v += step) out.push(Math.round(v * 1e6) / 1e6)
  return out
}

/** Line chart over categorical x labels (months, dates). */
export function LineChart({ labels, series, height = 220, format = money, markers, yMin }: { labels: string[]; series: LineSeries[]; height?: number; format?: (n: number) => string; markers?: { index: number; label: string }[]; yMin?: number }) {
  const [hover, setHover] = useState<number | null>(null)
  const W = 720, H = height, L = 64, Rm = 16, T = 14, B = 30
  const vals = series.flatMap((s) => s.values.filter((v): v is number => v != null))
  const max = Math.max(...vals, 0), min = Math.min(yMin ?? 0, ...vals)
  const ticks = niceTicks(min, max)
  const ymin = ticks[0], ymax = ticks[ticks.length - 1]
  const X = (i: number) => L + (labels.length <= 1 ? 0 : (i / (labels.length - 1)) * (W - L - Rm))
  const Y = (v: number) => T + (1 - (v - ymin) / (ymax - ymin || 1)) * (H - T - B)
  const every = Math.max(1, Math.ceil(labels.length / 8))
  return (
    <div className="relative">
      <svg viewBox={`0 0 ${W} ${H}`} className="w-full" role="img" onMouseLeave={() => setHover(null)}>
        {ticks.map((t) => <g key={t}><line x1={L} x2={W - Rm} y1={Y(t)} y2={Y(t)} stroke={AXIS} strokeWidth={1} /><text x={L - 8} y={Y(t) + 4} textAnchor="end" fontSize={11} fill={TEXT}>{format(t)}</text></g>)}
        {labels.map((l, i) => i % every === 0 && <text key={i} x={X(i)} y={H - 10} textAnchor="middle" fontSize={11} fill={TEXT}>{l}</text>)}
        {markers?.map((m) => <g key={m.index}><line x1={X(m.index)} x2={X(m.index)} y1={T} y2={H - B} stroke="rgb(var(--ink))" strokeDasharray="3 3" strokeWidth={1} /><text x={X(m.index) + 4} y={T + 10} fontSize={11} fill="rgb(var(--ink))">{m.label}</text></g>)}
        {series.map((s, si) => {

          let d = ''

          const valid = pts.filter(Boolean) as (readonly [number, number])[]
          const area = s.area && valid.length ? `M${valid[0][0]} ${Y(Math.max(ymin, 0))}L` + valid.map((p) => `${p[0]} ${p[1]}`).join('L') + `L${valid[valid.length - 1][0]} ${Y(Math.max(ymin, 0))}Z` : ''
          const last = valid[valid.length - 1]
          return <g key={s.name}>{area && <path d={area} fill={c} opacity={0.1} />}<path d={d} fill="none" stroke={c} strokeWidth={2} strokeDasharray={s.dashed ? '5 4' : undefined} strokeLinejoin="round" />{last && <circle cx={last[0]} cy={last[1]} r={3.5} fill={c} />}</g>
        })}

'2px dashed' : undefined }} />{s.name}{hover != null && s.values[hover] != null && <b className="tabular-nums text-ink">{format(s.values[hover] as number)}</b>}</span>)}
        {hover != null && <span className="text-faint">{labels[hover]}</span>}
      </div>
    </div>
  )
}

export function BarChart({ labels, series, height = 220, format = money, stacked, highlight }: { labels: string[]; series: { name: string; values: number[]; color?: string }[]; height?: number; format?: (n: number) => string; stacked?: boolean; highlight?: number }) {

      </svg>

/** Waterfall bridge from one total to another (estimate or baseline bridge). */
export function Waterfall({ start, end, steps, startLabel = 'From', endLabel = 'To', height = 240 }: { start: number; end: number; steps: { label: string; delta: number }[]; startLabel?: string; endLabel?: string; height?: number }) {
  const W = 720, H = height, L = 64, Rm = 12, T = 18, B = 46
  const bars: { label: string; y0: number; y1: number; kind: 'total' | 'up' | 'down' }[] = [{ label: startLabel, y0: 0, y1: start, kind: 'total' }]
  let acc = start
  for (const s of steps) { bars.push({ label: s.label, y0: acc, y1: acc + s.delta, kind: s.delta >= 0 ? 'up' : 'down' }); acc += s.delta }
  bars.push({ label: endLabel, y0: 0, y1: end, kind: 'total' })
  const hi = Math.max(...bars.map((b) => Math.max(b.y0, b.y1)))
  const lo = Math.min(0, ...bars.map((b) => Math.min(b.y0, b.y1)))
  const ticks = niceTicks(lo, hi)
  const ymin = ticks[0], ymax = ticks[ticks.length - 1]
  const Y = (v: number) => T + (1 - (v - ymin) / (ymax - ymin)) * (H - T - B)
  const bw = (W - L - Rm) / bars.length
  return (
    <svg viewBox={`0 0 ${W} ${H}`} className="w-full" role="img">
      {ticks.map((t) => <g key={t}><line x1={L} x2={W - Rm} y1={Y(t)} y2={Y(t)} stroke={AXIS} /><text x={L - 8} y={Y(t) + 4} textAnchor="end" fontSize={11} fill={TEXT}>{money(t)}</text></g>)}
      {bars.map((b, i) => {
        const x = L + i * bw + bw * 0.15, w = bw * 0.7
        const top = Y(Math.max(b.y0, b.y1)), h = Math.max(1.5, Math.abs(Y(b.y0) - Y(b.y1)))
        const c = b.kind === 'total' ? 'rgb(var(--ink))' : b.kind === 'up' ? 'rgb(var(--bad))' : 'rgb(var(--good))'
        const delta = b.y1 - b.y0
        return (
          <g key={i}>
            <rect x={x} y={top} width={w} height={h} fill={c} opacity={b.kind === 'total' ? 0.85 : 0.8} rx={1} />
            {i < bars.length - 1 && <line x1={x + w} x2={x + bw} y1={Y(b.y1)} y2={Y(b.y1)} stroke={TEXT} strokeDasharray="2 2" />}
            <text x={x + w / 2} y={top - 4} textAnchor="middle" fontSize={10.5} fill="rgb(var(--ink))">{b.kind === 'total' ? money(b.y1) : (delta > 0 ? '+' : '') + money(delta)}</text>
            <text x={x + w / 2} y={H - B + 14} textAnchor="middle" fontSize={10.5} fill={TEXT}>{b.label.length > 14 ? b.label.slice(0, 13) + '…' : b.label}</text>
          </g>
        )
      })}
    </svg>
  )
}

export function Sparkline({ values, width = 90, height = 24, color = SERIES[0] }: { values: number[]; width?: number; height?: number; color?: string }) {
  if (values.length < 2) return <svg width={width} height={height} />
  const mx = Math.max(...values), mn = Math.min(...values)
  const X = (i: number) => (i / (values.length - 1)) * (width - 4) + 2
  const Y = (v: number) => height - 3 - ((v - mn) / (mx - mn || 1)) * (height - 6)
  const d = values.map((v, i) => `${i ? 'L' : 'M'}${X(i).toFixed(1)} ${Y(v).toFixed(1)}`).join('')
  return <svg width={width} height={height} aria-hidden><path d={d} fill="none" stroke={color} strokeWidth={1.5} /><circle cx={X(values.length - 1)} cy={Y(values[values.length - 1])} r={2.5} fill={color} /></svg>
}

/** Histogram with percentile markers (Monte Carlo). */
export function Histogram({ bins, markers, format = money, height = 200 }: { bins: { x: number; n: number }[]; markers?: { x: number; label: string; color?: string }[]; format?: (n: number) => string; height?: number }) {

  if (!bins.length) return null
  const lo = bins[0].x, hi = bins[bins.length - 1].x
  const mx = Math.max(...bins.map((b) => b.n))
  const X = (v: number) => L + ((v - lo) / (hi - lo || 1)) * (W - L - Rm)
  const bw = (W - L - Rm) / bins.length
  return (
    <svg viewBox={`0 0 ${W} ${H}`} className="w-full" role="img">
      {bins.map((b, i) => { const h = (b.n / mx) * (H - T - B); return <rect key={i} x={L + i * bw} y={H - B - h} width={bw - 1} height={h} fill="rgb(var(--link))" opacity={0.55} /> })}
      <line x1={L} x2={W - Rm} y1={H - B} y2={H - B} stroke={AXIS} />

/** Map common statuses to tones so every module colors them the same way. */
export function statusTone(status: string): Tone {
  const s = status.toLowerCase()
  if (['approved', 'accepted', 'done', 'closed', 'issued', 'acquired', 'verified', 'complete', 'go', 'met', 'received', 'published', 'green', 'compliant', 'awarded', 'active-ok'].includes(s)) return 'good'

  if (['open', 'pending', 'identified', 'not-applied', 'not-started', 'draft', 'rfp', 'requested'].includes(s)) return 'neutral'
  return 'neutral'
}

// ───────────────────────────── Table
export interface Column<T> { key: string; header: React.ReactNode; render: (row: T) => React.ReactNode; num?: boolean; width?: string | number; sort?: (row: T) => number | string; className?: string }
export function Table<T>({ rows, columns, onRowClick, rowKey, empty, maxHeight, dense }: { rows: T[]; columns: Column<T>[]; onRowClick?: (r: T) => void; rowKey: (r: T) => string; empty?: React.ReactNode; maxHeight?: number; dense?: boolean }) {
  const [sort, setSort] = useState<{ key: string; dir: 1 | -1 } | null>(null)
  const col = sort && columns.find((c) => c.key === sort.key)
  const sorted = col?.sort ? rows.slice().sort((a, b) => { const x = col.sort!(a), y = col.sort!(b); return (x < y ? -1 : x > y ? 1 : 0) * sort!.dir }) : rows
  if (!rows.length && empty) return <>{empty}</>
  return (
    <div className="scroll-x" style={maxHeight ? { maxHeight, overflowY: 'auto' } : undefined}>
      <table className={cx('tbl', dense && '[&_td]:py-1')}>
        <thead>
          <tr>
            {columns.map((c) => (
              <th key={c.key} className={cx(c.num && 'num', c.sort && 'cursor-pointer select-none hover:text-ink')} style={{ width: c.width }} onClick={() => c.sort && setSort((s) => (s?.key === c.key ? { key: c.key, dir: (s.dir * -1) as 1 | -1 } : { key: c.key, dir: 1 }))}>
                {c.header}{sort?.key === c.key ? (sort.dir === 1 ? ' ▲' : ' ▼') : ''}
              </th>
            ))}
          </tr>
        </thead>
        <tbody>
          {sorted.map((r) => (
            <tr key={rowKey(r)} className={cx(onRowClick && 'clickable')} onClick={onRowClick ? () => onRowClick(r) : undefined}>
              {columns.map((c) => <td key={c.key} className={cx(c.num && 'num', c.className)}>{c.render(r)}</td>)}
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  )
}

export function GisMap(props: GisMapProps) {
  const el = useRef<HTMLDivElement>(null)
  const st = useRef<{ L?: any; esri?: any; map?: any; base?: any; ext: Map<string, any>; demo?: any; projects?: any; draft?: any; tileErrors: number; tileLoads: number }>({ ext: new Map(), tileErrors: 0, tileLoads: 0 })
  const [ready, setReady] = useState(false)
  const latest = useRef(props)
  latest.current = props

  // create map once
  useEffect(() => {
    let alive = true
    ;(async () => {
      const L = (await import('leaflet')).default
      const esri = await import('esri-leaflet')
      if (!alive || !el.current) return
      const map = L.map(el.current, { zoomControl: true, attributionControl: true, preferCanvas: false, doubleClickZoom: false }).setView(props.center, props.zoom)
      map.on('click', (e: any) => latest.current.onMapClick?.([+e.latlng.lat.toFixed(6), +e.latlng.lng.toFixed(6)]))
      Object.assign(st.current, { L, esri, map })
      setReady(true)
    })()
    return () => { alive = false; st.current.map?.remove(); st.current.map = undefined }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [])

  // basemap
  useEffect(() => {
    const { L, map } = st.current
    if (!ready || !L || !map) return
    st.current.base?.remove()
    const bm = props.external ? BASEMAPS.find((b) => b.id === props.basemap) : undefined
    st.current.tileErrors = 0; st.current.tileLoads = 0

import { create } from 'zustand'
import { cx } from '@/core/util'

interface T { id: number; text: string; tone: 'good' | 'bad' | 'info' }
const useToasts = create<{ items: T[] }>(() => ({ items: [] }))
let n = 0
/** toast('Change request submitted') — short confirmation of what just happened. */
export function toast(text: string, tone: T['tone'] = 'good') {
  const id = ++n
  useToasts.setState((s) => ({ items: [...s.items, { id, text, tone }] }))
  setTimeout(() => useToasts.setState((s) => ({ items: s.items.filter((x) => x.id !== id) })), 3800)
}
export function Toaster() {
  const items = useToasts((s) => s.items)
  return (
    <div className="pointer-events-none fixed bottom-4 right-4 z-[70] flex flex-col gap-2" style={{ bottom: 'max(16px, env(safe-area-inset-bottom))', right: 'max(16px, env(safe-area-inset-right))' }} aria-live="polite">
      {items.map((t) => (
        <div key={t.id} className={cx('fadein pointer-events-auto rounded-md border px-3 py-2 text-[13px] font-semibold shadow-lg', t.tone === 'good' ? 'border-good/40 bg-surface text-ink' : t.tone === 'bad' ? 'border-bad/50 bg-surface text-bad' : 'border-info/40 bg-surface text-ink')}>

  }
  const post = () => {

    mutate((d) => { res = postImport(d, imp.id, me.id) })
    if (!res.ok) { setBlocked(res.reason ?? 'Posting is blocked'); setTab('exceptions'); toast(`Not posted: ${res.reason}`, 'bad'); return }

    toast(`Posted to ${res.periods} project-period${res.periods === 1 ? '' : 's'} — forecast actuals updated`)
  }

  const exportCsv = async () => {
    const rows = txns.map((t) => ({ 'Document': t.docNo, 'Posting date': t.postingDate, Period: t.period, Project: s.projects.find((p) => p.id === t.projectId)?.code ?? '', 'WBS element': t.sapWbs, 'Cost element': t.costElement, 'Cost element name': t.costElementName, Vendor: t.vendor, Text: t.text, Amount: t.amount, Status: TXN_STATUS[t.status].label, Line: lineById(s, t.lineId)?.code ??             <StatusBadge status={IMP_TONE[imp.status]} label={IMP_LABEL[imp.status]} />
            {staged && F.exceptions > 0 && <Badge tone="bad">{F.exceptions} exception{F.exceptions === 1 ? '' : 's'} block posting</Badge>}
            {staged && F.exceptions === 0 && <Badge tone="good">Ready to post</Badge>}
          </div>
          <div className="mt-1 text-[12px] text-muted">
            {imp.source} · {F.periods.map(fmtMonth).join(', ')} · imported {fmtDateTime(imp.importedAt)} by {personName(s, imp.byId)}
            {imp.postedAt && <> · posted {fmtDateTime(imp.postedAt)} by {personName(s, imp.postedById)}</>}
            {imp.note && <> · {imp.note}</>}
          </div>
        </div>

        </div>

        <Tabs className="px-2 pt-1" value={tab} onChange={setTab} tabs={[

    </div>
  )

/** Dense SAP line-item table. With `share`, an extra column shows what each row contributes to that line / function. */
export function TxnTable({ s, txns, share, showProject, showStatus, limit = 200, onRow }: {
  s: AppState; txns: ActualTxn[]; share?: { lineId?: string; fnId?: string }; showProject?: boolean; showStatus?: boolean; limit?: number; onRow?: (t: ActualTxn) => void
}) {
  const [n, setN] = useState(limit)
  const rows = txns.slice(0, n)
  return (
    <div className="scroll-x">
      <table className="tbl [&_td]:py-1">
        <thead>
          <tr>
            <th>Doc no.</th><th>Posted</th>{showProject && <th>Project / WBS</th>}<th>Cost element</th><th>Vendor · text</th>
            <th className="num">Amount</th>{share && <th className="num" title="Amount, or the allocated share when a mapping rule allocates the row">To this {share.lineId ? 'line' : 'function'}</th>}
            {showStatus && <th>Status</th>}<th>Mapping</th>
          </tr>
        </thead>
        <tbody>
          {rows.map((t) => {
            const sh = share ? txnShare(s, t, share) : 0
            const line = lineById(s, t.lineId)
            return (
              <tr key={t.id} className={cx(onRow && 'clickable')} onClick={onRow ? () => onRow(t) : undefined}>
                <td className="whitespace-nowrap font-mono text-[11.5px]">{t.docNo || '—'}</td>
                <td className="whitespace-nowrap text-muted">{fmtDate(t.postingDate)}</td>
                {showProject && <td className="whitespace-nowrap"><span className="font-mono text-[11.5px]">{s.projects.find((p) => p.id === t.projectId)?.code ?? <span className="text-bad">none</span>}</span> <span className="font-mono text-[11px] text-faint">{t.sapWbs}</span></td>}
                <td className="whitespace-nowrap"><span className="font-mono text-[11.5px]">{t.costElement}</span><div className="max-w-[180px] truncate text-[11px] text-faint" title={t.costElementName}>{t.costElementName}</div></td>
                <td className="min-w-[160px] max-w-[280px]"><div className="truncate" title={t.vendor}>{t.vendor || <span className="text-faint">Internal</span>}</div><div className="truncate text-[11px] text-muted" title={t.text}>{t.text}</div></td>
                <td className={cx('num whitespace-nowrap', t.amount < 0 && 'text-bad')}>{$(t.amount)}{t.amount < 0 && <div className="text-[10.5px] text-faint">credit</div>}</td>
                {share && <td className="num whitespace-nowrap font-semibold">{$(sh)}{t.status === 'allocated' && t.amount ? <div className="text-[10.5px] font-normal text-faint">{((sh / t.amount) * 100).toFixed(1)}% share</div> : null}</td>}
                {showStatus && <td><Badge tone={TXN_STATUS[t.status].tone}>{TXN_STATUS[t.status].label}</Badge>{t.manual && <Badge tone="info" className="ml-1">Manual</Badge>}</td>}
                <td className="min-w-[180px] max-w-[300px] text-[11.5px] text-muted">
                  {t.status === 'mapped' && line && <div className="font-semibold text-ink">{line.code} {line.name}</div>}
                  {t.status === 'allocated' && <div className="font-semibold text-ink">Allocated over {Object.keys(t.allocation ?? {}).length || 'the period’s'} lines</div>}
                  <div className="line-clamp-2" title={howMapped(s, t)}>{howMapped(s, t)}</div>
                </td>

    </div>
  )
}

/** The SAP transactions behind one actual cell (a line or a function, one month or a collapsed year). */
export function ActualDrill({ projectId, lineId, fnId, months, cellAmount, onClose }: { projectId: string; lineId?: string; fnId?: string; months: string[]; cellAmount: number; onClose: () => void }) {
  const s = useStore()
  const p = s.projects.find((x) => x.id === projectId)
  const txns = useMemo(() => months.flatMap((m) => txnsFor(s, projectId, { lineId, fnId, period: m })).sort((a, b) => a.postingDate.localeCompare(b.postingDate)), [s, projectId, lineId, fnId, months])
  const share = { lineId, fnId: lineId ? undefined : fnId }
  const total = sum(txns.map((t) => txnShare(s, t, share)))
  const cps = s.costPeriods.filter((c) => c.projectId === projectId && months.includes(c.period))
  const summaryMonths = cps.filter((c) => c.source !== 'sap').map((c) => c.period).sort()
  const line = lineById(s, lineId)
  const fn = s.settings.wbsFunctions.find((f) => f.id === (fnId ?? line?.functionId))
  const what = line ? `${line.code} ${line.name}` : fn ? `${fn.code} ${fn.name}` : 'All lines'
  const when = months.length === 1 ? fmtMonth(months[0]) : `${fmtMonth(months[0])} – ${fmtMonth(months[months.length - 1])}`
  const diff = Math.round(cellAmount - total)
  return (
    <Drawer open onClose={onClose} width={860} title={<span>Actuals · {what}</span>}
      footer={<><a href="#actuals" className="mr-auto self-center text-[12.5px] font-semibold text-link hover:underline">Open SAP actuals</a><Button onClick={onClose}>Close</Button></>}>
      <div className="flex flex-col gap-3">
        <div className="grid grid-cols-2 gap-px overflow-hidden rounded-md border border-line bg-line sm:grid-cols-4">
          <Cell label="Project" value={p?.code ?? '—'} sub={p?.erpWbs} />
          <Cell label="Period" value={when} sub={`${months.length} month${months.length === 1 ? '' : 's'}`} />
          <Cell label="Actual on the grid" value={$(cellAmount)} />
          <Cell label="From SAP rows" value={$(total)} sub={`${txns.length} transaction${txns.length === 1 ? '' : 's'}`} />
        </div>
        {summaryMonths.length > 0 && (
          <Callout tone="info" title={`${summaryMonths.length === months.length ? 'This period predates' : 'Part of this period predates'} the SAP line-item feed`}>
            {summaryMonths.map(fmtMonth).join(', ')} {summaryMonths.length === 1 ? 'was' : 'were'} loaded as monthly summary actuals from the legacy ERP and split to lines by the recorded function split and budget share. There are no line items to show for {summaryMonths.length === 1 ? 'it' : 'them'}.
          </Callout>
        )}
        {diff !== 0 && summaryMonths.length === 0 && txns.length > 0 && <Callout tone="warn">The grid and the SAP rows differ by {$(diff)} — whole-dollar rounding of allocated rows across lines.</Callout>}
        {txns.length ? <TxnTable s={s} txns={txns} share={share} /> : summaryMonths.length === 0 && <Empty title="No SAP rows">Nothing was posted to {what} in {when}.</Empty>}
        <p className="text-[11.5px] text-faint">Rows come from posted SAP extracts. Each row reached this line through the SAP mapping table (cost element range, WBS suffix, text) or a manual resolution by project controls; loader settlements are allocated over the period’s direct lines pro-rata.</p>
      </div>
    </Drawer>
  )
}

function Cell({ label, value, sub }: { label: string; value: string; sub?: string }) {
  return <div className="min-w-0 bg-surface px-3 py-2"><div className="eyebrow truncate">{label}</div><div className="truncate font-semibold tabular-nums">{value}</div>{sub && <div className="truncate font-mono text-[11px] text-faint">{sub}</div>}</div>
}

export const TXN_STATUS: Record<TxnStatus, { label: string; tone: 'good' | 'info' | 'neutral' | 'bad' | 'warn'; hint: string }> = {
  mapped: { label: 'Mapped', tone: 'good', hint: 'A mapping rule (or a person) sent it to one forecast line' },
  allocated: { label: 'Allocated', tone: 'info', hint: 'Loader / overhead settlement spread over the period’s direct lines pro-rata' },
  excluded: { label: 'Excluded', tone: 'neutral', hint: 'Kept out of project actuals (balance-sheet or non-project cost)' },
  unmapped: { label: 'Unmapped', tone: 'bad', hint: 'No active rule covers the cost element — blocks posting' },
  'no-project': { label: 'No project', tone: 'bad', hint: 'The SAP WBS element matches no project — blocks posting' },
}

export const importTxns = (s: AppState, importId: string) => s.actualTxns.filter((t) => t.importId === importId)

export interface ImportFigures { rows: number; total: Money; mapped: Money; allocated: Money; excluded: Money; unmapped: Money; noProject: Money; exceptions: number; resolved: number; credits: number; projects: number; periods: string[] }
export function importFigures(txns: ActualTxn[]): ImportFigures {
  const t = (st: TxnStatus) => Math.round(sum(txns.filter((x) => x.status === st).map((x) => x.amount)))
  return {
    rows: txns.length,
    total: Math.round(sum(txns.map((x) => x.amount))),
    mapped: t('mapped'), allocated: t('allocated'), excluded: t('excluded'), unmapped: t('unmapped'), noProject: t('no-project'),
    exceptions: txns.filter(isException).length,
    resolved: txns.filter((x) => x.manual).length,
    credits: txns.filter((x) => x.amount < 0 && x.status !== 'excluded').length,
    projects: new Set(txns.map((x) => x.projectId).filter(Boolean)).size,
    periods: [...new Set(txns.map((x) => x.period))].sort(),
  }
}

export const isException = (t: ActualTxn) => t.status === 'unmapped' || t.status === 'no-project'

/** Plain-language reason a row needs a person. */
export function exceptionReason(s: AppState, t: ActualTxn): string {
  if (t.status === 'no-project') return `WBS element ${t.sapWbs} does not belong to any project (no project has that ERP WBS or a parent of it)`
  if (t.status === 'unmapped') {
    const r = s.settings.sapMappings.find((x) => x.id === t.ruleId)
    if (r?.target === 'line') return `Rule P${r.priority} matches but targets a line that is missing or inactive`
    return `No active mapping rule covers cost element ${t.costElement}${t.costElementName ? ` (${t.costElementName})` : ''}`
  }
  return ''
}

/** "P30 · 5200000–5200999 → 06.10 Contract construction" */
export function ruleLabel(s: AppState, r?: SapMapping): string {
  if (!r) return ''
  return `P${r.priority} · ${r.costElementFrom}–${r.costElementTo}${r.wbsSuffix ? ` on *${r.wbsSuffix}` : ''}${r.textContains ? ` text “${r.textContains}”` : ''} → ${targetLabel(s, r)}`
}
export function targetLabel(s: AppState, r: Pick<SapMapping, 'target' | 'lineId'>): string {
  if (r.target === 'allocate') return 'Allocate pro-rata'
  if (r.target === 'exclude') return 'Exclude'
  const l = lineById(s, r.lineId)
  return l ? `${l.code} ${l.name}` : 'Missing line'
}
export const ruleOf = (s: AppState, t: ActualTxn) => s.settings.sapMappings.find((x) => x.id === t.ruleId)

/** How a transaction landed: rule, manual resolution or status. */
export function howMapped(s: AppState, t: ActualTxn): string {
  if (t.manual) return `Resolved by hand${t.manual.note ? ` — ${t.manual.note}` : ''}`
  const r = ruleOf(s, t)
  return r ? ruleLabel(s, r) : TXN_STATUS[t.status].hint
}

export interface ProjectRoll {
  p?: Project
  projectId: string
  byLine: Record<string, { direct: Money; allocated: Money }>
  direct: Money; allocated: Money; excluded: Money; exceptions: Money
  rows: number
}
/**
 * Roll an import up by project and forecast line. Allocated rows use their posted allocation; on a staged import
 * they are previewed the same way posting will do it (pro-rata to the project-period's mapped lines).
 */
export function projectLineRollup(s: AppState, txns: ActualTxn[]): ProjectRoll[] {
  const out = new Map<string, ProjectRoll>()
  const get = (pid: string) => {
    let r = out.get(pid)
    if (!r) { r = { p: s.projects.find((x) => x.id === pid), projectId: pid, byLine: {}, direct: 0, allocated: 0, excluded: 0, exceptions: 0, rows: 0 }; out.set(pid, r) }
    return r
  }
  const cell = (r: ProjectRoll, lid: string) => (r.byLine[lid] ??= { direct: 0, allocated: 0 })
  for (const t of txns) {
    const r = get(t.projectId ?? '—')
    r.rows++
    if (t.status === 'mapped' && t.lineId) { cell(r, t.lineId).direct += t.amount; r.direct += t.amount }
    else if (t.status === 'excluded') r.excluded += t.amount
    else if (isException(t)) r.exceptions += t.amount
  }
  for (const t of txns.filter((x) => x.status === 'allocated')) {
    const r = get(t.projectId ?? '—')
    const share = t.allocation ?? previewAllocation(txns, t)
    for (const [lid, v] of Object.entries(share)) cell(r, lid).allocated += v
    r.allocated += t.amount
  }
  return [...out.values()].sort((a, b) => (a.p?.code ?? 'ZZZ').localeCompare(b.p?.code ?? 'ZZZ'))
}

/** Pro-rata split of an allocated row over its project-period's mapped lines (mirrors postImport). */
export function previewAllocation(txns: ActualTxn[], t: ActualTxn): Record<string, Money> {
  const direct: Record<string, Money> = {}
  for (const x of txns) if (x.status === 'mapped' && x.lineId && x.projectId === t.projectId && x.period === t.period) direct[x.lineId] = (direct[x.lineId] || 0) + x.amount
  const ids = Object.keys(direct)
  if (!ids.length) return {}
  const base = sum(Object.values(direct))
  const out: Record<string, Money> = {}
  let acc = 0
  ids.forEach((k, i) => { const v = i === ids.length - 1 ? Math.round((t.amount - acc) * 100) / 100 : Math.round(t.amount * (base ? direct[k] / base : 1 / ids.length) * 100) / 100; out[k] = v; acc += v })
  return out
}

export interface Overlap { a: SapMapping; b: SapMapping; kind: 'refines' | 'shadowed' | 'conflict'; text: string }
const n = (x: string) => Number(x)
const covers = (a: SapMapping, b: SapMapping) => n(a.costElementFrom) <= n(b.costElementFrom) && n(a.costElementTo) >= n(b.costElementTo)
const intersects = (a: SapMapping, b: SapMapping) => n(a.costElementFrom) <= n(b.costElementTo) && n(b.costElementFrom) <= n(a.costElementTo)
const hasFilter = (r: SapMapping) => !!(r.wbsSuffix || r.textContains)

/** Overlapping cost-element ranges among active rules, classified: a narrower rule first (fine), a rule that can never match, or a real conflict. */
export function ruleOverlaps(rules: SapMapping[]): Overlap[] {
  const act = rules.filter((r) => r.active).sort((a, b) => a.priority - b.priority)
  const out: Overlap[] = []
  for (let i = 0; i < act.length; i++) for (let j = i + 1; j < act.length; j++) {
    const a = act[i], b = act[j] // a runs first
    if (!intersects(a, b)) continue
    const sameTarget = a.target === b.target && a.lineId === b.lineId
    if (covers(a, b) && !hasFilter(a)) out.push({ a, b, kind: 'shadowed', text: `P${b.priority} never matches: P${a.priority} runs first and covers its whole range with no WBS or text filter` })
    else if (hasFilter(a) || covers(b, a)) out.push({ a, b, kind: 'refines', text: `P${a.priority} is narrower and runs first; P${b.priority} catches the rest` })
    else out.push({ a, b, kind: sameTarget ? 'refines' : 'conflict', text: sameTarget ? `Ranges overlap but both send cost to the same place` : `Ranges partly overlap with different targets — rows in ${a.costElementFrom > b.costElementFrom ? a.costElementFrom : b.costElementFrom}–${a.costElementTo < b.costElementTo ? a.costElementTo : b.costElementTo} follow P${a.priority}` })
  }
  return out
}

export const linesWithoutRule = (s: AppState) => forecastLines(s).filter((l) => !s.settings.sapMappings.some((r) => r.active && r.target === 'line' && r.lineId === l.id))

/** Cost elements in staged imports that no rule covers (what to add next), with row count and amount. */
export function unmappedElements(s: AppState): { costElement: string; name: string; rows: number; amount: Money }[] {
  const staged = new Set(s.actualsImports.filter((x) => x.status === 'staged').map((x) => x.id))
  const m = new Map<string, { costElement: string; name: string; rows: number; amount: Money }>()
  for (const t of s.actualTxns) if (staged.has(t.importId) && t.status === 'unmapped' && !t.manual) {
    const e = m.get(t.costElement) ?? { costElement: t.costElement, name: t.costElementName, rows: 0, amount: 0 }
    e.rows++; e.amount += t.amount
    m.set(t.costElement, e)
  }
  return [...m.values()]
}

/** Test one row against the mapping table: the winning rule, every rule that would match, and the project. */
export function testRow(s: AppState, row: { costElement: string; sapWbs: string; text: string }, rules = s.settings.sapMappings) {
  const r = { ...row, vendor: '' }
  const winner = matchRule(rules, r)
  const all = [...rules].filter((x) => x.active).sort((a, b) => a.priority - b.priority).filter((x) => matchRule([x], r))
  const p = row.sapWbs.trim() ? projectForWbs(s, row.sapWbs) : undefined
  const line = winner?.target === 'line' ? lineById(s, winner.lineId) : undefined
  return { winner, all, project: p, line, fnId: line ? functionOfLine(s, line.id) : undefined }
}

function StatusNote({ pp }: { pp: ReconPeriod }) {
  const un = pp.diff - pp.explained
  if (pp.status === 'ties') return <Callout tone="good">Crosswalk’s {fmtMonth(pp.period)} actuals tie to SAP in total and on every forecast line.</Callout>
  if (pp.status === 'not-posted') return <Callout tone="warn" title="Not posted">Crosswalk has no actuals for {fmtMonth(pp.period)} on this project. Post the extract to bring them in.</Callout>
  if (pp.status === 'lines-differ') return <Callout tone="warn" title="Total ties, lines differ">The total agrees with SAP, but some cost sits on different forecast lines than the mapping table sends it — usually a hand resolution. See Reconciling rows.</Callout>
  return (
    <Callout tone="bad" title={`Crosswalk ${pp.diff > 0 ? 'is higher' : 'is lower'} than SAP by ${$(Math.abs(pp.diff))}`}>
      {pp.source === 'summary' ? 'Crosswalk holds monthly summary actuals for this month, not SAP line items; posting the extract replaces them.' : Math.abs(un) <= RECON_TOLERANCE ? 'Every dollar of the gap is explained by the reconciling rows below.' : `${$(Math.abs(un))} is not explained by row-level matching — check for edits to the cost period outside an SAP import.`}
    </Callout>
  )
}

/** Local editable copy of a value; `dirty` when it differs from the source. Resets when the source changes. */
export function useDraft<T>(source: T): [T, React.Dispatch<React.SetStateAction<T>>, boolean, () => void] {
  const key = JSON.stringify(source) ?? 'null'
  const [d, setD] = useState<T>(() => JSON.parse(key))
  useEffect(() => { setD(JSON.parse(key)) }, [key])
  return [d, setD, JSON.stringify(d) !== key, () => setD(JSON.parse(key))]
}

export const clone = <T,>(x: T): T => JSON.parse(JSON.stringify(x))

/** Chip-style multi-select for short option lists. */
export function MultiCheck<T extends string>({ options, value, onChange, className }: { options: { id: T; label: string }[]; value: T[]; onChange: (v: T[]) => void; className?: string }) {

        <span className="font-mono text-ink">{describeCondition(value)}</span>
        <span className="mx-1.5">·</span>
        Applies to <b className="text-ink">{hits.length}</b> of {projects.length} projects on this template
        {value && hits.length > 0 && <span className="text-faint"> ({hits.slice(0, 8).map((p) => p.code).join(', ')}{hits.length > 8 ? '…' : ''})</span>}
      </div>
    </div>
  )
}

/** Construction window milestones (standard or condensed template). */
export function constructionStart(p: Project): string | undefined {
  const m = p.milestones.find((x) => x.name === 'Construction start') ?? p.milestones.find((x) => x.name === 'Construction authorized')
  return m?.actual ?? m?.forecast
}
export function inServiceDate(p: Project): string | undefined {
  const m = p.milestones.find((x) => x.name === 'In service') ?? p.milestones.find((x) => x.name === 'In service & closed')
  return m?.actual ?? m?.forecast
}
