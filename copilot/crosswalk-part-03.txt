// ── Unit-cost library (ES-03). p50 values are illustrative 2025$ planning rates.
let rid = 0
function rate(assetType: AssetType, workMethod: WorkMethod, categoryId: string, description: string, unit: string, p50: number, extra: { diameterIn?: number; terrain?: Terrain; n?: number } = {}): UnitCostRate {
  rid++
  return { id: `rate-${rid}`, assetType, workMethod, categoryId, description, unit, p50: Math.round(p50), p10: Math.round(p50 * 0.78), p90: Math.round(p50 * 1.34), diameterIn: extra.diameterIn, terrain: extra.terrain, sampleSize: extra.n ?? 4 + ((rid * 7) % 19), basisYear: 2025 }
}

// Built from the unit-rate library so the demo's Class 5 quick estimates agree with its parametric estimates.
export function buildHistoricalCosts(rates: UnitCostRate[]): HistoricalCost[] {
  const out: HistoricalCost[] = []
  let k = 0
  const codes = ['PRJ-0871', 'PRJ-0904', 'PRJ-0912', 'PRJ-0933', 'PRJ-0957', 'PRJ-0968', 'PRJ-0981', 'PRJ-0995', 'PRJ-1003', 'PRJ-1011', 'PRJ-1017', 'PRJ-1024']
  const add = (assetType: AssetType, workMethod: WorkMethod, diameterBand: DiameterBand, terrain: Terrain | 'any', unit: HistUnit, rep: { dia: number; terrain: Terrain; len: number; qty: number; sites?: number }) => {
    const proj = { asset: { type: assetType, diameterIn: rep.dia, lengthFt: rep.len }, workMethod, location: { terrain: rep.terrain }, scope: { quantities: [{ label: 'Valve sites', qty: rep.sites ?? 1, unit: 'ea' }] } } as unknown as Project
    const lines = parametricLines(proj, rates)
    if (!lines.length) return
    const byCat: Record<string, number> = {}
    for (const l of lines) byCat[l.categoryId] = (byCat[l.categoryId] || 0) + l.qty * l.unitCost
    const tot = Object.values(byCat).reduce((a, b) => a + b, 0)
    const per = tot / rep.qty
    k++
    const wob = 0.94 + ((k * 37) % 13) / 100 // history is never exactly the library
    const n = 3 + ((k * 7) % 11)
    out.push({
      id: `hc-${k}`, assetType, workMethod, diameterBand, terrain, unit,
      p50: Math.round(per * wob), p10: Math.round(per * wob * (0.7 + (k % 3) * 0.03)), p90: Math.round(per * wob * (1.38 + (k % 4) * 0.06)),
      n, baseYear: k % 3 ? 2025 : 2024,
      mix: Object.fromEntries(Object.entries(byCat).map(([c, v]) => [c, +(v / tot).toFixed(4)])),
      sources: codes.filter((_, i) => (i + k) % 3 === 0).slice(0, Math.min(n, 5)),
      updatedAt: '2026-07-15', updatedById: 'u-estl', active: true,
      note: n < 5 ? 'Small sample — confirm with the estimating lead' : undefined,
    })
  }
  const bands: [DiameterBand, number][] = [['10-16', 16], ['20-24', 24], ['26-36', 30]]
  for (const wm of ['replace-open-trench', 'replace-hdd'] as WorkMethod[])
    for (const [b, d] of bands) for (const t of ['urban', 'suburban', 'rural'] as Terrain[]) add('transmission-pipeline', wm, b, t, 'ft', { dia: d, terrain: t, len: 15_000, qty: 15_000 })
  for (const t of ['urban', 'suburban', 'rural'] as Terrain[]) add('transmission-pipeline', 'hydrotest', 'n/a', t, 'ft', { dia: 20, terrain: t, len: 25_000, qty: 25_000 })
  for (const t of ['urban', 'suburban', 'rural'] as Terrain[]) add('distribution-main', 'replace-open-trench', 'le-8', t, 'ft', { dia: 6, terrain: t, len: 8_000, qty: 8_000 })
  add('valve-site', 'valve-automation', 'n/a', 'any', 'site', { dia: 24, terrain: 'suburban', len: 0, qty: 3, sites: 3 })
  add('compressor-station', 'compressor-replacement', 'n/a', 'any', 'hp', { dia: 0, terrain: 'rural', len: 0, qty: 22_000 })
  add('compressor-station', 'station-rebuild', 'n/a', 'any', 'hp', { dia: 0, terrain: 'rural', len: 0, qty: 9_000 })
  k++
  out.push({ id: `hc-${k}`, assetType: 'regulator-station', workMethod: 'station-rebuild', diameterBand: 'n/a', terrain: 'any', unit: 'station', p10: 1_900_000, p50: 2_750_000, p90: 4_100_000, n: 9, baseYear: 2025, mix: { 'cat-equipment': 0.24, 'cat-contract': 0.38, 'cat-eng': 0.12, 'cat-co-labor': 0.1, 'cat-land': 0.04, 'cat-env': 0.03, 'cat-inspect': 0.03, 'cat-pmc': 0.06 }, sources: ['PRJ-0912', 'PRJ-0968', 'PRJ-1011'], updatedAt: '2026-07-15', updatedById: 'u-estl', active: true })
  return out
}

import type { Assignee, FormField, RecordType, RoleId, WfEdge, WfNode, WorkflowDef } from '../types'

type Step = { id: string; type: WfNode['type']; label: string; a?: Assignee; sla?: number; rule?: 'any' | 'all'; desc?: string; sys?: WfNode['systemAction'] }

const role = (roleId: RoleId): Assignee => ({ kind: 'role', roleId })
const team = (position: Extract<Assignee, { kind: 'team-position' }>['position']): Assignee => ({ kind: 'team-position', position })
const requester: Assignee = { kind: 'requester' }
const doa: Assignee = { kind: 'doa', field: 'amount' }

let eid = 0
const E = (from: string, to: string, extra: Partial<WfEdge> = {}): WfEdge => ({ id: `e${++eid}`, from, to, ...extra })

/** A straight-line process: start → steps → end. Approvals/reviews get a 'return' edge to the first form step. */
function linear(
  id: string, key: string, name: string, recordType: RecordType, ownerRole: RoleId, description: string, purpose: string, fields: FormField[], steps: Step[],
): WorkflowDef {
  const all: Step[] = [{ id: 'start', type: 'start', label: 'Start' }, ...steps, { id: 'end', type: 'end', label: 'Complete' }]
  const nodes: WfNode[] = all.map((s, i) => ({ id: s.id, type: s.type, label: s.label, x: 60 + i * 170, y: 160, assignee: s.a, slaDays: s.sla, rule: s.rule, description: s.desc, systemAction: s.sys }))
  const edges: WfEdge[] = []
  for (let i = 0; i < all.length - 1; i++) edges.push(E(all[i].id, all[i + 1].id, { outcome: all[i].type === 'approval' || all[i].type === 'review' ? 'approve' : undefined }))
  const firstForm = steps.find((s) => s.type === 'form')
  if (firstForm) for (const s of steps) if ((s.type === 'approval' || s.type === 'review') && s.id !== firstForm.id) edges.push(E(s.id, firstForm.id, { outcome: 'return', label: 'Return for revision' }))
  return { id, key, name, recordType, version: 1, status: 'published', effective: '2026-01-01', description, purpose, nodes, edges, fields, ownerRole }
}

  const [to, setTo] = useState(addDays(asOf, 7))
  const [delegate, setDelegate] = useState('')
  useEffect(() => {
    if (!open) return
    setFrom(me.outOfOffice?.from ?? asOf); setTo(me.outOfOffice?.to ?? addDays(asOf, 7)); setDelegate(me.outOfOffice?.delegateId ?? '')

const AXIS = 'rgb(var(--line))'
const TEXT = 'rgb(var(--muted))'

function niceTicks(min: number, max: number, n = 4): number[] {
  if (max === min) max = min + 1
  const span = max - min
  const step0 = span / n
  const mag = Math.pow(10, Math.floor(Math.log10(step0)))
  const step = [1, 2, 2.5, 5, 10].map((m) => m * mag).find((s) => span / s <= n) ?? mag * 10
  const lo = Math.floor(min / step) * step
  const out: number[] = []
  for (let v = lo; v <= max + step * 0.5; v += step) out.push(Math.round(v * 1e6) / 1e6)
  return out
}

export interface LineSeries { name: string; values: (number | null)[]; color?: string; dashed?: boolean; area?: boolean }

/** Line chart over categorical x labels (months, dates). */
export function LineChart({ labels, series, height = 220, format = money, markers, yMin }: { labels: string[]; series: LineSeries[]; height?: number; format?: (n: number) => string; markers?: { index: number; label: string }[]; yMin?: number }) {
  const [hover, setHover] = useState<number | null>(null)
  const W = 720, H = height, L = 64, Rm = 16, T = 14, B = 30
  const vals = series.flatMap((s) => s.values.filter((v): v is number => v != null))
  const max = Math.max(...vals, 0), min = Math.min(yMin ?? 0, ...vals)
  const ticks = niceTicks(min, max)
  const ymin = ticks[0], ymax = ticks[ticks.length - 1]
  const X = (i: number) => L + (labels.length <= 1 ? 0 : (i / (labels.length - 1)) * (W - L - Rm))
  const Y = (v: number) => T + (1 - (v - ymin) / (ymax - ymin || 1)) * (H - T - B)
  const every = Math.max(1, Math.ceil(labels.length / 8))
  return (
    <div className="relative">
      <svg viewBox={`0 0 ${W} ${H}`} className="w-full" role="img" onMouseLeave={() => setHover(null)}>
        {ticks.map((t) => <g key={t}><line x1={L} x2={W - Rm} y1={Y(t)} y2={Y(t)} stroke={AXIS} strokeWidth={1} /><text x={L - 8} y={Y(t) + 4} textAnchor="end" fontSize={11} fill={TEXT}>{format(t)}</text></g>)}
        {labels.map((l, i) => i % every === 0 && <text key={i} x={X(i)} y={H - 10} textAnchor="middle" fontSize={11} fill={TEXT}>{l}</text>)}
        {markers?.map((m) => <g key={m.index}><line x1={X(m.index)} x2={X(m.index)} y1={T} y2={H - B} stroke="rgb(var(--ink))" strokeDasharray="3 3" strokeWidth={1} /><text x={X(m.index) + 4} y={T + 10} fontSize={11} fill="rgb(var(--ink))">{m.label}</text></g>)}
        {series.map((s, si) => {
          const c = s.color ?? SERIES[si % SERIES.length]
          const pts = s.values.map((v, i) => (v == null ? null : [X(i), Y(v)] as const))
          let d = ''
          pts.forEach((p, i) => { if (!p) return; d += (d && pts[i - 1] ? 'L' : 'M') + p[0].toFixed(1) + ' ' + p[1].toFixed(1) })
          const valid = pts.filter(Boolean) as (readonly [number, number])[]
          const area = s.area && valid.length ? `M${valid[0][0]} ${Y(Math.max(ymin, 0))}L` + valid.map((p) => `${p[0]} ${p[1]}`).join('L') + `L${valid[valid.length - 1][0]} ${Y(Math.max(ymin, 0))}Z` : ''
          const last = valid[valid.length - 1]
          return <g key={s.name}>{area && <path d={area} fill={c} opacity={0.1} />}<path d={d} fill="none" stroke={c} strokeWidth={2} strokeDasharray={s.dashed ? '5 4' : undefined} strokeLinejoin="round" />{last && <circle cx={last[0]} cy={last[1]} r={3.5} fill={c} />}</g>
        })}
        {labels.map((_, i) => <rect key={i} x={X(i) - (W - L) / labels.length / 2} y={T} width={(W - L) / labels.length} height={H - T - B} fill="transparent" onMouseEnter={() => setHover(i)} />)}
        {hover != null && <line x1={X(hover)} x2={X(hover)} y1={T} y2={H - B} stroke={TEXT} strokeWidth={1} />}
      </svg>
      <div className="mt-1 flex flex-wrap gap-x-4 gap-y-1 text-xs text-muted">
        {series.map((s, si) => <span key={s.name} className="inline-flex items-center gap-1.5"><span className="inline-block h-0.5 w-4" style={{ background: s.color ?? SERIES[si % SERIES.length], borderTop: s.dashed ? '2px dashed' : undefined }} />{s.name}{hover != null && s.values[hover] != null && <b className="tabular-nums text-ink">{format(s.values[hover] as number)}</b>}</span>)}
        {hover != null && <span className="text-faint">{labels[hover]}</span>}
      </div>
    </div>
  )
}

/** Vertical bars; multiple series render side by side (grouped) or stacked. */
export function BarChart({ labels, series, height = 220, format = money, stacked, highlight }: { labels: string[]; series: { name: string; values: number[]; color?: string }[]; height?: number; format?: (n: number) => string; stacked?: boolean; highlight?: number }) {

  const totals = labels.map((_, i) => (stacked ? series.reduce((a, s) => a + Math.max(0, s.values[i] || 0), 0) : Math.max(...series.map((s) => s.values[i] || 0))))
  const ticks = niceTicks(0, Math.max(...totals, 1))
  const ymax = ticks[ticks.length - 1]
  const bw = (W - L - Rm) / labels.length
  const Y = (v: number) => T + (1 - v / ymax) * (H - T - B)
  const every = Math.max(1, Math.ceil(labels.length / 10))

        {labels.map((l, i) => {

          let acc = 0
          return (

              <title>{l + ': ' + series.map((s) => `${s.name} ${format(s.values[i] || 0)}`).join(', ')}</title>
              {series.map((s, si) => {
                const v = Math.max(0, s.values[i] || 0)
                const c = s.color ?? SERIES[si % SERIES.length]
                if (stacked) { const y = Y(acc + v), h = Y(acc) - y; acc += v; return <rect key={si} x={x0 + bw * 0.18} y={y} width={bw * 0.64} height={Math.max(0, h)} fill={c} /> }
                const gw = (bw * 0.7) / series.length
                return <rect key={si} x={x0 + bw * 0.15 + si * gw} y={Y(v)} width={gw - 1} height={Math.max(0, Y(0) - Y(v))} fill={c} rx={1} />
              })}
              {i % every === 0 && <text x={x0 + bw / 2} y={H - 10} textAnchor="middle" fontSize={11} fill={TEXT}>{l}</text>}
            </g>
          )
        })}
      </svg>
      {series.length > 1 && <div className="mt-1 flex flex-wrap gap-x-4 text-xs text-muted">{series.map((s, si) => <span key={s.name} className="inline-flex items-center gap-1.5"><span className="inline-block h-2.5 w-2.5 rounded-sm" style={{ background: s.color ?? SERIES[si % SERIES.length] }} />{s.name}</span>)}</div>}
    </div>
  )

/** Waterfall bridge from one total to another (estimate or baseline bridge). */
export function Waterfall({ start, end, steps, startLabel = 'From', endLabel = 'To', height = 240 }: { start: number; end: number; steps: { label: string; delta: number }[]; startLabel?: string; endLabel?: string; height?: number }) {
  const W = 720, H = height, L = 64, Rm = 12, T = 18, B = 46
  const bars: { label: string; y0: number; y1: number; kind: 'total' | 'up' | 'down' }[] = [{ label: startLabel, y0: 0, y1: start, kind: 'total' }]
  let acc = start
  for (const s of steps) { bars.push({ label: s.label, y0: acc, y1: acc + s.delta, kind: s.delta >= 0 ? 'up' : 'down' }); acc += s.delta }
  bars.push({ label: endLabel, y0: 0, y1: end, kind: 'total' })
  const hi = Math.max(...bars.map((b) => Math.max(b.y0, b.y1)))
  const lo = Math.min(0, ...bars.map((b) => Math.min(b.y0, b.y1)))
  const ticks = niceTicks(lo, hi)
  const ymin = ticks[0], ymax = ticks[ticks.length - 1]
  const Y = (v: number) => T + (1 - (v - ymin) / (ymax - ymin)) * (H - T - B)
  const bw = (W - L - Rm) / bars.length
  return (
    <svg viewBox={`0 0 ${W} ${H}`} className="w-full" role="img">
      {ticks.map((t) => <g key={t}><line x1={L} x2={W - Rm} y1={Y(t)} y2={Y(t)} stroke={AXIS} /><text x={L - 8} y={Y(t) + 4} textAnchor="end" fontSize={11} fill={TEXT}>{money(t)}</text></g>)}
      {bars.map((b, i) => {
        const x = L + i * bw + bw * 0.15, w = bw * 0.7
        const top = Y(Math.max(b.y0, b.y1)), h = Math.max(1.5, Math.abs(Y(b.y0) - Y(b.y1)))
        const c = b.kind === 'total' ? 'rgb(var(--ink))' : b.kind === 'up' ? 'rgb(var(--bad))' : 'rgb(var(--good))'
        const delta = b.y1 - b.y0
        return (
          <g key={i}>
            <rect x={x} y={top} width={w} height={h} fill={c} opacity={b.kind === 'total' ? 0.85 : 0.8} rx={1} />
            {i < bars.length - 1 && <line x1={x + w} x2={x + bw} y1={Y(b.y1)} y2={Y(b.y1)} stroke={TEXT} strokeDasharray="2 2" />}
            <text x={x + w / 2} y={top - 4} textAnchor="middle" fontSize={10.5} fill="rgb(var(--ink))">{b.kind === 'total' ? money(b.y1) : (delta > 0 ? '+' : '') + money(delta)}</text>
            <text x={x + w / 2} y={H - B + 14} textAnchor="middle" fontSize={10.5} fill={TEXT}>{b.label.length > 14 ? b.label.slice(0, 13) + '…' : b.label}</text>
          </g>
        )
      })}
    </svg>
  )
}

export function Sparkline({ values, width = 90, height = 24, color = SERIES[0] }: { values: number[]; width?: number; height?: number; color?: string }) {
  if (values.length < 2) return <svg width={width} height={height} />
  const mx = Math.max(...values), mn = Math.min(...values)
  const X = (i: number) => (i / (values.length - 1)) * (width - 4) + 2
  const Y = (v: number) => height - 3 - ((v - mn) / (mx - mn || 1)) * (height - 6)
  const d = values.map((v, i) => `${i ? 'L' : 'M'}${X(i).toFixed(1)} ${Y(v).toFixed(1)}`).join('')
  return <svg width={width} height={height} aria-hidden><path d={d} fill="none" stroke={color} strokeWidth={1.5} /><circle cx={X(values.length - 1)} cy={Y(values[values.length - 1])} r={2.5} fill={color} /></svg>
}

/** Histogram with percentile markers (Monte Carlo). */
export function Histogram({ bins, markers, format = money, height = 200 }: { bins: { x: number; n: number }[]; markers?: { x: number; label: string; color?: string }[]; format?: (n: number) => string; height?: number }) {
  const W = 720, H = height, L = 16, Rm = 16, T = 22, B = 28
  if (!bins.length) return null
  const lo = bins[0].x, hi = bins[bins.length - 1].x
  const mx = Math.max(...bins.map((b) => b.n))
  const X = (v: number) => L + ((v - lo) / (hi - lo || 1)) * (W - L - Rm)
  const bw = (W - L - Rm) / bins.length
  return (
    <svg viewBox={`0 0 ${W} ${H}`} className="w-full" role="img">
      {bins.map((b, i) => { const h = (b.n / mx) * (H - T - B); return <rect key={i} x={L + i * bw} y={H - B - h} width={bw - 1} height={h} fill="rgb(var(--link))" opacity={0.55} /> })}
      <line x1={L} x2={W - Rm} y1={H - B} y2={H - B} stroke={AXIS} />
      {[0, 0.25, 0.5, 0.75, 1].map((f) => <text key={f} x={L + f * (W - L - Rm)} y={H - 10} textAnchor={f === 0 ? 'start' : f === 1 ? 'end' : 'middle'} fontSize={11} fill={TEXT}>{format(lo + f * (hi - lo))}</text>)}
      {markers?.map((m, i) => <g key={m.label}><line x1={X(m.x)} x2={X(m.x)} y1={T - 4} y2={H - B} stroke={m.color ?? 'rgb(var(--ink))'} strokeWidth={1.5} strokeDasharray="4 3" /><text x={X(m.x)} y={T - 8 + (i % 2) * 0} textAnchor="middle" fontSize={11} fontWeight={600} fill={m.color ?? 'rgb(var(--ink))'}>{m.label}</text></g>)}
    </svg>
  )
}

/** Map common statuses to tones so every module colors them the same way. */
export function statusTone(status: string): Tone {
  const s = status.toLowerCase()
  if (['approved', 'accepted', 'done', 'closed', 'issued', 'acquired', 'verified', 'complete', 'go', 'met', 'received', 'published', 'green', 'compliant', 'awarded', 'active-ok'].includes(s)) return 'good'
  if (['in-review', 'submitted', 'running', 'mitigating', 'applied', 'negotiating', 'ordered', 'shipped', 'go-with-conditions', 'attached', 'scheduled', 'amber', 'conditional', 'in-progress', 'draft-review'].includes(s)) return 'info'
  if (['required', 'returned', 'recycle', 'hold', 'on-hold', 'expired', 'missing', 'realized', 'overdue', 'red', 'rejected', 'cancelled', 'cancel', 'non-compliant', 'open-high'].includes(s)) return 'bad'
  if (['open', 'pending', 'identified', 'not-applied', 'not-started', 'draft', 'rfp', 'requested'].includes(s)) return 'neutral'
  return 'neutral'
}

// ───────────────────────────── Table
export interface Column<T> { key: string; header: React.ReactNode; render: (row: T) => React.ReactNode; num?: boolean; width?: string | number; sort?: (row: T) => number | string; className?: string }
export function Table<T>({ rows, columns, onRowClick, rowKey, empty, maxHeight, dense }: { rows: T[]; columns: Column<T>[]; onRowClick?: (r: T) => void; rowKey: (r: T) => string; empty?: React.ReactNode; maxHeight?: number; dense?: boolean }) {
  const [sort, setSort] = useState<{ key: string; dir: 1 | -1 } | null>(null)
  const col = sort && columns.find((c) => c.key === sort.key)
  const sorted = col?.sort ? rows.slice().sort((a, b) => { const x = col.sort!(a), y = col.sort!(b); return (x < y ? -1 : x > y ? 1 : 0) * sort!.dir }) : rows
  if (!rows.length && empty) return <>{empty}</>
  return (
    <div className="scroll-x" style={maxHeight ? { maxHeight, overflowY: 'auto' } : undefined}>
      <table className={cx('tbl', dense && '[&_td]:py-1')}>
        <thead>
          <tr>
            {columns.map((c) => (
              <th key={c.key} className={cx(c.num && 'num', c.sort && 'cursor-pointer select-none hover:text-ink')} style={{ width: c.width }} onClick={() => c.sort && setSort((s) => (s?.key === c.key ? { key: c.key, dir: (s.dir * -1) as 1 | -1 } : { key: c.key, dir: 1 }))}>
                {c.header}{sort?.key === c.key ? (sort.dir === 1 ? ' ▲' : ' ▼') : ''}
              </th>
            ))}
          </tr>
        </thead>
        <tbody>
          {sorted.map((r) => (
            <tr key={rowKey(r)} className={cx(onRowClick && 'clickable')} onClick={onRowClick ? () => onRowClick(r) : undefined}>
              {columns.map((c) => <td key={c.key} className={cx(c.num && 'num', c.className)}>{c.render(r)}</td>)}
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  )
}

    mutate((d) => { id = stageImport(d, rows, name, me.id).id })
    setSelId(id)
    setUpload(null)
    setPasteOpen(false)
    const t = useStore.getState().actualTxns.filter((x) => x.importId === id)
    const ex = t.filter(isException).length
    toast(ex ? `Staged ${rows.length} rows · ${ex} exception${ex === 1 ? '' : 's'} to resolve` : `Staged ${rows.length} rows · all mapped`, ex ? 'info' : 'good')
  }

    { key: 'period', header: 'Period', sort: (r) => r.period, render: (r) => <span className="whitespace-nowrap">{fmtMonth(r.period)}</span> },
    { key: 'status', header: 'Status', sort: (r) => r.status, render: (r) => <StatusBadge status={IMP_TONE[r.status]} label={IMP_LABEL[r.status]} /> },

    { key: 'rows', header: 'Rows', num: true, sort: (r) => r.rows, render: (r) => r.rows.toLocaleString('en-US') },

  ]

  return (
    <div>
      <PageHeader eyebrow="Cost control" title="SAP actuals"
        subtitle={`Actual cost arrives from SAP as line items by WBS element and cost element. The mapping table translates each row to a forecast line item, project controls resolves what doesn’t map, and posting rebuilds each project-period’s actuals by line — the actuals every forecast and EAC is built on.`}
        actions={<>
          {!uploadsLocked && <FileButton label={<><Icon name="download" className="h-3.5 w-3.5 rotate-180" />Upload extract</>} accept=".csv,.txt,.tsv,text/csv,text/plain" onFile={onFile} variant="primary" />}
          {!uploadsLocked && <Button size="sm" onClick={() => setPasteOpen(true)}>Paste rows</Button>}
          <Button size="sm" variant="ghost" onClick={loadSample} title={`Bundled with the demo: samples/${FICTIONAL_SAMPLE_FILE}`}>Load the fictional sample file</Button>
        </>} />

      <FictionalOnly />

      <Tabs className="mb-4" value={page} onChange={setPage} tabs={[
        { id: 'imports', label: 'Imports & exceptions', count: staged.length || undefined },
        { id: 'recon', label: 'Reconciliation' },
      ]} />

      {page === 'recon' ? <Reconciliation s={s} source={reconSource} onSource={setReconSource} projectId={props.projectId ?? handoff?.projectId} /> : <>
      <HowItWorks s={s} />

      <div className="mb-4 grid grid-cols-2 gap-3 lg:grid-cols-4">
        <Stat label="Staged extracts" value={staged.length} tone={staged.length ? 'warn' : undefined} sub={staged.length ? 'Waiting to be posted' : 'Nothing waiting'} onClick={staged[0] ? () => setSelId(staged[0].id) : undefined} />
        <Stat label="Open exceptions" value={openExceptions} tone={openExceptions ? 'bad' : 'good'} sub={openExceptions ? 'Block posting until resolved' : 'All rows map'} />
        <Stat label="Posted through" value={lastPosted ? fmtMonth([...posted].sort((a, b) => b.period.localeCompare(a.period))[0].period) : '—'} sub={lastPosted ? `last posted ${fmtDate(lastPosted.postedAt?.slice(0, 10))}` : 'No posted extracts'} />
        <Stat label="Mapping rules" value={s.settings.sapMappings.filter((r) => r.active).length} sub={<a href="#admin" onClick={() => lsSet('crosswalk:admin-tab', 'sap')} className="text-link hover:underline">{forecastLines(s).length} forecast lines · edit in Configuration</a>} />
      </div>

      {sel ? <ImportReview key={sel.id} s={s} imp={sel} canPost={canPost} onReconcile={() => reconcileImport(sel.id)} /> : <Panel title="No extracts yet"><Empty title="Load an extract">Load the fictional sample file, or paste or upload rows you made up in the same column layout (CSV or tab-delimited text).</Empty></Panel>}

      <Panel className="mt-4" title="Import history" subtitle={`${imports.length} extracts · click one to review it`} pad={false}>
        <Table rows={imports} rowKey={(r) => r.id} columns={cols} dense onRowClick={(r) => { setSelId(r.id); document.getElementById('main')?.scrollTo?.({ top: 0, behavior: 'smooth' }) }} empty={<div className="p-4"><Empty title="No imports yet" /></div>} />
      </Panel>
      </>}

      {upload && <StageModal s={s} name={upload.name} text={upload.text} onClose={() => setUpload(null)} onStage={stage} />}
      {pasteOpen && <PasteModal s={s} onClose={() => setPasteOpen(false)} onStage={stage} />}
    </div>
  )
}

// ───────────────────────────── Review one import
type RTab = 'exceptions' | 'rollup' | 'txns'

function ImportReview({ s, imp, canPost, onReconcile }: { s: AppState; imp: ActualsImport; canPost: boolean; onReconcile: () => void }) {
  const me = useMe()
  const txns = useMemo(() => importTxns(s, imp.id), [s.actualTxns, imp.id])
  const F = importFigures(txns)
  const ex = txns.filter(isException)
  const review = txns.filter((t) => !isException(t) && (t.manual || t.status === 'excluded' || t.amount < 0))
  const [tab, setTab] = useState<RTab>(imp.status === 'staged' && (ex.length || review.length) ? 'exceptions' : 'rollup')
  const [blocked, setBlocked] = useState<string | null>(null)
  const staged = imp.status === 'staged'

  const rerun = () => {
    const before = ex.length
    mutate((d) => remap(d, imp.id), { entity: 'actuals-import', entityId: imp.id, action: 'remap', summary: `Re-ran the SAP mapping table over ${imp.fileName}` })
    const after = useStore.getState().actualTxns.filter((t) => t.importId === imp.id && isException(t)).length
    toast(after < before ? `Mapping re-run · ${before - after} exception${before - after === 1 ? '' : 's'} cleared` : 'Mapping re-run · no change', after < before ? 'good' : 'info')
  }
  const post = () => {
    let res: ReturnType<typeof postImport> = { ok: false, periods: 0 }
    mutate((d) => { res = postImport(d, imp.id, me.id) })
    if (!res.ok) { setBlocked(res.reason ?? 'Posting is blocked'); setTab('exceptions'); toast(`Not posted: ${res.reason}`, 'bad'); return }
    setBlocked(null)
    toast(`Posted to ${res.periods} project-period${res.periods === 1 ? '' : 's'} — forecast actuals updated`)
  }
  const discard = () => { mutate((d) => discardImport(d, imp.id, me.id)); toast('Extract discarded', 'info') }
  const exportCsv = async () => {
    const rows = txns.map((t) => ({ 'Document': t.docNo, 'Posting date': t.postingDate, Period: t.period, Project: s.projects.find((p) => p.id === t.projectId)?.code ?? '', 'WBS element': t.sapWbs, 'Cost element': t.costElement, 'Cost element name': t.costElementName, Vendor: t.vendor, Text: t.text, Amount: t.amount, Status: TXN_STATUS[t.status].label, Line: lineById(s, t.lineId)?.code ?? '', Rule: s.settings.sapMappings.find((r) => r.id === t.ruleId)?.priority ?? '', Manual: t.manual?.note ?? '' }))
    const r = await saveFile(`${imp.fileName.replace(/\.\w+$/, '')}-mapped.csv`, toCSV(rows), 'text/csv')
    if (r === 'saved') toast('Mapped extract exported')
  }

  return (
    <Panel pad={false}>
      <div className="flex flex-col gap-3 p-4 lg:flex-row lg:items-start lg:justify-between">
        <div className="min-w-0">
          <div className="flex flex-wrap items-center gap-2">
            <h2 className="min-w-0 truncate font-mono text-[15px] font-semibold">{imp.fileName}</h2>
            <StatusBadge status={IMP_TONE[imp.status]} label={IMP_LABEL[imp.status]} />
            {staged && F.exceptions > 0 && <Badge tone="bad">{F.exceptions} exception{F.exceptions === 1 ? '' : 's'} block posting</Badge>}
            {staged && F.exceptions === 0 && <Badge tone="good">Ready to post</Badge>}
          </div>
          <div className="mt-1 text-[12px] text-muted">
            {imp.source} · {F.periods.map(fmtMonth).join(', ')} · imported {fmtDateTime(imp.importedAt)} by {personName(s, imp.byId)}
            {imp.postedAt && <> · posted {fmtDateTime(imp.postedAt)} by {personName(s, imp.postedById)}</>}
            {imp.note && <> · {imp.note}</>}
          </div>
        </div>
        <div className="flex flex-wrap items-center gap-2">
          {staged && <Button size="sm" variant="ghost" onClick={rerun} title="Run the current mapping table again; rows resolved by hand are kept">Re-run mapping</Button>}
          {staged && <ConfirmButton size="sm" confirmLabel="Discard extract" onConfirm={discard}>Discard</ConfirmButton>}
          {txns.length > 0 && <Button size="sm" variant="ghost" onClick={exportCsv} icon={<Icon name="download" className="h-3.5 w-3.5" />}>CSV</Button>}
          {txns.length > 0 && <Button size="sm" variant="ghost" onClick={onReconcile} title="Compare this extract with Crosswalk's posted actuals and the forecast">Reconcile</Button>}
          {staged && <Button size="sm" variant={F.exceptions ? 'secondary' : 'marker'} disabled={!canPost} title={canPost ? undefined : 'Project controls, PMO or finance post actuals'} onClick={post}>Post to project actuals</Button>}
        </div>
      </div>
      {blocked && staged && <div className="px-4 pb-3"><Callout tone="bad" title="Posting is blocked">{blocked}. Map each exception below to a line item, assign a project, or exclude it with a note — then post again.</Callout></div>}
      {imp.status === 'discarded' && <div className="px-4 pb-3"><Callout tone="info">This extract was discarded; its rows were removed and nothing reached project actuals.</Callout></div>}

      <div className="grid grid-cols-2 gap-px border-y border-line bg-line sm:grid-cols-4 xl:grid-cols-7">
        <Fig label="Rows" value={F.rows.toLocaleString('en-US')} sub={`${F.projects} project${F.projects === 1 ? '' : 's'}`} />
        <Fig label="Total" value={money(F.total)} sub={$(F.total)} />
        <Fig label="Mapped to a line" value={money(F.mapped)} sub={`${txns.filter((t) => t.status === 'mapped').length} rows`} tone="good" />
        <Fig label="Allocated" value={money(F.allocated)} sub="spread pro-rata by rule" />
        <Fig label="Excluded" value={money(F.excluded)} sub={`${txns.filter((t) => t.status === 'excluded').length} rows · not project cost`} />
        <Fig label="Unmapped" value={money(F.unmapped)} sub={`${txns.filter((t) => t.status === 'unmapped').length} rows`} tone={F.unmapped ? 'bad' : undefined} />
        <Fig label="No project" value={money(F.noProject)} sub={`${txns.filter((t) => t.status === 'no-project').length} rows`} tone={F.noProject ? 'bad' : undefined} />
      </div>

      {txns.length === 0 ? <div className="p-4"><Empty title="No rows">{imp.status === 'discarded' ? 'Discarded extracts keep only their summary.' : 'This extract has no rows.'}</Empty></div> : <>
        <Tabs className="px-2 pt-1" value={tab} onChange={setTab} tabs={[
          { id: 'exceptions', label: staged ? 'Exceptions & checks' : 'Resolutions & checks', count: ex.length + review.length },
          { id: 'rollup', label: 'By project & line item', count: F.projects },
          { id: 'txns', label: 'Transactions', count: txns.length },
        ]} />
        <div className="p-3 sm:p-4">
          {tab === 'exceptions' && <Exceptions s={s} imp={imp} ex={ex} review={review} />}
          {tab === 'rollup' && <Rollup s={s} txns={txns} posted={imp.status === 'posted'} />}
          {tab === 'txns' && <Txns s={s} txns={txns} />}
        </div>
      </>}
    </Panel>
  )
}

function Fig({ label, value, sub, tone }: { label: string; value: string; sub?: string; tone?: 'good' | 'bad' }) {
  return (
    <div className="min-w-0 bg-surface px-4 py-2.5">
      <div className="eyebrow truncate">{label}</div>
      <div className={cx('mt-0.5 font-display text-[19px] font-bold leading-tight tabular-nums', tone === 'bad' ? 'text-bad' : 'text-ink')}>{value}</div>
      {sub && <div className="truncate text-[11.5px] text-muted" title={sub}>{sub}</div>}
    </div>
  )
}

  return (

        ? <Callout tone="bad" title={`${ex.length} row${ex.length === 1 ? '' : 's'} need project controls`}>Each one blocks posting. Map it to a forecast line, assign the right project, or exclude it — every resolution keeps a note and goes to the audit log. If a whole cost element is missing from the mapping table, add a rule in Configuration → SAP mapping and re-run mapping instead.</Callout>

function ExceptionRow({ s, t, editable }: { s: AppState; t: ActualTxn; editable: boolean }) {
  const me = useMe()
  const noProject = t.status === 'no-project'
  const [projectId, setProjectId] = useState('')
  const rule = useMemo(() => testRow(s, { costElement: t.costElement, sapWbs: t.sapWbs, text: t.text }), [s, t])
  const [mode, setMode] = useState<'line' | 'rule' | 'exclude'>(noProject && rule.winner ? 'rule' : 'line')
  const [lineId, setLineId] = useState(rule.line?.id ?? '')
  const [note, setNote] = useState('')
  const lines = forecastLines(s)
  const fnCode = (id: string) => s.settings.wbsFunctions.find((f) => f.id === id)?.code ?? ''
  const resolve = () => {
    if (noProject && !projectId) { toast('Choose the project this cost belongs to', 'bad'); return }
    if (mode === 'line' && !lineId) { toast('Choose a forecast line', 'bad'); return }
    if (!note.trim()) { toast('Add a note — why this resolution is right', 'bad'); return }
    const to = mode === 'exclude' ? { exclude: true, projectId: projectId || undefined } : mode === 'rule' ? { projectId } : { lineId, projectId: projectId || undefined }

          </div>
          <div className="mt-1 text-[12.5px]">{exceptionReason(s, t)}</div>
        </div>
        <div className={cx('shrink-0 font-display text-[18px] font-bold tabular-nums', t.amount < 0 && 'text-bad')}>{$(t.amount)}</div>
      </div>
      <div className="grid grid-cols-2 gap-x-4 gap-y-1 px-3 py-2 text-[12px] sm:grid-cols-4">
        <KVs k="WBS element" v={<span className="font-mono">{t.sapWbs}</span>} />

                <Select value={projectId} onChange={(e) => setProjectId(e.target.value)} options={[{ value: '', label: '— Select project —' }, ...s.projects.filter((p) => p.status !== 'closed').map((p) => ({ value: p.id, label: `${p.code} · ${p.erpWbs} · ${p.name}` }))]} />
              </Field>
            )}
            <Field label="Resolution">
              <div className="flex flex-col gap-1.5">
                <SegButtons value={mode} onChange={setMode} options={[...(noProject && rule.winner ? [{ id: 'rule' as const, label: 'Use mapping rule' }] : []), { id: 'line' as const, label: 'Map to a line' }, { id: 'exclude' as const, label: 'Exclude' }]} />
                {mode === 'rule' && rule.winner && <span className="text-[11.5px] text-muted">{ruleLabel(s, rule.winner)}</span>}

                {mode === 'exclude' && <span className="text-[11.5px] text-muted">Kept in the import record but out of project actuals.</span>}

  )

/** Dense SAP line-item table. With `share`, an extra column shows what each row contributes to that line / function. */
export function TxnTable({ s, txns, share, showProject, showStatus, limit = 200, onRow }: {
  s: AppState; txns: ActualTxn[]; share?: { lineId?: string; fnId?: string }; showProject?: boolean; showStatus?: boolean; limit?: number; onRow?: (t: ActualTxn) => void
}) {
  const [n, setN] = useState(limit)
  const rows = txns.slice(0, n)
  return (

/** The SAP transactions behind one actual cell (a line or a function, one month or a collapsed year). */
export function ActualDrill({ projectId, lineId, fnId, months, cellAmount, onClose }: { projectId: string; lineId?: string; fnId?: string; months: string[]; cellAmount: number; onClose: () => void }) {
  const s = useStore()
  const p = s.projects.find((x) => x.id === projectId)
  const txns = useMemo(() => months.flatMap((m) => txnsFor(s, projectId, { lineId, fnId, period: m })).sort((a, b) => a.postingDate.localeCompare(b.postingDate)), [s, projectId, lineId, fnId, months])
  const share = { lineId, fnId: lineId ? undefined : fnId }
  const total = sum(txns.map((t) => txnShare(s, t, share)))
  const cps = s.costPeriods.filter((c) => c.projectId === projectId && months.includes(c.period))
  const summaryMonths = cps.filter((c) => c.source !== 'sap').map((c) => c.period).sort()
  const line = lineById(s, lineId)
  const fn = s.settings.wbsFunctions.find((f) => f.id === (fnId ?? line?.functionId))
  const what = line ? `${line.code} ${line.name}` : fn ? `${fn.code} ${fn.name}` : 'All lines'
  const when = months.length === 1 ? fmtMonth(months[0]) : `${fmtMonth(months[0])} – ${fmtMonth(months[months.length - 1])}`
  const diff = Math.round(cellAmount - total)
  return (
    <Drawer open onClose={onClose} width={860} title={<span>Actuals · {what}</span>}
      footer={<><a href="#actuals" className="mr-auto self-center text-[12.5px] font-semibold text-link hover:underline">Open SAP actuals</a><Button onClick={onClose}>Close</Button></>}>
      <div className="flex flex-col gap-3">
        <div className="grid grid-cols-2 gap-px overflow-hidden rounded-md border border-line bg-line sm:grid-cols-4">
          <Cell label="Project" value={p?.code ?? '—'} sub={p?.erpWbs} />
          <Cell label="Period" value={when} sub={`${months.length} month${months.length === 1 ? '' : 's'}`} />
          <Cell label="Actual on the grid" value={$(cellAmount)} />
          <Cell label="From SAP rows" value={$(total)} sub={`${txns.length} transaction${txns.length === 1 ? '' : 's'}`} />
        </div>
        {summaryMonths.length > 0 && (
          <Callout tone="info" title={`${summaryMonths.length === months.length ? 'This period predates' : 'Part of this period predates'} the SAP line-item feed`}>
            {summaryMonths.map(fmtMonth).join(', ')} {summaryMonths.length === 1 ? 'was' : 'were'} loaded as monthly summary actuals from the legacy ERP and split to lines by the recorded function split and budget share. There are no line items to show for {summaryMonths.length === 1 ? 'it' : 'them'}.
          </Callout>
        )}
        {diff !== 0 && summaryMonths.length === 0 && txns.length > 0 && <Callout tone="warn">The grid and the SAP rows differ by {$(diff)} — whole-dollar rounding of allocated rows across lines.</Callout>}
        {txns.length ? <TxnTable s={s} txns={txns} share={share} /> : summaryMonths.length === 0 && <Empty title="No SAP rows">Nothing was posted to {what} in {when}.</Empty>}
        <p className="text-[11.5px] text-faint">Rows come from posted SAP extracts. Each row reached this line through the SAP mapping table (cost element range, WBS suffix, text) or a manual resolution by project controls; loader settlements are allocated over the period’s direct lines pro-rata.</p>
      </div>
    </Drawer>
  )
}
